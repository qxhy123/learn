\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=11cm, height=11cm,
  xmin=-3.3, xmax=5.3, ymin=-3.3, ymax=5.3,
  axis lines=middle,
  axis equal image,
  xlabel={$x$}, ylabel={$y$},
  xlabel style={at={(axis description cs:1,0.5)},anchor=west},
  ylabel style={at={(axis description cs:0.5,1)},anchor=south},
  xtick={-3,-2,-1,1,2,3,4,5}, ytick={-3,-2,-1,1,2,3,4,5},
  tick label style={font=\footnotesize},
  clip=false,
]
  % y = x (mirror line)
  \addplot[dashed, gray, thick, domain=-3.2:5.2]{x};

  % y = 2^x
  \addplot[very thick, blue, domain=-3.2:2.32, samples=200]{2^x};
  % y = log_2 x
  \addplot[very thick, red!80!black, domain=0.09:5.2, samples=200]{ln(x)/ln(2)};

  % labels
  \node[blue, font=\small] at (axis cs:1.55,4.55) {$y=2^{x}$};
  \node[red!80!black, font=\small] at (axis cs:4.55,2.05) {$y=\log_2 x$};
  \node[gray, font=\small] at (axis cs:4.45,4.95) {$y=x$};

  % a pair of mirror points: (1,2) on 2^x and (2,1) on log_2 x
  \addplot[only marks, mark=*, mark size=2pt, blue] coordinates {(1,2)};
  \addplot[only marks, mark=*, mark size=2pt, red!80!black] coordinates {(2,1)};
  \node[blue, font=\footnotesize, anchor=south east] at (axis cs:1,2.1) {$(1,2)$};
  \node[red!80!black, font=\footnotesize, anchor=north west] at (axis cs:2.05,1.0) {$(2,1)$};
  % thin connector showing reflection across y=x
  \addplot[densely dotted, gray] coordinates {(1,2) (2,1)};

  % annotation box: empty lower-right region
  \node[draw, rounded corners, fill=yellow!18, fill opacity=0.80, text opacity=1,
        align=left, font=\small, inner sep=5pt]
    at (axis cs:2.55,-2.05)
    {$y=2^{x}$ 与 $y=\log_2 x$\\ 关于直线 $y=x$ 对称\\（互为反函数）};
\end{axis}
\end{tikzpicture}
\end{document}
