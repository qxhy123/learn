\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=13cm, height=9cm,
  xmin=-0.02, xmax=1.05, ymin=-0.12, ymax=1.24,
  axis lines=middle,
  xlabel={$p$}, ylabel={$H(p)$（bit）},
  xlabel style={at={(axis description cs:1,0.10)},anchor=north east},
  ylabel style={at={(axis description cs:0.02,1)},anchor=north west},
  xtick={0,0.25,0.5,0.75,1},
  xticklabels={$0$,$\tfrac14$,$\tfrac12$,$\tfrac34$,$1$},
  ytick={0.25,0.5,0.75,1},
  tick label style={font=\small},
  domain=0.0001:0.9999, samples=400,
  clip=false,
]
  % binary entropy H(p) = -p log2 p - (1-p) log2 (1-p)
  \addplot[very thick, blue]
    {-x*ln(x)/ln(2) - (1-x)*ln(1-x)/ln(2)};

  % endpoints H(0)=H(1)=0
  \addplot[only marks, mark=*, mark size=2.2pt, black]
    coordinates {(0,0) (1,0)};
  \node[anchor=north east, font=\footnotesize] at (axis cs:0,0) {$0$};
  \node[anchor=north, font=\footnotesize] at (axis cs:1,-0.01) {$1$};

  % maximum at p=1/2, H=1
  \addplot[only marks, mark=*, mark size=2.4pt, red!80!black]
    coordinates {(0.5,1)};
  \draw[gray, dashed] (axis cs:0.5,0) -- (axis cs:0.5,1);
  \draw[gray, dashed] (axis cs:0,1) -- (axis cs:0.5,1);
  \node[red!80!black, font=\small, anchor=south] at (axis cs:0.5,1.02)
    {最大值 $H(\tfrac12)=1$ bit};

  % endpoint annotation
  \node[font=\footnotesize, align=center, anchor=north] at (axis cs:0.12,0.30)
    {$p\to0$\\$H\to0$};
  \node[font=\footnotesize, align=center, anchor=north] at (axis cs:0.88,0.30)
    {$p\to1$\\$H\to0$};

  % annotation box (semi-transparent), in empty lower-right, avoiding peak
  \node[draw, rounded corners, fill=yellow!18, fill opacity=0.80, text opacity=1,
        align=left, font=\small, inner sep=5pt]
    at (axis cs:0.80,0.66)
    {二元熵 $H(p)=$\\ $-p\log_2 p-(1-p)\log_2(1-p)$\\[2pt]
     最公平时（$p=\tfrac12$）最不确定};
\end{axis}
\end{tikzpicture}
\end{document}
