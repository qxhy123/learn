\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
% ---- LEFT: original data y = C e^{kx}, curved ----
\begin{axis}[
  name=left,
  width=8.4cm, height=8.4cm,
  title={\textbf{原始坐标：弯曲}},
  title style={yshift=1.3cm},
  xmin=0, xmax=5.4, ymin=0, ymax=125,
  axis lines=left,
  xlabel={$x$}, ylabel={$y$},
  xtick={0,1,2,3,4,5}, ytick={0,30,60,90,120},
  tick label style={font=\footnotesize},
  clip=false,
]
  % y = 10 e^{0.6 x}
  \addplot[very thick, blue, domain=0:5, samples=120]{10*exp(0.6*x)};
  % data points on the curve
  \addplot[only marks, mark=*, mark size=2pt, blue]
    coordinates {(0,10) (1,18.2) (2,33.2) (3,60.5) (4,110.2)};
  \node[blue, font=\small, anchor=west] at (axis cs:0.5,95)
    {$y=Ce^{kx}$};
\end{axis}

% arrow between panels
\node at ($(left.east)+(1.05,-0.2)$) {};
\draw[-{Stealth[length=4mm]}, very thick, black!70]
  ($(left.east)+(0.25,0.0)$) -- ++(1.55,0)
  node[midway, above, font=\footnotesize, align=center] {纵轴\\取 $\ln y$};

% ---- RIGHT: semi-log, straight line ----
\begin{axis}[
  name=right,
  at={($(left.east)+(3.1cm,0)$)}, anchor=west,
  width=8.4cm, height=8.4cm,
  title={\textbf{半对数坐标：变直线}},
  title style={yshift=1.3cm},
  xmin=0, xmax=5.4, ymin=1.8, ymax=5.4,
  axis lines=left,
  xlabel={$x$}, ylabel={$\ln y$},
  xtick={0,1,2,3,4,5}, ytick={2,3,4,5},
  tick label style={font=\footnotesize},
  clip=false,
]
  % ln y = ln C + k x  ->  ln 10 + 0.6 x
  \addplot[very thick, red!80!black, domain=0:5, samples=2]{ln(10)+0.6*x};
  \addplot[only marks, mark=*, mark size=2pt, red!80!black]
    coordinates {(0,2.303) (1,2.901) (2,3.502) (3,4.103) (4,4.702)};
  \node[red!80!black, font=\small, anchor=west] at (axis cs:0.35,5.05)
    {$\ln y=\ln C+kx$};

  % slope annotation, lower-right empty area
  \node[draw, rounded corners, fill=yellow!18, fill opacity=0.80, text opacity=1,
        align=left, font=\footnotesize, inner sep=4pt]
    at (axis cs:3.55,2.55)
    {斜率 $=k$\\ 截距 $=\ln C$};
\end{axis}
\end{tikzpicture}
\end{document}
