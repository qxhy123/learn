\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=13cm, height=9.5cm,
  xlabel={$x$}, ylabel={$y$},
  xmin=0, xmax=10, ymin=0, ymax=10,
  axis lines=left,
  xtick={1,2,4,6,8,10},
  ytick={1,2,4,6,8,10},
  tick label style={font=\small},
  label style={font=\small},
  domain=0.05:10, samples=200,
  enlargelimits=false,
  clip=true,
  restrict y to domain=-20:60,
  legend style={font=\small, at={(0.02,0.98)}, anchor=north west,
                draw=black!40, fill=white, fill opacity=0.85, text opacity=1},
]
  % 指数 2^x（限制定义域避免数值溢出）
  \addplot[red, very thick, domain=0.05:3.33, samples=120] {2^x};
  \addlegendentry{$2^x$（指数，最快）}
  % 幂 sqrt(x)
  \addplot[blue, very thick] {sqrt(x)};
  \addlegendentry{$\sqrt{x}$（幂）}
  % 对数 ln x
  \addplot[teal!80!black, very thick] {ln(x)};
  \addlegendentry{$\ln x$（对数，最慢）}
\end{axis}
% 在轴外放层级关系标注，半透明
\node[draw=black!40, fill=yellow!18, fill opacity=0.9, text opacity=1,
      rounded corners, font=\small] at (8.1,1.1)
  {$\ln x \ll \sqrt{x} \ll 2^x$};
\end{tikzpicture}
\end{document}
