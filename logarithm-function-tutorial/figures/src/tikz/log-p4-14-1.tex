\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=13cm, height=8.5cm,
  xlabel={$t$}, ylabel={$y=\dfrac1t$},
  xmin=0, xmax=5.2, ymin=0, ymax=2.3,
  axis lines=left,
  xtick={1,2,3,4,5},
  xticklabels={$1$,$2$,$3$,$x=4$,$5$},
  ytick={0.25,0.5,1,2},
  yticklabels={$\frac14$,$\frac12$,$1$,$2$},
  tick label style={font=\small},
  label style={font=\small},
  domain=0.5:5.2, samples=250,
  enlargelimits=false,
  clip=true,
]
  % 阴影区域：1 到 x(=4) 之间 1/t 下方
  \addplot[draw=none, fill=teal!25, domain=1:4] {1/x} \closedcycle;

  % 曲线 y = 1/t
  \addplot[teal!80!black, very thick, domain=0.55:5.2] {1/x};
  \node[teal!60!black, anchor=south west, font=\small] at (axis cs:4.3,1/4.3) {$y=\dfrac1t$};

  % 竖直边界线 t=1 与 t=x(=4)
  \draw[gray!70, dashed] (axis cs:1,0) -- (axis cs:1,1);
  \draw[gray!70, dashed] (axis cs:4,0) -- (axis cs:4,0.25);

  % 阴影面积标注（半透明，放在阴影区域内）
  \node[font=\large, teal!35!black] at (axis cs:2.05,0.30) {面积 $=\ln x$};

  % 公式注释框，半透明，放右上避开曲线
  \node[draw=black!40, fill=yellow!18, fill opacity=0.9, text opacity=1,
        rounded corners, align=center, font=\small, anchor=north east]
        at (axis cs:5.1,2.25)
    {$\displaystyle \ln x=\int_{1}^{x}\frac{1}{t}\,dt$\\[3pt]
     （此处取 $x=4$）};
\end{axis}
\end{tikzpicture}
\end{document}
