\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
% Top: function-graph view of log_2 x < 3
\begin{axis}[
  name=graph,
  width=12cm, height=7.2cm,
  xmin=-0.6, xmax=11.5, ymin=-1.6, ymax=4.3,
  axis lines=middle,
  xlabel={$x$}, ylabel={$y$},
  xlabel style={at={(axis description cs:1,0.34)},anchor=west},
  ylabel style={at={(axis description cs:0.5,1)},anchor=south},
  xtick={1,8}, ytick={3},
  tick label style={font=\small},
  clip=false,
]
  % y = log_2 x
  \addplot[very thick, blue, domain=0.1:11.3, samples=250]{ln(x)/ln(2)};
  \node[blue, font=\small] at (axis cs:9.6,3.65) {$y=\log_2 x$};

  % horizontal line y = 3
  \addplot[very thick, red!80!black, domain=0:11.3]{3};
  \node[red!80!black, font=\small, anchor=south east] at (axis cs:3.0,3.05) {$y=3$};

  % intersection at (8,3)
  \addplot[only marks, mark=*, mark size=2pt, black] coordinates {(8,3)};
  \draw[densely dashed, gray] (axis cs:8,0) -- (axis cs:8,3);

  % shade region where curve is below the line: 0 < x < 8
  \addplot[draw=none, fill=blue!12, fill opacity=0.6, domain=0.1:8, samples=120]
    {ln(x)/ln(2)} \closedcycle;

  % annotation box, upper-left empty area
  \node[draw, rounded corners, fill=yellow!18, fill opacity=0.80, text opacity=1,
        align=left, font=\small, inner sep=5pt]
    at (axis cs:2.7,2.0)
    {曲线在 $y=3$ 下方\\ 对应 $0<x<8$};
\end{axis}

% Bottom: number line of the solution set
\begin{scope}[shift={(0.2,-2.7)}]
  \draw[->, thick] (-0.7,0) -- (11.6,0) node[right, font=\small] {$x$};
  % ticks
  \foreach \x/\lab in {0/0, 8/8}{
    \draw[thick] (\x*1.0,0.12) -- (\x*1.0,-0.12);
  }
  \node[font=\small, anchor=north] at (0,-0.2) {$0$};
  \node[font=\small, anchor=north] at (8,-0.2) {$8$};
  % open endpoints at 0 and 8, shaded solution segment
  \draw[very thick, green!55!black] (0,0) -- (8,0);
  \fill[white] (0,0) circle (2.4pt);
  \draw[very thick, green!55!black] (0,0) circle (2.4pt);
  \fill[white] (8,0) circle (2.4pt);
  \draw[very thick, green!55!black] (8,0) circle (2.4pt);
  \node[green!45!black, font=\small, anchor=south] at (4,0.18) {解集 $(0,\,8)$};
\end{scope}
\end{tikzpicture}
\end{document}
