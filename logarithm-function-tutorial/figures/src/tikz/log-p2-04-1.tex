\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=12cm, height=9cm,
    axis lines=middle,
    xlabel={$x$}, ylabel={$y$},
    xlabel style={at={(ticklabel* cs:1)}, anchor=west},
    ylabel style={at={(ticklabel* cs:1)}, anchor=south},
    xmin=-0.6, xmax=6.4, ymin=-3.2, ymax=3.2,
    xtick={1,2,3,4,5,6},
    ytick={-3,-2,-1,1,2,3},
    domain=0.04:6.2, samples=300,
    legend pos=south east,
    legend cell align=left,
    clip=false,
]
% 竖直渐近线 x=0 (y轴)
\addplot[gray, dashed, thick, domain=-3:3, forget plot] ({0},{x});
\node[gray, anchor=west, font=\small] at (axis cs:0.12,-2.7) {渐近线 $x=0$};

% a>1: y=log_2 x  递增
\addplot[blue, very thick, domain=0.06:6.2] {ln(x)/ln(2)};
\addlegendentry{$y=\log_2 x$\quad($a>1$，递增)}

% 0<a<1: y=log_{1/2} x  递减
\addplot[red, very thick, domain=0.06:6.2] {ln(x)/ln(0.5)};
\addlegendentry{$y=\log_{1/2} x$\quad($0<a<1$，递减)}

% 过 (1,0)
\filldraw[black] (axis cs:1,0) circle (2pt);
\node[anchor=north west, font=\small] at (axis cs:1.05,-0.12) {$(1,0)$};

% 关键点 (2,1)
\filldraw[blue] (axis cs:2,1) circle (1.6pt);
\node[blue, anchor=south, font=\footnotesize] at (axis cs:2,1.12) {$(2,1)$};

% 关键点 (2,-1)
\filldraw[red] (axis cs:2,-1) circle (1.6pt);
\node[red, anchor=north, font=\footnotesize] at (axis cs:2,-1.12) {$(2,-1)$};

% 半透明注释框：定义域与值域
\node[fill=yellow!18, draw=black!35, rounded corners, align=left,
      fill opacity=0.75, text opacity=1, font=\small, anchor=north east]
      at (axis cs:6.2,3.1)
      {定义域 $(0,+\infty)$\\值域 $\mathbb{R}$};
\end{axis}
\end{tikzpicture}
\end{document}
