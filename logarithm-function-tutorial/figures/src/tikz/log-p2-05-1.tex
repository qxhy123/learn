\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=11cm, height=11cm,
    axis equal image,
    axis lines=middle,
    xlabel={$x$}, ylabel={$y$},
    xlabel style={at={(ticklabel* cs:1)}, anchor=west},
    ylabel style={at={(ticklabel* cs:1)}, anchor=south},
    xmin=-3.2, xmax=4.2, ymin=-3.2, ymax=4.2,
    xtick={-3,-2,-1,1,2,3,4},
    ytick={-3,-2,-1,1,2,3,4},
    samples=300,
    legend pos=north west,
    legend cell align=left,
    clip=false,
]
% 对称轴 y=x
\addplot[black!55, dashed, thick, domain=-3:4, forget plot] {x};
\node[black!60, anchor=south west, font=\small] at (axis cs:3.0,3.0) {$y=x$};

% 指数函数 y=2^x
\addplot[blue, very thick, domain=-3.1:2.05] {2^x};
\addlegendentry{$y=2^x$}

% 对数函数 y=log_2 x
\addplot[red, very thick, domain=0.06:4.1] {ln(x)/ln(2)};
\addlegendentry{$y=\log_2 x$}

% 对称点对 (0,1) <-> (1,0)
\filldraw[blue] (axis cs:0,1) circle (1.7pt);
\node[blue, anchor=east, font=\footnotesize] at (axis cs:-0.1,1.15) {$(0,1)$};
\filldraw[red] (axis cs:1,0) circle (1.7pt);
\node[red, anchor=north west, font=\footnotesize] at (axis cs:1.1,-0.1) {$(1,0)$};
\draw[black!45, dotted, thick] (axis cs:0,1) -- (axis cs:1,0);

% 对称点对 (1,2) <-> (2,1)
\filldraw[blue] (axis cs:1,2) circle (1.7pt);
\node[blue, anchor=south east, font=\footnotesize] at (axis cs:0.95,2.12) {$(1,2)$};
\filldraw[red] (axis cs:2,1) circle (1.7pt);
\node[red, anchor=north west, font=\footnotesize] at (axis cs:2.1,0.95) {$(2,1)$};
\draw[black!45, dotted, thick] (axis cs:1,2) -- (axis cs:2,1);

% 半透明注释框
\node[fill=yellow!18, draw=black!35, rounded corners, align=left,
      fill opacity=0.78, text opacity=1, font=\small, anchor=south east]
      at (axis cs:4.1,-3.1)
      {$y=2^x$ 与 $y=\log_2 x$ 互为反函数，\\图像关于直线 $y=x$ 对称};
\end{axis}
\end{tikzpicture}
\end{document}
