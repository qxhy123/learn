\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=13cm, height=9cm,
  xlabel={$x$}, ylabel={$y$},
  xmin=0, xmax=8, ymin=-1.6, ymax=2.4,
  axis lines=left,
  xtick={1,2,3,4,5,6,7},
  ytick={-1,0,1,2},
  tick label style={font=\small},
  label style={font=\small},
  domain=0.12:8, samples=200,
  enlargelimits=false,
  clip=true,
]
  % y = ln x
  \addplot[teal!80!black, very thick] {ln(x)};
  \node[teal!60!black, anchor=east, font=\small] at (axis cs:7.6,1.78) {$y=\ln x$};

  % 切点 (2, ln2), 斜率 1/2: 切线 y = ln2 + (1/2)(x-2)
  % ln2 ≈ 0.6931
  \addplot[red, thick, domain=0.2:7.6] {0.6931472 + 0.5*(x-2)};
  \node[red, anchor=west, font=\small] at (axis cs:6.5,3.0) {};

  % 切点
  \addplot[only marks, mark=*, mark size=2pt, black] coordinates {(2,0.6931472)};
  \node[anchor=south east, font=\footnotesize] at (axis cs:1.9,0.78) {$(x_0,\ln x_0)=(2,\ \ln 2)$};

  % 斜率直角三角形：从 (2, ln2) 水平到 (4, ln2) 再竖直到 (4, ln2+1)
  \draw[blue!70!black, thick] (axis cs:2,0.6931472) -- (axis cs:4,0.6931472)
        node[midway, below, font=\footnotesize, blue!70!black] {$\Delta x=2$};
  \draw[blue!70!black, thick] (axis cs:4,0.6931472) -- (axis cs:4,1.6931472)
        node[midway, right, font=\footnotesize, blue!70!black] {$\Delta y=1$};

  % 斜率标注框，半透明，避开曲线峰值
  \node[draw=black!40, fill=yellow!18, fill opacity=0.9, text opacity=1,
        rounded corners, align=left, font=\footnotesize, anchor=north west]
        at (axis cs:0.25,2.3)
    {切线斜率 $=(\ln x)'\big|_{x_0=2}=\dfrac{1}{x_0}=\dfrac12$\\[2pt]
     即 $\dfrac{\Delta y}{\Delta x}=\dfrac12$};

  % 红色切线标签
  \node[red, anchor=south east, font=\footnotesize] at (axis cs:7.6,3.6) {};
  \node[red, anchor=west, font=\small] at (axis cs:5.6,2.55) {切线};
\end{axis}
\end{tikzpicture}
\end{document}
