\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=13cm, height=9cm,
  xmin=0, xmax=10.5, ymin=-1.6, ymax=3.7,
  axis lines=middle,
  xlabel={$x$}, ylabel={$y$},
  xlabel style={at={(axis description cs:1,0.32)},anchor=west},
  ylabel style={at={(axis description cs:0.5,1)},anchor=south},
  xtick={1,2,4,10}, ytick={-1,1,2,3},
  tick label style={font=\small},
  domain=0.12:10.4, samples=300,
  clip=false,
]
  % three log curves with different bases (换底: log_a x = ln x / ln a)
  \addplot[very thick, blue]   {ln(x)/ln(2)};
  \addplot[very thick, red!80!black]    {ln(x)/ln(4)};
  \addplot[very thick, green!55!black]  {ln(x)/ln(10)};

  % common point (1,0)
  \addplot[only marks, mark=*, mark size=2pt, black] coordinates {(1,0)};
  \node[anchor=north west, font=\small] at (axis cs:1.05,-0.08) {$(1,0)$};

  % curve labels
  \node[blue, font=\small] at (axis cs:3.0,2.55) {$y=\log_2 x$};
  \node[red!80!black, font=\small] at (axis cs:7.6,2.25) {$y=\log_4 x$};
  \node[green!55!black, font=\small] at (axis cs:8.7,1.18) {$y=\log_{10} x$};

  % annotation box (semi-transparent), placed in empty lower-right
  \node[draw, rounded corners, fill=yellow!18, fill opacity=0.78, text opacity=1,
        align=left, font=\small, inner sep=5pt]
    at (axis cs:7.3,-0.95)
    {底数越大，曲线越平缓\\（增长越慢）：\\ $\log_2 x>\log_4 x>\log_{10}x\ (x>1)$};
\end{axis}
\end{tikzpicture}
\end{document}
