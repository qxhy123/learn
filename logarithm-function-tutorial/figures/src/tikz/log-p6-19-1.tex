\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
  \begin{axis}[
      width=12cm, height=8.4cm,
      xmin=0, xmax=8.6, ymin=-0.6, ymax=2.4,
      axis lines=middle,
      xlabel={$x$}, ylabel={$y$},
      xtick={1,2,4,7}, xticklabels={$1$,$a$,$\tfrac{a+b}{2}$,$b$},
      ytick=\empty,
      clip=false,
      every axis x label/.style={at={(rel axis cs:1,0)},anchor=west},
      every axis y label/.style={at={(rel axis cs:0,1)},anchor=south},
    ]

    % 曲线 y = ln x
    \addplot[domain=0.18:8.4,samples=200,thick,blue!70!black] {ln(x)};
    \node[blue!70!black] at (axis cs:7.6,2.18) {$y=\ln x$};

    % a, b 处坐标
    \def\xa{2}\def\xb{7}
    % ln2 = 0.6931, ln7 = 1.9459, midpoint x = 4.5 -> ln4.5=1.504
    \coordinate (A) at (axis cs:2,0.6931);
    \coordinate (B) at (axis cs:7,1.9459);

    % 割线（弦）a 到 b
    \draw[red!75!black,thick] (A) -- (B);
    \node[red!70!black,font=\small] at (axis cs:3.0,0.78) {弦（割线）};

    % 端点
    \filldraw[blue!70!black] (A) circle (1.6pt);
    \filldraw[blue!70!black] (B) circle (1.6pt);

    % 中点横坐标 m = (a+b)/2 = 4.5
    \def\xm{4.5}
    % 曲线上的点：(m, ln m)，ln 4.5 = 1.5041
    \coordinate (Ccurve) at (axis cs:4.5,1.5041);
    % 弦上的点：纵坐标 = (ln a + ln b)/2 = (0.6931+1.9459)/2 = 1.3195
    \coordinate (Cchord) at (axis cs:4.5,1.3195);

    \filldraw[blue!70!black] (Ccurve) circle (1.8pt);
    \filldraw[red!75!black] (Cchord) circle (1.8pt);

    % 竖直辅助线
    \draw[gray!55,dashed] (axis cs:4.5,0) -- (Ccurve);
    \draw[gray!55,dashed] (axis cs:2,0) -- (A);
    \draw[gray!55,dashed] (axis cs:7,0) -- (B);

    % 标注两点纵坐标差（凹性：曲线在弦上方）
    \draw[<->,teal!60!black,line width=0.8pt]
      (axis cs:4.78,1.3195) -- (axis cs:4.78,1.5041);
    \node[teal!55!black,right,font=\small,align=left]
      at (axis cs:4.86,1.41) {凹性间隙};

    % 标签：曲线点
    \node[blue!70!black,above left=-1pt,font=\small,align=center]
      at (Ccurve) {$\ln\dfrac{a+b}{2}$};
    % 标签：弦点（移到点的左侧，避开“弦（割线）”标签）
    \node[red!70!black,left=2pt,font=\small,align=center]
      at (Cchord) {$\dfrac{\ln a+\ln b}{2}$};

    % 半透明注释框：Jensen 不等式（放在右下空白处，避开曲线/弦）
    \node[fill=orange!12,draw=orange!55!black,rounded corners,
          align=center,font=\small,inner sep=5pt,fill opacity=0.85,text opacity=1]
      at (axis cs:6.0,0.28)
      {弦在曲线下方 $\Rightarrow$ 凹函数\\[2pt]
       $\ln\dfrac{a+b}{2}\ \ge\ \dfrac{\ln a+\ln b}{2}$\\[2pt]
       （Jensen $\Rightarrow$ AM--GM）};

  \end{axis}
\end{tikzpicture}
\end{document}
