% 例 1（平行投影：旗杆相似）：竹竿 1.5/1，旗杆 h/8
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[thick, scale=0.7]
  \draw[gray] (-0.5,0) -- (12,0);
  % 竹竿 1.5, 影 1
  \coordinate (D) at (1,0);
  \coordinate (C) at (1,1.5);
  \draw[very thick] (D) -- (C);
  \coordinate (Ds) at (2,0);
  \draw[red, thick] (D) -- (Ds);
  \draw[->, orange, thick] ($(C)+(-1,1)$) -- (C);
  \draw[->, orange, thick] ($(Ds)+(-1,1)$) -- (Ds);
  \draw (C) -- (Ds);
  \node[left, font=\footnotesize] at ($(D)!0.5!(C)$) {$1.5$};
  \node[below, font=\footnotesize] at ($(D)!0.5!(Ds)$) {$1$};

  % 旗杆 h, 影 8
  \coordinate (B) at (4,0);
  \coordinate (A) at (4,4.5);  % 比例 1.5/1 = h/8 => h=12，画 4.5 单位高即可
  \draw[very thick] (B) -- (A);
  \coordinate (Bs) at (10,0);
  \draw[red, thick] (B) -- (Bs);
  \draw[->, orange, thick] ($(A)+(-1,1)$) -- (A);
  \draw[->, orange, thick] ($(Bs)+(-1,1)$) -- (Bs);
  \draw (A) -- (Bs);
  \node[left, font=\footnotesize] at ($(B)!0.5!(A)$) {$h$};
  \node[below, font=\footnotesize] at ($(B)!0.5!(Bs)$) {$8$};

  \node[below, font=\footnotesize] at (6,-1) {$\dfrac{h}{8} = \dfrac{1.5}{1} \Rightarrow h = 12$};
\end{tikzpicture}
\end{document}
