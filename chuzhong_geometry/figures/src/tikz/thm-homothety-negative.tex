% 反向位似：异侧，k<0，△ABC 与 △A'B'C' (k=-1)
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[scale=0.9, thick]
  \coordinate (O) at (0,0);
  \coordinate (A) at (1.4, 0.6);
  \coordinate (B) at (2.4, 1.6);
  \coordinate (C) at (0.8, 1.8);
  % k=-1：A' = -A
  \coordinate (Ap) at (-1.4, -0.6);
  \coordinate (Bp) at (-2.4, -1.6);
  \coordinate (Cp) at (-0.8, -1.8);

  \draw (A) -- (B) -- (C) -- cycle;
  \draw[blue, dashed] (Ap) -- (Bp) -- (Cp) -- cycle;
  \draw[gray, dashed] (Ap) -- (A);
  \draw[gray, dashed] (Bp) -- (B);
  \draw[gray, dashed] (Cp) -- (C);

  \fill (O) circle (1.4pt);
  \fill (A) circle (1.2pt); \fill (B) circle (1.2pt); \fill (C) circle (1.2pt);
  \fill (Ap) circle (1.2pt); \fill (Bp) circle (1.2pt); \fill (Cp) circle (1.2pt);
  \node[above right] at (O) {$O$};
  \node[below right] at (A) {$A$};
  \node[right] at (B) {$B$};
  \node[above] at (C) {$C$};
  \node[above left] at (Ap) {$A'$};
  \node[left] at (Bp) {$B'$};
  \node[below] at (Cp) {$C'$};
  \node[below] at (0,-2.4) {\footnotesize $k<0$（异侧）};
\end{tikzpicture}
\end{document}
