% D.29 将军饮马：l 同侧两点 A、B；A 到 l 距 2，B 距 4，投影距 6
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[scale=0.7, thick]
  \draw[gray] (-1, 0) -- (8, 0) node[right] {$l$};
  \coordinate (Ap) at (1, 0);
  \coordinate (Bp) at (7, 0);
  \coordinate (A)  at (1, 2);
  \coordinate (B)  at (7, 4);
  \coordinate (As) at (1, -2);   % A 关于 l 的对称点
  \coordinate (P)  at (intersection of As--B and Ap--Bp);

  \draw[gray, dashed] (A) -- (Ap);
  \draw[gray, dashed] (B) -- (Bp);
  \draw[red] (A) -- (P) -- (B);
  \draw[blue, dashed] (As) -- (P);

  \fill (A) circle (1.5pt); \node[above] at (A) {$A$};
  \fill (B) circle (1.5pt); \node[above] at (B) {$B$};
  \fill (P) circle (1.5pt); \node[below] at (P) {$P$};
  \fill (As) circle (1.5pt); \node[below] at (As) {$A'$};
\end{tikzpicture}
\end{document}
