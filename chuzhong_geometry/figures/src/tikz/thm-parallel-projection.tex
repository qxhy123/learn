% 平行投影：太阳光下旗杆与小棍的影子方向相同
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[thick, scale=1.0]
  % 地面
  \draw[gray] (-0.5,0) -- (9.5,0);
  % 旗杆（高 3）
  \coordinate (P1) at (1,0);
  \coordinate (P1T) at (1,3);
  \draw[very thick] (P1) -- (P1T);
  \node[left, font=\footnotesize] at (P1T) {旗杆};
  % 影子末端：影长 2
  \coordinate (S1) at ($(P1) + (2,0)$);
  \draw[red, thick] (P1) -- (S1);
  \node[below, font=\footnotesize, red] at ($(P1)!0.5!(S1)$) {影};
  % 光线（平行斜线）
  \draw[->, orange, thick] ($(P1T)+(-0.8,0.8)$) -- (P1T);
  \draw[->, orange, thick] ($(S1)+(-0.8,0.8)$) -- (S1);

  % 小棍（高 1.5）位置远处
  \coordinate (P2) at (6,0);
  \coordinate (P2T) at (6,1.5);
  \draw[very thick] (P2) -- (P2T);
  \node[left, font=\footnotesize] at (P2T) {小棍};
  \coordinate (S2) at ($(P2) + (1,0)$);
  \draw[red, thick] (P2) -- (S2);
  \draw[->, orange, thick] ($(P2T)+(-0.8,0.8)$) -- (P2T);
  \draw[->, orange, thick] ($(S2)+(-0.8,0.8)$) -- (S2);
  % 额外几条平行光线
  \draw[->, orange, thick] (3, 4) -- ($(3, 4)+(0.8,-0.8)$);
  \node[above right, orange, font=\footnotesize] at (3,4) {太阳光（平行）};

  \node[below, font=\footnotesize, gray] at (4.5, -0.4) {同一时刻影子方向相同，$h:s$ 为常数};
\end{tikzpicture}
\end{document}
