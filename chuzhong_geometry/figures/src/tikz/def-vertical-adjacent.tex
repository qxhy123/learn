% 对顶角与邻补角：两直线 AB, CD 交于 O，4 个角
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \coordinate (O) at (0,0);
  % 直线 AB 水平稍倾斜，CD 斜向上
  \pgfmathsetmacro{\anga}{15}
  \pgfmathsetmacro{\angc}{75}
  \coordinate (A) at ({3*cos(\anga+180)}, {3*sin(\anga+180)});
  \coordinate (B) at ({3*cos(\anga)}, {3*sin(\anga)});
  \coordinate (C) at ({3*cos(\angc)}, {3*sin(\angc)});
  \coordinate (D) at ({3*cos(\angc+180)}, {3*sin(\angc+180)});
  \draw[-{Stealth[length=3pt]}] (A) -- (B);
  \draw[-{Stealth[length=3pt]}] (D) -- (C);
  % 4 个角弧 + 颜色：对顶角同色
  \draw[red] (0.9,0) ++(\anga:0) arc (\anga:\angc:0.9);
  \node[red, font=\footnotesize] at ({0.6*cos((\anga+\angc)/2)}, {0.6*sin((\anga+\angc)/2)}) {$\angle 1$};
  \draw[blue] ({0.9*cos(\angc)}, {0.9*sin(\angc)}) arc (\angc:\anga+180:0.9);
  \node[blue, font=\footnotesize] at ({0.6*cos((\angc+\anga+180)/2)}, {0.6*sin((\angc+\anga+180)/2)}) {$\angle 2$};
  \draw[red] ({0.9*cos(\anga+180)}, {0.9*sin(\anga+180)}) arc (\anga+180:\angc+180:0.9);
  \node[red, font=\footnotesize] at ({0.6*cos((\anga+\angc)/2+180)}, {0.6*sin((\anga+\angc)/2+180)}) {$\angle 3$};
  \draw[blue] ({0.9*cos(\angc+180)}, {0.9*sin(\angc+180)}) arc (\angc+180:\anga+360:0.9);
  \node[blue, font=\footnotesize] at ({0.6*cos((\angc+360+\anga)/2)}, {0.6*sin((\angc+360+\anga)/2)}) {$\angle 4$};
  \filldraw (O) circle (1.5pt);
  \node[below left] at (O) {$O$};
  \node[left] at (A) {$A$}; \node[right] at (B) {$B$};
  \node[above] at (C) {$C$}; \node[below] at (D) {$D$};
  \node[below, font=\small, align=center] at (0,-3.2) {对顶角：$\angle 1=\angle 3$、$\angle 2=\angle 4$（同色）\\ 邻补角：$\angle 1+\angle 2=180°$ 等};
\end{tikzpicture}
\end{document}
