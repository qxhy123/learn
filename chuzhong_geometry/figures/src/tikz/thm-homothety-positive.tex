% 正位似：位似中心 O 同侧，△ABC 与 △A'B'C' (k=2)
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[scale=0.9, thick]
  \coordinate (O) at (0,0);
  \coordinate (A) at (1.4, 0.5);
  \coordinate (B) at (1.8, 1.6);
  \coordinate (C) at (0.7, 1.3);
  % k=2 正向：A'=2A=(2.8,1.0)
  \coordinate (Ap) at (2.8, 1.0);
  \coordinate (Bp) at (3.6, 3.2);
  \coordinate (Cp) at (1.4, 2.6);

  \draw (A) -- (B) -- (C) -- cycle;
  \draw[blue, dashed] (Ap) -- (Bp) -- (Cp) -- cycle;
  \draw[gray, dashed] (O) -- (Ap);
  \draw[gray, dashed] (O) -- (Bp);
  \draw[gray, dashed] (O) -- (Cp);

  \fill (O) circle (1.4pt);
  \fill (A) circle (1.2pt); \fill (B) circle (1.2pt); \fill (C) circle (1.2pt);
  \fill (Ap) circle (1.2pt); \fill (Bp) circle (1.2pt); \fill (Cp) circle (1.2pt);
  \node[below left] at (O) {$O$};
  \node[below right] at (A) {$A$};
  \node[right] at (B) {$B$};
  \node[above left] at (C) {$C$};
  \node[below right] at (Ap) {$A'$};
  \node[right] at (Bp) {$B'$};
  \node[above] at (Cp) {$C'$};
  \node[below] at (3,-0.3) {\footnotesize $k>0$（同侧）};
\end{tikzpicture}
\end{document}
