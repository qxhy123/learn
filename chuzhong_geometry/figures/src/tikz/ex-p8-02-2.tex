% 例 2（中心投影：路灯人影）：AB=6, CD=1.5, BD=3, DE=x
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[thick, scale=0.9]
  \draw[gray] (-0.3,0) -- (7,0);
  % 灯柱 AB
  \coordinate (B) at (0,0);
  \coordinate (A) at (0,6);
  \draw[very thick] (B) -- (A);
  \filldraw[orange] (A) circle (3pt);
  \node[left, font=\footnotesize] at (A) {$A$};
  \node[below, font=\footnotesize] at (B) {$B$};
  \node[left, font=\footnotesize] at (0,3) {$6$};
  % 人 CD: D 在 (3,0), C=(3,1.5)
  \coordinate (D) at (3,0);
  \coordinate (C) at (3,1.5);
  \draw[very thick] (D) -- (C);
  \node[below, font=\footnotesize] at (D) {$D$};
  \node[right, font=\footnotesize] at (C) {$C$};
  \node[right, font=\footnotesize] at (3,0.75) {$1.5$};
  % BD 长 3
  \node[below, font=\footnotesize] at (1.5,-0.05) {$3$};
  % 影末 E: A,C 共线延到 y=0
  % A=(0,6), C=(3,1.5), 方向(3,-4.5), 到 y=0 时 t=6/4.5=4/3, x=3*4/3=4
  \coordinate (E) at (4,0);
  \draw[red, thick] (D) -- (E);
  \node[below, font=\footnotesize] at ($(D)!0.5!(E)$) {$x$};
  \node[below, font=\footnotesize] at (E) {$E$};
  % 光线 A->E
  \draw[dashed, orange] (A) -- (E);
  % 三角形 EBA 标注
  \draw[very thin, gray] (E) -- (A);
\end{tikzpicture}
\end{document}
