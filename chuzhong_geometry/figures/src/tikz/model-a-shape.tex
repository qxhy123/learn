% A 字型相似：△ABC 中 DE ∥ BC，D∈AB、E∈AC → △ADE ∼ △ABC
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \coordinate (A) at (1, 4);
  \coordinate (B) at (-2, 0);
  \coordinate (C) at (4, 0);
  % D 在 AB 上、E 在 AC 上：取 AD:DB = 3:2 → D = A + 3/5*(B-A)
  \coordinate (D) at ($(A)!0.4!(B)$);
  \coordinate (E) at ($(A)!0.4!(C)$);

  \draw (A) -- (B);
  \draw (A) -- (C);
  \draw (B) -- (C);

  \draw[blue, thick] (D) -- (E);

  % 长度标注
  \node at ($(A)!0.2!(B) + (-0.3, 0.1)$) {$3$};
  \node at ($(D)!0.5!(B) + (-0.3, 0)$)   {$2$};

  \node[above]       at (A) {$A$};
  \node[below left]  at (B) {$B$};
  \node[below right] at (C) {$C$};
  \node[left]        at (D) {$D$};
  \node[right]       at (E) {$E$};
\end{tikzpicture}
\end{document}
