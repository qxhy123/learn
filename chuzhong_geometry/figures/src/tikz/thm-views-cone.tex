% 圆锥三视图
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[thick, scale=1.0]
  \pgfmathsetmacro{\r}{0.9}
  \pgfmathsetmacro{\h}{2.4}
  \pgfmathsetmacro{\yr}{0.3}
  % --- 立体 ---
  \draw (-\r,0) -- (0,\h) -- (\r,0);
  \draw (-\r,0) arc (180:360:{\r} and {\yr});
  \draw[dashed, gray] (-\r,0) arc (180:0:{\r} and {\yr});
  \node[below, font=\footnotesize] at (0,-0.8) {（a）圆锥};
  \filldraw (0,\h) circle (1pt);

  % --- 主视：等腰三角形 ---
  \begin{scope}[xshift=3.5cm]
    \draw (0,0) -- (2*\r,0) -- (\r,\h) -- cycle;
    % 高线（虚）
    \draw[dashed, gray] (\r,0) -- (\r,\h);
    \node[font=\footnotesize, below] at (\r,0) {主视};
  \end{scope}

  % --- 左视 ---
  \begin{scope}[xshift=6.2cm]
    \draw (0,0) -- (2*\r,0) -- (\r,\h) -- cycle;
    \draw[dashed, gray] (\r,0) -- (\r,\h);
    \node[font=\footnotesize, below] at (\r,0) {左视};
  \end{scope}

  % --- 俯视：圆 + 中心点 ---
  \begin{scope}[xshift=9cm, yshift=0.6cm]
    \draw (0,\h/2) circle (\r);
    \filldraw (0,\h/2) circle (1.5pt);
    \node[font=\footnotesize, below] at (0, \h/2 - \r - 0.2) {俯视（圆+顶点投影）};
  \end{scope}
\end{tikzpicture}
\end{document}
