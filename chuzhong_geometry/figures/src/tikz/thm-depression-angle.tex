% 俯角示意
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \coordinate (T) at (0,2.4);  % 塔顶（高处观察者）
  \coordinate (G) at (4,0);    % 地面目标
  \coordinate (F) at (0,0);    % 塔底
  \draw (T) -- (5,2.4) node[right]{\footnotesize 水平线};
  \draw (F) -- (T);
  \draw (F) -- (G);
  \draw[red, thick] (T) -- (G);
  % 俯角 β：在 T 处水平线下方
  \draw (0.9,2.4) arc (0:-31:0.9);
  \node at (1.1, 2.15) {\footnotesize $\beta$};
  \fill (T) circle (1.3pt);
  \node[above left] at (T) {\footnotesize 观察者};
  \node[below] at (G) {\footnotesize 目标};
  \node[above] at (2,2.4) {\footnotesize 水平线};
  \node[above right] at (2,1.2) {\footnotesize 视线};
\end{tikzpicture}
\end{document}
