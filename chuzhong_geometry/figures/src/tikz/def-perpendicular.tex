% 垂线：两条直线交于 O，标记直角
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \draw[-{Stealth[length=3pt]}] (-3,0) -- (3,0) node[right] {$l_1$};
  \draw[-{Stealth[length=3pt]}] (0,-2.5) -- (0,2.5) node[above] {$l_2$};
  \draw (0.25,0) -- (0.25,0.25) -- (0,0.25);
  \filldraw (0,0) circle (1.5pt);
  \node[below left] at (0,0) {$O$};
  \node[below, font=\small] at (0,-2.8) {$l_1 \perp l_2$，垂足 $O$};
\end{tikzpicture}
\end{document}
