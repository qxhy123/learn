\documentclass[tikz,border=10pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc, positioning, arrows.meta, decorations.pathreplacing}

% 正定：A=[[2,0],[0,1]], x^TAx=1 => 2x1^2+x2^2=1 (标准椭圆)
% 负定：A=[[-2,0],[0,-1]], x^TAx=-1 => 2x1^2+x2^2=1 (同样是椭圆，但二次型为负)
% 不定：A=[[2,0],[0,-1]], x^TAx=1 => 2x1^2-x2^2=1 (双曲线)

\begin{document}
\begin{tikzpicture}[>=Stealth]

%% ============================
%%  左子图：正定 — x^TAx=1，椭圆
%%  A = diag(2,1), lambda1=2>0, lambda2=1>0
%% ============================
\begin{scope}[xshift=0cm]

  \begin{axis}[
    name=ax1,
    width=6.0cm, height=6.0cm,
    xmin=-1.3, xmax=1.3,
    ymin=-1.3, ymax=1.3,
    axis lines=center,
    xlabel={$x_1$},
    ylabel={$x_2$},
    xlabel style={font=\small, right},
    ylabel style={font=\small, above},
    xtick={-1,1},
    xticklabels={$-1$,$1$},
    ytick={-1,1},
    yticklabels={$-1$,$1$},
    tick label style={font=\scriptsize},
    grid=both,
    grid style={gray!15},
    clip=false,
    axis equal image,
  ]

    % 椭圆: 2x1^2+x2^2=1 => x1=(1/sqrt2)cos(t), x2=sin(t)
    \addplot[blue!70!black, very thick, fill=blue!8,
             domain=0:360, samples=120]
      ({0.70711*cos(x)}, {sin(x)});

    % 半轴标注
    \draw[<->,blue!60,thin]
      (axis cs:0,0) -- (axis cs:0.70711,0)
      node[midway,above,font=\scriptsize,blue!60] {$\tfrac{1}{\sqrt{2}}$};
    \draw[<->,blue!60,thin]
      (axis cs:0,0) -- (axis cs:0,1)
      node[midway,right,font=\scriptsize,blue!60] {$1$};

    % 特征值标注
    \node[font=\scriptsize, blue!70!black,
          fill=blue!10, draw=blue!40, rounded corners=2pt, inner sep=3pt,
          align=center]
      at (axis cs:0, -1.1) {
        $\lambda_1=2>0$\\
        $\lambda_2=1>0$
      };

  \end{axis}

  % 正定标签
  \node[font=\small\bfseries, blue!70!black,
        fill=blue!10, draw=blue!50, rounded corners=4pt,
        inner sep=5pt, align=center]
    at ([yshift=0.4cm]ax1.north) {\textbf{正定}（Positive Definite）};

  % 公式
  \node[font=\small, align=center] at ([yshift=-0.7cm]ax1.south)
    {$\mathbf{x}^TA\mathbf{x}>0$（$\forall\mathbf{x}\neq\mathbf{0}$）\\
     二次型曲线 $=$ 实椭圆};

\end{scope}

%% ============================
%%  中子图：负定 — x^TAx=-1（等价展示2x^2+y^2=1），
%%  A = diag(-2,-1), lambda1=-2<0, lambda2=-1<0
%% ============================
\begin{scope}[xshift=7.8cm]

  \begin{axis}[
    name=ax2,
    width=6.0cm, height=6.0cm,
    xmin=-1.3, xmax=1.3,
    ymin=-1.3, ymax=1.3,
    axis lines=center,
    xlabel={$x_1$},
    ylabel={$x_2$},
    xlabel style={font=\small, right},
    ylabel style={font=\small, above},
    xtick={-1,1},
    xticklabels={$-1$,$1$},
    ytick={-1,1},
    yticklabels={$-1$,$1$},
    tick label style={font=\scriptsize},
    grid=both,
    grid style={gray!15},
    clip=false,
    axis equal image,
  ]

    % 负定: x^T A x = -1 => -2x1^2 - x2^2 = -1 => 2x1^2+x2^2=1 (同形椭圆)
    \addplot[red!70!black, very thick, fill=red!8,
             domain=0:360, samples=120]
      ({0.70711*cos(x)}, {sin(x)});

    % 半轴标注
    \draw[<->,red!60,thin]
      (axis cs:0,0) -- (axis cs:0.70711,0)
      node[midway,above,font=\scriptsize,red!60] {$\tfrac{1}{\sqrt{2}}$};
    \draw[<->,red!60,thin]
      (axis cs:0,0) -- (axis cs:0,1)
      node[midway,right,font=\scriptsize,red!60] {$1$};

    % 向下凹面示意（所有值≤0）
    \node[font=\scriptsize, red!70!black,
          fill=red!8, draw=red!40, rounded corners=2pt, inner sep=3pt,
          align=center]
      at (axis cs:0, -1.1) {
        $\lambda_1=-2<0$\\
        $\lambda_2=-1<0$
      };

  \end{axis}

  % 负定标签
  \node[font=\small\bfseries, red!70!black,
        fill=red!10, draw=red!50, rounded corners=4pt,
        inner sep=5pt, align=center]
    at ([yshift=0.4cm]ax2.north) {\textbf{负定}（Negative Definite）};

  % 公式
  \node[font=\small, align=center] at ([yshift=-0.7cm]ax2.south)
    {$\mathbf{x}^TA\mathbf{x}<0$（$\forall\mathbf{x}\neq\mathbf{0}$）\\
     曲线 $\mathbf{x}^TA\mathbf{x}=-1$ 为实椭圆};

\end{scope}

%% ============================
%%  右子图：不定 — x^TAx=1，双曲线
%%  A = diag(2,-1), lambda1=2>0, lambda2=-1<0
%%  2x1^2 - x2^2 = 1 → 双曲线，实轴在x1方向
%% ============================
\begin{scope}[xshift=15.6cm]

  \begin{axis}[
    name=ax3,
    width=6.0cm, height=6.0cm,
    xmin=-1.8, xmax=1.8,
    ymin=-2.2, ymax=2.2,
    axis lines=center,
    xlabel={$x_1$},
    ylabel={$x_2$},
    xlabel style={font=\small, right},
    ylabel style={font=\small, above},
    xtick={-1,1},
    xticklabels={$-1$,$1$},
    ytick={-2,-1,1,2},
    yticklabels={$-2$,$-1$,$1$,$2$},
    tick label style={font=\scriptsize},
    grid=both,
    grid style={gray!15},
    clip=true,
    axis equal image,
  ]

    % 双曲线右支: 2x1^2 - x2^2 = 1 => x2 = sqrt(2x1^2-1)，x1>=1/sqrt2
    % 用 x2 作为参数: x1=sqrt((x2^2+1)/2)
    \addplot[purple!70!black, very thick, domain=-2.0:2.0, samples=80]
      ({sqrt((x*x+1)/2)}, {x});
    % 左支
    \addplot[purple!70!black, very thick, domain=-2.0:2.0, samples=80]
      ({-sqrt((x*x+1)/2)}, {x});

    % 渐近线 x2 = ±sqrt(2)*x1
    \addplot[gray!50, dashed, domain=-1.6:1.6]
      {sqrt(2)*x};
    \addplot[gray!50, dashed, domain=-1.6:1.6]
      {-sqrt(2)*x};

    % 顶点标注
    \addplot[only marks, mark=*, mark size=2pt, purple!70!black]
      coordinates {(0.70711,0) (-0.70711,0)};
    \node[font=\scriptsize,purple!70!black,above]
      at (axis cs:0.72,0.1) {$\tfrac{1}{\sqrt{2}}$};

    % 特征值标注
    \node[font=\scriptsize, purple!70!black,
          fill=purple!8, draw=purple!40, rounded corners=2pt, inner sep=3pt,
          align=center]
      at (axis cs:0, 1.9) {
        $\lambda_1=2>0$\\
        $\lambda_2=-1<0$
      };

    % 渐近线标注
    \node[font=\scriptsize, gray!60] at (axis cs:1.2,1.5)
      {渐近线};

  \end{axis}

  % 不定标签
  \node[font=\small\bfseries, purple!70!black,
        fill=purple!10, draw=purple!50, rounded corners=4pt,
        inner sep=5pt, align=center]
    at ([yshift=0.4cm]ax3.north) {\textbf{不定}（Indefinite）};

  % 公式
  \node[font=\small, align=center] at ([yshift=-0.7cm]ax3.south)
    {$\mathbf{x}^TA\mathbf{x}$ 可正可负\\
     曲线 $\mathbf{x}^TA\mathbf{x}=1$ 为双曲线};

\end{scope}

%% ===== 底部判定准则框 =====
\node[font=\small, fill=white, draw=gray!50, thin, rounded corners=4pt,
      inner sep=10pt, align=center]
  at (7.8, -2.6) {
    \textbf{判定准则（实对称矩阵）：}
    \quad 正定 $\Leftrightarrow$ 所有特征值 $>0$
    \quad 负定 $\Leftrightarrow$ 所有特征值 $<0$
    \quad 不定 $\Leftrightarrow$ 特征值有正有负
  };

%% 整体标题
\node[font=\large\bfseries] at (7.8, 5.4)
  {二次型的正定性：椭圆与双曲线};

\end{tikzpicture}
\end{document}
