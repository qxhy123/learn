\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{ctex}
\usetikzlibrary{calc, positioning, arrows.meta, decorations.pathreplacing}

\begin{document}
\begin{tikzpicture}[>=Stealth]

%% ===== 参数：旋转角 30° =====
% cos30 ≈ 0.866, sin30 = 0.5
% e1=(1,0) → T(e1)=(cos30, sin30)=(0.866, 0.5)
% e2=(0,1) → T(e2)=(-sin30, cos30)=(-0.5, 0.866)

%% ===== 左图：标准基 =====
\begin{scope}[xshift=0cm]
  \draw[gray!25, thin] (-1.3,-1.3) grid (1.6,1.6);
  \draw[->,gray!60, thin] (-1.4,0) -- (1.8,0) node[right,font=\scriptsize] {$x$};
  \draw[->,gray!60, thin] (0,-1.4) -- (0,1.8) node[above,font=\scriptsize] {$y$};

  % e1
  \draw[->,thick,blue!80!black] (0,0) -- (1,0)
    node[below right,font=\small] {$\mathbf{e}_1=(1,0)$};
  % e2
  \draw[->,thick,teal] (0,0) -- (0,1)
    node[left,font=\small] {$\mathbf{e}_2=(0,1)$};

  % 单位圆（虚线）
  \draw[gray!40, dashed, thin] (0,0) circle (1cm);

  \fill (0,0) circle (1.5pt) node[below left,font=\scriptsize] {$O$};
  \node[font=\small\bfseries] at (0.15,-1.85) {标准基};
\end{scope}

%% ===== 中间箭头 =====
\draw[->,very thick,gray!60] (2.1,0) -- (3.3,0)
  node[midway,above,font=\normalsize\bfseries] {$T$}
  node[midway,below,font=\scriptsize,gray] {旋转 $\theta=30°$};

%% ===== 右图：旋转后的像（两列就是矩阵列）=====
\begin{scope}[xshift=4.6cm]
  \draw[gray!25, thin] (-1.3,-1.3) grid (1.6,1.6);
  \draw[->,gray!60, thin] (-1.4,0) -- (1.8,0) node[right,font=\scriptsize] {$x$};
  \draw[->,gray!60, thin] (0,-1.4) -- (0,1.8) node[above,font=\scriptsize] {$y$};

  % T(e1) = (cos30, sin30)
  \draw[->,thick,blue!80!black,dashed] (0,0) -- (1,0);  % 原 e1（淡）
  \draw[->,thick,blue!80!black] (0,0) -- (0.866,0.5)
    node[right,font=\scriptsize] {$T(\mathbf{e}_1)=(\cos\theta,\sin\theta)$};

  % T(e2) = (-sin30, cos30)
  \draw[->,thick,teal,dashed] (0,0) -- (0,1);  % 原 e2（淡）
  \draw[->,thick,teal] (0,0) -- (-0.5,0.866)
    node[left,font=\scriptsize] {$T(\mathbf{e}_2)=(-\sin\theta,\cos\theta)$};

  % 旋转角弧度标注
  \draw[gray!50, thin] (0.4,0) arc (0:30:0.4cm);
  \node[font=\scriptsize,gray!70] at (0.55,0.14) {$\theta$};

  % 单位圆
  \draw[gray!40, dashed, thin] (0,0) circle (1cm);

  \fill (0,0) circle (1.5pt) node[below left,font=\scriptsize] {$O$};
  \node[font=\small\bfseries] at (0.15,-1.85) {旋转后的像};
\end{scope}

%% ===== 矩阵展示框 =====
\node[font=\small, fill=white, draw=gray!50, thin, rounded corners=4pt,
      inner sep=8pt, align=center]
  at (3.2, -3.1) {
  \textbf{旋转矩阵}（列 = $T$ 作用在标准基上的像）\\[4pt]
  $[T] = \begin{pmatrix} \cos\theta & -\sin\theta \\ \sin\theta & \cos\theta \end{pmatrix}
   = \begin{pmatrix} \frac{\sqrt{3}}{2} & -\frac{1}{2} \\[4pt] \frac{1}{2} & \frac{\sqrt{3}}{2} \end{pmatrix}
   \quad (\theta = 30°)$
};

%% ===== 整体标题 =====
\node[font=\large\bfseries] at (3.2, 2.7) {矩阵的列 = 标准基像（旋转 $\theta=30°$）};

\end{tikzpicture}
\end{document}
