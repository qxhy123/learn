\documentclass[tikz,border=10pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{ctex}

\begin{document}
\begin{tikzpicture}[>=Stealth]

%% ============================
%%  左子图：凸函数 f(x)=x^2，H>0
%% ============================
\begin{scope}[xshift=0cm]
  \begin{axis}[
    name=ax1,
    width=6.0cm, height=5.5cm,
    xmin=-2.3, xmax=2.3,
    ymin=-0.5, ymax=5.2,
    axis lines=center,
    xlabel={$x$}, ylabel={$f(x)$},
    xlabel style={font=\small, right},
    ylabel style={font=\small, above},
    xtick={-2,-1,1,2},
    ytick={1,2,3,4},
    tick label style={font=\scriptsize},
    grid=both, grid style={gray!12},
    clip=false,
  ]
    \addplot[blue!70!black, very thick, domain=-2.2:2.2, samples=80] {x^2};

    % 弦：(-1.5, 2.25) -- (1.5, 2.25)
    \addplot[orange!80!black, dashed, thick]
      coordinates {(-1.5, 2.25) (1.5, 2.25)};

    % 切线 y=0
    \addplot[green!60!black, thin, domain=-1.6:1.6] {0};
    \node[font=\scriptsize, green!60!black, right] at (axis cs:1.3,0.15) {切线};

    % 弦标注
    \draw[<-,orange!80!black,thin]
      (axis cs:0,2.25) -- (axis cs:0.7,3.6)
      node[right,font=\scriptsize,orange!80!black] {弦 $\geq$ 曲线};

    % 极小值点
    \addplot[only marks,mark=*,mark size=2.5pt,red!70!black]
      coordinates {(0,0)};
    \node[font=\scriptsize,red!70!black,above right] at (axis cs:0.08,0.1) {极小};

    % H>0 标注框
    \node[font=\small, fill=blue!10, draw=blue!50, rounded corners=4pt,
          inner sep=5pt, align=center]
      at (axis cs:1.55,4.2) {$H>0$\\$f''=2$};
  \end{axis}

  \node[font=\small\bfseries, blue!70!black,
        fill=blue!10, draw=blue!50, rounded corners=4pt,
        inner sep=5pt, align=center]
    at ([yshift=0.35cm]ax1.north) {\textbf{凸函数}（Convex）};

  \node[font=\small, align=center] at ([yshift=-1.1cm]ax1.south)
    {$f(x)=x^2$，\ $H=f''>0$\\
     弦在曲线上方，全局极小值};
\end{scope}

%% ============================
%%  中子图：鞍点 f=x^2-y^2（两方向截面）
%% ============================
\begin{scope}[xshift=7.2cm]
  \begin{axis}[
    name=ax2,
    width=6.0cm, height=5.5cm,
    xmin=-2.3, xmax=2.3,
    ymin=-4.8, ymax=4.8,
    axis lines=center,
    xlabel={$x$}, ylabel={$f$},
    xlabel style={font=\small, right},
    ylabel style={font=\small, above},
    xtick={-2,-1,1,2},
    ytick={-4,-2,2,4},
    tick label style={font=\scriptsize},
    grid=both, grid style={gray!12},
    clip=false,
  ]
    % x 方向截面 f=x^2
    \addplot[blue!60!black, thick, dashed, domain=-2.1:2.1, samples=60] {x^2};
    \node[font=\scriptsize,blue!70!black] at (axis cs:1.6,2.8) {$f_{xx}=2$};

    % y 方向截面 f=-y^2
    \addplot[red!60!black, thick, domain=-2.1:2.1, samples=60] {-x^2};
    \node[font=\scriptsize,red!70!black] at (axis cs:1.6,-2.8) {$f_{yy}=-2$};

    % 鞍点
    \addplot[only marks,mark=*,mark size=3pt,orange!80!black]
      coordinates {(0,0)};
    \node[font=\scriptsize, orange!80!black, right] at (axis cs:0.1,0.45) {鞍点};

    % H 矩阵框
    \node[font=\small, fill=orange!10, draw=orange!50, rounded corners=4pt,
          inner sep=5pt, align=center]
      at (axis cs:-1.35,3.6) {
        $H=\begin{pmatrix}2&0\\0&{-2}\end{pmatrix}$\\不定
      };
  \end{axis}

  \node[font=\small\bfseries, orange!80!black,
        fill=orange!10, draw=orange!50, rounded corners=4pt,
        inner sep=5pt, align=center]
    at ([yshift=0.35cm]ax2.north) {\textbf{鞍点}（Saddle Point）};

  \node[font=\small, align=center] at ([yshift=-1.1cm]ax2.south)
    {$f=x^2-y^2$，\ $\det(H)<0$\\
     $x$方向凸（蓝虚），$y$方向凹（红实）};
\end{scope}

%% ============================
%%  右子图：凹函数 f(x)=-x^2，H<0
%% ============================
\begin{scope}[xshift=14.4cm]
  \begin{axis}[
    name=ax3,
    width=6.0cm, height=5.5cm,
    xmin=-2.3, xmax=2.3,
    ymin=-5.2, ymax=0.5,
    axis lines=center,
    xlabel={$x$}, ylabel={$f(x)$},
    xlabel style={font=\small, right},
    ylabel style={font=\small, above},
    xtick={-2,-1,1,2},
    ytick={-4,-3,-2,-1},
    tick label style={font=\scriptsize},
    grid=both, grid style={gray!12},
    clip=false,
  ]
    \addplot[red!70!black, very thick, domain=-2.2:2.2, samples=80] {-x^2};

    % 弦：(-1.5,-2.25) -- (1.5,-2.25)，在曲线下方
    \addplot[orange!80!black, dashed, thick]
      coordinates {(-1.5, -2.25) (1.5, -2.25)};
    \draw[<-,orange!80!black,thin]
      (axis cs:0,-2.25) -- (axis cs:0.7,-3.6)
      node[right,font=\scriptsize,orange!80!black] {弦 $\leq$ 曲线};

    % 切线 y=0
    \addplot[green!60!black, thin, domain=-1.6:1.6] {0};
    \node[font=\scriptsize, green!60!black, right] at (axis cs:1.3,-0.2) {切线};

    % 极大值点
    \addplot[only marks,mark=*,mark size=2.5pt,red!70!black]
      coordinates {(0,0)};
    \node[font=\scriptsize,red!70!black,above right] at (axis cs:0.08,-0.1) {极大};

    % H<0 标注框
    \node[font=\small, fill=red!10, draw=red!50, rounded corners=4pt,
          inner sep=5pt, align=center]
      at (axis cs:-1.55,-4.2) {$H<0$\\$f''=-2$};
  \end{axis}

  \node[font=\small\bfseries, red!70!black,
        fill=red!10, draw=red!50, rounded corners=4pt,
        inner sep=5pt, align=center]
    at ([yshift=0.35cm]ax3.north) {\textbf{凹函数}（Concave）};

  \node[font=\small, align=center] at ([yshift=-1.1cm]ax3.south)
    {$f(x)=-x^2$，\ $H=f''<0$\\
     弦在曲线下方，全局极大值};
\end{scope}

%% ===== 底部公式框 =====
\node[font=\normalsize, fill=white, draw=gray!50, thin, rounded corners=5pt,
      inner sep=10pt, align=center]
  at (10.8, -4.0) {
    \textbf{Hessian 矩阵 $H = \nabla^2 f$ 与凸性：}
    \quad $H \succ 0$（正定）$\Rightarrow$ $f$ 严格凸，极小值为全局极小值
    \quad $H \prec 0$（负定）$\Rightarrow$ $f$ 严格凹，极大值为全局极大值
    \quad $H$ 不定 $\Rightarrow$ 鞍点
  };

%% ===== 整体标题 =====
\node[font=\large\bfseries] at (10.8, 5.2)
  {Hessian 矩阵与函数凹凸性};
