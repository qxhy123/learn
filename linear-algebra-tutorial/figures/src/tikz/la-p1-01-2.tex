\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}
\usetikzlibrary{calc, positioning, arrows.meta, decorations.pathreplacing, angles, quotes}

\begin{document}
\begin{tikzpicture}[scale=1.1]

  % 背景网格（淡）
  \draw[gray!20, thin] (-0.3,-0.3) grid (5.3,3.8);
  \draw[->,gray!50, thin] (-0.3,0) -- (5.5,0) node[right,font=\scriptsize] {$x$};
  \draw[->,gray!50, thin] (0,-0.3) -- (0,4.0) node[above,font=\scriptsize] {$y$};

  % 定义坐标
  % u = (4, 3), v = (5, 1)
  % proj of u onto v: scalar = (u.v)/(v.v) = (20+3)/(25+1) = 23/26
  % proj vector = (23/26)*(5,1) = (115/26, 23/26) ≈ (4.423, 0.885)
  \coordinate (O) at (0,0);
  \coordinate (U) at (4,3);
  \coordinate (V) at (5,1);
  % foot of perpendicular from U onto line of V
  \coordinate (F) at (4.423, 0.885);

  % 向量 u
  \draw[->,thick,blue!80!black,>=Stealth] (O) -- (U)
    node[above left,font=\small] {$\mathbf{u}$};
  % 向量 v
  \draw[->,thick,teal!80!black,>=Stealth] (O) -- (V)
    node[below right,font=\small] {$\mathbf{v}$};

  % 投影虚线（从 U 到 F，垂直线）
  \draw[densely dashed, orange!80!black, thick] (U) -- (F);

  % 投影向量（从 O 到 F）
  \draw[->,thick,orange!70!black,>=Stealth] (O) -- (F)
    node[below,font=\small,yshift=-4pt]
      {$\dfrac{\mathbf{u}\cdot\mathbf{v}}{\|\mathbf{v}\|^2}\mathbf{v}$};

  % 直角标记
  \draw[orange!60, thin]
    ($(F)!0.18!(U)$) --
    ($(F)!0.18!(U) + ($(F)!0.18!(V)$) - (F)$) --
    ($(F)!0.18!(V)$);

  % 夹角 theta 弧线
  \draw[red!70!black, thick] (0.9,0) arc[start angle=0, end angle=36.87, radius=0.9]
    node[right=2pt, midway, font=\small] {$\theta$};

  % 点标注
  \fill (O) circle (1.5pt) node[below left,font=\scriptsize] {$O$};

  % 公式框（放在右下角，避免遮盖向量）
  \node[font=\small, align=center, fill=white, draw=gray!50, thin,
        rounded corners=2pt, inner sep=6pt]
    at (4.0, -0.7) {$\mathbf{u}\cdot\mathbf{v} = \|\mathbf{u}\|\,\|\mathbf{v}\|\cos\theta$};

  % 标题
  \node[font=\large\bfseries] at (2.5, -1.8) {内积的几何意义};

\end{tikzpicture}
\end{document}
