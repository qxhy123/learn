\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\usepackage{amsmath}
\pgfplotsset{compat=1.18}

\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=12cm, height=7cm,
  axis lines=left,
  xlabel={$x$},
  ylabel={$f(x)$},
  xlabel style={right},
  ylabel style={above},
  xmin=-2.3, xmax=2.3,
  ymin=-0.4, ymax=3.0,
  xtick={-1.5, -1, 0, 1, 1.5},
  xticklabels={$-1.5$, $-1$, $0$, $1$, $1.5$},
  ytick=\empty,
  samples=300,
  clip=false,
  title={动量穿越局部极小：GD 被困 vs 动量法越过势垒},
  title style={font=\small},
]

% 双峰函数 f(x) = (x^2-1)^2 （极小值 x=±1，局部极大 x=0）
\addplot[black, thick, domain=-2.1:2.1] {(x^2-1)^2};

% 标记两个极小值
\fill[black] (-1, 0) circle (2.5pt);
\node[anchor=north, font=\small] at (-1, 0) {局部极小 $x=-1$};
\fill[black] (1, 0) circle (2.5pt);
\node[anchor=north, font=\small] at (1, 0) {全局极小 $x=1$};

% 标记局部极大值（势垒）
\fill[gray] (0, 1) circle (2pt);
\node[anchor=west, font=\scriptsize, gray] at (0.05, 1.05) {势垒 $x=0$};
\draw[gray, dashed] (0, 0) -- (0, 1);

% GD 轨迹：从 x0=-0.3 出发，收敛到 x=-1（局部极小）
% f'(x) = 4x(x^2-1)，x=-0.3 时 f'(-0.3)=4(-0.3)(0.09-1)=-4*0.3*0.91 < 0，向左走
\addplot[blue, very thick, dashed, -{Stealth[scale=0.8]},
  domain=-0.3:-0.95, samples=20]
  ({x}, {(x^2-1)^2});
% 沿函数曲线画箭头，简化为分段线段
\draw[blue, very thick, -{Stealth[scale=0.9]}]
  (-0.3,  0.826) -- (-0.6,  0.4096) -- (-0.85, 0.0596) -- (-0.98, 0.0004);

\fill[blue] (-0.3, 0.826) circle (2.5pt);
\node[blue, anchor=south, font=\scriptsize] at (-0.3, 0.85) {$x_0$};

% GD 终点标注
\fill[blue] (-1, 0) circle (3pt);
\draw[blue, -{Stealth[scale=0.8]}]
  (-1.3, 0.4) node[blue, anchor=east, font=\scriptsize] {GD 卡在局部极小}
  -- (-1.05, 0.05);

% 动量法轨迹：从 x0=-0.3 出发，积累动量越过 x=0，到达 x=+1
% 先向左一点，然后由于惯性向右冲过势垒
\draw[orange!80!red, very thick, -{Stealth[scale=0.9]}]
  (-0.3,  0.826) -- (-0.05, 0.985) -- (0.2, 0.7056) --
  (0.55, 0.155)  -- (0.85,  0.0596) -- (0.98, 0.0004);

\fill[orange!80!red] (-0.3, 0.826) circle (2.5pt);

% 动量法终点标注
\fill[orange!80!red] (1, 0) circle (3pt);
\draw[orange!80!red, -{Stealth[scale=0.8]}]
  (1.5, 0.5) node[orange!80!red, anchor=west, font=\scriptsize] {动量法越过势垒}
  -- (1.05, 0.05);

% 图例
\draw[blue, very thick, dashed] (-2.1, 2.7) -- (-1.7, 2.7);
\node[blue, anchor=west, font=\footnotesize] at (-1.65, 2.7) {GD（被困）};

\draw[orange!80!red, very thick] (-2.1, 2.4) -- (-1.7, 2.4);
\node[orange!80!red, anchor=west, font=\footnotesize] at (-1.65, 2.4) {动量法（越过）};

% 惯性注释
\node[font=\scriptsize, gray!80] at (0.15, 1.6)
  {动量积累$\to$\\ 冲过势垒};

\end{axis}
\end{tikzpicture}
\end{document}
