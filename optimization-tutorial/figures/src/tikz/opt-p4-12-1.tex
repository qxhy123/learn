\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{document}
\begin{tikzpicture}

\begin{axis}[
  width=10cm, height=9cm,
  xmin=-4.5, xmax=4.5,
  ymin=-4.0, ymax=4.0,
  xlabel={$x$},
  ylabel={$\mathcal{S}_\lambda(x) = \mathrm{sign}(x)\max(|x|-\lambda,\,0)$},
  xlabel style={font=\small},
  ylabel style={font=\small, yshift=4pt},
  axis lines=middle,
  axis line style={-Stealth},
  tick style={font=\small},
  xtick={-4,-3,-2,-1,0,1,2,3,4},
  ytick={-4,-3,-2,-1,0,1,2,3,4},
  extra x ticks={-1.5, 1.5},
  extra x tick labels={$-\lambda$, $\lambda$},
  extra x tick style={
    tick style={red!70, thick},
    tick label style={font=\small, text=red!80!black},
  },
  grid=both,
  grid style={gray!15},
  clip=false,
]

% λ = 1.5

% 三段折线
% 段1：x < -1.5，y = x + 1.5
\addplot[blue, very thick, domain=-4.5:-1.5, samples=2] {x + 1.5};

% 段2：|x| <= 1.5，y = 0
\addplot[blue, very thick, domain=-1.5:1.5, samples=2] {0};

% 段3：x > 1.5，y = x - 1.5
\addplot[blue, very thick, domain=1.5:4.5, samples=2] {x - 1.5};

% 折点（空心圆标注连续性）
\fill[blue] (axis cs:-1.5, 0) circle (2.5pt);
\fill[blue] (axis cs:1.5, 0)  circle (2.5pt);

% 参考线：恒等映射 y = x（灰色虚线）
\addplot[gray!60, dashed, domain=-4.5:4.5, samples=2] {x};
\node[font=\tiny, gray!60, anchor=south west] at (axis cs:3.2, 3.2) {$y=x$（无收缩）};

% 阈值标注（垂直红色虚线）
\draw[red!70, dashed, thick]
  (axis cs:-1.5, -3.5) -- (axis cs:-1.5, 0.2);
\draw[red!70, dashed, thick]
  (axis cs:1.5, -0.2) -- (axis cs:1.5, 3.5);

% 阈值标注（水平方向箭头）
\draw[{Stealth[length=4pt]}-{Stealth[length=4pt]}, red!80, thick]
  (axis cs:-1.5, -1.0) -- (axis cs:1.5, -1.0);
\node[font=\small, red!80!black, anchor=north] at (axis cs:0, -1.05)
  {死区：$|x| \leq \lambda$，输出为 $0$};

% 段落标注
\node[font=\footnotesize, blue!80, align=center] at (axis cs:-3.2, -2.5) {
  $x + \lambda$\\（向右收缩 $\lambda$）
};
\node[font=\footnotesize, blue!80, align=center] at (axis cs:3.2, 2.5) {
  $x - \lambda$\\（向左收缩 $\lambda$）
};

% 收缩量标注（右侧）
\draw[{Stealth[length=4pt]}-{Stealth[length=4pt]}, orange, thick]
  (axis cs:3.5, 3.5) -- (axis cs:3.5, 2.0);
\node[font=\small, orange, anchor=west] at (axis cs:3.55, 2.75)
  {$\lambda$};

\draw[{Stealth[length=4pt]}-{Stealth[length=4pt]}, orange, thick]
  (axis cs:3.5, 2.0) -- (axis cs:3.5, 1.5);
\node[font=\tiny, orange, anchor=west] at (axis cs:3.55, 1.75)
  {$\lambda = 1.5$};

% 点 (3.5, 3.5) 在 y=x 上，(3.5, 2.0) 在软阈值上
\fill[gray!60] (axis cs:3.5, 3.5) circle (1.5pt);
\fill[blue] (axis cs:3.5, 2.0) circle (2pt);

% 标题注释
\node[font=\small, align=center, draw=gray!40, fill=white, rounded corners]
  at (axis cs:-2.8, 2.8) {
    软阈值算子\\
    $\lambda = 1.5$
  };

\end{axis}
\end{tikzpicture}
\end{document}
