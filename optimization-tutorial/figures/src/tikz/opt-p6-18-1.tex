\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\usepackage{pgfplotstable}
\usepackage{amsmath}
\pgfplotsset{compat=1.18}

\begin{document}
\begin{tikzpicture}

% 模拟有效学习率曲线
% AdaGrad: eff_lr = alpha / sqrt(sum_g^2) ，设 alpha=0.1，g恒为1
%   step k: v_k = k, eff = 0.1/sqrt(k)
% RMSprop: rho=0.9, g=1: v_k趋向1，eff趋向 0.1/1=0.1
% Adam: 有偏差修正，前期接近 alpha=0.1，后期根据信噪比自适应

\begin{axis}[
  width=13cm, height=8cm,
  axis lines=left,
  xlabel={迭代步数 $k$},
  ylabel={有效学习率 $\tilde{\alpha}_k$},
  xlabel style={right},
  ylabel style={above, yshift=4pt},
  xmin=0, xmax=200,
  ymin=0, ymax=0.18,
  ytick={0, 0.04, 0.08, 0.10, 0.12, 0.16},
  yticklabels={$0$, $0.04$, $0.08$, $\alpha$, $0.12$, $0.16$},
  xtick={0, 50, 100, 150, 200},
  grid=both,
  grid style={gray!20, line width=0.5pt},
  legend pos=north east,
  legend style={font=\small, fill=white, fill opacity=0.9, draw=gray!50},
  title={AdaGrad / RMSprop / Adam 有效学习率对比（$\alpha=0.1$）},
  title style={font=\small},
  clip=false,
]

% AdaGrad: eff = 0.1 / (sqrt(k) + 1e-8)
\addplot[blue, thick, domain=1:200, samples=200]
  {0.1 / (sqrt(x) + 0.00000001)};
\addlegendentry{AdaGrad（单调递减至 $0$）}

% RMSprop: rho=0.9, g恒为1 => v趋向1/(1-rho)=10*0.1^2=...
% v_k = (1-rho)*sum_{j=1}^k rho^{k-j} * 1 ≈ 1 (稳态)
% 用近似公式 v_k ≈ 1 - 0.9^k
% eff = 0.1 / sqrt(1 - 0.9^k + 1e-6)
\addplot[orange!80!red, thick, domain=1:200, samples=200]
  {0.1 / (sqrt(1 - 0.9^x + 0.000001) + 0.00000001)};
\addlegendentry{RMSprop（趋向稳定值 $\alpha$）}

% Adam: 带偏差修正
% v_k = 1 - 0.999^k (稳态二阶矩=1)
% m_k = 1 - 0.9^k (稳态一阶矩=1)
% 有效步长 = alpha * |m_hat| / sqrt(v_hat)
% 近似为 alpha * SNR, SNR = (1-0.9^k) / sqrt(1-0.999^k)
% 初期 k=1: SNR = 0.1/sqrt(0.001) ≈ 3.16, eff = 0.316
% 但实际 eff = alpha (由偏差修正)
% 真实 eff_k = alpha * (1-0.9^k)/sqrt(1-0.999^k)
\addplot[green!60!black, thick, domain=1:200, samples=200]
  {min(0.16, 0.1 * (1 - 0.9^x) / (sqrt(max(1-0.999^x, 0.00001)) + 0.00000001))};
\addlegendentry{Adam（偏差修正后趋向 $\alpha$）}

% 参考线 alpha = 0.1
\addplot[black, dashed, domain=0:200] {0.10};
\node[anchor=west, font=\scriptsize] at (axis cs:202, 0.10) {$\alpha=0.1$};

\end{axis}
\end{tikzpicture}
\end{document}
