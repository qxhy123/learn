% KKT 条件四要素流程图
\documentclass[tikz,border=8pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usetikzlibrary{calc,arrows.meta,shapes.geometric,shapes.misc,fit,positioning}

\begin{document}
\begin{tikzpicture}[
  boxstyle/.style={
    draw, rounded corners=6pt, thick,
    minimum width=4.2cm, minimum height=1.1cm,
    font=\small, align=center, inner sep=5pt
  },
  arrstyle/.style={->, >=Stealth, thick},
  scale=1.0
]

%% === 中央标题 ===
\node[font=\large\bfseries] (title) at (0, 4.0) {KKT 条件（一阶必要条件）};

%% === 四个条件方框（上下左右布局） ===

%% (1) 驻点条件（顶部）
\node[boxstyle, fill=red!10, draw=red!60!black] (stationarity) at (0, 2.4)
  {\textbf{(1) 驻点条件}\\（Stationarity）\\
   $\nabla f + \displaystyle\sum_i\mu_i\nabla g_i + \displaystyle\sum_j\lambda_j\nabla h_j = \mathbf{0}$};

%% (2) 原始可行（左）
\node[boxstyle, fill=blue!10, draw=blue!60!black] (primal) at (-3.8, 0.0)
  {\textbf{(2) 原始可行}\\（Primal Feasibility）\\
   $g_i(\mathbf{x}) \leq 0$，$h_j(\mathbf{x}) = 0$};

%% (3) 对偶可行（右）
\node[boxstyle, fill=green!10, draw=green!60!black] (dual) at (3.8, 0.0)
  {\textbf{(3) 对偶可行}\\（Dual Feasibility）\\
   $\mu_i \geq 0$};

%% (4) 互补松弛（底部）
\node[boxstyle, fill=orange!10, draw=orange!60!black] (cs) at (0, -2.4)
  {\textbf{(4) 互补松弛}\\（Complementary Slackness）\\
   $\mu_i g_i(\mathbf{x}) = 0,\quad \forall i$};

%% === 连接箭头（四条件均需满足）===
\draw[arrstyle] (stationarity.west) -- (primal.north east);
\draw[arrstyle] (stationarity.east) -- (dual.north west);
\draw[arrstyle] (primal.south east) -- (cs.west);
\draw[arrstyle] (dual.south west) -- (cs.east);

%% === 中央 KKT 点说明 ===
\node[draw=gray!50, fill=yellow!8, rounded corners=4pt,
      font=\small, align=center, inner sep=6pt] (kkt) at (0, 0)
  {\textbf{KKT 点 $\mathbf{x}^*$}\\
   四条件\textbf{同时}满足\\
   （$\Rightarrow$ 局部极小的必要条件）};

%% 从四方框指向中央
\draw[arrstyle, gray!60, dashed] (stationarity.south) -- (kkt.north);
\draw[arrstyle, gray!60, dashed] (primal.east) -- (kkt.west);
\draw[arrstyle, gray!60, dashed] (dual.west) -- (kkt.east);
\draw[arrstyle, gray!60, dashed] (cs.north) -- (kkt.south);

%% === 注释框（右下角）===
\node[draw=gray!40, fill=gray!5, rounded corners, font=\scriptsize,
      align=left, inner sep=5pt] at (0, -4.2)
  {注：等式约束乘子 $\lambda_j\in\mathbb{R}$（无符号限制）；不等式约束乘子 $\mu_i\geq 0$（对偶可行）；互补松弛 $\Leftrightarrow$ $\mu_i=0$ 或 $g_i(\mathbf{x})=0$（二选一）。};

\end{tikzpicture}
\end{document}
