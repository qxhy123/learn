\documentclass[tikz,border=8pt]{standalone}
\usepackage{ctex}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,positioning,arrows.meta,decorations.pathreplacing}

\begin{document}
\begin{tikzpicture}

%% ===== 线搜索 Wolfe 条件示意图 =====
%% 横轴 alpha（步长），纵轴 phi(alpha) = f(x_k + alpha * p_k)
%% 展示：Armijo 条件（充分下降）+ 曲率条件（斜率改善）可接受区间

\begin{axis}[
  width=11cm, height=8cm,
  xmin=-0.05, xmax=2.2,
  ymin=-1.4, ymax=3.6,
  axis lines=left,
  xlabel={$\alpha$（步长）},
  ylabel={$\phi(\alpha) = f(\mathbf{x}_k + \alpha\mathbf{p}_k)$},
  xlabel style={font=\small},
  ylabel style={font=\small},
  xtick={0, 0.4, 0.8, 1.0, 1.2, 1.6, 2.0},
  xticklabels={$0$, , , $\alpha^*$, , , },
  ytick=\empty,
  clip=false,
]

%% ---- 真实曲线 phi(alpha) = 2.5 - 3*alpha + 1.5*alpha^2 ----
%% phi(0) = 2.5，phi'(0) = -3，精确最小值在 alpha* = 1.0，phi(1.0) = 1.0
\addplot[blue!70!black, very thick, domain=0:2.1, samples=120]
  {2.5 + (-3)*x + 1.5*x^2}
  node[pos=0.88, above right, font=\small, blue!70!black] {$\phi(\alpha)$（真实函数）};

%% ---- Armijo 界：phi(0) + c1 * alpha * phi'(0) ----
%% c1 = 0.2，斜率 = c1 * phi'(0) = 0.2 * (-3) = -0.6
\addplot[red!65!black, thick, dashed, domain=0:2.2, samples=2]
  {2.5 + (-0.6)*x}
  node[pos=0.82, above right, font=\small, red!65!black]
  {Armijo 界 $\phi(0)+c_1\alpha\phi'(0)$};

%% ---- 切线（真实初始斜率 phi'(0)=-3）----
\addplot[gray!55, thin, dotted, domain=0:0.9, samples=2]
  {2.5 + (-3)*x}
  node[pos=0.95, below right, font=\scriptsize, gray!70!black]
  {切线（斜率 $\phi'(0)$）};

%% ---- phi(0) 标记 ----
\fill[black] (axis cs:0, 2.5) circle (2.5pt);
\node[left, font=\small] at (axis cs:0, 2.5) {$\phi(0)$};

%% ---- 精确最小值点 alpha* = 1.0 ----
\fill[green!55!black] (axis cs:1.0, 1.0) circle (3pt);
\node[below right, font=\small, green!55!black] at (axis cs:1.0, 1.0)
  {精确最小值 $\alpha^*$};
\draw[dotted, green!40!black, thin] (axis cs:1.0, -0.5) -- (axis cs:1.0, 1.0);

%% ---- Armijo 可接受区间右端点（曲线与 Armijo 界交点）----
%% 2.5 - 3a + 1.5a^2 = 2.5 - 0.6a => 1.5a^2 - 2.4a = 0 => a = 1.6
\fill[red!60!black] (axis cs:1.6, 1.54) circle (2pt);
\draw[dotted, red!50!black, thin] (axis cs:1.6, -0.5) -- (axis cs:1.6, 1.54);

%% ---- Armijo 可接受区间（双向箭头）----
\draw[<->, red!60!black, thick]
  (axis cs:0.0, -0.7) -- (axis cs:1.6, -0.7);
\node[below, font=\small, red!65!black] at (axis cs:0.8, -0.75)
  {Armijo 可接受区间（充分下降）};

%% ---- 曲率条件：phi'(alpha) >= c2 * phi'(0) ----
%% phi'(a) = -3 + 3a，c2 = 0.6 => -3 + 3a >= 0.6*(-3) = -1.8 => 3a >= 1.2 => a >= 0.4
\draw[dotted, blue!50!black, thin]
  (axis cs:0.4, -1.2) -- (axis cs:0.4, {2.5 - 3*0.4 + 1.5*0.16});
\fill[blue!60!black] (axis cs:0.4, {2.5 - 3*0.4 + 1.5*0.16}) circle (2pt);

%% ---- Wolfe 可接受区间（双向箭头）----
\draw[<->, blue!65!black, thick]
  (axis cs:0.4, -1.2) -- (axis cs:1.6, -1.2);
\node[below, font=\small, blue!65!black] at (axis cs:1.0, -1.25)
  {Wolfe 可接受区间（充分下降 + 曲率条件）};

%% ---- 注释文字框 ----
\node[draw=red!35, fill=red!5, rounded corners=3pt,
      font=\scriptsize, inner sep=5pt, align=left, anchor=north west]
  at (axis cs:0.05, 3.55) {
    \textbf{Armijo 充分下降条件}：\\
    $\phi(\alpha) \leq \phi(0) + c_1\alpha\phi'(0)$\\
    参数：$c_1 \in (0,1)$，典型 $10^{-4}$\\
    防止步长太大（函数值不够下降）
  };

\node[draw=blue!35, fill=blue!5, rounded corners=3pt,
      font=\scriptsize, inner sep=5pt, align=left, anchor=north east]
  at (axis cs:2.15, 3.55) {
    \textbf{曲率条件（Wolfe 第二条件）}：\\
    $\phi'(\alpha) \geq c_2\phi'(0)$\\
    参数：$c_2 \in (c_1,1)$，典型 $0.9$\\
    防止步长太小（远未到达极小值）
  };

\end{axis}

\node[font=\small\bfseries] at (5.8, -0.7)
  {线搜索 Wolfe 条件：Armijo 区间 $\subset$ 可接受区间 $\subset$ Wolfe 区间};

\end{tikzpicture}
\end{document}
