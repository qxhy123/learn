\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}

  \begin{axis}[
      width=9.5cm, height=9.5cm,
      axis lines=middle,
      axis equal image,
      xmin=-1.35, xmax=1.35, ymin=-1.35, ymax=1.35,
      xtick={-1,1}, ytick={-1,1},
      xlabel={\small $x$}, ylabel={\small $y$},
      every axis x label/.style={at={(ticklabel* cs:1.0)},anchor=west},
      every axis y label/.style={at={(ticklabel* cs:1.0)},anchor=south},
      title={\small 李萨如图形 $\;x=\sin 3t,\ \ y=\sin 2t$},
      clip=false,
    ]
    % 李萨如曲线 x=sin(3t), y=sin(2t), t in [0, 2pi]
    \addplot[domain=0:360,samples=600,smooth,blue!70!black,very thick]
      ({sin(3*x)},{sin(2*x)});

    % 标注起点 t=0 处 (0,0)
    \fill[red!75!black] (axis cs:0,0) circle (1.6pt);

    % 半透明注释框（放右下角，避开曲线最密区）
    \node[align=left,fill=blue!8,fill opacity=0.85,text opacity=1,
          rounded corners,draw=blue!35,inner sep=4pt]
      at (axis cs:0.80,-1.12)
      {\small 频率比 $3:2$};
  \end{axis}

\end{tikzpicture}
\end{document}
