\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=15cm, height=8cm,
    xmin=0, xmax=6.2832, ymin=-2.6, ymax=2.6,
    axis lines=middle,
    xlabel={$t$}, ylabel={},
    xtick={0,1.5708,3.1416,4.7124,6.2832},
    xticklabels={$0$,,$\pi$,,$2\pi$},
    ytick={-2,-1,1,2},
    enlargelimits=false,
    clip=false,
    samples=600, domain=0:6.2832,
    tick label style={font=\small},
    label style={font=\small},
]
  % 合成波 sin(2 pi f1 t)+sin(2 pi f2 t)，取 f1=20/(2pi)≈ -> 用 f1t=20t? 简化：sin(11 t)+sin(9 t)
  % 11与9 -> 中心频率10，频率差1 => 包络 2cos(t)
  \addplot[blue!70,very thick] {sin(deg(11*x))+sin(deg(9*x))};

  % 包络 +/- 2cos(t)
  \addplot[red!75,thick,dashed] {2*cos(deg(x))};
  \addplot[red!75,thick,dashed] {-2*cos(deg(x))};

  % 包络标注
  \node[red!75,font=\small] at (axis cs:0.9,2.25) {包络 $2\cos t$};

  % 半透明注释框（放在波节附近的上方空白，避开峰值）
  \node[fill=white,fill opacity=0.80,text opacity=1,rounded corners,
        draw=blue!45,inner sep=4pt,align=center,font=\small]
        at (axis cs:3.1416,2.05)
        {和差化积：\\[1pt]
         $\sin 11t+\sin 9t=2\cos t\,\sin 10t$\\[1pt]
         快速振荡 $\sin 10t$ $\times$ 慢变包络 $2\cos t$};
\end{axis}
\end{tikzpicture}
