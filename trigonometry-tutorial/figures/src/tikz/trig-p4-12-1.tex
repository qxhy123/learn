\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}

\node[font=\bfseries, anchor=south] at (6.0cm,7.6cm)
  {$a\sin x+b\cos x=R\sin(x+\varphi)$ 的合成曲线与最值（$a=3,\ b=4,\ R=5$）};

\begin{axis}[
    width=14cm, height=8.2cm,
    axis lines=middle,
    xlabel={$x$}, ylabel={$y$},
    xtick={-3.14159,0,3.14159,6.28319},
    xticklabels={$-\pi$,$0$,$\pi$,$2\pi$},
    ytick={-5,-4,-3,3,4,5}, yticklabels={$-5$,$-4$,$-3$,$3$,$4$,$5$},
    ymajorgrids=false,
    xmin=-3.8, xmax=7.4, ymin=-6.6, ymax=6.0,
    enlargelimits=false,
    tick label style={font=\small},
    label style={font=\small},
    legend style={at={(0.5,-0.11)}, anchor=north, legend columns=-1,
                  font=\small, draw=none, fill=none, column sep=1.4em},
  ]
  % R 与 -R 水平参考线
  \addplot[domain=-3.8:7.4, samples=2, red!70!black, dashed, forget plot] {5};
  \addplot[domain=-3.8:7.4, samples=2, red!70!black, dashed, forget plot] {-5};

  % 三条分量与合成
  \addplot[domain=-3.8:7.4, samples=300, gray!55, thin] {3*sin(deg(x))};
  \addlegendentry{$3\sin x$}
  \addplot[domain=-3.8:7.4, samples=300, blue!45!gray, thin, densely dotted] {4*cos(deg(x))};
  \addlegendentry{$4\cos x$}
  \addplot[domain=-3.8:7.4, samples=400, very thick, black] {5*sin(deg(x)+53.1301)};
  \addlegendentry{$5\sin(x+\varphi)$}

  % 最大值点：5 sin(x+phi)=5 当 x+phi=pi/2 => x = pi/2 - phi, phi=53.13deg=0.9273rad
  % x_max = 1.5708 - 0.9273 = 0.6435; 周期 2pi
  \addplot[only marks, mark=*, red!75!black, mark size=2pt, forget plot]
    coordinates {(0.6435,5) (-5.6397,5)};
  % 最小值点：x+phi = -pi/2 => x = -pi/2 - phi = -2.498; +2pi=3.785
  \addplot[only marks, mark=*, red!75!black, mark size=2pt, forget plot]
    coordinates {(3.7851,-5) (-2.4981,-5)};

  % 最值标注（半透明，避开峰）
  \node[font=\small, red!70!black, fill=white, fill opacity=0.8, text opacity=1,
        inner sep=1.5pt, anchor=south west] at (axis cs:0.9,5.15)
    {最大值 $R=\sqrt{a^2+b^2}=5$};
  \node[font=\small, red!70!black, fill=white, fill opacity=0.8, text opacity=1,
        inner sep=1.5pt, anchor=north] at (axis cs:3.7851,-5.15)
    {最小值 $-R=-5$};
\end{axis}

\end{tikzpicture}
\end{document}
