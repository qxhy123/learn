\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}[scale=3.0,>=Stealth]
  % 直角三角形：theta 在 O 处，对边 x（竖直），邻边 sqrt(a^2-x^2)（水平），斜边 a
  \coordinate (O) at (0,0);          % 角 theta 顶点
  \coordinate (A) at (1.732,0);      % 直角顶点，邻边长 sqrt(a^2-x^2)
  \coordinate (B) at (1.732,1);      % 对边端点，对边长 x

  % 三角形
  \draw[very thick] (O) -- (A) -- (B) -- cycle;

  % 直角标记
  \draw (1.617,0) -- (1.617,0.115) -- (1.732,0.115);

  % 角 theta
  \draw pic[draw,->,"$\theta$",angle eccentricity=1.3,angle radius=0.85cm,font=\large]
        {angle=A--O--B};

  % 边标注
  \node[below,font=\large] at (0.866,-0.02) {$\sqrt{a^2-x^2}$};   % 邻边
  \node[right,font=\large] at (1.732,0.5) {$x$};                  % 对边
  \node[above left,font=\large,blue!70!black] at (0.866,0.5) {$a$}; % 斜边

  % 斜边加粗显色
  \draw[very thick,blue!70!black] (O) -- (B);

  % 关系说明（半透明框，放在三角形外侧空白区，避开图形）
  \node[align=left,fill=yellow!18,fill opacity=0.85,text opacity=1,
        draw=gray!50,rounded corners,inner sep=5pt,font=\small]
        at (0.92,1.28)
        {代换 $x=a\sin\theta$：\\[2pt]
         $\sin\theta=\dfrac{x}{a},\quad
          \cos\theta=\dfrac{\sqrt{a^2-x^2}}{a}$\\[3pt]
         于是 $\sqrt{a^2-x^2}=a\cos\theta$，根式被三角化简};
\end{tikzpicture}
\end{document}
