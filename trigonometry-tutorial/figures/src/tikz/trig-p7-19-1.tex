\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}[scale=3.2,>=Latex]
  \def\ang{52}
  % axes
  \draw[->,gray!70] (-1.35,0) -- (1.45,0) node[right,black] {$\mathrm{Re}$};
  \draw[->,gray!70] (0,-1.35) -- (0,1.45) node[above,black] {$\mathrm{Im}$};
  % unit circle
  \draw[thick,blue!60] (0,0) circle (1);
  % radius to point
  \coordinate (O) at (0,0);
  \coordinate (P) at ({cos(\ang)},{sin(\ang)});
  \coordinate (Px) at ({cos(\ang)},0);
  \coordinate (Py) at (0,{sin(\ang)});
  % projections
  \draw[dashed,red!70] (P) -- (Px);
  \draw[dashed,orange!80!black] (P) -- (Py);
  % cos segment (real part) on Re axis
  \draw[very thick,red!75] (O) -- (Px);
  \node[red!75,below] at (0.5,-0.04) {$\cos\theta$};
  % sin segment (imaginary part) on Im axis
  \draw[very thick,orange!80!black] (O) -- (Py);
  \node[orange!80!black,left] at (-0.02,0.42) {$\sin\theta$};
  % radius
  \draw[very thick,black] (O) -- (P);
  \node[above right=1pt and -1pt] at (0.5,0.62) {$1$};
  % angle arc
  \draw[->,thick,purple] (0.42,0) arc (0:\ang:0.42);
  \node[purple] at ({0.56*cos(\ang/2)},{0.56*sin(\ang/2)}) {$\theta$};
  % the point
  \filldraw[black] (P) circle (0.018);
  \node[above right=1pt] at (P) {$e^{i\theta}=\cos\theta+i\sin\theta$};
  % small tick marks at 1 and i
  \filldraw[blue!60] (1,0) circle (0.012) node[below right=0pt] {\small $1$};
  \filldraw[blue!60] (0,1) circle (0.012) node[above left=0pt] {\small $i$};
\end{tikzpicture}
\end{document}
