\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}[scale=1.0]
  % 三角形 ABC：底 a = BC，从顶点 A 作高 h = b*sinC，b = CA
  % 取 C 在原点，B 在 x 轴正向（a=BC 沿底边），A 由 b 与角 C 决定
  \pgfmathsetmacro{\aside}{6}     % 底 a = BC
  \pgfmathsetmacro{\bside}{4}     % 边 b = CA
  \pgfmathsetmacro{\Cang}{52}     % 角 C

  \coordinate (C) at (0,0);
  \coordinate (B) at (\aside,0);
  \coordinate (A) at ({\bside*cos(\Cang)},{\bside*sin(\Cang)});
  \coordinate (H) at ({\bside*cos(\Cang)},0);   % 高的垂足

  % 三角形
  \draw[very thick] (A) -- (B) -- (C) -- cycle;

  % 高 h（从 A 到底边 BC 的垂线），虚线
  \draw[dashed,thick] (A) -- (H);
  % 直角标记
  \draw ($(H)+(-0.3,0)$) -- ($(H)+(-0.3,0.3)$) -- ($(H)+(0,0.3)$);

  % 角 C 标记
  \pic[draw,thick,"$C$",angle radius=0.95cm,angle eccentricity=1.5]
       {angle = B--C--A};

  % 顶点
  \fill (A) circle (1.8pt) node[above] {$A$};
  \fill (B) circle (1.8pt) node[below right] {$B$};
  \fill (C) circle (1.8pt) node[below left] {$C$};
  \fill (H) circle (1.4pt) node[below=2pt] {$H$};

  % 边 b = CA
  \node[fill=white,inner sep=1.5pt] at ($(C)!0.5!(A)+(-0.32,0.12)$) {$b$};
  % 高 h = b sin C
  \node[fill=white,inner sep=1.5pt] at ($(H)!0.5!(A)+(0.42,0)$) {$h=b\sin C$};
  % 底 a = BC
  \draw[<->,thin] (0,-0.7) -- (\aside,-0.7)
        node[midway,fill=white,inner sep=1.5pt] {底 $a$};

  % 半透明面积说明框
  \node[fill=green!16,fill opacity=0.78,text opacity=1,rounded corners,
        inner sep=4pt,align=left] at (5.0,3.55)
       {$S=\dfrac12\,a\,h=\dfrac12\,ab\sin C$};
\end{tikzpicture}
\end{document}
