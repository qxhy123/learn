\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}[>=Stealth,scale=1.0]

  % ===== 原向量 v 与旋转角 theta =====
  \def\vx{2.7}
  \def\vy{0.85}
  \def\th{55}   % 旋转角 theta（度）

  % 旋转后向量 v' = R(theta) v
  \pgfmathsetmacro\wx{\vx*cos(\th) - \vy*sin(\th)}
  \pgfmathsetmacro\wy{\vx*sin(\th) + \vy*cos(\th)}
  \pgfmathsetmacro\vlen{sqrt(\vx*\vx+\vy*\vy)}

  % 坐标轴
  \draw[->,gray!70] (-1.0,0) -- (4.0,0) node[right,black]{\small $x$};
  \draw[->,gray!70] (0,-0.8) -- (0,3.9) node[above,black]{\small $y$};

  % 圆弧（标示旋转半径不变，模长保持）
  \draw[gray!40,densely dashed] (0,0) circle (\vlen);

  % 原向量 v
  \draw[->,blue!70!black,very thick] (0,0) -- (\vx,\vy)
    node[below right=-2pt,blue!70!black]{\small $\mathbf v$};

  % 旋转后向量 v'（标签放右侧，避开左上角的矩阵框）
  \draw[->,red!70!black,very thick] (0,0) -- (\wx,\wy);
  \node[red!70!black,right=1pt] at (\wx+0.05,\wy+0.18) {\small $\mathbf v' = R(\theta)\,\mathbf v$};

  % 旋转角弧 theta（从 v 到 v'）
  \pgfmathsetmacro\angv{atan2(\vy,\vx)}
  \pgfmathsetmacro\angw{atan2(\wy,\wx)}
  \draw[->,orange!85!black,thick] (\angv:1.5) arc (\angv:\angw:1.5);
  \node[orange!85!black] at ({(\angv+\angw)/2}:1.85) {\small $\theta$};

  % 模长保持标注
  \node[gray!45!black] at (\angv:\vlen+0.0) [below=1pt] {};

  % ===== 旋转矩阵注释框（半透明，放左上角空白区，避开向量）=====
  \node[align=center,fill=blue!8,fill opacity=0.9,text opacity=1,
        rounded corners,draw=blue!40,inner sep=5pt]
    at (-1.85,3.15)
    {\small $R(\theta)=\begin{pmatrix}\cos\theta & -\sin\theta\\[2pt]\sin\theta & \cos\theta\end{pmatrix}$};

  % 模长不变说明（放右下，避开峰值）
  \node[align=left,fill=gray!10,fill opacity=0.85,text opacity=1,
        rounded corners,inner sep=3pt]
    at (2.95,-1.05)
    {\small 旋转保持模长：$|\mathbf v'|=|\mathbf v|$};

\end{tikzpicture}
\end{document}
