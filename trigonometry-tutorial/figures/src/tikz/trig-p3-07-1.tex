\begin{document}
\begin{tikzpicture}

% ===== 面板一：y = sin x =====
\begin{scope}[yshift=0cm]
\begin{axis}[
  width=14cm, height=5.2cm,
  axis lines=middle,
  xmin=-3.6, xmax=7.4, ymin=-1.6, ymax=1.9,
  xtick={-3.14159,0,1.5708,3.14159,4.71239,6.28319},
  xticklabels={$-\pi$,$0$,$\frac{\pi}{2}$,$\pi$,$\frac{3\pi}{2}$,$2\pi$},
  ytick={-1,1}, yticklabels={$-1$,$1$},
  xlabel={$x$}, ylabel={$y$},
  xlabel style={anchor=west}, ylabel style={anchor=south},
  tick label style={font=\footnotesize},
  title={\bfseries $y=\sin x$（周期 $2\pi$，过原点，有界）},
  title style={yshift=2pt},
  clip=false,
]
  \addplot[blue,very thick,smooth,samples=200,domain=-3.4:6.6] {sin(deg(x))};
  % 周期标注：0 -> 2pi
  \draw[<->,red!70!black,thick] (axis cs:0,1.45) -- (axis cs:6.28319,1.45)
    node[midway,fill=white,fill opacity=0.75,text opacity=1,inner sep=1.5pt,font=\footnotesize]{周期 $T=2\pi$};
  \draw[red!70!black,dashed,thin] (axis cs:0,0) -- (axis cs:0,1.45);
  \draw[red!70!black,dashed,thin] (axis cs:6.28319,0) -- (axis cs:6.28319,1.45);
  % 峰点
  \filldraw[blue] (axis cs:1.5708,1) circle (1.6pt);
\end{axis}
\end{scope}

% ===== 面板三：y = tan x =====
\begin{scope}[yshift=-12cm]
\begin{axis}[
  width=14cm, height=5.6cm,
  axis lines=middle,
  xmin=-3.6, xmax=7.4, ymin=-4, ymax=4.5,
  xtick={-3.14159,-1.5708,0,1.5708,3.14159,4.71239,6.28319},
  xticklabels={$-\pi$,$-\frac{\pi}{2}$,$0$,$\frac{\pi}{2}$,$\pi$,$\frac{3\pi}{2}$,$2\pi$},
  ytick={-2,2}, yticklabels={$-2$,$2$},
  xlabel={$x$}, ylabel={$y$},
  xlabel style={anchor=west}, ylabel style={anchor=south},
  tick label style={font=\footnotesize},
  title={\bfseries $y=\tan x$（周期 $\pi$，无界，竖直渐近线 $x=\frac{\pi}{2}+k\pi$）},
  title style={yshift=2pt},
  clip=false,
]
  % 各分支（避开渐近线）
  \addplot[violet,very thick,smooth,samples=120,domain=-1.4:1.4] {tan(deg(x))};
  \addplot[violet,very thick,smooth,samples=120,domain=1.74159:4.4823] {tan(deg(x))};
  \addplot[violet,very thick,smooth,samples=120,domain=4.88:6.6] {tan(deg(x))};
  % 渐近线
  \draw[red!70!black,dashed,thick] (axis cs:-1.5708,-4) -- (axis cs:-1.5708,4.5);
  \draw[red!70!black,dashed,thick] (axis cs:1.5708,-4) -- (axis cs:1.5708,4.5);
  \draw[red!70!black,dashed,thick] (axis cs:4.71239,-4) -- (axis cs:4.71239,4.5);
  % 周期标注：相邻渐近线间距 = pi
  \draw[<->,orange!80!black,thick] (axis cs:1.5708,-3.4) -- (axis cs:4.71239,-3.4)
    node[midway,fill=white,fill opacity=0.78,text opacity=1,inner sep=1.5pt,font=\footnotesize]{周期 $T=\pi$};
\end{axis}
\end{scope}
