\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}[scale=3.6,>=Stealth]
  % 角度
  \def\ang{37}

  % 坐标轴
  \draw[->,gray!70] (-1.25,0) -- (1.35,0) node[below right,gray!80] {$x$};
  \draw[->,gray!70] (0,-1.25) -- (0,1.35) node[above left,gray!80] {$y$};

  % 单位圆
  \draw[thick,blue!55] (0,0) circle (1);

  % 圆上点 P
  \coordinate (O) at (0,0);
  \coordinate (P) at (\ang:1);
  \coordinate (F) at ({cos(\ang)},0); % 横轴上的垂足

  % 直角三角形：填充
  \fill[orange!18] (O) -- (F) -- (P) -- cycle;

  % 半径=斜边 1
  \draw[very thick,red!75] (O) -- (P) node[midway,above left=-1pt] {$1$};
  % 底边 = cos theta
  \draw[very thick,teal!80] (O) -- (F) node[midway,below=1pt] {$\cos\theta$};
  % 竖边 = sin theta
  \draw[very thick,violet!80] (F) -- (P) node[midway,right=1pt] {$\sin\theta$};

  % 直角标记
  \draw (F) ++(-0.075,0) -- ++(0,0.075) -- ++(0.075,0);

  % 角 theta
  \draw[->,thick] (0.45,0) arc (0:\ang:0.45);
  \node at (\ang*0.5:0.30) {$\theta$};

  % 点 P 标注
  \fill[red!75] (P) circle (0.018);
  \node[above right] at (P) {$P(\cos\theta,\,\sin\theta)$};

  % 半透明注释框（放在第三象限空白处，避开图形）
  \node[fill=white,fill opacity=0.78,text opacity=1,rounded corners,
        draw=blue!40,inner sep=4pt,align=center,font=\small]
        at (-0.62,-0.66)
        {勾股定理 $\Rightarrow$\\[1pt]$\cos^2\theta+\sin^2\theta=1$};
\end{tikzpicture}
\end{document}
