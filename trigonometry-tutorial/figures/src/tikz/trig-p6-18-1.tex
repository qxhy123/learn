\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=12cm, height=7.5cm,
    xmin=0, xmax=6.2832,
    ymin=-1.45, ymax=1.45,
    samples=300,
    axis lines=middle,
    xlabel={$x$}, ylabel={$y$},
    xtick={0,1.5708,3.1416,4.7124,6.2832},
    xticklabels={$0$,$\frac{\pi}{2}$,$\pi$,$\frac{3\pi}{2}$,$2\pi$},
    ytick={-1,-0.5,0.5,1},
    every axis x label/.style={at={(ticklabel* cs:1)},anchor=west},
    every axis y label/.style={at={(ticklabel* cs:1)},anchor=south},
    legend pos=south east,
    legend cell align=left,
    legend style={font=\small,fill=white,fill opacity=0.85,
                  text opacity=1,draw=gray!50},
    grid=major, grid style={gray!18},
]
  % cos 3x —— 蓝色实线
  \addplot[blue!70!black,very thick,domain=0:6.2832] {cos(deg(3*x))};
  \addlegendentry{$y=\cos 3x$}

  % 4cos^3 x - 3cos x —— 橙色虚线（应与上式完全重合）
  \addplot[orange!90!black,thick,dashed,domain=0:6.2832,samples=160]
        {4*(cos(deg(x)))^3 - 3*cos(deg(x))};
  \addlegendentry{$y=4\cos^3 x-3\cos x$}
\end{axis}

% 半透明注释框（放在左上空白处，避开曲线峰值）
\node[align=left,fill=yellow!18,fill opacity=0.85,text opacity=1,
      draw=gray!50,rounded corners,inner sep=5pt,font=\small]
      at (3.0,6.55)
      {两条曲线完全重合，可视化恒等式\\[2pt]
       $\cos 3x=4\cos^3x-3\cos x=T_3(\cos x)$};
\end{tikzpicture}
