\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  scale=1.2, thick,
  vec/.style={-{Stealth[length=7pt, width=4pt]}, thick},
]

% 三角形顶点
\coordinate (A) at (0, 0);
\coordinate (B) at (4, 0);
\coordinate (C) at (1.2, 2.8);

% BC 的中点 D
\coordinate (D) at ({(4+1.2)/2}, {(0+2.8)/2});  % (2.6, 1.4)

% 三角形边（黑色实线）
\draw[black, thick] (A) -- (B) -- (C) -- cycle;

% 中线 AD（红色）
\draw[red, thick] (A) -- (D);

% 等长边标记：BD = DC（tickone style 手动画）
% BD 中点
\coordinate (MBD) at ({(4+2.6)/2}, {(0+1.4)/2});  % (3.3, 0.7)
\draw[thin, black] ($(MBD) + (-1.5pt,-3pt)$) -- ($(MBD) + (1.5pt,3pt)$);
% DC 中点
\coordinate (MDC) at ({(2.6+1.2)/2}, {(1.4+2.8)/2});  % (1.9, 2.1)
\draw[thin, black] ($(MDC) + (-1.5pt,-3pt)$) -- ($(MDC) + (1.5pt,3pt)$);

% 顶点标签
\node[below left]  at (A) {$A$};
\node[below right] at (B) {$B$};
\node[above]       at (C) {$C$};
\node[right, font=\small] at (D) {$D$（中点）};

% 向量标注
\draw[vec, blue, thick] (A) -- (B) node[midway, below, yshift=-3pt] {$\vec{AB}$};
\draw[vec, blue, thick] (A) -- (C) node[midway, left, xshift=-2pt]  {$\vec{AC}$};
\draw[vec, red,  thick] (A) -- (D) node[midway, right, xshift=12pt, yshift=4pt] {$\vec{AD}$};

% 顶点圆点
\filldraw[black] (A) circle (1.5pt);
\filldraw[black] (B) circle (1.5pt);
\filldraw[black] (C) circle (1.5pt);
\filldraw[red]   (D) circle (1.5pt);

% 公式说明
\node[font=\small, align=center] at (2.0, -0.65)
  {$\vec{AD} = \dfrac{1}{2}(\vec{AB} + \vec{AC})$};

\end{tikzpicture}
\end{document}
