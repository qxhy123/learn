\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=0.9, thick]

% 左图：焦点在 x 轴的双曲线  x^2/4 - y^2/3 = 1  (a=2, b=sqrt(3), c=sqrt(7))
% 右图：焦点在 y 轴的双曲线  y^2/4 - x^2/3 = 1  (a=2, b=sqrt(3), c=sqrt(7))

%% -------- 左图（焦点在 x 轴）--------
\begin{scope}[xshift=-4.5cm]
  % 坐标轴
  \draw[->, thin, gray] (-4.0, 0) -- (4.0, 0) node[right, font=\small] {$x$};
  \draw[->, thin, gray] (0, -3.2) -- (0, 3.2) node[above, font=\small] {$y$};
  \node[font=\footnotesize, below right] at (0,0) {$O$};

  % 双曲线右支 x^2/4 - y^2/3 = 1，a=2, b=sqrt(3)
  \draw[black, thick, domain=-2.0:2.0, smooth, samples=80]
      plot ({2*cosh(\x)}, {sqrt(3)*sinh(\x)});
  % 左支
  \draw[black, thick, domain=-2.0:2.0, smooth, samples=80]
      plot ({-2*cosh(\x)}, {sqrt(3)*sinh(\x)});

  % 渐近线 y=±(sqrt(3)/2)x
  \draw[dashed, gray, thin] (-3.8, {-3.8*sqrt(3)/2}) -- (3.8, {3.8*sqrt(3)/2});
  \draw[dashed, gray, thin] (-3.8, {3.8*sqrt(3)/2}) -- (3.8, {-3.8*sqrt(3)/2});

  % 焦点 (±sqrt(7),0)
  \fill[black] ({sqrt(7)},0) circle [radius=2pt];
  \fill[black] ({-sqrt(7)},0) circle [radius=2pt];
  \node[font=\footnotesize, below right] at ({sqrt(7)},0) {$F_2$};
  \node[font=\footnotesize, below left] at ({-sqrt(7)},0) {$F_1$};

  % 顶点
  \fill[black] (2,0) circle [radius=2pt];
  \fill[black] (-2,0) circle [radius=2pt];
  \node[font=\footnotesize, below] at (2,0) {$(a,0)$};
  \node[font=\footnotesize, below] at (-2,0) {$(-a,0)$};

  % 标题
  \node[font=\small, below] at (0, -3.8)
      {$\dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1$（焦点在 $x$ 轴）};
\end{scope}

%% -------- 右图（焦点在 y 轴）--------
\begin{scope}[xshift=4.5cm]
  % 坐标轴
  \draw[->, thin, gray] (-4.0, 0) -- (4.0, 0) node[right, font=\small] {$x$};
  \draw[->, thin, gray] (0, -3.2) -- (0, 3.2) node[above, font=\small] {$y$};
  \node[font=\footnotesize, below right] at (0,0) {$O$};

  % 双曲线 y^2/4 - x^2/3 = 1，上支（cosh 方向沿 y 轴）
  \draw[black, thick, domain=-2.0:2.0, smooth, samples=80]
      plot ({sqrt(3)*sinh(\x)}, {2*cosh(\x)});
  % 下支
  \draw[black, thick, domain=-2.0:2.0, smooth, samples=80]
      plot ({sqrt(3)*sinh(\x)}, {-2*cosh(\x)});

  % 渐近线 y=±(2/sqrt(3))x
  \draw[dashed, gray, thin] (-3.0, {-3.0*2/sqrt(3)}) -- (3.0, {3.0*2/sqrt(3)});
  \draw[dashed, gray, thin] (-3.0, {3.0*2/sqrt(3)}) -- (3.0, {-3.0*2/sqrt(3)});

  % 焦点 (0,±sqrt(7))
  \fill[black] (0,{sqrt(7)}) circle [radius=2pt];
  \fill[black] (0,{-sqrt(7)}) circle [radius=2pt];
  \node[font=\footnotesize, right] at (0,{sqrt(7)}) {$F_2$};
  \node[font=\footnotesize, right] at (0,{-sqrt(7)}) {$F_1$};

  % 顶点
  \fill[black] (0,2) circle [radius=2pt];
  \fill[black] (0,-2) circle [radius=2pt];
  \node[font=\footnotesize, right] at (0,2) {$(0,a)$};
  \node[font=\footnotesize, right] at (0,-2) {$(0,-a)$};

  % 标题
  \node[font=\small, below] at (0, -3.8)
      {$\dfrac{y^2}{a^2}-\dfrac{x^2}{b^2}=1$（焦点在 $y$ 轴）};
\end{scope}

\end{tikzpicture}
\end{document}
