\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  scale=0.9, thick,
  vec/.style={-{Stealth[length=7pt, width=4pt]}, thick},
  axis/.style={-{Stealth[length=6pt, width=3.5pt]}, thin, gray},
]

% 直角坐标轴
\draw[axis] (-0.3, 0) -- (4.2, 0) node[right] {$x$};
\draw[axis] (0, -0.3) -- (0, 3.2) node[above] {$y$};
\node[below left] at (0,0) {$O$};

% 基底单位向量 e1=(1,0), e2=(0,1)
\draw[vec, blue] (0,0) -- (1,0) node[below, midway, yshift=-3pt]
  {\small $\vec{e}_1=(1,0)$};
\draw[vec, blue] (0,0) -- (0,1) node[left, midway, xshift=-3pt]
  {\small $\vec{e}_2=(0,1)$};

% 向量 a = (3, 2)
\coordinate (A) at (3, 2);
\draw[vec, black, very thick] (0,0) -- (A)
  node[above right] {$\vec{a} = (3,\,2)$};

% x 投影（红色虚线）
\draw[dashed, red, thick] (A) -- (3, 0);
\draw[{Stealth[length=5pt]}-{Stealth[length=5pt]}, red, thin] (0, -0.55) -- (3, -0.55)
  node[midway, below] {\small $x$ 分量 $= 3$};

% y 投影（蓝色虚线）
\draw[dashed, blue!70!black, thick] (A) -- (0, 2);
\draw[{Stealth[length=5pt]}-{Stealth[length=5pt]}, blue!70!black, thin]
  (3.6, 0) -- (3.6, 2)
  node[midway, right] {\small $y$ 分量 $= 2$};

% 坐标刻度
\foreach \x in {1,2,3} {
  \draw[thin, gray] (\x, 0.05) -- (\x, -0.05);
  \node[below, gray, font=\footnotesize] at (\x, -0.08) {$\x$};
}
\foreach \y in {1,2} {
  \draw[thin, gray] (0.05, \y) -- (-0.05, \y);
  \node[left, gray, font=\footnotesize] at (-0.08, \y) {$\y$};
}

% 端点标记
\filldraw[black] (A) circle (1.5pt);
\filldraw[black] (0,0) circle (1.5pt);

% 模长公式
\node[font=\small] at (1.5, -1.1)
  {$|\vec{a}| = \sqrt{3^2 + 2^2} = \sqrt{13}$};

\end{tikzpicture}
\end{document}
