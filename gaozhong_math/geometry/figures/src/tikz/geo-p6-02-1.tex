\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta, decorations.pathreplacing}

\begin{document}
\begin{tikzpicture}[scale=1.0, thick]

% 双曲线几何性质图：顶点 + 焦点 + 渐近线 + a,b,c 标注
% 方程：x^2/4 - y^2/9 = 1  (a=2, b=3, c=sqrt(13)≈3.606)

% 坐标轴
\draw[->, thin, gray] (-4.8, 0) -- (4.8, 0) node[right, font=\small] {$x$};
\draw[->, thin, gray] (0, -4.0) -- (0, 4.0) node[above, font=\small] {$y$};
\node[font=\footnotesize, below right] at (0,0) {$O$};

% 渐近线 y = ±(3/2)x
\draw[dashed, gray] (-4.5, {-4.5*3/2}) -- (4.5, {4.5*3/2})
    node[above right, font=\footnotesize, gray] {$y=\dfrac{b}{a}x$};
\draw[dashed, gray] (-4.5, {4.5*3/2}) -- (4.5, {-4.5*3/2})
    node[below right, font=\footnotesize, gray] {$y=-\dfrac{b}{a}x$};

% 双曲线（a=2, b=3）右支
\draw[black, thick, domain=-1.5:1.5, smooth, samples=80]
    plot ({2*cosh(\x)}, {3*sinh(\x)});
% 左支
\draw[black, thick, domain=-1.5:1.5, smooth, samples=80]
    plot ({-2*cosh(\x)}, {3*sinh(\x)});

% 顶点 A1(-2,0), A2(2,0)
\fill[black] (2,0) circle [radius=2.5pt];
\fill[black] (-2,0) circle [radius=2.5pt];
\node[font=\small, below right] at (2,0) {$A_2(a,0)$};
\node[font=\small, below left] at (-2,0) {$A_1(-a,0)$};

% 焦点 F1(-c,0), F2(c,0)，c=sqrt(13)≈3.606
\fill[black] ({sqrt(13)},0) circle [radius=2.5pt];
\fill[black] ({-sqrt(13)},0) circle [radius=2.5pt];
\node[font=\small, below right] at ({sqrt(13)},0) {$F_2(c,0)$};
\node[font=\small, below left] at ({-sqrt(13)},0) {$F_1(-c,0)$};

% 辅助点 B(0,b) 用于标注 b
\fill[gray] (0,3) circle [radius=2pt];
\node[font=\footnotesize, right] at (0,3) {$B(0,b)$};

% 辅助矩形（虚线）显示 a, b 关系
\draw[gray, dashed, thin] (0,0) -- (2,0);
\draw[gray, dashed, thin] (0,0) -- (0,3);
\draw[gray, dashed, thin] (2,0) -- (2,3);
\draw[gray, dashed, thin] (0,3) -- (2,3);

% 标注 a（水平）
\draw[<->, blue, thin] (0,-0.35) -- (2,-0.35)
    node[midway, below, font=\footnotesize, blue] {$a$};

% 标注 b（垂直）
\draw[<->, blue, thin] (2.35,0) -- (2.35,3)
    node[midway, right, font=\footnotesize, blue] {$b$};

% 标注 c（焦点到原点）
\draw[<->, red, thin] (0,0.25) -- ({sqrt(13)},0.25)
    node[midway, above, font=\footnotesize, red] {$c$};

% 关系式
\node[font=\small, align=center] at (0, -4.7)
    {$c^2 = a^2 + b^2$（注意是加号！）};

\end{tikzpicture}
\end{document}
