\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.0, thick]

% 双曲线定义图：两支 + 两焦点 + 动点 P + 焦半径
% 参数：a=1.5, b=1.0, c=sqrt(a^2+b^2)=sqrt(3.25)≈1.803

% 坐标轴
\draw[->, thin, gray] (-4.2, 0) -- (4.2, 0) node[right, font=\small] {$x$};
\draw[->, thin, gray] (0, -3.2) -- (0, 3.2) node[above, font=\small] {$y$};
\node[font=\footnotesize, below right] at (0,0) {$O$};

% 双曲线右支：参数方程 (a*cosh(t), b*sinh(t))，a=1.5, b=1.0
\draw[black, thick, domain=-2.2:2.2, smooth, samples=80]
    plot ({1.5*cosh(\x)}, {1.0*sinh(\x)});

% 双曲线左支：(-a*cosh(t), b*sinh(t))
\draw[black, thick, domain=-2.2:2.2, smooth, samples=80]
    plot ({-1.5*cosh(\x)}, {1.0*sinh(\x)});

% 渐近线 y = ±(b/a)x = ±(2/3)x
\draw[dashed, gray, thin] (-4.0, {-4.0*1.0/1.5}) -- (4.0, {4.0*1.0/1.5});
\draw[dashed, gray, thin] (-4.0, {4.0*1.0/1.5}) -- (4.0, {-4.0*1.0/1.5});

% 焦点 F1=(-c,0), F2=(c,0)，c≈1.803
\fill[black] ({-sqrt(3.25)}, 0) circle [radius=2.5pt];
\fill[black] ({sqrt(3.25)}, 0) circle [radius=2.5pt];
\node[font=\small, below left] at ({-sqrt(3.25)}, -0.05) {$F_1$};
\node[font=\small, below right] at ({sqrt(3.25)}, -0.05) {$F_2$};

% 动点 P（在右支，取参数 t=1.1）
% cosh(1.1)≈1.668, sinh(1.1)≈1.336 → P=(2.502, 1.336)
\coordinate (P) at ({1.5*cosh(1.1)}, {1.0*sinh(1.1)});
\fill[black] (P) circle [radius=2.5pt];
\node[above right, font=\small] at (P) {$P$};

% 焦半径 PF1（蓝色，较长）
\draw[blue, thick] (P) -- ({-sqrt(3.25)}, 0)
    node[midway, above, font=\footnotesize, sloped] {$|PF_1|$};

% 焦半径 PF2（红色，较短）
\draw[red, thick] (P) -- ({sqrt(3.25)}, 0)
    node[midway, above, font=\footnotesize, sloped] {$|PF_2|$};

% 定义标注
\node[font=\small, align=center] at (0, -3.8)
    {$\bigl||PF_1| - |PF_2|\bigr| = 2a$};

\end{tikzpicture}
\end{document}
