\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.0, thick]

% 直线与双曲线 4 种位置关系
% 双曲线 x^2/a^2 - y^2/b^2 = 1，a=1.2，b=0.9
% 渐近线斜率 = ±b/a = ±0.75

% 每栏宽约 4 cm，共 4 栏

% ══════════════════════════════════════════════
% 栏 1：相离（Δ < 0）
% ══════════════════════════════════════════════
\begin{scope}[xshift=0cm]
  \node[font=\small\bfseries] at (1.8, 3.6) {相离};
  \node[font=\footnotesize] at (1.8, 3.1) {$\Delta < 0$，无交点};

  % 双曲线右支：参数画法 x = a cosh(t), y = b sinh(t)
  % a=1.2, b=0.9；中心在 (1.8, 1.8)
  \def\cx{1.8}\def\cy{1.8}\def\a{1.2}\def\b{0.9}

  % 右支
  \draw[black, thick, domain=-1.3:1.3, samples=60]
    plot ({(\cx) + \a * cosh(\x)}, {(\cy) + \b * sinh(\x)});
  % 左支
  \draw[black, thick, domain=-1.3:1.3, samples=60]
    plot ({(\cx) - \a * cosh(\x)}, {(\cy) + \b * sinh(\x)});

  % 渐近线（灰色虚线）
  \draw[gray, thin, dashed] ({\cx - 2.5}, {\cy - 2.5*\b/\a}) -- ({\cx + 2.5}, {\cy + 2.5*\b/\a});
  \draw[gray, thin, dashed] ({\cx - 2.5}, {\cy + 2.5*\b/\a}) -- ({\cx + 2.5}, {\cy - 2.5*\b/\a});

  % 直线（在双曲线两支之间的"空隙"水平穿过，y 值在 |y-cy| < b*小量 范围，不与双曲线相交）
  % 选 y = cy + 0.0（x 轴方向），但 x 范围只在两支之间，整体不相交：
  % 实际取一条斜率适中但不相交的直线，设 y = 0.5(x - cx) + cy + 0.5
  % 判断：在椭圆/双曲线之外，Delta<0
  % 取水平线 y = cy + 0.5 高于两渐近线，平行于 x 轴且在两支之间"上方"区域
  \draw[blue, thick] ({(\cx) - 2.2}, {(\cy) + 2.0}) -- ({(\cx) + 2.2}, {(\cy) + 2.0});
  \node[blue, font=\small, right] at ({(\cx)+2.2}, {(\cy)+2.0}) {$l$};
\end{scope}

% 分隔线
\draw[gray, thin, dashed] (3.8, -0.2) -- (3.8, 4.0);

% ══════════════════════════════════════════════
% 栏 2：相切（Δ = 0，单交点①）
% ══════════════════════════════════════════════
\begin{scope}[xshift=4.2cm]
  \node[font=\small\bfseries] at (1.8, 3.6) {相切};
  \node[font=\footnotesize] at (1.8, 3.1) {$\Delta = 0$，切点唯一};

  \def\cx{1.8}\def\cy{1.8}\def\a{1.2}\def\b{0.9}

  % 右支
  \draw[black, thick, domain=-1.3:1.3, samples=60]
    plot ({(\cx) + \a * cosh(\x)}, {(\cy) + \b * sinh(\x)});
  % 左支
  \draw[black, thick, domain=-1.3:1.3, samples=60]
    plot ({(\cx) - \a * cosh(\x)}, {(\cy) + \b * sinh(\x)});

  % 渐近线
  \draw[gray, thin, dashed] ({\cx - 2.5}, {\cy - 2.5*\b/\a}) -- ({\cx + 2.5}, {\cy + 2.5*\b/\a});
  \draw[gray, thin, dashed] ({\cx - 2.5}, {\cy + 2.5*\b/\a}) -- ({\cx + 2.5}, {\cy - 2.5*\b/\a});

  % 切线：在右支顶点 (cx+a, cy) 处的水平切线 y = cy
  \draw[blue, thick] ({(\cx) - 0.3}, {(\cy)}) -- ({(\cx) + 2.6}, {(\cy)});
  \node[blue, font=\small, right] at ({(\cx)+2.6}, {(\cy)}) {$l$};

  % 切点
  \fill[red] ({(\cx)+\a}, {(\cy)}) circle [radius=1.8pt];
  \node[font=\footnotesize, above right] at ({(\cx)+\a}, {(\cy)}) {$T$};
\end{scope}

% 分隔线
\draw[gray, thin, dashed] (8.0, -0.2) -- (8.0, 4.0);

% ══════════════════════════════════════════════
% 栏 3：相交两支（Δ > 0，两个交点分属两支）
% ══════════════════════════════════════════════
\begin{scope}[xshift=8.4cm]
  \node[font=\small\bfseries] at (1.8, 3.6) {相交（两支）};
  \node[font=\footnotesize] at (1.8, 3.1) {$\Delta > 0$，两交点};

  \def\cx{1.8}\def\cy{1.8}\def\a{1.2}\def\b{0.9}

  % 右支
  \draw[black, thick, domain=-1.3:1.3, samples=60]
    plot ({(\cx) + \a * cosh(\x)}, {(\cy) + \b * sinh(\x)});
  % 左支
  \draw[black, thick, domain=-1.3:1.3, samples=60]
    plot ({(\cx) - \a * cosh(\x)}, {(\cy) + \b * sinh(\x)});

  % 渐近线
  \draw[gray, thin, dashed] ({\cx - 2.5}, {\cy - 2.5*\b/\a}) -- ({\cx + 2.5}, {\cy + 2.5*\b/\a});
  \draw[gray, thin, dashed] ({\cx - 2.5}, {\cy + 2.5*\b/\a}) -- ({\cx + 2.5}, {\cy - 2.5*\b/\a});

  % 直线穿过两支：y = 0（即 y = cy 偏移 = cy，这里中心在 cy=1.8，取实际 y=1.8 经过中心的水平线
  % 水平线 y=cy 与右支顶点相切；改用斜率更大的直线穿两支
  % 直线 y = 0.3(x - cx) + cy；在右支 x≈cx+a=3.0 时 y≈cy+0.36，在左支 x≈cx-a=0.6 时 y≈cy-0.36
  % 需要 Delta>0，即斜率绝对值 < b/a = 0.75，选 k=0.3
  % 交点：代入双曲线 (x-cx)^2/a^2 - (y-cy)^2/b^2=1，y-cy=0.3(x-cx)
  % (x-cx)^2*(1/a^2 - k^2/b^2) = 1
  % 1/1.44 - 0.09/0.81 = 0.6944 - 0.1111 = 0.5833
  % x-cx = ±1/sqrt(0.5833) ≈ ±1.309
  % 右侧交点：x = cx+1.309=3.109, y = cy+0.3*1.309 = cy+0.393
  % 左侧交点：x = cx-1.309=0.491, y = cy-0.393
  \coordinate (A3) at ({1.8 + 1.309}, {1.8 + 0.393});
  \coordinate (B3) at ({1.8 - 1.309}, {1.8 - 0.393});
  \draw[blue, thick] ({1.8 - 2.2}, {1.8 - 0.3*2.2}) -- ({1.8 + 2.2}, {1.8 + 0.3*2.2});
  \node[blue, font=\small, right] at ({1.8+2.2}, {1.8+0.66}) {$l$};
  \fill[black] (A3) circle [radius=1.8pt];
  \fill[black] (B3) circle [radius=1.8pt];
  \node[font=\footnotesize, above right] at (A3) {$A$};
  \node[font=\footnotesize, below left] at (B3) {$B$};
\end{scope}

% 分隔线
\draw[gray, thin, dashed] (12.2, -0.2) -- (12.2, 4.0);

% ══════════════════════════════════════════════
% 栏 4：平行渐近线（单交点②，不是切点）
% ══════════════════════════════════════════════
\begin{scope}[xshift=12.6cm]
  \node[font=\small\bfseries, align=center] at (1.8, 3.6) {平行渐近线};
  \node[font=\footnotesize, align=center] at (1.8, 3.1) {斜率 $k=\pm\frac{b}{a}$，单交点};

  \def\cx{1.8}\def\cy{1.8}\def\a{1.2}\def\b{0.9}

  % 右支
  \draw[black, thick, domain=-1.3:1.3, samples=60]
    plot ({(\cx) + \a * cosh(\x)}, {(\cy) + \b * sinh(\x)});
  % 左支
  \draw[black, thick, domain=-1.3:1.3, samples=60]
    plot ({(\cx) - \a * cosh(\x)}, {(\cy) + \b * sinh(\x)});

  % 渐近线 y-cy = (b/a)(x-cx)（灰色虚线）
  \draw[gray, thin, dashed] ({\cx - 2.5}, {\cy - 2.5*\b/\a}) -- ({\cx + 2.5}, {\cy + 2.5*\b/\a});
  \draw[gray, thin, dashed] ({\cx - 2.5}, {\cy + 2.5*\b/\a}) -- ({\cx + 2.5}, {\cy - 2.5*\b/\a});

  % 平行于渐近线的直线：斜率 = b/a = 0.75，过点 (cx-a, cy) + 右移偏移
  % 直线 y - cy = 0.75*(x - (cx + 0.0)) + 0.6（向上平移，只与右支相交一次）
  % 交点：代入双曲线，系数为 0，方程变为一次（只有一个解）
  % 取直线 y = 0.75*(x - cx) + cy + 1.2
  % 与右支交点：代入后系数为 0，解：
  % -(1/a^2)*(2*m + 2*a^2*k^2/b^2 * (—)) ... 复杂，直接估算
  % 图示意即可，取直线平移量 = +1.2 上移
  % 估算交点：y-cy = 0.75*(x-cx)+1.2，右支 x=cx+a*cosh(t)≈cx+1.5 时 y≈cy+0.375+1.2=cy+1.575
  % 设交点大约在 x ≈ cx+1.5, y ≈ cy+1.575
  \coordinate (P4) at ({1.8 + 1.5}, {1.8 + 1.575});
  \draw[blue, thick] ({\cx - 0.4}, {\cy - 0.3 + 1.2}) -- ({(\cx)+2.5}, {\cy + 0.75*2.5 + 1.2 - 0.75*0.0});
  % y = 0.75*(x-cx)+cy+1.2; at x=cx-0.4: y=cy+0.75*(-0.4)+1.2=cy+0.9; at x=cx+2.5: y=cy+0.75*2.5+1.2=cy+3.075
  % 修正：直接画精确端点
  \draw[blue, thick] ({1.8 - 0.4}, {1.8 - 0.3 + 1.2}) -- ({1.8 + 2.0}, {1.8 + 1.5 + 1.2});
  \node[blue, font=\small, right] at ({1.8+2.0}, {1.8+2.7}) {$l$};

  % 单交点（大约位置）
  \fill[red] (P4) circle [radius=1.8pt];
  \node[font=\footnotesize, right] at (P4) {$P$};

  % 注释
  \node[font=\footnotesize, align=center] at (1.8, 0.0)
    {$\Delta=0$ 时含 $k=\pm\frac{b}{a}$};
  \node[font=\footnotesize, align=center] at (1.8, -0.5)
    {此情形不算相切！};
\end{scope}

\end{tikzpicture}
