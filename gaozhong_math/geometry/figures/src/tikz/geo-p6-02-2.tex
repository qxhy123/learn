\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=0.85, thick]

% 三条双曲线对比：e=1.5, 2, 3（焦点在 x 轴）
% 统一令 a=1，则：
%   e=1.5: c=1.5, b^2=c^2-a^2=1.25, b=sqrt(1.25)≈1.118
%   e=2:   c=2,   b^2=4-1=3,        b=sqrt(3)≈1.732
%   e=3:   c=3,   b^2=9-1=8,        b=sqrt(8)≈2.828

% 坐标轴
\draw[->, thin, gray] (-4.5, 0) -- (4.5, 0) node[right, font=\small] {$x$};
\draw[->, thin, gray] (0, -4.0) -- (0, 4.0) node[above, font=\small] {$y$};
\node[font=\footnotesize, below right] at (0,0) {$O$};

%% e=1.5（最窄，黑色实线）b=sqrt(1.25)≈1.118
\draw[black, thick, domain=-1.8:1.8, smooth, samples=80]
    plot ({cosh(\x)}, {sqrt(1.25)*sinh(\x)});
\draw[black, thick, domain=-1.8:1.8, smooth, samples=80]
    plot ({-cosh(\x)}, {sqrt(1.25)*sinh(\x)});
% 渐近线
\draw[black!40, dashed, thin] (-4.2, {-4.2*sqrt(1.25)}) -- (4.2, {4.2*sqrt(1.25)});
\draw[black!40, dashed, thin] (-4.2, {4.2*sqrt(1.25)}) -- (4.2, {-4.2*sqrt(1.25)});
\node[font=\footnotesize, black] at (3.2, 1.5) {$e=1.5$};

%% e=2（中等，蓝色虚线）b=sqrt(3)≈1.732
\draw[blue, thick, dashed, domain=-1.8:1.8, smooth, samples=80]
    plot ({cosh(\x)}, {sqrt(3)*sinh(\x)});
\draw[blue, thick, dashed, domain=-1.8:1.8, smooth, samples=80]
    plot ({-cosh(\x)}, {sqrt(3)*sinh(\x)});
% 渐近线
\draw[blue!40, dashed, thin] (-2.8, {-2.8*sqrt(3)}) -- (2.8, {2.8*sqrt(3)});
\draw[blue!40, dashed, thin] (-2.8, {2.8*sqrt(3)}) -- (2.8, {-2.8*sqrt(3)});
\node[font=\footnotesize, blue] at (3.2, 2.5) {$e=2$};

%% e=3（最宽，红色点线）b=sqrt(8)≈2.828
\draw[red, thick, loosely dotted, domain=-1.8:1.8, smooth, samples=80]
    plot ({cosh(\x)}, {sqrt(8)*sinh(\x)});
\draw[red, thick, loosely dotted, domain=-1.8:1.8, smooth, samples=80]
    plot ({-cosh(\x)}, {sqrt(8)*sinh(\x)});
% 渐近线
\draw[red!40, dashed, thin] (-1.5, {-1.5*sqrt(8)}) -- (1.5, {1.5*sqrt(8)});
\draw[red!40, dashed, thin] (-1.5, {1.5*sqrt(8)}) -- (1.5, {-1.5*sqrt(8)});
\node[font=\footnotesize, red] at (3.2, 3.5) {$e=3$};

% 共同顶点
\fill[gray] (1,0) circle [radius=2pt];
\fill[gray] (-1,0) circle [radius=2pt];
\node[font=\footnotesize, below] at (1,0) {$(a,0)$};

% 说明
\node[font=\small, align=center] at (0, -4.7)
    {$e$ 越大，双曲线开口越宽（渐近线越陡）};

\end{tikzpicture}
\end{document}
