\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.0, thick]

% 三栏：圆 / 点 / 空集
% 判别式 D^2 + E^2 - 4F 的三种情形

% ── 栏 1：D²+E²-4F > 0 → 圆 ──
\begin{scope}[xshift=0cm]
  \node[font=\small\bfseries] at (1.5, 3.6) {圆};
  % 画圆
  \draw[black, thick] (1.5, 1.8) circle [radius=1.3];
  % 圆心
  \fill[black] (1.5, 1.8) circle [radius=1.5pt];
  \node[font=\footnotesize, below] at (1.5, 1.8) {圆心 $\left(-\dfrac{D}{2},\,-\dfrac{E}{2}\right)$};
  % 半径标注
  \draw[red, thick] (1.5, 1.8) -- (2.8, 1.8);
  \node[red, font=\footnotesize, above] at (2.15, 1.8) {$r$};
  % 条件
  \node[font=\footnotesize, align=center] at (1.5, 0.1)
    {$D^2+E^2-4F > 0$};
  % r 公式
  \node[font=\footnotesize, align=center] at (1.5, -0.45)
    {$r=\dfrac{1}{2}\sqrt{D^2+E^2-4F}$};
\end{scope}

% 分隔线
\draw[gray, thin, dashed] (3.5, -0.7) -- (3.5, 4.0);

% ── 栏 2：D²+E²-4F = 0 → 点（退化圆）──
\begin{scope}[xshift=4.0cm]
  \node[font=\small\bfseries] at (1.5, 3.6) {点（退化圆）};
  % 仅画一个点
  \fill[black] (1.5, 1.8) circle [radius=2.5pt];
  \node[font=\footnotesize, right] at (1.5, 1.8)
    {\quad $\left(-\dfrac{D}{2},\,-\dfrac{E}{2}\right)$};
  % 半径为 0 的说明
  \draw[red!40, dashed] (1.5, 1.8) circle [radius=0.01];
  % 条件
  \node[font=\footnotesize, align=center] at (1.5, 0.1)
    {$D^2+E^2-4F = 0$};
  \node[font=\footnotesize, align=center] at (1.5, -0.45)
    {$r = 0$（半径为零）};
\end{scope}

% 分隔线
\draw[gray, thin, dashed] (7.5, -0.7) -- (7.5, 4.0);

% ── 栏 3：D²+E²-4F < 0 → 空集 ──
\begin{scope}[xshift=8.0cm]
  \node[font=\small\bfseries] at (1.5, 3.6) {空集};
  % 画一个虚圆（表示不存在）
  \draw[gray, dashed, thick] (1.5, 1.8) circle [radius=1.3];
  % 画叉号
  \draw[red, very thick] (0.7, 1.0) -- (2.3, 2.6);
  \draw[red, very thick] (0.7, 2.6) -- (2.3, 1.0);
  % 条件
  \node[font=\footnotesize, align=center] at (1.5, 0.1)
    {$D^2+E^2-4F < 0$};
  \node[font=\footnotesize, align=center] at (1.5, -0.45)
    {无实数解，图形为空集};
\end{scope}

\end{tikzpicture}
\end{document}
