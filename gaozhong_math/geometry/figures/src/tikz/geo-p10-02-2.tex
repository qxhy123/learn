\documentclass[tikz,border=8pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[thick]

% 含参圆锥曲线分类：x^2/4 + y^2/m = 1，三种情形
% 一图三栏展示

%────────────────────────────────────
% 栏 1：m=4 → 圆 x^2+y^2=4
%────────────────────────────────────
\begin{scope}[xshift=0cm]
  % 标题框
  \node[draw=purple!60, fill=purple!8, rounded corners=3pt,
        font=\small\bfseries, inner sep=5pt, align=center] at (0,3.0)
    {$m=4$};
  % 坐标轴
  \draw[->, gray, thin] (-2.4,0) -- (2.4,0) node[right, font=\scriptsize] {$x$};
  \draw[->, gray, thin] (0,-2.4) -- (0,2.4) node[above, font=\scriptsize] {$y$};
  \node[font=\scriptsize, below left] at (0,0) {$O$};
  % 圆
  \draw[purple, very thick] (0,0) circle [radius=2.0];
  % 顶点
  \fill[purple] (2,0) circle [radius=2pt];
  \fill[purple] (-2,0) circle [radius=2pt];
  \fill[purple] (0,2) circle [radius=2pt];
  \fill[purple] (0,-2) circle [radius=2pt];
  \node[font=\scriptsize, below] at (2,0) {$2$};
  \node[font=\scriptsize, below] at (-2,0) {$-2$};
  \node[font=\scriptsize, right] at (0,2) {$2$};
  % 方程标注
  \node[font=\small, purple, align=center] at (0,-2.85)
    {$x^2+y^2=4$\\（圆，$r=2$）};
\end{scope}

%────────────────────────────────────
% 栏 2：0<m<4（取 m=1）→ 横轴椭圆
%────────────────────────────────────
\begin{scope}[xshift=5.5cm]
  \node[draw=blue!60, fill=blue!8, rounded corners=3pt,
        font=\small\bfseries, inner sep=5pt, align=center] at (0,3.0)
    {$0<m<4$（如 $m=1$）};
  \draw[->, gray, thin] (-2.4,0) -- (2.4,0) node[right, font=\scriptsize] {$x$};
  \draw[->, gray, thin] (0,-1.8) -- (0,1.8) node[above, font=\scriptsize] {$y$};
  \node[font=\scriptsize, below left] at (0,0) {$O$};
  % 椭圆 x^2/4 + y^2 = 1，a=2, b=1
  \draw[blue, very thick] (0,0) ellipse [x radius=2.0, y radius=1.0];
  % 顶点
  \fill[blue] (2,0) circle [radius=2pt];
  \fill[blue] (-2,0) circle [radius=2pt];
  \fill[blue] (0,1) circle [radius=2pt];
  \fill[blue] (0,-1) circle [radius=2pt];
  \node[font=\scriptsize, below] at (2,0) {$2$};
  \node[font=\scriptsize, below] at (-2,0) {$-2$};
  \node[font=\scriptsize, right] at (0,1) {$1$};
  % 焦点（c^2 = 4-1=3, c=√3≈1.732）
  \fill[red] (1.732,0) circle [radius=2.2pt];
  \fill[red] (-1.732,0) circle [radius=2.2pt];
  \node[font=\scriptsize, below, red] at (1.732,0) {$F_1$};
  \node[font=\scriptsize, below, red] at (-1.732,0) {$F_2$};
  % 方程标注
  \node[font=\small, blue, align=center] at (0,-2.85)
    {$\dfrac{x^2}{4}+y^2=1$\\（横轴椭圆，$a>b$）};
\end{scope}

%────────────────────────────────────
% 栏 3：m<0（取 m=-3）→ 双曲线
%────────────────────────────────────
\begin{scope}[xshift=11.5cm]
  \node[draw=orange!70, fill=orange!8, rounded corners=3pt,
        font=\small\bfseries, inner sep=5pt, align=center] at (0,3.0)
    {$m<0$（如 $m=-3$）};
  \draw[->, gray, thin] (-2.6,0) -- (2.6,0) node[right, font=\scriptsize] {$x$};
  \draw[->, gray, thin] (0,-2.4) -- (0,2.4) node[above, font=\scriptsize] {$y$};
  \node[font=\scriptsize, below left] at (0,0) {$O$};
  % 双曲线 x^2/4 - y^2/3 = 1（y=±sqrt(3*(x^2/4-1))，用 x 作参数）
  % 右支上半
  \draw[orange!80!black, very thick, domain=2.0:2.5, samples=30, smooth]
    plot (\x, {sqrt(3*(\x*\x/4-1))});
  % 右支下半
  \draw[orange!80!black, very thick, domain=2.0:2.5, samples=30, smooth]
    plot (\x, {-sqrt(3*(\x*\x/4-1))});
  % 左支上半
  \draw[orange!80!black, very thick, domain=-2.5:-2.0, samples=30, smooth]
    plot (\x, {sqrt(3*(\x*\x/4-1))});
  % 左支下半
  \draw[orange!80!black, very thick, domain=-2.5:-2.0, samples=30, smooth]
    plot (\x, {-sqrt(3*(\x*\x/4-1))});
  % 渐近线 y = ±(√3/2)x
  \draw[gray, dashed, thin] (-2.5, {-sqrt(3)/2*2.5}) -- (2.5, {sqrt(3)/2*2.5});
  \draw[gray, dashed, thin] (-2.5, {sqrt(3)/2*2.5}) -- (2.5, {-sqrt(3)/2*2.5});
  \node[font=\scriptsize, gray, right] at (2.35,2.0) {渐近线};
  % 顶点
  \fill[orange!80!black] (2,0) circle [radius=2pt];
  \fill[orange!80!black] (-2,0) circle [radius=2pt];
  \node[font=\scriptsize, below] at (2,0) {$2$};
  \node[font=\scriptsize, below] at (-2,0) {$-2$};
  % 焦点（c^2 = 4+3=7, c≈2.646）
  \fill[red] (2.646,0) circle [radius=2.2pt];
  \fill[red] (-2.646,0) circle [radius=2.2pt];
  \node[font=\scriptsize, below, red] at (2.646,0) {$F_1$};
  \node[font=\scriptsize, below, red] at (-2.646,0) {$F_2$};
  % 方程标注
  \node[font=\small, orange!80!black, align=center] at (0,-2.85)
    {$\dfrac{x^2}{4}-\dfrac{y^2}{3}=1$\\（双曲线，$x$ 轴实轴）};
\end{scope}

% 总标题
\node[font=\large\bfseries, align=center] at (5.75,4.2)
  {$\dfrac{x^2}{4}+\dfrac{y^2}{m}=1$ 随 $m$ 变化的三种形态};

% 分割线
\draw[gray!50, dashed, thin] (2.8,-3.3) -- (2.8,3.4);
\draw[gray!50, dashed, thin] (8.6,-3.3) -- (8.6,3.4);

\end{tikzpicture}
\end{document}
