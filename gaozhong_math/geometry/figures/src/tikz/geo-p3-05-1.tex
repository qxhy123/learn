\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.0, thick]

% 坐标轴
\draw[->, thin, gray] (-0.5,0) -- (5.0,0) node[right, font=\small, gray] {$x$};
\draw[->, thin, gray] (0,-0.5) -- (0,5.0) node[above, font=\small, gray] {$y$};
\node[font=\footnotesize, gray] at (-0.2,-0.2) {$O$};

% 直线 l: y=x（蓝色）
\draw[blue, thick] (-0.3,-0.3) -- (4.8,4.8);
\node[blue, font=\small] at (4.6, 5.0) {$l\colon y=x$};

% P(3,1) 和 P'(1,3)
\coordinate (P)  at (3,1);
\coordinate (Pp) at (1,3);

% 连线 PP'（黑实线）
\draw[black, thick] (P) -- (Pp);

% 中点 M（在 y=x 上，(2,2)）
\coordinate (M) at (2,2);
\filldraw[black] (M) circle (1.8pt);
\node[font=\footnotesize, above right] at (M) {$M(2,2)$};

% 垂直标记（直角小方块）
% PP' 方向：(1-3, 3-1) = (-2, 2)，单位化 (-1/√2, 1/√2)
% l 方向：(1, 1)/√2
% 在中点 M 处画直角标记，边长 0.18
\draw[gray, thin]
  ($(M) + 0.18*({-1/sqrt(2)},{1/sqrt(2)})$) --
  ($(M) + 0.18*({-1/sqrt(2)},{1/sqrt(2)}) + 0.18*({1/sqrt(2)},{1/sqrt(2)})$) --
  ($(M) + 0.18*({1/sqrt(2)},{1/sqrt(2)})$);

% 垂直符号标注
\node[font=\footnotesize, gray] at (1.55, 2.55) {$\perp$};

% 点 P
\filldraw[red, thick] (P) circle (2.2pt);
\node[font=\small, red, below right] at (P) {$P(3,1)$};

% 点 P'
\filldraw[blue, thick] (Pp) circle (2.2pt);
\node[font=\small, blue, above left] at (Pp) {$P'(1,3)$};

% 中点说明
\node[font=\footnotesize, align=left] at (3.8, 1.0) {
  中点 $M\in l$\\
  $PP'\perp l$
};

% 原点点
\filldraw[gray] (0,0) circle (1.5pt);

\end{tikzpicture}
\end{document}
