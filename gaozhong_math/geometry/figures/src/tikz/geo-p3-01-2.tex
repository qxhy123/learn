\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  scale=0.9,
  axis/.style={-{Stealth[length=6pt, width=3.5pt]}, thin},
  myline/.style={thick, black},
]

% 四个子图排成 2x2
% 子图间距：水平 5.5，垂直 4.0
% 左上 (0,4)：水平线 α=0, k=0
% 右上 (5.5,4)：锐角 0<α<π/2, k>0
% 左下 (0,0)：垂直线 α=π/2, k 不存在
% 右下 (5.5,0)：钝角 π/2<α<π, k<0

%--- 左上：水平线 α=0, k=0 ---
\begin{scope}[shift={(0,4.2)}]
  \draw[axis] (-0.3,0) -- (3.0,0) node[right] {\footnotesize $x$};
  \draw[axis] (0,-0.3) -- (0,2.0) node[above] {\footnotesize $y$};
  \node[below left, font=\footnotesize] at (0,0) {$O$};
  % 水平直线
  \draw[myline] (-0.2,1.0) -- (2.8,1.0);
  % 倾斜角标注（无弧，α=0）
  \node[red, font=\small] at (1.2, 0.6) {$\alpha = 0$};
  \node[blue, font=\small] at (1.5, 1.35) {$k = 0$};
  \node[font=\footnotesize, below] at (1.3,-0.4) {（水平线）};
\end{scope}

%--- 右上：锐角 0<α<π/2, k>0 ---
\begin{scope}[shift={(4.8,4.2)}]
  \draw[axis] (-0.3,0) -- (3.0,0) node[right] {\footnotesize $x$};
  \draw[axis] (0,-0.3) -- (0,2.2) node[above] {\footnotesize $y$};
  \node[below left, font=\footnotesize] at (0,0) {$O$};
  % 斜率为正，倾斜角约50°，过 (0.4,0)
  \draw[myline] (-0.1, {-0.839*0.5}) -- (2.8, {0.839*(2.8-0.4)});
  % 角弧
  \coordinate (Qb) at (0.4, 0);
  \draw[red, thick] (Qb) ++(0.65,0) arc[start angle=0, end angle=40, radius=0.65];
  \node[red, font=\small] at ($(Qb)+(0.9,0.22)$) {$\alpha$};
  \node[blue, font=\small] at (2.2, 1.8) {$k > 0$};
  \node[font=\footnotesize, below] at (1.3,-0.4) {（锐角）};
\end{scope}

%--- 左下：垂直线 α=π/2, k 不存在 ---
\begin{scope}[shift={(0,0)}]
  \draw[axis] (-0.3,0) -- (3.0,0) node[right] {\footnotesize $x$};
  \draw[axis] (0,-0.3) -- (0,2.2) node[above] {\footnotesize $y$};
  \node[below left, font=\footnotesize] at (0,0) {$O$};
  % 垂直线 x=1.2
  \draw[myline] (1.2,-0.2) -- (1.2,2.0);
  % 角弧（90°）
  \coordinate (Qc) at (1.2, 0);
  \draw[red, thick] (Qc) ++(0.55,0) arc[start angle=0, end angle=90, radius=0.55];
  \node[red, font=\small] at ($(Qc)+(0.18,0.72)$) {$\dfrac{\pi}{2}$};
  \node[blue, font=\small] at (2.2, 1.4) {$k$ 不存在};
  \node[font=\footnotesize, below] at (1.3,-0.4) {（垂直线）};
\end{scope}

%--- 右下：钝角 π/2<α<π, k<0 ---
\begin{scope}[shift={(4.8,0)}]
  \draw[axis] (-0.3,0) -- (3.0,0) node[right] {\footnotesize $x$};
  \draw[axis] (0,-0.3) -- (0,2.2) node[above] {\footnotesize $y$};
  \node[below left, font=\footnotesize] at (0,0) {$O$};
  % 斜率为负，倾斜角约130°，过 (2.2,0)
  % tan(130°)≈-1.19，直线 y = -1.19*(x-2.2)
  \draw[myline] (-0.1, {1.19*(2.2+0.1)}) -- (2.8, {-1.19*(2.8-2.2)});
  % 角弧（从 x 轴正方向到直线，逆时针到130°）
  \coordinate (Qd) at (2.2, 0);
  \draw[red, thick] (Qd) ++(0.6,0) arc[start angle=0, end angle=130, radius=0.6];
  \node[red, font=\small] at ($(Qd)+(-0.05,0.75)$) {$\alpha$};
  \node[blue, font=\small] at (0.5, 1.6) {$k < 0$};
  \node[font=\footnotesize, below] at (1.3,-0.4) {（钝角）};
\end{scope}

\end{tikzpicture}
\end{document}
