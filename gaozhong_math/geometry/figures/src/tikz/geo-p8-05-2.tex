\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\begin{document}
\begin{tikzpicture}[
    line cap=round, line join=round,
    every label/.style={font=\small}
]

% ============ 左：圆柱 ============
\begin{scope}[xshift=0cm]
  % 顶椭圆
  \draw[thick] (0,3.5) ellipse (1 and 0.35);
  % 底椭圆（前半实线）
  \draw[thick] (-1,0) arc (180:360:1 and 0.35);
  \draw[dashed] (-1,0) arc (180:0:1 and 0.35);
  % 母线
  \draw[thick] (-1,0) -- (-1,3.5);
  \draw[thick] (1,0) -- (1,3.5);
  % 半径/高
  \draw[red,->,thick] (0,3.5) -- (1,3.5) node[midway,above,red] {$r$};
  \draw[blue,<->,thick] (1.3,0) -- (1.3,3.5) node[midway,right,blue] {$h$};
  \node[font=\bfseries] at (0,-0.9) {圆柱};
  \node[font=\small] at (0,-1.4) {$V = \pi r^2 h$};
  \node[font=\small] at (0,-1.9) {$S_\text{侧} = 2\pi r h$};
\end{scope}

% ============ 中：圆锥 ============
\begin{scope}[xshift=5cm]
  % 顶点
  \coordinate (V) at (0,3.5);
  % 底椭圆
  \draw[thick] (-1.2,0) arc (180:360:1.2 and 0.4);
  \draw[dashed] (-1.2,0) arc (180:0:1.2 and 0.4);
  % 母线
  \draw[thick] (-1.2,0) -- (V);
  \draw[thick] (1.2,0) -- (V);
  % 高（虚线）
  \draw[blue,dashed,thick] (V) -- (0,0) node[midway,right,blue] {$h$};
  % 半径
  \draw[red,->,thick] (0,0) -- (1.2,0) node[midway,below,red] {$r$};
  % 母线标 l
  \draw[purple,thick,dotted] (V) -- (1.2,0);
  \node[purple] at (0.85,1.85) {$l$};
  \node[above] at (V) {$V$};
  \node[font=\bfseries] at (0,-0.9) {圆锥};
  \node[font=\small] at (0,-1.4) {$V = \tfrac{1}{3}\pi r^2 h$};
  \node[font=\small] at (0,-1.9) {$S_\text{侧} = \pi r l$};
\end{scope}

% ============ 右：圆台 ============
\begin{scope}[xshift=10cm]
  % 上椭圆
  \draw[thick] (0,3) ellipse (0.6 and 0.22);
  % 下椭圆
  \draw[thick] (-1.2,0) arc (180:360:1.2 and 0.4);
  \draw[dashed] (-1.2,0) arc (180:0:1.2 and 0.4);
  % 母线
  \draw[thick] (-1.2,0) -- (-0.6,3);
  \draw[thick] (1.2,0) -- (0.6,3);
  % 高
  \draw[blue,dashed,thick] (0,3) -- (0,0) node[midway,right,blue] {$h$};
  % 半径
  \draw[red,->,thick] (0,3) -- (0.6,3) node[midway,above,red] {$r_1$};
  \draw[red,->,thick] (0,0) -- (1.2,0) node[midway,below,red] {$r_2$};
  \node[font=\bfseries] at (0,-0.9) {圆台};
  \node[font=\small] at (0,-1.4) {$V = \tfrac{1}{3}\pi h(r_1^2+r_1 r_2+r_2^2)$};
\end{scope}

\end{tikzpicture}
\end{document}
