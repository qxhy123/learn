\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[
    x={(1cm,0cm)}, y={(0,1cm)}, z={(-0.5cm,-0.4cm)},
    line cap=round, line join=round,
    every label/.style={font=\small}
]

% ============ 左：三棱柱 ============
\begin{scope}[xshift=0cm]
  \coordinate (A) at (0,0,0);
  \coordinate (B) at (2,0,0);
  \coordinate (C) at (1,0,1.6);
  \coordinate (Ap) at (0,2,0);
  \coordinate (Bp) at (2,2,0);
  \coordinate (Cp) at (1,2,1.6);
  % 实线
  \draw[thick] (A) -- (B) -- (C) -- cycle;
  \draw[thick] (Ap) -- (Bp) -- (Cp) -- cycle;
  \draw[thick] (A) -- (Ap);
  \draw[thick] (B) -- (Bp);
  \draw[thick] (C) -- (Cp);
  % 标签
  \node[below left] at (A) {$A$};
  \node[below right] at (B) {$B$};
  \node[right] at (C) {$C$};
  \node[above] at (Ap) {$A'$};
  \node[above] at (Bp) {$B'$};
  \node[above] at (Cp) {$C'$};
  \node[font=\bfseries] at (1, -0.8, 0.8) {三棱柱};
  \node[font=\small] at (1, -1.4, 0.8) {$V = S_\text{底} \cdot h$};
\end{scope}

% ============ 中：四棱锥 ============
\begin{scope}[xshift=5cm]
  \coordinate (E) at (0,0,0);
  \coordinate (F) at (2,0,0);
  \coordinate (G) at (2,0,1.6);
  \coordinate (H) at (0,0,1.6);
  \coordinate (P) at (1,2.2,0.8);
  % 底面
  \draw[thick] (E) -- (F) -- (G) -- (H) -- cycle;
  % 侧棱
  \draw[thick] (E) -- (P);
  \draw[thick] (F) -- (P);
  \draw[thick] (G) -- (P);
  \draw[thick,dashed] (H) -- (P);
  % 高
  \draw[blue,dashed,thick] (P) -- (1,0,0.8) node[midway, right, blue] {$h$};
  \node[below left] at (E) {$E$};
  \node[below right] at (F) {$F$};
  \node[right] at (G) {$G$};
  \node[left] at (H) {$H$};
  \node[above] at (P) {$P$};
  \node[font=\bfseries] at (1, -0.8, 0.8) {四棱锥};
  \node[font=\small] at (1, -1.4, 0.8) {$V = \tfrac{1}{3} S_\text{底} \cdot h$};
\end{scope}

% ============ 右：球 ============
\begin{scope}[xshift=10cm]
  \coordinate (O) at (1,1,1);
  \draw[thick] (O) circle (1.0);
  % 经线
  \draw[dashed] (O) ++(0,-1) arc (-90:90:0.45 and 1.0);
  \draw[thick] (O) ++(0,-1) arc (270:90:-0.45 and 1.0);
  % 赤道
  \draw[dashed] (O) ++(-1,0) arc (180:360:1.0 and 0.35);
  \draw[thick] (O) ++(-1,0) arc (180:0:1.0 and 0.35);
  % 半径
  \draw[red,thick,->] (O) -- ++(0.7,0.7) node[midway,above left,red] {$r$};
  \node[below] at (1,-0.2,1) {球心 $O$};
  \node[font=\bfseries] at (1, -0.8, 0.8) {球};
  \node[font=\small] at (1, -1.4, 0.8) {$S = 4\pi r^2,\; V = \tfrac{4}{3}\pi r^3$};
\end{scope}

\end{tikzpicture}
\end{document}
