\documentclass[tikz,border=10pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
% 面面平行的向量判定示意图
% α ∥ β ⟺ 两面法向量 n1 ∥ n2
\begin{tikzpicture}[
  x={(1.0cm,0cm)},
  y={(0.40cm,0.30cm)},
  z={(0cm,0.90cm)},
  thick,
  dot/.style={fill=black, circle, inner sep=1.5pt},
]

% 下方平面 β
\fill[green!10, draw=green!60!black, fill opacity=0.75]
  (-2.0,0,0) -- (2.0,0,0) -- (2.8,3.0,0) -- (-1.2,3.0,0) -- cycle;
\node[green!60!black, font=\small\bfseries] at (2.5, 1.5, 0) {$\beta$};

% 上方平面 α（平行，抬高 z=1.8）
\fill[blue!10, draw=blue!50, fill opacity=0.75]
  (-2.0,0,1.8) -- (2.0,0,1.8) -- (2.8,3.0,1.8) -- (-1.2,3.0,1.8) -- cycle;
\node[blue!70, font=\small\bfseries] at (2.5, 1.5, 1.8) {$\alpha$};

% β 面内取一点 Q，画法向量 n2
\coordinate (Q) at (0.2, 1.5, 0);
\node[dot] at (Q) {};

\draw[green!60!black, very thick, -Stealth] (Q) -- (0.2, 1.5, 1.4)
  node[right, font=\small\bfseries, green!60!black] {$\vec{n}_2$};

% α 面内取一点 P，画法向量 n1
\coordinate (Pal) at (0.2, 1.5, 1.8);
\node[dot] at (Pal) {};

\draw[blue!70, very thick, -Stealth] (Pal) -- (0.2, 1.5, 3.2)
  node[right, font=\small\bfseries, blue!70] {$\vec{n}_1$};

% 连接两法向量端点（虚线，表示平行）
\draw[gray, dashed, thin] (0.2, 1.5, 1.4) -- (0.2, 1.5, 1.8);

% 平行标记（在两法向量中间）
\draw[orange, thick] (0.05,1.5,0.6) -- (0.35,1.5,0.6);
\draw[orange, thick] (0.05,1.5,0.72) -- (0.35,1.5,0.72);
\draw[orange, thick] (0.05,1.5,2.4) -- (0.35,1.5,2.4);
\draw[orange, thick] (0.05,1.5,2.52) -- (0.35,1.5,2.52);

% 公式框
\node[draw=blue!40, rounded corners, fill=blue!5, font=\small, align=center,
      inner sep=4pt] at (3.8, 4.5, 0.9)
  {$\alpha \parallel \beta$\\[2pt]
   $\Leftrightarrow\;\vec{n}_1 \parallel \vec{n}_2$\\[2pt]
   $\Leftrightarrow\;\vec{n}_1 = \lambda\vec{n}_2$};

% 两平面间连接虚线（示意平行间距）
\draw[gray, dashed, thin] (-2.0,0,0) -- (-2.0,0,1.8);
\draw[gray, dashed, thin]  (2.0,0,0) --  (2.0,0,1.8);
\draw[gray, dashed, thin] (-1.2,3.0,0) -- (-1.2,3.0,1.8);

% 底部说明
\node[font=\small, align=center] at (0.5, -0.7, 0)
  {面面平行：两平面法向量 $\vec{n}_1 \parallel \vec{n}_2$};
\node[font=\small, align=center] at (0.5, -1.05, 0)
  {$\vec{n}_1 \cdot \vec{n}_2 \neq 0$ 且 $\vec{n}_1 = \lambda\vec{n}_2$（方向平行或反向）};

\end{tikzpicture}
\end{document}
