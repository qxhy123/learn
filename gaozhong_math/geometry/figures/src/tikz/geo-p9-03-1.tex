\documentclass[tikz,border=10pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
% 长方体建系示意图：以顶点 A 为原点，沿 AB=x轴，AD=y轴，AA1=z轴
% 等距投影风格
\begin{tikzpicture}[
  x={(1.0cm,0cm)},
  y={(0.40cm,0.32cm)},
  z={(0cm,0.90cm)},
  thick,
  dot/.style={fill=black, circle, inner sep=1.5pt},
]

% 长方体参数：AB=2, AD=3, AA1=1
% 坐标：A=(0,0,0), B=(2,0,0), D=(0,3,0), A1=(0,0,1)
\coordinate (A)  at (0,0,0);
\coordinate (B)  at (2,0,0);
\coordinate (C)  at (2,3,0);
\coordinate (D)  at (0,3,0);
\coordinate (A1) at (0,0,1);
\coordinate (B1) at (2,0,1);
\coordinate (C1) at (2,3,1);
\coordinate (D1) at (0,3,1);

% 填充面（半透明）
\fill[blue!8, fill opacity=0.7] (A) -- (B) -- (C) -- (D) -- cycle;       % 底面
\fill[blue!12, fill opacity=0.6] (A1) -- (B1) -- (C1) -- (D1) -- cycle;  % 顶面
\fill[blue!10, fill opacity=0.5] (A) -- (B) -- (B1) -- (A1) -- cycle;    % 前面
\fill[blue!10, fill opacity=0.5] (B) -- (C) -- (C1) -- (B1) -- cycle;    % 右侧面
\fill[blue!6,  fill opacity=0.4] (D) -- (C) -- (C1) -- (D1) -- cycle;    % 后面
\fill[blue!6,  fill opacity=0.4] (A) -- (D) -- (D1) -- (A1) -- cycle;    % 左侧面

% 实线棱（可见）
\draw[black!70] (A) -- (B) -- (C) -- (D) -- (A) -- (A1);
\draw[black!70] (B) -- (B1) -- (A1);
\draw[black!70] (B1) -- (C1) -- (D1) -- (A1);
\draw[black!70] (C) -- (C1);

% 虚线棱（被遮挡）
\draw[black!40, dashed] (D) -- (D1);

% 坐标轴（红色箭头，延伸超出长方体）
\draw[red, very thick, -Stealth] (A) -- (2.8, 0, 0)
  node[right, font=\small\bfseries, red] {$x$};
\draw[red, very thick, -Stealth] (A) -- (0, 3.8, 0)
  node[above right, font=\small\bfseries, red] {$y$};
\draw[red, very thick, -Stealth] (A) -- (0, 0, 1.6)
  node[above, font=\small\bfseries, red] {$z$};

% 原点标记
\node[dot] at (A) {};
\node[font=\small\bfseries, below left] at (A) {$A(0,0,0)$};
\node[font=\small, align=left] at ($(A)+(-0.3,-0.18)$) {$O$};

% 顶点标记 + 坐标
\node[dot] at (B)  {};
\node[font=\small, below right] at (B)  {$B(2,0,0)$};

\node[dot] at (D)  {};
\node[font=\small, above left] at (D)  {$D(0,3,0)$};

\node[dot] at (C)  {};
\node[font=\small, right] at (C)  {$C(2,3,0)$};

\node[dot] at (A1) {};
\node[font=\small, left] at (A1) {$A_1(0,0,1)$};

\node[dot] at (B1) {};
\node[font=\small, right] at (B1) {$B_1(2,0,1)$};

\node[dot] at (D1) {};
\node[font=\small, above left] at (D1) {$D_1(0,3,1)$};

\node[dot] at (C1) {};
\node[font=\small, above right] at (C1) {$C_1(2,3,1)$};

% 底部说明
\node[font=\small, align=center] at (1.0, -0.7, 0)
  {长方体 $ABCD$-$A_1B_1C_1D_1$（$AB=2,\,AD=3,\,AA_1=1$）};
\node[font=\small, align=center] at (1.0, -1.1, 0)
  {以 $A$ 为原点建立空间直角坐标系};

\end{tikzpicture}
\end{document}
