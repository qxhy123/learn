\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  x={(1.8cm,0cm)},
  y={(0cm,1.8cm)},
  z={(-0.52cm,-0.42cm)},
  line cap=round,
  vec/.style={-{Stealth[length=7pt, width=4pt]}, thick},
  vecred/.style={-{Stealth[length=7pt, width=4pt]}, thick, red},
  vecblue/.style={-{Stealth[length=7pt, width=4pt]}, thick, blue},
  vecgreen/.style={-{Stealth[length=7pt, width=4pt]}, thick, teal},
]

% 平行六面体法则：三条共顶点向量 a,b,c 相加
% 原点 O，三条棱方向分别是 x,y,z

\coordinate (O)  at (0,0,0);
\coordinate (A)  at (1,0,0);   % 沿 x 方向：\vec{a}
\coordinate (B)  at (0,1,0);   % 沿 y 方向：\vec{b}
\coordinate (C)  at (0,0,1);   % 沿 z 方向：\vec{c}

% 六面体其余顶点
\coordinate (AB)  at (1,1,0);
\coordinate (AC)  at (1,0,1);
\coordinate (BC)  at (0,1,1);
\coordinate (ABC) at (1,1,1);  % 对角顶点

% 辅助棱（灰色虚线，背景层）
\draw[gray, dashed] (A)  -- (AB);
\draw[gray, dashed] (A)  -- (AC);
\draw[gray, dashed] (B)  -- (AB);
\draw[gray, dashed] (B)  -- (BC);
\draw[gray, dashed] (C)  -- (AC);
\draw[gray, dashed] (C)  -- (BC);
\draw[gray, dashed] (AB) -- (ABC);
\draw[gray, dashed] (AC) -- (ABC);
\draw[gray, dashed] (BC) -- (ABC);

% 三条主向量：\vec{a}, \vec{b}, \vec{c}
\draw[vec, black] (O) -- (A);
\node[below right] at ($(O)!0.6!(A)$) {$\vec{a}$};

\draw[vecblue] (O) -- (B);
\node[above left] at ($(O)!0.6!(B)$) {$\vec{b}$};

\draw[vecgreen] (O) -- (C);
\node[below left] at ($(O)!0.6!(C)$) {$\vec{c}$};

% 和向量：从 O 沿对角线到 ABC（红色）
\draw[vecred] (O) -- (ABC);
\node[above right] at ($(O)!0.55!(ABC)$) {$\vec{a}+\vec{b}+\vec{c}$};

% 标注原点
\node[below left] at (O) {$O$};

% 底部说明
\node[below] at (0.5,-0.15,0) {\small 平行六面体法则：三条共顶点向量之和沿对角线};

\end{tikzpicture}
\end{document}
