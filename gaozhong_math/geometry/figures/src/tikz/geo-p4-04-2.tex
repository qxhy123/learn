\documentclass[tikz,border=8pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=0.85, thick]

% 坐标轴
\draw[->, thin, gray] (-4.5,0) -- (10.5,0) node[right, font=\small, gray] {$x$};
\draw[->, thin, gray] (0,-4.5) -- (0,4.5) node[above, font=\small, gray] {$y$};
\node[font=\footnotesize, gray] at (-0.3,-0.25) {$O$};

% 阿波罗尼斯圆：圆心 C(5,0)，半径 4
\draw[black, thick] (5,0) circle [radius=4];

% 圆心
\coordinate (Cc) at (5,0);
\filldraw[black] (Cc) circle (2pt);
\node[font=\small, below] at (Cc) {$C(5,0)$};

% 两定点 A(-3,0) 和 B(3,0)
\coordinate (A) at (-3,0);
\coordinate (B) at (3,0);

\filldraw[black] (A) circle (2.2pt);
\node[font=\small, below left] at (A) {$A(-3,0)$};

\filldraw[black] (B) circle (2.2pt);
\node[font=\small, below right] at (B) {$B(3,0)$};

% 动点 P（在圆上，取上方一点）
% 取 P = C + 4*(cos 110°, sin 110°) ≈ (5+4*(-0.342), 4*0.940) = (3.633, 3.759)
\coordinate (P) at ({5+4*cos(110)}, {4*sin(110)});
\filldraw[red, thick] (P) circle (2.2pt);
\node[red, font=\small, above] at (P) {$P$};

% PA, PB 辅助线
\draw[gray, dashed, thin] (P) -- (A);
\draw[gray, dashed, thin] (P) -- (B);

% 标 |PA|, |PB|
\node[font=\footnotesize, gray] at ($(P)!0.5!(A) + (-0.3,0.15)$) {$|PA|$};
\node[font=\footnotesize, gray] at ($(P)!0.5!(B) + (0.4,0.1)$) {$|PB|$};

% 公式及说明
\node[font=\small, align=center] at (5, -3.3) {
  轨迹圆：圆心 $C(5,0)$，半径 $r=4$
};
\node[font=\small] at (5, 4.8) {$\dfrac{|PA|}{|PB|}=2$};

\end{tikzpicture}
\end{document}
