\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.1, thick]

% 椭圆：a=3, b=2，焦点在 y 轴
% 方程：x^2/b^2 + y^2/a^2 = 1，即 x^2/4 + y^2/9 = 1
% 半长轴在 y 方向，a=3；半短轴在 x 方向，b=2
% c=sqrt(a^2-b^2)=sqrt(5)≈2.236，焦点：(0,±sqrt(5))

% 坐标轴
\draw[->, thin, gray] (-3.2, 0) -- (3.2, 0) node[right, font=\small] {$x$};
\draw[->, thin, gray] (0, -4.0) -- (0, 4.0) node[above, font=\small] {$y$};
\node[font=\footnotesize, below right] at (0,0) {$O$};

% 椭圆（x 方向短轴 b=2，y 方向长轴 a=3）
\draw[black, thick] (0,0) ellipse (2 and 3);

% 四个顶点
\fill[black] (-2, 0) circle [radius=2pt];
\fill[black] (2, 0) circle [radius=2pt];
\fill[black] (0, 3) circle [radius=2pt];
\fill[black] (0, -3) circle [radius=2pt];
\node[below left, font=\small] at (-2, 0) {$(-b,0)$};
\node[below right, font=\small] at (2, 0) {$(b,0)$};
\node[right, font=\small] at (0.1, 3) {$(0,a)$};
\node[right, font=\small] at (0.1, -3) {$(0,-a)$};

% 焦点（在 y 轴上）
\fill[black] (0, {sqrt(5)}) circle [radius=2pt];
\fill[black] (0, {-sqrt(5)}) circle [radius=2pt];
\node[right, font=\small] at (0.1, {sqrt(5)}) {$F_2(0,c)$};
\node[right, font=\small] at (0.1, {-sqrt(5)}) {$F_1(0,-c)$};

% 长半轴标注 a（红色，沿 y 轴）
\draw[red, thick, <->] (0.35, 0) -- (0.35, 3);
\node[red, font=\footnotesize, right] at (0.35, 1.5) {$a$};

% 短半轴标注 b（红色，沿 x 轴）
\draw[red, thick, <->] (0, -0.35) -- (2, -0.35);
\node[red, font=\footnotesize, below] at (1.0, -0.35) {$b$};

% 标注说明
\node[font=\small, align=center] at (0, -4.5)
  {焦点在 $y$ 轴：$\dfrac{x^2}{b^2}+\dfrac{y^2}{a^2}=1$};

\end{tikzpicture}
\end{document}
