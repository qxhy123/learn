\documentclass[tikz,border=8pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=0.9, thick]

% 坐标轴
\draw[->, thin, gray] (-3.8,0) -- (3.8,0) node[right, font=\small, gray] {$x$};
\draw[->, thin, gray] (0,-2.5) -- (0,2.5) node[above, font=\small, gray] {$y$};
\node[font=\footnotesize, gray] at (-0.25,-0.2) {$O$};

% 椭圆 x^2/9 + y^2/4 = 1，a=3, b=2
% c = sqrt(9-4) = sqrt(5) ≈ 2.236
\draw[black, thick] (0,0) ellipse [x radius=3, y radius=2];

% 焦点 F1(-√5,0) 和 F2(√5,0)
\coordinate (F1) at ({-sqrt(5)},0);
\coordinate (F2) at ({sqrt(5)},0);

\filldraw[black] (F1) circle (2pt);
\node[font=\small, below left] at (F1) {$F_1$};

\filldraw[black] (F2) circle (2pt);
\node[font=\small, below right] at (F2) {$F_2$};

% 椭圆上点 P（取上方偏右，参数角 70°）
% P = (3cos70°, 2sin70°) ≈ (1.026, 1.879)
\coordinate (P) at ({3*cos(70)}, {2*sin(70)});
\filldraw[black] (P) circle (2pt);
\node[font=\small, above right] at (P) {$P$};

% 光线：F1 -> P（红色入射）
\draw[red, thick, -{Stealth[length=6pt,width=3.5pt]}]
  (F1) -- (P);

% 光线：P -> F2（红色反射）
\draw[red, thick, -{Stealth[length=6pt,width=3.5pt]}]
  (P) -- (F2);

% 椭圆在 P 点的切线（用法线辅助）
% 椭圆法线方向：(x/a^2, y/b^2) = (cos70/3, sin70/2)
% 切线方向（垂直于法线）
% 法线方向向量：(3cos70/9, 2sin70/4) = (cos70/3, sin70/2)
% 简化：法线方向 (b^2*x, a^2*y) = (4*P.x, 9*P.y)
\pgfmathsetmacro{\px}{3*cos(70)}
\pgfmathsetmacro{\py}{2*sin(70)}
% 切线方向向量（垂直法向量）：(-9*py, 4*px)
\pgfmathsetmacro{\tx}{-9*\py}
\pgfmathsetmacro{\ty}{4*\px}
\pgfmathsetmacro{\tlen}{sqrt(\tx*\tx+\ty*\ty)}
% 切线长度 1.4，单位向量
\pgfmathsetmacro{\tux}{\tx/\tlen}
\pgfmathsetmacro{\tuy}{\ty/\tlen}
\draw[blue, dashed, thick]
  ($(\px,\py) - 1.4*(\tux,\tuy)$) -- ($(\px,\py) + 1.4*(\tux,\tuy)$);
\node[blue, font=\footnotesize] at ({3*cos(70)+1.6}, {2*sin(70)+0.3}) {切线};

% 入射角和反射角标记（在 P 处用小弧）
% 法线方向（指向外侧）：(4*px, 9*py) 单位化
\pgfmathsetmacro{\nx}{4*\px}
\pgfmathsetmacro{\ny}{9*\py}
\pgfmathsetmacro{\nlen}{sqrt(\nx*\nx+\ny*\ny)}
\pgfmathsetmacro{\nux}{\nx/\nlen}
\pgfmathsetmacro{\nuy}{\ny/\nlen}

% 标两角相等符号（文字说明）
\node[font=\footnotesize, align=left] at (-2.8, -1.8) {
  入射角 $=$ 反射角
};

% 标 a 和 c
\node[font=\footnotesize, gray] at (0, -0.25) {$2c$};
\draw[gray, thin, <->] (F1) -- (F2);

\end{tikzpicture}
\end{document}
