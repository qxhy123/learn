\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, angles, quotes, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  scale=1.2, thick,
  vec/.style={-{Stealth[length=7pt, width=4pt]}, thick},
]

% 原点
\coordinate (O) at (0, 0);

% 向量 a：水平向右，长度 3.5
\coordinate (A) at (3.5, 0);

% 向量 b：与 a 成 θ 角，长度 2.5
\coordinate (B) at ({2.5*cos(50)}, {2.5*sin(50)});

% 画向量
\draw[vec, black, very thick] (O) -- (A) node[right] {$\vec{a}$};
\draw[vec, red, very thick]   (O) -- (B) node[above right] {$\vec{b}$};

% 角度弧 θ
\draw[black, thin] (0.6, 0) arc[start angle=0, end angle=50, radius=0.6];
\node[font=\small] at (0.82, 0.25) {$\theta$};

% 投影：b 在 a 上的投影（b 的终点向 a 上作垂线）
\coordinate (Bproj) at ({2.5*cos(50)}, 0);
\draw[dashed, gray, thick] (B) -- (Bproj);

% 直角标记
\draw[gray, thin]
  ($(Bproj) + (0.12, 0)$) --
  ($(Bproj) + (0.12, 0.12)$) --
  ($(Bproj) + (0, 0.12)$);

% 投影线段标注（红色双箭头）
\draw[{Stealth[length=5pt]}-{Stealth[length=5pt]}, red, thin]
  (0, -0.45) -- ({2.5*cos(50)}, -0.45)
  node[midway, below, font=\small] {$|\vec{b}|\cos\theta$};

% 原点标记
\filldraw[black] (O) circle (1.5pt) node[below left] {$O$};
\filldraw[black] (A) circle (1.2pt);
\filldraw[black] (B) circle (1.2pt);

% 数量积公式说明
\node[font=\small, align=center] at (1.7, -1.1)
  {$\vec{a}\cdot\vec{b} = |\vec{a}||\vec{b}|\cos\theta$\\[2pt]
   $=|\vec{a}|\times(\vec{b}\text{ 在 }\vec{a}\text{ 上的投影})$};

\end{tikzpicture}
\end{document}
