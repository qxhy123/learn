\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, angles, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  scale=1.0, thick,
  axis/.style={-{Stealth[length=7pt, width=4pt]}, thin},
  pt/.style={circle, fill=black, inner sep=1.5pt},
]

% 坐标轴
\draw[axis] (-0.5,0) -- (5.5,0) node[right] {$x$};
\draw[axis] (0,-0.5) -- (0,4.0) node[above] {$y$};
\node[below left] at (0,0) {$O$};

% 直线：斜率 k = 0.75，过定点 P(1.5, 1.2)
% y - 1.2 = 0.75*(x - 1.5)  =>  y = 0.75*x + 0.075
\coordinate (LStart) at (-0.2, {0.75*(-0.2)+0.075});
\coordinate (LEnd) at (5.2, {0.75*5.2+0.075});
\draw[thick, black] (LStart) -- (LEnd);

% 定点 P(x_0, y_0) = (1.5, 1.2)
\coordinate (P) at (1.5, 1.2);
\node[pt] at (P) {};
\node[left] at (P) {$P(x_0,\, y_0)$};

% 动点 (x, y) 在直线上，取 (3.8, {0.75*3.8+0.075})
\coordinate (M) at (3.8, {0.75*3.8+0.075});
\node[pt] at (M) {};
\node[right] at (M) {$(x,\, y)$};

% 辅助虚线：水平和垂直分量
\draw[dashed, gray] (P) -- ($(M|-P)$) node[midway, below] {\footnotesize $x - x_0$};
\draw[dashed, gray] ($(M|-P)$) -- (M) node[midway, right] {\footnotesize $y - y_0$};

% 斜率 k 标注（在直线旁）
\node[above left] at ($(LStart)!0.55!(LEnd)$) {斜率 $k$};

% 公式框
\node[draw=blue!70, rounded corners=3pt, fill=blue!5, inner sep=4pt]
  at (4.0, 0.8)
  {$y - y_0 = k(x - x_0)$};

\end{tikzpicture}
\end{document}
