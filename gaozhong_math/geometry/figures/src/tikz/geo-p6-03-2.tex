\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.2, thick]

% 双曲线焦点弦示意图
% 双曲线 x^2/a^2 - y^2/b^2 = 1，a=2, b=1.5
% c = sqrt(a^2+b^2) = sqrt(4+2.25) = sqrt(6.25) = 2.5
% 焦点 F1(-2.5,0), F2(2.5,0)
% 渐近线 y = ±(b/a)x = ±0.75x

\def\a{2.0}
\def\b{1.5}
\def\c{2.5}

% 坐标轴
\draw[->, gray, thin] (-3.6, 0) -- (3.8, 0) node[right, font=\footnotesize] {$x$};
\draw[->, gray, thin] (0, -2.8) -- (0, 2.8) node[above, font=\footnotesize] {$y$};
\node[font=\footnotesize, below right] at (0,0) {$O$};

% 渐近线 y = ±0.75x（灰色虚线）
\draw[gray, thin, dashed] (-3.4, {-3.4*\b/\a}) -- (3.4, {3.4*\b/\a})
  node[right, font=\footnotesize, gray] {$y=\frac{b}{a}x$};
\draw[gray, thin, dashed] (-3.4, {3.4*\b/\a}) -- (3.4, {-3.4*\b/\a})
  node[right, font=\footnotesize, gray] {$y=-\frac{b}{a}x$};

% 双曲线右支
\draw[black, thick, domain=-1.4:1.4, samples=80]
  plot ({\a * cosh(\x)}, {\b * sinh(\x)});

% 双曲线左支
\draw[black, thick, domain=-1.4:1.4, samples=80]
  plot ({-\a * cosh(\x)}, {\b * sinh(\x)});

% 顶点
\fill[gray] (\a, 0) circle [radius=1.5pt];
\fill[gray] (-\a, 0) circle [radius=1.5pt];
\node[font=\footnotesize, below right] at (\a, 0) {$a$};
\node[font=\footnotesize, below left] at (-\a, 0) {$-a$};

% 焦点 F1, F2
\coordinate (F1) at (-\c, 0);
\coordinate (F2) at (\c, 0);
\fill[black] (F1) circle [radius=2pt];
\fill[black] (F2) circle [radius=2pt];
\node[font=\footnotesize, below] at (F1) {$F_1(-c,0)$};
\node[font=\footnotesize, below] at (F2) {$F_2(c,0)$};

% 焦点弦：过 F2(2.5,0)，斜率 k=1.0
% 代入双曲线：(x^2/4) - (x-2.5)^2/2.25 = 1
% 令 t = x：0.25*t^2 - (t-2.5)^2/2.25 = 1
% 乘以 9：2.25 t^2 - 4(t-2.5)^2 = 9
% 2.25 t^2 - 4(t^2 - 5t + 6.25) = 9
% 2.25 t^2 - 4t^2 + 20t - 25 = 9
% -1.75 t^2 + 20t - 34 = 0
% 1.75 t^2 - 20t + 34 = 0
% t = (20 ± sqrt(400 - 4*1.75*34))/(2*1.75) = (20 ± sqrt(400-238))/3.5 = (20 ± sqrt(162))/3.5
% sqrt(162) ≈ 12.728
% t1 = (20+12.728)/3.5 ≈ 9.351  (很大，右支远端)
% t2 = (20-12.728)/3.5 ≈ 2.078  (右支近端)
% 斜率 k=1 时两交点都在右支，不是典型焦点弦（跨两支）
%
% 改用斜率 k=0.5（< b/a = 0.75，穿两支）：
% 直线 y = 0.5*(x - 2.5)，代入 x^2/4 - y^2/2.25 = 1
% x^2/4 - 0.25*(x-2.5)^2/2.25 = 1
% 乘以 9：2.25x^2 - (x-2.5)^2 = 9
% 2.25x^2 - x^2 + 5x - 6.25 = 9
% 1.25x^2 + 5x - 15.25 = 0
% x = (-5 ± sqrt(25 + 4*1.25*15.25))/(2*1.25) = (-5 ± sqrt(25+76.25))/2.5 = (-5 ± sqrt(101.25))/2.5
% sqrt(101.25) ≈ 10.062
% x1 = (-5+10.062)/2.5 ≈ 2.025  (右支)
% x2 = (-5-10.062)/2.5 ≈ -6.025  (左支)
% y1 = 0.5*(2.025-2.5) = -0.2375
% y2 = 0.5*(-6.025-2.5) = -4.2625  (超出图范围，调整斜率)
%
% 改用更小斜率，或者选同支焦点弦（同一右支，k 较大）
% 取 k=2.0（> b/a=0.75），过 F2，两点都在右支：
% y = 2(x-2.5)
% x^2/4 - 4(x-2.5)^2/2.25 = 1
% 乘以 9：2.25x^2 - 16(x-2.5)^2 = 9
% 2.25x^2 - 16x^2 + 80x - 100 = 9
% -13.75x^2 + 80x - 109 = 0
% 13.75x^2 - 80x + 109 = 0
% x = (80 ± sqrt(6400 - 4*13.75*109))/(2*13.75)
%   = (80 ± sqrt(6400 - 5995))/27.5 = (80 ± sqrt(405))/27.5
% sqrt(405) ≈ 20.125
% x1 = (80+20.125)/27.5 ≈ 3.641, y1 = 2*(3.641-2.5) = 2.282
% x2 = (80-20.125)/27.5 ≈ 2.177, y2 = 2*(2.177-2.5) = -0.646

\coordinate (A) at (3.641, 2.282);
\coordinate (B) at (2.177, -0.646);

% 焦点弦 AB（红色）
\draw[red, thick] (A) -- (B);
\fill[black] (A) circle [radius=2pt];
\fill[black] (B) circle [radius=2pt];
\node[font=\footnotesize, above right] at (A) {$A(x_1,y_1)$};
\node[font=\footnotesize, below right] at (B) {$B(x_2,y_2)$};

% 焦半径：F2→A, F2→B（灰色虚线）
\draw[gray, dashed] (F2) -- (A);
\draw[gray, dashed] (F2) -- (B);

% 焦半径：F1→A, F1→B（灰色虚线）
\draw[gray, dashed] (F1) -- (A);
\draw[gray, dashed] (F1) -- (B);

% 标注 |AF2|, |BF2|
\node[red!70, font=\footnotesize] at (3.4, 0.8) {$|AF_2|$};
\node[red!70, font=\footnotesize] at (2.8, -0.5) {$|BF_2|$};

% 焦点弦公式注释（图下方）
\node[font=\small, align=center] at (0, -2.5)
  {右支焦点弦（过 $F_2$，两端在同一支）};
\node[font=\footnotesize, align=center] at (0, -3.0)
  {焦半径：$|PF_2| = |ex_P - a|$，\ $|PF_1| = ex_P + a$};

\end{tikzpicture}
\end{document}
