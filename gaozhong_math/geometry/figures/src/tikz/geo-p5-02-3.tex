\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.1, thick]

% 焦半径示意图：椭圆 + P点 + 两条焦半径
% a=3, b=2, c=sqrt(5)≈2.236

% 坐标轴
\draw[->, thin, gray] (-3.8, 0) -- (3.8, 0) node[right, font=\small] {$x$};
\draw[->, thin, gray] (0, -2.6) -- (0, 2.6) node[above, font=\small] {$y$};
\node[font=\footnotesize, below right] at (0,0) {$O$};

% 椭圆
\draw[black, thick] (0,0) ellipse (3 and 2);

% 焦点
\fill[black] ({-sqrt(5)}, 0) circle [radius=2pt];
\fill[black] ({sqrt(5)}, 0) circle [radius=2pt];
\node[below, font=\small] at ({-sqrt(5)}, 0) {$F_1$};
\node[below, font=\small] at ({sqrt(5)}, 0) {$F_2$};

% 动点 P（第一象限，参数角约65°）
\coordinate (P) at ({3*cos(65)}, {2*sin(65)});
\fill[black] (P) circle [radius=2pt];
\node[above right, font=\small] at (P) {$P(x_0,\,y_0)$};

% 焦半径 r1 = |PF1|（红色）
\draw[red, thick] (P) -- ({-sqrt(5)}, 0);
\node[red, font=\footnotesize, above] at ({(-sqrt(5)+3*cos(65))/2 - 0.2}, {2*sin(65)/2 + 0.1})
  {$r_1=|PF_1|=a+ex_0$};

% 焦半径 r2 = |PF2|（蓝色虚线）
\draw[blue, dashed, thick] (P) -- ({sqrt(5)}, 0);
\node[blue, font=\footnotesize, right] at ({(sqrt(5)+3*cos(65))/2 + 0.1}, {2*sin(65)/2})
  {$r_2=|PF_2|=a-ex_0$};

% 辅助：P 到 x 轴的垂线（灰色）
\draw[gray, thin, dashed] (P) -- ({3*cos(65)}, 0);
\fill[gray] ({3*cos(65)}, 0) circle [radius=1.5pt];
\node[below, font=\footnotesize, gray] at ({3*cos(65)}, 0) {$x_0$};

% 底部公式
\node[font=\small, align=center] at (0, -3.2)
  {$r_1+r_2=2a$ \quad $r_1-r_2=2ex_0$（$x_0>0$）};

\end{tikzpicture}
\end{document}
