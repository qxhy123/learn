\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  scale=1.0, thick,
  vec/.style={-{Stealth[length=7pt, width=4pt]}, thick},
  vecblue/.style={-{Stealth[length=7pt, width=4pt]}, thick, blue, dashed},
  vecred/.style={-{Stealth[length=7pt, width=4pt]}, thick, red},
]

%----------------------------------------------
% 左图：平行四边形法则  \vec{AB} + \vec{AD} = \vec{AC}
%----------------------------------------------

\coordinate (O_par) at (0,0);  % O as origin for left figure
% A at bottom-left, B at bottom-right, D at top-left, C at top-right
\coordinate (A) at (0.3, 0.5);
\coordinate (B) at (3.3, 0.5);
\coordinate (D) at (1.3, 2.5);
\coordinate (C) at (4.3, 2.5);   % C = B + (D - A)

% 平行四边形四边（辅助线灰色虚线）
\draw[gray, dashed] (B) -- (C);
\draw[gray, dashed] (D) -- (C);

% \vec{AB}
\draw[vec, black] (A) -- (B);
\node[below] at ($(A)!0.5!(B)$) {$\vec{AB}$};

% \vec{AD}
\draw[vec, black] (A) -- (D);
\node[left] at ($(A)!0.5!(D)$) {$\vec{AD}$};

% \vec{AC}（对角线，和向量，红色）
\draw[vecred] (A) -- (C);
\node[right] at ($(A)!0.6!(C)$) {$\vec{AC}$};

% 顶点
\node[below left] at (A) {$A$};
\node[below right] at (B) {$B$};
\node[above left] at (D) {$D$};
\node[above right] at (C) {$C$};

% 公式
\node[below] at (2.3, -0.2) {\small $\vec{AB} + \vec{AD} = \vec{AC}$};
\node[below] at (2.3, -0.75) {\small（平行四边形法则）};

%----------------------------------------------
% 右图：减法几何意义 \vec{OA} - \vec{OB} = \vec{BA}
%----------------------------------------------
\begin{scope}[xshift=6.5cm]

\coordinate (Orig) at (0, 0);
\coordinate (Av) at (3.5, 1.5);
\coordinate (Bv) at (1.5, 2.5);

% \vec{OA}（黑色）
\draw[vec, black] (Orig) -- (Av);
\node[below right] at ($(Orig)!0.55!(Av)$) {$\vec{OA}$};

% \vec{OB}（蓝色虚线）
\draw[vecblue] (Orig) -- (Bv);
\node[above left] at ($(Orig)!0.55!(Bv)$) {$\vec{OB}$};

% \vec{BA} = \vec{OA} - \vec{OB}（红色）
\draw[vecred] (Bv) -- (Av);
\node[right] at ($(Bv)!0.5!(Av)$) {$\vec{BA} = \vec{OA} - \vec{OB}$};

% 顶点
\node[below left] at (Orig) {$O$};
\node[below right] at (Av) {$A$};
\node[above left] at (Bv) {$B$};

% 公式
\node[below] at (1.8, -0.2) {\small 减法：$\vec{OA} - \vec{OB} = \vec{BA}$};
\node[below] at (1.8, -0.75) {\small（从减数终点指向被减数终点）};

\end{scope}

\end{tikzpicture}
\end{document}
