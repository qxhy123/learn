\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  scale=1.0, thick,
  vec/.style={-{Stealth[length=7pt, width=4pt]}, thick},
  vecblue/.style={-{Stealth[length=7pt, width=4pt]}, thick, blue, dashed},
  vecred/.style={-{Stealth[length=7pt, width=4pt]}, thick, red},
]

%----------------------------------------------
% 上半部分：相等向量示意（不同位置，同模同向）
%----------------------------------------------
\node[anchor=west] at (-0.5, 3.2) {\textbf{相等向量}（模相同、方向相同，与起点无关）};

% 向量 1：\vec{a}，位置偏左
\coordinate (P1) at (0, 2.2);
\coordinate (P2) at (2.5, 2.2);
\draw[vec, black] (P1) -- (P2);
\node[below] at ($(P1)!0.5!(P2)$) {\small $\vec{a}$};

% 向量 2：同模同向，位置偏右下
\coordinate (Q1) at (3.2, 1.5);
\coordinate (Q2) at (5.7, 1.5);
\draw[vecblue] (Q1) -- (Q2);
\node[below] at ($(Q1)!0.5!(Q2)$) {\small $\vec{b}$};

% 向量 3：再换一个位置
\coordinate (R1) at (1.0, 0.7);
\coordinate (R2) at (3.5, 0.7);
\draw[vecblue] (R1) -- (R2);
\node[below] at ($(R1)!0.5!(R2)$) {\small $\vec{c}$};

% 说明：三者相等
\node[right] at (6.0, 1.5) {\small $\vec{a} = \vec{b} = \vec{c}$};

% 分隔线
\draw[gray, thin, dashed] (-0.5, 0.1) -- (8.5, 0.1);

%----------------------------------------------
% 下半部分：共线向量示意（方向相同或相反）
%----------------------------------------------
\node[anchor=west] at (-0.5, -0.3) {\textbf{共线向量}（方向相同或相反）};

% 同向共线
\coordinate (S1) at (0.2, -1.0);
\coordinate (S2) at (2.2, -1.0);
\draw[vec, black] (S1) -- (S2);
\node[below left] at (S1) {\small $\vec{m}$};

\coordinate (T1) at (3.0, -1.0);
\coordinate (T2) at (5.0, -1.0);
\draw[vec, black] (T1) -- (T2);
\node[below left] at (T1) {\small $\vec{n}$};

\node[right] at (5.2, -1.0) {\small 同向（$\vec{m} \parallel \vec{n}$）};

% 反向共线
\coordinate (U1) at (0.2, -2.0);
\coordinate (U2) at (2.2, -2.0);
\draw[vec, black] (U1) -- (U2);
\node[below left] at (U1) {\small $\vec{p}$};

\coordinate (V1) at (5.0, -2.0);
\coordinate (V2) at (3.0, -2.0);
\draw[vecred] (V1) -- (V2);
\node[below right] at (V1) {\small $-\vec{p}$};

\node[right] at (5.2, -2.0) {\small 反向（$\vec{p}$ 与 $-\vec{p}$ 共线）};

% 零向量说明
\node[anchor=west] at (-0.5, -2.8) {\small 注：零向量 $\vec{0}$ 与任意向量共线。};

\end{tikzpicture}
\end{document}
