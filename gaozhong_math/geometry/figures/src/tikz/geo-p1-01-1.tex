\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta, decorations.markings}

\begin{document}
\begin{tikzpicture}[
  scale=1.0, thick,
  vec/.style={-{Stealth[length=7pt, width=4pt]}, thick},
]

% 有向线段表示向量 \vec{AB}
% 起点 A，终点 B
\coordinate (A) at (0,0);
\coordinate (B) at (4,0);

% 画向量箭头
\draw[vec, black] (A) -- (B);

% 标注起点和终点
\node[below left] at (A) {$A$（起点）};
\node[below right] at (B) {$B$（终点）};

% 向量名称标注在线段上方
\node[above] at ($(A)!0.5!(B)$) {$\vec{AB}$};

% 模长标注（用双箭头在线段下方）
\draw[{Stealth[length=5pt]}-{Stealth[length=5pt]}, gray, thin]
    (0,-0.55) -- (4,-0.55)
    node[midway, below, black] {\footnotesize $|\vec{AB}|$（模）};

% 小竖线标记端点对齐
\draw[gray, thin] (0,-0.35) -- (0,-0.75);
\draw[gray, thin] (4,-0.35) -- (4,-0.75);

% 说明文字
\node[below] at (2,-1.1) {\small 有向线段 $\vec{AB}$：方向由 $A$ 指向 $B$，长度为模 $|\vec{AB}|$};

\end{tikzpicture}
\end{document}
