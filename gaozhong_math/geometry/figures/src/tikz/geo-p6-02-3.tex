\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.0, thick]

% 等轴双曲线：a=b=2，方程 x^2/4 - y^2/4 = 1，即 x^2-y^2=4
% c=sqrt(a^2+b^2)=sqrt(8)=2sqrt(2)≈2.828
% 离心率 e=c/a=sqrt(2)≈1.414
% 渐近线 y=±x（斜率±1）

% 坐标轴
\draw[->, thin, gray] (-4.5, 0) -- (4.5, 0) node[right, font=\small] {$x$};
\draw[->, thin, gray] (0, -4.0) -- (0, 4.0) node[above, font=\small] {$y$};
\node[font=\footnotesize, below right] at (0,0) {$O$};

% 渐近线 y=x 和 y=-x（等轴双曲线特征）
\draw[dashed, gray] (-3.8, -3.8) -- (3.8, 3.8)
    node[above right, font=\small, gray] {$y=x$};
\draw[dashed, gray] (-3.8, 3.8) -- (3.8, -3.8)
    node[below right, font=\small, gray] {$y=-x$};

% 等轴双曲线右支：a=b=2，cosh×2, sinh×2
\draw[black, thick, domain=-1.8:1.8, smooth, samples=80]
    plot ({2*cosh(\x)}, {2*sinh(\x)});
% 左支
\draw[black, thick, domain=-1.8:1.8, smooth, samples=80]
    plot ({-2*cosh(\x)}, {2*sinh(\x)});

% 顶点 (±a,0) = (±2,0)
\fill[black] (2,0) circle [radius=2.5pt];
\fill[black] (-2,0) circle [radius=2.5pt];
\node[font=\small, below right] at (2,0) {$A_2(a,0)$};
\node[font=\small, below left] at (-2,0) {$A_1(-a,0)$};

% 焦点 (±2sqrt(2), 0)
\fill[black] ({2*sqrt(2)},0) circle [radius=2.5pt];
\fill[black] ({-2*sqrt(2)},0) circle [radius=2.5pt];
\node[font=\small, below right] at ({2*sqrt(2)},0) {$F_2$};
\node[font=\small, below left] at ({-2*sqrt(2)},0) {$F_1$};

% 标注 a=b
\draw[<->, blue, thin] (0, 0.25) -- (2, 0.25)
    node[midway, above, font=\footnotesize, blue] {$a$};
\draw[<->, blue, thin] (2.25, 0) -- (2.25, 2)
    node[midway, right, font=\footnotesize, blue] {$b=a$};
% 虚线辅助
\draw[gray, dashed, thin] (0,0) -- (0,2) -- (2,2) -- (2,0);

% 45度角标记（渐近线与x轴夹角为45°）
\draw[gray, thin] (0.7,0) arc (0:45:0.7);
\node[font=\footnotesize, gray] at (0.95, 0.22) {$45°$};

% 关键结论
\node[font=\small, align=center] at (0, -4.7)
    {等轴双曲线：$a=b$，$e=\sqrt{2}$，渐近线 $y=\pm x$};

\end{tikzpicture}
\end{document}
