\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  scale=1.0, thick,
  axis/.style={-{Stealth[length=7pt, width=4pt]}, thin},
  pt/.style={circle, fill=black, inner sep=1.8pt},
]

% 坐标轴
\draw[axis] (-0.8,0) -- (4.5,0) node[right] {$x$};
\draw[axis] (0,-0.8) -- (0,4.2) node[above] {$y$};
\node[below left] at (0,0) {$O$};

% 截距式直线：x/a + y/b = 1
% 取 a = 3.0, b = 2.5
% 两端点: A(a,0) = (3,0), B(0,b) = (0,2.5)
\coordinate (A) at (3.0, 0);
\coordinate (B) at (0, 2.5);

% 延伸直线超出截距点一点
\coordinate (LBeyondA) at (3.8, {2.5 - 2.5/3.0 * 3.8});
\coordinate (LBeyondB) at (-0.5, {2.5 - 2.5/3.0 * (-0.5)});
\draw[thick, black] (LBeyondB) -- (LBeyondA);

% 截距点
\node[pt] at (A) {};
\node[below right] at (A) {$A(a,\;0)$};
\node[pt] at (B) {};
\node[left] at (B) {$B(0,\;b)$};

% x 轴截距标注
\draw[{Stealth[length=5pt]}-{Stealth[length=5pt]}, red, thin]
    (0,-0.45) -- (3.0,-0.45);
\node[below, red] at (1.5,-0.45) {\footnotesize $x$ 轴截距 $a$};

% y 轴截距标注
\draw[{Stealth[length=5pt]}-{Stealth[length=5pt]}, blue, thin]
    (-0.55,0) -- (-0.55,2.5);
\node[left, blue] at (-0.55,1.25) {\footnotesize $y$ 轴截距 $b$};

% 公式框
\node[draw=red!60, rounded corners=3pt, fill=red!5, inner sep=4pt]
  at (3.2, 3.5)
  {$\dfrac{x}{a} + \dfrac{y}{b} = 1$};

% 条件注释
\node[font=\footnotesize, gray] at (2.5, 1.4) {$a \neq 0,\; b \neq 0$};

\end{tikzpicture}
\end{document}
