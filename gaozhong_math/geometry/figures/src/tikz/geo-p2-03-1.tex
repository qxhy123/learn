\documentclass[tikz,border=8pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  scale=1.0, thick,
  vec/.style={-{Stealth[length=7pt, width=4pt]}, thick},
]

% 直线方向向量参数
\def\dx{2.5} \def\dy{1.2}

% 定点 P0
\coordinate (P0) at (1.5, 1.2);

% 直线上若干点（参数 t）
% t = -1.1
\coordinate (Lstart) at ($(P0) + (-1.1*\dx, -1.1*\dy)$);
% t = 1.3
\coordinate (Lend)   at ($(P0) + (1.3*\dx, 1.3*\dy)$);

% t = 0.85（动点 P）
\coordinate (P) at ($(P0) + (0.85*\dx, 0.85*\dy)$);

% 画直线（实线，黑色）
\draw[black, thick] (Lstart) -- (Lend);

% 方向向量 d：从 P0 出发
\coordinate (Dtip) at ($(P0) + (\dx, \dy)$);
\draw[vec, red, thick] (P0) -- (Dtip) node[above right] {$\vec{d}$};

% 向量 OP0（从原点到 P0，灰色虚线）
\coordinate (O) at (0, 0);
\draw[vec, gray, dashed] (O) -- (P0);
\node[above, xshift=-4pt] at ($(O)!0.5!(P0)$) {\small $\vec{OP_0}$};

% 向量 OP（从原点到 P，灰色虚线）
\draw[vec, gray, dashed] (O) -- (P);
\node[below right, xshift=2pt] at ($(O)!0.5!(P)$) {\small $\vec{OP}$};

% 向量 P0->P（蓝色，表示 t*d）
\draw[vec, blue, dashed, thick] (P0) -- (P)
  node[midway, below right, xshift=2pt] {\small $t\vec{d}$};

% 点：原点 O
\filldraw[black] (O)  circle (1.8pt) node[below left] {$O$};

% 点：P0
\filldraw[black] (P0) circle (2pt);
\node[below left] at (P0) {$P_0(x_0, y_0)$};

% 点：P（动点）
\filldraw[black] (P)  circle (2pt);
\node[above right] at (P) {$P(x, y)$};

% 参数方程说明文字（放在图下方）
\node[align=left, font=\small] at (4.5, -0.8) {
  $P = P_0 + t\vec{d}$，$t \in \mathbb{R}$\\[3pt]
  即 $\begin{cases} x = x_0 + at \\ y = y_0 + bt \end{cases}$，$\vec{d} = (a,b)$
};

\end{tikzpicture}
\end{document}
