\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.2, thick]

% 中点弦（点差法）几何示意
% 椭圆 x^2/4 + y^2/1 = 1，a=2, b=1
% 弦 AB 的中点 M，椭圆中心 O，连 OM 与弦关系（斜率积 = -b^2/a^2）

\def\a{2.0}
\def\b{1.0}

% 坐标轴
\draw[->, gray, thin] (-2.8, 0) -- (2.8, 0) node[right, font=\footnotesize] {$x$};
\draw[->, gray, thin] (0, -1.5) -- (0, 1.8) node[above, font=\footnotesize] {$y$};
\node[font=\footnotesize, below right] at (0,0) {$O$};

% 椭圆
\draw[black, thick] (0,0) ellipse [x radius=\a, y radius=\b];

% 选取弦 AB：设中点 M(0.8, 0.5)
% 对椭圆 x^2/4 + y^2 = 1，点差法得 k_AB = -b^2 x_0 / (a^2 y_0) = -1*0.8/(4*0.5) = -0.4
% 直线过 M(0.8,0.5)，斜率 -0.4：y - 0.5 = -0.4(x - 0.8)
% y = -0.4x + 0.32 + 0.5 = -0.4x + 0.82
% 代入椭圆：x^2/4 + (-0.4x+0.82)^2 = 1
% x^2/4 + 0.16x^2 - 0.656x + 0.6724 = 1
% 0.25x^2 + 0.16x^2 - 0.656x + 0.6724 - 1 = 0
% 0.41x^2 - 0.656x - 0.3276 = 0
% x = (0.656 ± sqrt(0.430336 + 4*0.41*0.3276)) / (2*0.41)
% x = (0.656 ± sqrt(0.430336 + 0.537264)) / 0.82
% x = (0.656 ± sqrt(0.9676)) / 0.82
% sqrt(0.9676) ≈ 0.9837
% x1 = (0.656 + 0.9837)/0.82 ≈ 1.999 ≈ 2.0
% x2 = (0.656 - 0.9837)/0.82 ≈ -0.399 ≈ -0.4
% y1 = -0.4*2.0 + 0.82 = 0.02 ≈ 0
% y2 = -0.4*(-0.4) + 0.82 = 0.16+0.82 = 0.98 ≈ 1.0
% 验证中点：x_mid=(2.0-0.4)/2=0.8 ✓, y_mid=(0+1.0)/2=0.5 ✓

\coordinate (A) at (2.0, 0.02);
\coordinate (B) at (-0.4, 0.98);
\coordinate (M) at (0.8, 0.5);   % 弦中点

% 弦 AB（红色）
\draw[red, thick] (A) -- (B);
\fill[black] (A) circle [radius=1.8pt];
\fill[black] (B) circle [radius=1.8pt];
\node[font=\footnotesize, right] at (A) {$A(x_1,y_1)$};
\node[font=\footnotesize, left] at (B) {$B(x_2,y_2)$};

% 弦中点 M
\fill[red] (M) circle [radius=2pt];
\node[font=\footnotesize, above right] at (M) {$M(x_0,y_0)$};

% 椭圆中心 O 到中点 M 的连线（蓝色虚线）
\draw[blue, dashed, thick] (0,0) -- (M);

% 中心标注
\fill[black] (0,0) circle [radius=1.8pt];

% 斜率关系标注
% k_OM = y0/x0 = 0.5/0.8 = 0.625
% k_AB * k_OM = -0.4 * 0.625 = -0.25 = -b^2/a^2 = -1/4
% 注意：k_AB = -b^2*x0/(a^2*y0)，不是 -1/k_OM（除非 a=b 圆）

% 在 OM 延长处标注斜率
\node[blue, font=\footnotesize] at (0.2, 0.55) {$k_{OM}$};
\node[red, font=\footnotesize] at (1.3, 0.55) {$k_{AB}$};

% 点差法公式（图下方）
\node[font=\small, align=center] at (0, -2.0)
  {点差法：$k_{AB} = -\dfrac{b^2 x_0}{a^2 y_0}$（中点 $M(x_0,y_0)$）};

% 辅助：标出半轴
\fill[gray] (\a, 0) circle [radius=1.2pt];
\fill[gray] (-\a, 0) circle [radius=1.2pt];
\fill[gray] (0, \b) circle [radius=1.2pt];
\fill[gray] (0, -\b) circle [radius=1.2pt];
\node[font=\footnotesize, below right] at (\a, 0) {$a$};
\node[font=\footnotesize, below left] at (-\a, 0) {$-a$};

\end{tikzpicture}
\end{document}
