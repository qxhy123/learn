\documentclass[tikz,border=10pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
% 正三棱锥建系示意图
% 底面边长 a，取底面中心 O 为原点，OA 方向为 x 轴，z 轴垂直底面向上
% 正三棱锥 P-ABC，底面正三角形
\begin{tikzpicture}[
  thick,
  dot/.style={fill=black, circle, inner sep=1.5pt},
]

% 设底面边长 = 2，外接圆半径 R = 2/sqrt(3)
% A、B、C 顶点（平面坐标，用等距投影）
% A 在 x 轴正方向，B、C 对称
% 使用等距投影近似
\def\R{1.5}    % 外接圆半径（图上单位）
\def\H{2.6}    % 棱锥高度（图上单位）

% 底面中心
\coordinate (O)  at (0, 0);
% 底面三顶点：A 在右（x轴方向），B、C 各偏左
\coordinate (A)  at (\R, 0);
\coordinate (B)  at (-0.75*\R, -0.65*\R);
\coordinate (C)  at (-0.75*\R,  0.65*\R);
% 顶点 P
\coordinate (P)  at (0, \H);

% 填充底面（浅蓝）
\fill[blue!10, draw=blue!50, fill opacity=0.7] (A) -- (B) -- (C) -- cycle;

% 画底面三边
\draw[blue!60] (A) -- (B);
\draw[blue!60] (B) -- (C);
\draw[blue!60] (C) -- (A);

% 侧棱 PA、PB、PC
\draw[black!70] (P) -- (A);
\draw[black!70] (P) -- (B);
\draw[black!70] (P) -- (C);

% 顶点到底面中心的高线（虚线）
\draw[gray, dashed, thin] (O) -- (P);

% 坐标轴：x 轴沿 OA 方向，y 轴在底面内垂直 x，z 轴向上（即 OP 方向）
\draw[red, very thick, -Stealth] (O) -- (2.2, 0)
  node[right, font=\small\bfseries, red] {$x$};
\draw[red, very thick, -Stealth] (O) -- (0, -1.5)
  node[below, font=\small\bfseries, red] {$y$};
\draw[red, very thick, -Stealth] (O) -- (0, \H+0.5)
  node[above, font=\small\bfseries, red] {$z$};

% 在 O 处画垂直标记（高线与底面垂直）
\draw[gray, thin]
  ($(O)+(0.12,0)$) -- ($(O)+(0.12,0.12)$) -- ($(O)+(0,0.12)$);

% 标注顶点
\node[dot] at (O) {};
\node[font=\small\bfseries, below left] at (O)
  {$O(0,0,0)$};

\node[dot] at (A) {};
\node[font=\small, right] at (A)
  {$A\!\left(\dfrac{\sqrt{3}}{3}a,0,0\right)$};

\node[dot] at (B) {};
\node[font=\small, below left] at (B)
  {$B$};

\node[dot] at (C) {};
\node[font=\small, above left] at (C)
  {$C$};

\node[dot] at (P) {};
\node[font=\small, above right] at (P)
  {$P\!\left(0,0,h\right)$};

% 标注高
\node[font=\small, right, gray] at ($(O)!0.5!(P)+(0.12,0)$) {$h$};

% 底面标注边长
\node[font=\small, below right, blue!60] at ($(A)!0.5!(B)$) {$a$};

% 底部说明
\node[font=\small, align=center] at (0, -2.2)
  {正三棱锥 $P$-$ABC$，底面边长 $a$，高 $h$};
\node[font=\small, align=center] at (0, -2.6)
  {以底面中心 $O$ 为原点建立坐标系};

\end{tikzpicture}
\end{document}
