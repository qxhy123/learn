\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.0, thick]

% 三栏：相离 / 相切 / 相交

% ── 栏 1：相离 d > r ──
\begin{scope}[xshift=0cm]
  \node[font=\small\bfseries] at (1.5, 3.8) {相离};

  % 圆
  \draw[black, thick] (1.5, 1.8) circle [radius=1.1];
  \fill[black] (1.5, 1.8) circle [radius=1.5pt];
  \node[font=\footnotesize, below] at (1.5, 1.8) {$O$};

  % 直线（在圆外，不相交）
  \draw[blue, thick] (0.0, 3.2) -- (3.0, 3.2);
  \node[blue, font=\small] at (3.15, 3.2) {$l$};

  % 圆心到直线距离 d（灰色虚线）
  \draw[gray, dashed] (1.5, 1.8) -- (1.5, 3.2);
  \node[gray, font=\footnotesize, right] at (1.5, 2.5) {$d$};

  % 半径标注
  \draw[red, thick] (1.5, 1.8) -- (2.6, 1.8);
  \node[red, font=\footnotesize, above] at (2.05, 1.8) {$r$};

  % 条件
  \node[font=\footnotesize, align=center] at (1.5, 0.2) {$d > r$};
  \node[font=\footnotesize, align=center] at (1.5, -0.3) {无交点};
\end{scope}

% 分隔线
\draw[gray, thin, dashed] (3.5, -0.5) -- (3.5, 4.2);

% ── 栏 2：相切 d = r ──
\begin{scope}[xshift=4.0cm]
  \node[font=\small\bfseries] at (1.5, 3.8) {相切};

  % 圆
  \draw[black, thick] (1.5, 1.8) circle [radius=1.1];
  \fill[black] (1.5, 1.8) circle [radius=1.5pt];
  \node[font=\footnotesize, below left] at (1.5, 1.8) {$O$};

  % 直线恰好与圆相切（切线在圆顶 y = 1.8 + 1.1 = 2.9）
  \draw[blue, thick] (0.0, 2.9) -- (3.0, 2.9);
  \node[blue, font=\small] at (3.15, 2.9) {$l$};

  % 切点
  \fill[black] (1.5, 2.9) circle [radius=1.5pt];
  \node[font=\footnotesize, above right] at (1.5, 2.9) {$T$};

  % 圆心到切线距离 = r
  \draw[gray, dashed] (1.5, 1.8) -- (1.5, 2.9);
  \node[gray, font=\footnotesize, right] at (1.5, 2.35) {$d=r$};

  % 半径与切线垂直的小方块
  \draw[gray, thin]
    (1.5, 2.9) -- (1.5+0.12, 2.9) -- (1.5+0.12, 2.9-0.12) -- (1.5, 2.9-0.12);

  % 条件
  \node[font=\footnotesize, align=center] at (1.5, 0.2) {$d = r$};
  \node[font=\footnotesize, align=center] at (1.5, -0.3) {1 个切点 $T$};
\end{scope}

% 分隔线
\draw[gray, thin, dashed] (7.5, -0.5) -- (7.5, 4.2);

% ── 栏 3：相交 d < r ──
\begin{scope}[xshift=8.0cm]
  \node[font=\small\bfseries] at (1.5, 3.8) {相交};

  % 圆
  \draw[black, thick] (1.5, 1.8) circle [radius=1.3];
  \fill[black] (1.5, 1.8) circle [radius=1.5pt];
  \node[font=\footnotesize, below left] at (1.5, 1.8) {$O$};

  % 直线穿过圆（y = 2.5，在圆内部）
  \draw[blue, thick] (0.0, 2.5) -- (3.0, 2.5);
  \node[blue, font=\small] at (3.15, 2.5) {$l$};

  % 两个交点：x = 1.5 ± sqrt(1.3^2 - (2.5-1.8)^2) = 1.5 ± sqrt(1.69-0.49) = 1.5 ± sqrt(1.2) ≈ 1.5 ± 1.095
  \coordinate (A) at ({1.5 - sqrt(1.2)}, 2.5);
  \coordinate (B) at ({1.5 + sqrt(1.2)}, 2.5);
  \fill[black] (A) circle [radius=1.5pt];
  \fill[black] (B) circle [radius=1.5pt];
  \node[font=\footnotesize, above left] at (A) {$A$};
  \node[font=\footnotesize, above right] at (B) {$B$};

  % 圆心到直线距离
  \draw[gray, dashed] (1.5, 1.8) -- (1.5, 2.5);
  \node[gray, font=\footnotesize, right] at (1.5, 2.15) {$d$};

  % 半径
  \draw[red, thick] (1.5, 1.8) -- ({1.5 + sqrt(1.2)}, 2.5);
  \node[red, font=\footnotesize, below right] at ({1.5 + sqrt(1.2)/2}, 2.15) {$r$};

  % 弦长标注
  \draw[<->, red!60, thin] ($(A) + (0, 0.22)$) -- ($(B) + (0, 0.22)$);
  \node[red!70, font=\footnotesize, above] at (1.5, 2.72) {$|AB|=2\sqrt{r^2-d^2}$};

  % 条件
  \node[font=\footnotesize, align=center] at (1.5, 0.2) {$d < r$};
  \node[font=\footnotesize, align=center] at (1.5, -0.3) {2 个交点 $A, B$};
\end{scope}

\end{tikzpicture}
\end{document}
