\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.0, thick]

% 三栏：直线与椭圆 3 种位置（相离 / 相切 / 相交）
% 椭圆半轴 a=1.6, b=1.0

% ── 栏 1：相离 Δ < 0 ──
\begin{scope}[xshift=0cm]
  \node[font=\small\bfseries] at (1.6, 3.8) {相离};

  % 椭圆：中心 (1.6, 1.8)，a=1.6，b=1.0
  \draw[black, thick] (1.6, 1.8) ellipse [x radius=1.4, y radius=0.9];
  \fill[black] (1.6, 1.8) circle [radius=1.5pt];
  \node[font=\footnotesize, below] at (1.6, 1.8) {$O$};

  % 直线（在椭圆外上方，不相交）
  \draw[blue, thick] (0.0, 3.2) -- (3.2, 3.2);
  \node[blue, font=\small] at (3.4, 3.2) {$l$};

  % 条件
  \node[font=\footnotesize] at (1.6, 0.3) {$\Delta < 0$};
  \node[font=\footnotesize] at (1.6, -0.2) {无交点};
\end{scope}

% 分隔线
\draw[gray, thin, dashed] (3.6, -0.5) -- (3.6, 4.2);

% ── 栏 2：相切 Δ = 0 ──
\begin{scope}[xshift=4.0cm]
  \node[font=\small\bfseries] at (1.6, 3.8) {相切};

  % 椭圆：中心 (1.6, 1.8)
  \draw[black, thick] (1.6, 1.8) ellipse [x radius=1.4, y radius=0.9];
  \fill[black] (1.6, 1.8) circle [radius=1.5pt];
  \node[font=\footnotesize, below left] at (1.6, 1.8) {$O$};

  % 切线（直线恰好与椭圆顶部 y = 1.8+0.9 = 2.7 相切）
  \draw[blue, thick] (0.0, 2.7) -- (3.2, 2.7);
  \node[blue, font=\small] at (3.4, 2.7) {$l$};

  % 切点
  \fill[red] (1.6, 2.7) circle [radius=1.5pt];
  \node[font=\footnotesize, above right] at (1.6, 2.7) {$T$};

  % 切点到中心的虚线（辅助）
  \draw[gray, dashed] (1.6, 1.8) -- (1.6, 2.7);

  % 条件
  \node[font=\footnotesize] at (1.6, 0.3) {$\Delta = 0$};
  \node[font=\footnotesize] at (1.6, -0.2) {1 个切点 $T$};
\end{scope}

% 分隔线
\draw[gray, thin, dashed] (7.6, -0.5) -- (7.6, 4.2);

% ── 栏 3：相交 Δ > 0 ──
\begin{scope}[xshift=8.0cm]
  \node[font=\small\bfseries] at (1.6, 3.8) {相交};

  % 椭圆：中心 (1.6, 1.8)，a=1.4，b=0.9
  \draw[black, thick] (1.6, 1.8) ellipse [x radius=1.4, y radius=0.9];
  \fill[black] (1.6, 1.8) circle [radius=1.5pt];
  \node[font=\footnotesize, below left] at (1.6, 1.8) {$O$};

  % 直线穿过椭圆，y = 2.1（在 b 范围内）
  % 椭圆方程 (x-1.6)^2/1.96 + (y-1.8)^2/0.81 = 1
  % 令 y=2.1: (y-1.8)^2/0.81 = 0.09/0.81 = 1/9, (x-1.6)^2/1.96 = 8/9
  % x - 1.6 = ±1.4*sqrt(8)/3 ≈ ±1.32
  \draw[blue, thick] (0.0, 2.3) -- (3.2, 2.3);
  \node[blue, font=\small] at (3.4, 2.3) {$l$};

  % 两交点（直线 y=2.3, 椭圆方程求 x）
  % (y-1.8)^2/0.81 = 0.25/0.81, (x-1.6)^2 = 1.96*(1 - 0.25/0.81) = 1.96*0.5617 ≈ 1.101
  % x = 1.6 ± 1.049
  \coordinate (A) at ({1.6 - 1.049}, 2.3);
  \coordinate (B) at ({1.6 + 1.049}, 2.3);
  \fill[black] (A) circle [radius=1.5pt];
  \fill[black] (B) circle [radius=1.5pt];
  \node[font=\footnotesize, above left] at (A) {$A$};
  \node[font=\footnotesize, above right] at (B) {$B$};

  % 弦长标注
  \draw[<->, red!70, thin] ($(A)+(0,0.2)$) -- ($(B)+(0,0.2)$);
  \node[red!80, font=\footnotesize] at (1.6, 2.75) {$|AB|$};

  % 条件
  \node[font=\footnotesize] at (1.6, 0.3) {$\Delta > 0$};
  \node[font=\footnotesize] at (1.6, -0.2) {2 个交点 $A, B$};
\end{scope}

\end{tikzpicture}
\end{document}
