\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}

\begin{document}
\begin{tikzpicture}[
  scale=1.2, thick,
]

% 三角形顶点
\coordinate (A) at (0.5, 3.0);
\coordinate (B) at (0, 0);
\coordinate (C) at (4.0, 0);

% 各边中点
\coordinate (MA) at ({(0+4.0)/2}, {(0+0)/2});   % BC 中点 (2.0, 0)
\coordinate (MB) at ({(0.5+4.0)/2}, {(3.0+0)/2}); % AC 中点 (2.25, 1.5)
\coordinate (MC) at ({(0.5+0)/2}, {(3.0+0)/2});   % AB 中点 (0.25, 1.5)

% 重心（三坐标平均）
\coordinate (G) at ({(0.5+0+4.0)/3}, {(3.0+0+0)/3});  % (1.5, 1.0)

% 三角形边
\draw[black, thick] (A) -- (B) -- (C) -- cycle;

% 三条中线（灰色）
\draw[gray, thick] (A) -- (MA);
\draw[gray, thick] (B) -- (MB);
\draw[gray, thick] (C) -- (MC);

% 顶点标签
\node[above]       at (A) {$A$};
\node[below left]  at (B) {$B$};
\node[below right] at (C) {$C$};

% 中点标签
\node[below, font=\footnotesize] at (MA) {$M_A$};
\node[right, font=\footnotesize] at (MB) {$M_B$};
\node[left,  font=\footnotesize] at (MC) {$M_C$};

% 重心 G
\filldraw[red] (G) circle (2pt);
\node[below right, font=\small] at (G) {$G$（重心）};

% 顶点圆点
\filldraw[black] (A) circle (1.5pt);
\filldraw[black] (B) circle (1.5pt);
\filldraw[black] (C) circle (1.5pt);
\filldraw[gray]  (MA) circle (1.2pt);
\filldraw[gray]  (MB) circle (1.2pt);
\filldraw[gray]  (MC) circle (1.2pt);

% 公式说明
\node[font=\small, align=center] at (2.0, -0.65)
  {$\vec{OG} = \dfrac{1}{3}(\vec{OA} + \vec{OB} + \vec{OC})$};

\end{tikzpicture}
