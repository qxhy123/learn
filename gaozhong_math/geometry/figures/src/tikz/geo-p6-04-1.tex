\documentclass[tikz,border=8pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=0.85, thick]

% 坐标轴
\draw[->, thin, gray] (-4.5,0) -- (4.5,0) node[right, font=\small, gray] {$x$};
\draw[->, thin, gray] (0,-3.5) -- (0,3.5) node[above, font=\small, gray] {$y$};
\node[font=\footnotesize, gray] at (-0.25,-0.25) {$O$};

% 双曲线 x^2/4 - y^2/3 = 1，a=2, b=√3, c=√7≈2.646
% 取 a=2, b=sqrt(3), c=sqrt(7)
% 渐近线 y = ±(√3/2)x
\draw[gray, dashed, thin] (-4.2,{-4.2*sqrt(3)/2}) -- (4.2,{4.2*sqrt(3)/2});
\draw[gray, dashed, thin] (-4.2,{4.2*sqrt(3)/2}) -- (4.2,{-4.2*sqrt(3)/2});

% 右支（声呐问题：|PA|-|PB|=2a > 0 对应右支）
\draw[black, thick, domain=2:4.2, samples=80, variable=\x]
  plot ({\x}, {sqrt(3*((\x)^2/4-1))});
\draw[black, thick, domain=2:4.2, samples=80, variable=\x]
  plot ({\x}, {-sqrt(3*((\x)^2/4-1))});

% 左支（灰色，仅示意）
\draw[gray, thin, domain=2:4.2, samples=80, variable=\x]
  plot ({-\x}, {sqrt(3*((\x)^2/4-1))});
\draw[gray, thin, domain=2:4.2, samples=80, variable=\x]
  plot ({-\x}, {-sqrt(3*((\x)^2/4-1))});

% 两焦点（声源接收站 A, B）
\coordinate (FA) at ({-sqrt(7)},0);
\coordinate (FB) at ({sqrt(7)},0);

\filldraw[blue, thick] (FA) circle (2.5pt);
\node[blue, font=\small, below left] at (FA) {$A$};

\filldraw[blue, thick] (FB) circle (2.5pt);
\node[blue, font=\small, below right] at (FB) {$B$};

% 动点 P（在右支上）
% 取双曲线上点：x=3.0, y=sqrt(3*(9/4-1))=sqrt(3*5/4)=sqrt(15/4)=√15/2≈1.936
\coordinate (P) at ({3.0}, {sqrt(15)/2});
\filldraw[red, thick] (P) circle (2.2pt);
\node[red, font=\small, right] at (P) {$P$（声源）};

% PA, PB 辅助线
\draw[gray, dashed, thin] (P) -- (FA);
\draw[gray, dashed, thin] (P) -- (FB);

% 标 |PA|, |PB|
\node[font=\footnotesize, gray] at ($(P)!0.5!(FA) + (-0.4,0.2)$) {$|PA|$};
\node[font=\footnotesize, gray] at ($(P)!0.5!(FB) + (0.3,0.2)$) {$|PB|$};

% 标注距离差
\node[font=\small, align=center] at (0, -3.1) {$|PA|-|PB|=2a$（常数）};

% 顶点 a
\filldraw[black] (2,0) circle (1.8pt);
\node[font=\footnotesize, below] at (2,0) {$a$};
\filldraw[black] (-2,0) circle (1.8pt);

% 标记 2a
\draw[gray, thin, <->] (-0.05,0.35) -- (1.95,0.35);
\node[font=\footnotesize, gray, above] at (1.0,0.35) {$a$};

\end{tikzpicture}
\end{document}
