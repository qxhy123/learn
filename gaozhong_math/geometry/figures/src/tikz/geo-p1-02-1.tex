\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  scale=1.0, thick,
  vec/.style={-{Stealth[length=7pt, width=4pt]}, thick},
  vecred/.style={-{Stealth[length=7pt, width=4pt]}, thick, red},
]

% 三角形法则：首尾相接 \vec{AB} + \vec{BC} = \vec{AC}
%
%  A --\vec{AB}--> B --\vec{BC}--> C
%  A ---------\vec{AC}---------> C

\coordinate (A) at (0,0);
\coordinate (B) at (3.5, 1.2);
\coordinate (C) at (6.0, 0);

% \vec{AB}
\draw[vec, black] (A) -- (B);
\node[above left] at ($(A)!0.5!(B)$) {$\vec{AB}$};

% \vec{BC}
\draw[vec, black] (B) -- (C);
\node[above right] at ($(B)!0.5!(C)$) {$\vec{BC}$};

% \vec{AC}（和向量，用红色）
\draw[vecred] (A) -- (C);
\node[below] at ($(A)!0.5!(C)$) {$\vec{AC}$（和向量）};

% 顶点标注
\node[left] at (A) {$A$};
\node[above] at (B) {$B$};
\node[right] at (C) {$C$};

% 公式
\node[below] at (3.0, -0.7) {$\vec{AB} + \vec{BC} = \vec{AC}$\quad（首尾相接法则）};

\end{tikzpicture}
\end{document}
