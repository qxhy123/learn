\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, angles, quotes, arrows.meta}
\begin{document}
\begin{tikzpicture}[thick]
% ===================== 情形 1：两解 =====================
\begin{scope}[scale=1.0, xshift=0cm]
  % 给定 A = 30°, c = AB = 5, a = BC = 3 (a > h, a < c → 两解)
  \def\Aang{30}
  \def\cAB{5}
  \def\aa{3}
  \coordinate (A) at (0,0);
  \coordinate (B) at (\cAB,0);
  % 高 h = c sin A = 5*0.5 = 2.5
  \pgfmathsetmacro{\hh}{\cAB*sin(\Aang)}
  % BC 长 = a，从 B 向上 a 的圆弧与终边的两个交点
  % 直接算两交点：t1, t2 满足距离公式
  \pgfmathsetmacro{\dd}{sqrt(\aa*\aa - \hh*\hh)}
  \pgfmathsetmacro{\fAB}{\cAB - \dd*cos(\Aang) - \hh*cos(90-\Aang)} % 不用，下面手算
  % 实际上 C 在 A 的终边上，距离 A 为 t；BC^2 = t^2 - 2 t c cos A + c^2 = a^2
  % t^2 - 2 t c cos A + c^2 - a^2 = 0
  \pgfmathsetmacro{\bb}{2*\cAB*cos(\Aang)}
  \pgfmathsetmacro{\disc}{\bb*\bb - 4*(\cAB*\cAB - \aa*\aa)}
  \pgfmathsetmacro{\sqd}{sqrt(\disc)}
  \pgfmathsetmacro{\tone}{(\bb + \sqd)/2}
  \pgfmathsetmacro{\ttwo}{(\bb - \sqd)/2}
  \coordinate (C1) at ({\tone*cos(\Aang)}, {\tone*sin(\Aang)});
  \coordinate (C2) at ({\ttwo*cos(\Aang)}, {\ttwo*sin(\Aang)});
  % A 处终边（点划线，到最远的 C1 略远）
  \draw[gray, thin, dashed] (A) -- ({(\tone+0.5)*cos(\Aang)}, {(\tone+0.5)*sin(\Aang)});
  % AB
  \draw[black, very thick] (A) -- (B);
  % B 处圆弧（半径 = a）
  \draw[gray, thin, dashed] (B) circle (\aa);
  % 两条 BC
  \draw[red, very thick] (B) -- (C1);
  \draw[blue, very thick] (B) -- (C2);
  % AC1, AC2 (实线突出)
  \draw[red, very thick] (A) -- (C1);
  \draw[blue, very thick] (A) -- (C2);
  % 标
  \node[below left, font=\footnotesize] at (A) {$A$};
  \node[below right, font=\footnotesize] at (B) {$B$};
  \node[above right, font=\footnotesize, red] at (C1) {$C_1$};
  \node[above, font=\footnotesize, blue] at (C2) {$C_2$};
  % 角弧
  \draw[teal, thick] (0.5,0) arc (0:\Aang:0.5);
  \node[teal, font=\footnotesize] at (0.75,0.16) {$A$};
  % 标题
  \node[font=\small, align=center] at (\cAB/2,-1.2) {两解：$a>h$ 且 $a<c$};
\end{scope}

% ===================== 情形 2：一解 =====================
\begin{scope}[scale=1.0, xshift=7.5cm]
  \def\AangB{30}
  \def\cABB{5}
  \def\aaB{2.5}  % = h，恰好一解（直角）
  \coordinate (A) at (0,0);
  \coordinate (B) at (\cABB,0);
  % 解：t = c cos A
  \pgfmathsetmacro{\tt}{\cABB*cos(\AangB)}
  \coordinate (C) at ({\tt*cos(\AangB)}, {\tt*sin(\AangB)});
  % 终边
  \draw[gray, thin, dashed] (A) -- ({(\tt+0.5)*cos(\AangB)}, {(\tt+0.5)*sin(\AangB)});
  \draw[black, very thick] (A) -- (B);
  \draw[gray, thin, dashed] (B) circle (\aaB);
  \draw[red, very thick] (A) -- (C);
  \draw[red, very thick] (B) -- (C);
  \node[below left, font=\footnotesize] at (A) {$A$};
  \node[below right, font=\footnotesize] at (B) {$B$};
  \node[above right, font=\footnotesize, red] at (C) {$C$};
  \draw[teal, thick] (0.5,0) arc (0:\AangB:0.5);
  \node[teal, font=\footnotesize] at (0.75,0.16) {$A$};
  \node[font=\small, align=center] at (\cABB/2,-1.2) {一解：$a=h$（直角）};
\end{scope}

% ===================== 情形 3：无解 =====================
\begin{scope}[scale=1.0, xshift=15cm]
  \def\AangC{30}
  \def\cABC{5}
  \def\aaC{1.5}  % < h = 2.5 → 无解
  \coordinate (A) at (0,0);
  \coordinate (B) at (\cABC,0);
  \draw[gray, thin, dashed] (A) -- ({(\cABC+0.5)*cos(\AangC)}, {(\cABC+0.5)*sin(\AangC)});
  \draw[black, very thick] (A) -- (B);
  \draw[gray, thin, dashed] (B) circle (\aaC);
  % 高线 h
  \pgfmathsetmacro{\hC}{\cABC*sin(\AangC)}
  \pgfmathsetmacro{\fxC}{\cABC*cos(\AangC)*cos(\AangC)}
  \pgfmathsetmacro{\fyC}{\cABC*cos(\AangC)*sin(\AangC)}
  \draw[orange, dashed, thick] (B) -- (\fxC,\fyC);
  \node[font=\footnotesize, orange] at ({(\fxC+\cABC)/2+0.05},{\fyC/2+0.1}) {$h$};
  \node[below left, font=\footnotesize] at (A) {$A$};
  \node[below right, font=\footnotesize] at (B) {$B$};
  \draw[teal, thick] (0.5,0) arc (0:\AangC:0.5);
  \node[teal, font=\footnotesize] at (0.75,0.16) {$A$};
  \node[font=\small, align=center] at (\cABC/2,-1.2) {无解：$a<h$};
\end{scope}
\end{tikzpicture}
\end{document}
