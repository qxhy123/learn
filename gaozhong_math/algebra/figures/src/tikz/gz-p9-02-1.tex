% 等差 vs 等比数列：数轴标点对比
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta, decorations.pathreplacing}
\begin{document}
\begin{tikzpicture}[thick]
  % ====== 左图：等差数列 2,5,8,11,14 ======
  \begin{scope}[shift={(0,0)}]
    % 数轴
    \draw[->, gray!60!black] (-0.3, 0) -- (16.5, 0) node[right, black, font=\small] {数值};
    \foreach \x in {0,2,...,16} \draw[gray] (\x, 0.1) -- (\x, -0.1);
    \foreach \x in {0,4,...,16} \node[below, font=\scriptsize, gray!40!black] at (\x, -0.1) {$\x$};

    % 标点 2,5,8,11,14
    \foreach \x/\n in {2/1, 5/2, 8/3, 11/4, 14/5} {
      \fill[blue!70!black] (\x, 0) circle (3pt);
      \node[above, font=\footnotesize, blue!70!black] at (\x, 0.1) {$a_{\n}=\x$};
    }

    % 公差弧线：+3
    \foreach \xa/\xb in {2/5, 5/8, 8/11, 11/14} {
      \draw[red, ->, thick] (\xa, 0.85) to[bend left=50] node[midway, above, font=\scriptsize] {$+3$} (\xb, 0.85);
    }

    \node[font=\small, anchor=west] at (0, 2.3) {\textbf{等差数列}：相邻项之差 $=$ 公差 $d=3$（加法递增）};
    \node[font=\footnotesize, anchor=west, gray!40!black] at (0, 1.85) {$a_n = 2 + 3(n-1) = 3n-1$};
  \end{scope}

  % ====== 右图：等比数列 2,4,8,16,32 ======
  \begin{scope}[shift={(0, -4)}]
    % 数轴
    \draw[->, gray!60!black] (-0.3, 0) -- (16.5, 0) node[right, black, font=\small] {数值};
    \foreach \x in {0,2,...,16} \draw[gray] (\x, 0.1) -- (\x, -0.1);
    \foreach \x in {0,4,...,16} \node[below, font=\scriptsize, gray!40!black] at (\x, -0.1) {$\x$};

    % 标点 2,4,8,16；32 落在右边外面用箭头表示
    \foreach \x/\n in {2/1, 4/2, 8/3, 16/4} {
      \fill[orange!80!black] (\x, 0) circle (3pt);
      \node[above, font=\footnotesize, orange!70!black] at (\x, 0.1) {$b_{\n}=\x$};
    }
    \node[right, font=\footnotesize, orange!70!black] at (16.5, 0.3) {$b_5=32 \to$};

    % 公比弧线：×2
    \foreach \xa/\xb in {2/4, 4/8, 8/16} {
      \draw[red, ->, thick] (\xa, 0.85) to[bend left=50] node[midway, above, font=\scriptsize] {$\times 2$} (\xb, 0.85);
    }

    \node[font=\small, anchor=west] at (0, 2.3) {\textbf{等比数列}：相邻项之比 $=$ 公比 $q=2$（乘法递增）};
    \node[font=\footnotesize, anchor=west, gray!40!black] at (0, 1.85) {$b_n = 2 \cdot 2^{n-1} = 2^n$};
  \end{scope}

\end{tikzpicture}
\end{document}
