% gz-p10-05-1: "在"vs"过"——两类切线对比
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc}

\begin{document}
\begin{tikzpicture}
% 左图：在曲线上一点 P 的切线（P 是切点）
\begin{axis}[
  name=left,
  width=7.5cm, height=6cm,
  axis lines=center,
  xlabel={$x$}, ylabel={$y$},
  every axis x label/.style={at={(axis cs:2.4,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,5.5)}, anchor=south},
  xmin=-0.5, xmax=2.4,
  ymin=-0.5, ymax=5.5,
  xtick=\empty, ytick=\empty,
  clip=true,
  title={\small 在 $P$ 点的切线},
]
  % 曲线 y = x^2
  \addplot[black, thick, domain=-0.5:2.3, samples=80] {x^2};
  \node[right, black, font=\small] at (axis cs:1.8, 4.2) {$y=f(x)$};

  % 切点 P(1.2, 1.44)，切线斜率 = 2x = 2.4，切线: y - 1.44 = 2.4(x - 1.2)
  \addplot[only marks, mark=*, mark size=2.5pt, red] coordinates {(1.2,1.44)};
  \node[above left, red, font=\small] at (axis cs:1.2,1.44) {$P$};

  % 切线 y = 2.4x - 1.44
  \addplot[red, thick, domain=0.0:2.3] {2.4*x - 1.44};

  \node[font=\footnotesize, red, align=left] at (axis cs:0.8, 5.0)
    {$P$ 是切点};
\end{axis}

% 右图：过曲线外一点 Q 的切线（Q 不是切点）
\begin{axis}[
  at={($(left.east)+(2cm,0)$)},
  anchor=west,
  width=7.5cm, height=6cm,
  axis lines=center,
  xlabel={$x$}, ylabel={$y$},
  every axis x label/.style={at={(axis cs:2.4,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,5.5)}, anchor=south},
  xmin=-0.5, xmax=2.4,
  ymin=-0.5, ymax=5.5,
  xtick=\empty, ytick=\empty,
  clip=true,
  title={\small 过 $Q$ 点的切线},
]
  % 曲线 y = x^2
  \addplot[black, thick, domain=-0.5:2.3, samples=80] {x^2};
  \node[right, black, font=\small] at (axis cs:1.8, 4.2) {$y=f(x)$};

  % 切点 T(0.8, 0.64)，切线: y = 1.6x - 0.64
  \addplot[only marks, mark=*, mark size=2.5pt, red] coordinates {(0.8,0.64)};
  \node[below right, red, font=\small] at (axis cs:0.8,0.64) {$T$ (切点)};

  % 切线
  \addplot[red, thick, domain=-0.2:2.3] {1.6*x - 0.64};

  % 外点 Q(2, 2.56)，在切线上但不在曲线上
  \addplot[only marks, mark=*, mark size=2.5pt, blue] coordinates {(2, 2.56)};
  \node[above left, blue, font=\small] at (axis cs:2,2.56) {$Q$};

  \node[font=\footnotesize, blue, align=left] at (axis cs:0.4, 5.0)
    {$Q$ 不是切点,\\切线过 $Q$};
\end{axis}
\end{tikzpicture}
\end{document}
