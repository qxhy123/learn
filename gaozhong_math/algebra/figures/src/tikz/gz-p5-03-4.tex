\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}
\begin{document}
\begin{tikzpicture}
\begin{groupplot}[
    group style={group size=1 by 4, vertical sep=0.9cm},
    axis lines=middle,
    xmin=-4.0, xmax=7.5,
    width=14cm, height=4.5cm,
    samples=400,
    grid=both,
    grid style={gray!20},
    xlabel={$x$}, ylabel={$y$},
    xlabel style={right}, ylabel style={above},
    xtick={-3.1416,-1.5708,0,1.5708,3.1416,4.7124,6.2832},
    xticklabels={$-\pi$,$-\tfrac{\pi}{2}$,$0$,$\tfrac{\pi}{2}$,$\pi$,$\tfrac{3\pi}{2}$,$2\pi$},
    tick label style={font=\scriptsize},
    title style={font=\small},
]
% step 1: y = sin x
\nextgroupplot[ymin=-1.6, ymax=1.6, ytick={-1,1}, title={\textbf{step 1:} $y=\sin x$}]
\addplot[blue, very thick, domain=-3.5:6.7] {sin(deg(x))};

% step 2: y = sin(2x)
\nextgroupplot[ymin=-1.6, ymax=1.6, ytick={-1,1}, title={\textbf{step 2:} $y=\sin 2x$ {\textemdash} 横向压缩 $\tfrac{1}{2}$ 倍}]
\addplot[teal, very thick, domain=-3.5:6.7] {sin(deg(2*x))};

% step 3: y = sin(2x - pi/3)
\nextgroupplot[ymin=-1.6, ymax=1.6, ytick={-1,1}, title={\textbf{step 3:} $y=\sin(2x-\tfrac{\pi}{3})$ {\textemdash} 右移 $\tfrac{\pi}{6}$}]
\addplot[orange, very thick, domain=-3.5:6.7] {sin(deg(2*x - pi/3))};

% step 4: y = 2 sin(2x - pi/3) + 1
\nextgroupplot[ymin=-2.0, ymax=4.0, ytick={-1,1,3}, title={\textbf{step 4:} $y=2\sin(2x-\tfrac{\pi}{3})+1$ {\textemdash} 纵向拉伸 $2$ 倍\; 上移 $1$}]
\addplot[red, very thick, domain=-3.5:6.7] {2*sin(deg(2*x - pi/3)) + 1};
\end{groupplot}
\end{tikzpicture}
\end{document}
