\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, angles, quotes, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=3.0, thick]
  % 坐标轴
  \draw[->, gray] (-1.45,0) -- (1.45,0) node[right, black] {$x$};
  \draw[->, gray] (0,-1.4) -- (0,1.45) node[above, black] {$y$};
  % 单位圆
  \draw[black] (0,0) circle (1);
  % 原角 alpha = 50°
  \def\ang{50}
  \coordinate (O) at (0,0);
  \coordinate (P) at ({cos(\ang)}, {sin(\ang)});  % alpha 终边点
  \coordinate (Q) at ({cos(180-\ang)}, {sin(180-\ang)}); % pi - alpha 终边点
  \coordinate (Xp) at ({cos(\ang)}, 0);
  \coordinate (Xq) at ({cos(180-\ang)}, 0);
  % 终边 alpha (蓝)
  \draw[blue, very thick] (O) -- (P);
  % 终边 pi - alpha (红)
  \draw[red, very thick] (O) -- (Q);
  % 角弧
  \draw[blue, thick] (0.22,0) arc (0:\ang:0.22);
  \node[blue, font=\footnotesize] at ({0.34*cos(\ang/2)}, {0.34*sin(\ang/2)}) {$\alpha$};
  \draw[red, thick] (0.32,0) arc (0:{180-\ang}:0.32);
  \node[red, font=\footnotesize] at (0.05,0.46) {$\pi-\alpha$};
  % P / Q 点
  \fill (P) circle (0.018);
  \node[above right, font=\footnotesize] at (P) {$P$};
  \fill (Q) circle (0.018);
  \node[above left, font=\footnotesize] at (Q) {$P'$};
  % y 投影（关键：两点 y 坐标相同）
  \draw[teal, dashed, thick] (P) -- (0,{sin(\ang)});
  \draw[teal, dashed, thick] (Q) -- (0,{sin(\ang)});
  \fill (0,{sin(\ang)}) circle (0.018);
  \node[left, font=\footnotesize, teal] at (-0.04,{sin(\ang)+0.04}) {$\sin\alpha$};
  % x 投影对称
  \draw[gray, dashed] (P) -- (Xp);
  \draw[gray, dashed] (Q) -- (Xq);
  % 原点
  \node[below left, font=\footnotesize] at (O) {$O$};
  % 底部公式说明
  \node[font=\footnotesize, align=center] at (0,-1.3) {$P$ 与 $P'$ 关于 $y$ 轴对称 $\Rightarrow \sin(\pi-\alpha)=\sin\alpha$};
\end{tikzpicture}
\end{document}
