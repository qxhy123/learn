\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, angles, quotes, decorations.pathreplacing, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=2.8, thick]
  % 坐标轴
  \draw[->, gray] (-1.35,0) -- (1.4,0) node[right, black] {$x$};
  \draw[->, gray] (0,-1.35) -- (0,1.4) node[above, black] {$y$};
  % 单位圆
  \draw[black, thick] (0,0) circle (1);
  % 半径标记 (1 弧度 ≈ 57.3°)
  \coordinate (O) at (0,0);
  \coordinate (A) at (1,0);
  \coordinate (B) at ({cos(57.2958)}, {sin(57.2958)});
  % 圆心角弧线（红色）
  \draw[red, very thick] (0.32,0) arc (0:57.2958:0.32);
  \node[red, font=\footnotesize] at (0.5,0.18) {$1$ rad};
  % 半径 OA
  \draw[blue, very thick] (O) -- (A);
  \node[below, font=\footnotesize, blue] at (0.5,-0.03) {半径 $r$};
  % 半径 OB
  \draw[blue, very thick] (O) -- (B);
  % 弧 AB（突出显示弧长 = r）
  \draw[red, very thick] (1,0) arc (0:57.2958:1);
  \node[red, font=\footnotesize, right] at ({1.02*cos(28.6)}, {1.02*sin(28.6)+0.04}) {弧长 $=r$};
  % 点标
  \node[below right, font=\footnotesize] at (A) {$A$};
  \node[above right, font=\footnotesize] at (B) {$B$};
  \node[below left, font=\footnotesize] at (O) {$O$};
  % 提示文字
  \node[font=\footnotesize, align=center] at (-0.55,-1.1) {弧长 $=$ 半径 $\Rightarrow$ 圆心角为 $1$ 弧度};
\end{tikzpicture}
\end{document}
