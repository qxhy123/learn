% 不同 σ 的正态曲线对比（均值同为 0）
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=12cm, height=7cm,
    samples=200,
    domain=-5:5,
    xmin=-5, xmax=5,
    ymin=0, ymax=0.95,
    xtick={-4,-3,-2,-1,0,1,2,3,4},
    ytick={0,0.2,0.4,0.6,0.8},
    xlabel={$x$}, ylabel={$\varphi(x)$},
    title={同均值 $\mu=0$、不同 $\sigma$ 的正态曲线对比},
    axis lines=middle,
    legend style={at={(0.97,0.97)}, anchor=north east, font=\footnotesize},
    enlarge x limits=0.02,
]
\addplot[red, thick, domain=-3:3] {1/(0.5*sqrt(2*pi))*exp(-x^2/(2*0.5^2))};
\addlegendentry{$\sigma=0.5$（最尖）}
\addplot[blue, thick, domain=-4:4] {1/sqrt(2*pi)*exp(-x^2/2)};
\addlegendentry{$\sigma=1$}
\addplot[green!50!black, thick, domain=-5:5] {1/(2*sqrt(2*pi))*exp(-x^2/8)};
\addlegendentry{$\sigma=2$（最扁）}

\node[font=\footnotesize, align=center] at (axis cs:-2.3, 0.55)
  {$\sigma$ 越小\\ 曲线越尖};
\draw[->, gray] (axis cs:-1.5, 0.55) -- (axis cs:-0.6, 0.75);
\end{axis}
\end{tikzpicture}
\end{document}
