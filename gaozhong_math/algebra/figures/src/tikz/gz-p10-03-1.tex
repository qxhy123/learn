% gz-p10-03-1: 单调性与 f' 符号——上图 y=f(x) 先增后减；下图 y=f'(x) 正变负
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc}

\begin{document}
\begin{tikzpicture}
% 上图: y = f(x) = -x^3/3 + x + 1（在 x=-1 和 x=1 处极值）
% 此处简化：用 f(x) = -(x-1)^2 + 4，先增后减，顶点 x=1
\begin{axis}[
  name=top,
  width=10cm, height=5cm,
  axis lines=center,
  xlabel={$x$}, ylabel={$y$},
  every axis x label/.style={at={(axis cs:3.4,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,4.6)}, anchor=south},
  xmin=-1.5, xmax=3.4,
  ymin=-0.5, ymax=4.6,
  xtick={1},
  xticklabels={$x_0$},
  ytick=\empty,
  tick style={color=black},
  ticklabel style={font=\small},
  clip=true,
]
  % 曲线
  \addplot[black, thick, domain=-1.2:3.2, samples=80] {-(x-1)^2 + 4};
  \node[right, black] at (axis cs:2.6, 0.5) {$y=f(x)$};

  % 顶点
  \addplot[only marks, mark=*, mark size=2pt, red] coordinates {(1,4)};
  \node[above, red, font=\small] at (axis cs:1,4) {极大值};

  % 虚线连接 x_0
  \draw[gray, dashed] (axis cs:1,4) -- (axis cs:1,0);

  % 增减标注
  \node[font=\footnotesize, blue] at (axis cs:-0.3, 1.8) {$f$ 增};
  \node[font=\footnotesize, blue] at (axis cs:2.3, 1.8) {$f$ 减};
\end{axis}

% 下图: y = f'(x) = -2(x-1) = -2x + 2
\begin{axis}[
  at={($(top.south)+(0,-1.0cm)$)},
  anchor=north,
  width=10cm, height=4cm,
  axis lines=center,
  xlabel={$x$}, ylabel={$y$},
  every axis x label/.style={at={(axis cs:3.4,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,4.2)}, anchor=south},
  xmin=-1.5, xmax=3.4,
  ymin=-3.2, ymax=4.2,
  xtick={1},
  xticklabels={$x_0$},
  ytick=\empty,
  tick style={color=black},
  ticklabel style={font=\small},
  clip=true,
]
  % 导数曲线
  \addplot[red, thick, domain=-1.2:3.2, samples=2] {-2*x + 2};
  \node[right, red] at (axis cs:2.6, -2.5) {$y=f'(x)$};

  % 零点
  \addplot[only marks, mark=*, mark size=2pt, red] coordinates {(1,0)};

  % 虚线连接
  \draw[gray, dashed] (axis cs:1,0) -- (axis cs:1,4);

  % 符号标注
  \node[font=\footnotesize, blue] at (axis cs:-0.3, 2.0) {$f'>0$};
  \node[font=\footnotesize, blue] at (axis cs:2.3, -1.5) {$f'<0$};
\end{axis}
\end{tikzpicture}
\end{document}
