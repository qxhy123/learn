% gz-p8-02-1: 几何概型——大圆 + 中心小圆，P = 小圆面积 / 大圆面积
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}

\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  % 大圆（半径 R = 2.5），蓝色虚线
  \fill[blue!10] (0, 0) circle (2.5);
  \draw[blue, thick] (0, 0) circle (2.5);
  \node[font=\small, blue] at (-1.8, 1.8) {$\Omega$};

  % 小圆（半径 r = 1.2），红色填充
  \fill[red!40] (0, 0) circle (1.2);
  \draw[red, thick] (0, 0) circle (1.2);
  \node[font=\small, red] at (0, 0) {$A$};

  % 标注半径
  \draw[gray, dashed, thin] (0, 0) -- (2.5, 0);
  \node[above, font=\footnotesize, gray] at (1.85, 0) {$R$};

  \draw[gray, dashed, thin] (0, 0) -- ({1.2*cos(120)}, {1.2*sin(120)});
  \node[font=\footnotesize, gray] at ({0.6*cos(120) - 0.1}, {0.6*sin(120) + 0.2}) {$r$};

  % 中心点
  \filldraw (0, 0) circle (1.2pt);

  % 公式
  \node[font=\small, below] at (0, -3.0)
    {$P(A)=\dfrac{\text{小圆面积}}{\text{大圆面积}}=\dfrac{\pi r^2}{\pi R^2}=\dfrac{r^2}{R^2}$};
\end{tikzpicture}
\end{document}
