% gz-p7-02-1: 频率分布直方图——6 组，组距 5，纵轴=频率/组距
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  % 坐标轴
  \draw[-{Stealth[length=4pt]}] (0, 0) -- (10.5, 0) node[right, font=\small] {成绩};
  \draw[-{Stealth[length=4pt]}] (0, 0) -- (0, 5.5) node[above, font=\small] {频率/组距};

  % 6 组数据：组距 5，频率/组距 分别为 0.010, 0.025, 0.040, 0.055, 0.045, 0.025
  % 用条形高度 ×100 显示
  % 区间 [50,55), [55,60), [60,65), [65,70), [70,75), [75,80)
  \def\bw{1.5}  % 条宽

  % 第 1 组 [50, 55), h=1.0
  \draw[fill=blue!20] (0.5, 0) rectangle (0.5+\bw, 1.0);
  % 第 2 组 [55, 60), h=2.5
  \draw[fill=blue!30] (0.5+\bw, 0) rectangle (0.5+2*\bw, 2.5);
  % 第 3 组 [60, 65), h=4.0
  \draw[fill=blue!40] (0.5+2*\bw, 0) rectangle (0.5+3*\bw, 4.0);
  % 第 4 组 [65, 70), h=5.0 (最高)
  \draw[fill=red!50] (0.5+3*\bw, 0) rectangle (0.5+4*\bw, 5.0);
  % 第 5 组 [70, 75), h=4.0
  \draw[fill=blue!40] (0.5+4*\bw, 0) rectangle (0.5+5*\bw, 4.0);
  % 第 6 组 [75, 80), h=2.5
  \draw[fill=blue!30] (0.5+5*\bw, 0) rectangle (0.5+6*\bw, 2.5);

  % x 轴刻度
  \foreach \i/\v in {0/50, 1/55, 2/60, 3/65, 4/70, 5/75, 6/80} {
    \draw (0.5+\i*\bw, 0) -- (0.5+\i*\bw, -0.1);
    \node[below, font=\footnotesize] at (0.5+\i*\bw, -0.1) {\v};
  }

  % y 轴刻度
  \foreach \v in {1, 2, 3, 4, 5} {
    \draw (0, \v) -- (-0.1, \v);
    \node[left, font=\footnotesize] at (-0.1, \v) {0.0\v};
  }

  % 顶部数值标注
  \node[above, font=\tiny, blue] at (0.5+0.5*\bw, 1.0) {0.01};
  \node[above, font=\tiny, blue] at (0.5+1.5*\bw, 2.5) {0.025};
  \node[above, font=\tiny, blue] at (0.5+2.5*\bw, 4.0) {0.04};
  \node[above, font=\tiny, red] at (0.5+3.5*\bw, 5.0) {0.05};
  \node[above, font=\tiny, blue] at (0.5+4.5*\bw, 4.0) {0.04};
  \node[above, font=\tiny, blue] at (0.5+5.5*\bw, 2.5) {0.025};

  % 标题
  \node[font=\small] at (5, -1.0) {组距 $=5$，频率 $=$ 频率/组距 $\times$ 组距};
\end{tikzpicture}
\end{document}
