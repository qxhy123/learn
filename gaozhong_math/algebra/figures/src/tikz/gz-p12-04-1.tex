% 标准正态分布 N(0,1) 钟形曲线，3σ 区间三色阴影
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=12cm, height=7cm,
    samples=200,
    domain=-4:4,
    xmin=-4, xmax=4,
    ymin=0, ymax=0.45,
    xtick={-3,-2,-1,0,1,2,3},
    xticklabels={$-3\sigma$,$-2\sigma$,$-\sigma$,$0$,$\sigma$,$2\sigma$,$3\sigma$},
    ytick={0,0.1,0.2,0.3,0.4},
    xlabel={$x$}, ylabel={$\varphi(x)$},
    title={标准正态分布 $N(0,1)$ 的 $3\sigma$ 区间},
    axis lines=middle,
    enlarge x limits=0.02,
]
% 三层阴影：从外到内
\addplot[fill=blue!10, draw=none, domain=-3:3] {1/sqrt(2*pi)*exp(-x^2/2)} \closedcycle;
\addplot[fill=blue!25, draw=none, domain=-2:2] {1/sqrt(2*pi)*exp(-x^2/2)} \closedcycle;
\addplot[fill=blue!50, draw=none, domain=-1:1] {1/sqrt(2*pi)*exp(-x^2/2)} \closedcycle;
% 钟形曲线
\addplot[blue!70!black, thick, domain=-4:4] {1/sqrt(2*pi)*exp(-x^2/2)};

% 标注概率
\node[font=\footnotesize] at (axis cs:0, 0.18) {$68.27\%$};
\node[font=\footnotesize] at (axis cs:1.55, 0.07) {$95.45\%$};
\node[font=\footnotesize] at (axis cs:2.55, 0.025) {$99.73\%$};
\end{axis}
\end{tikzpicture}
