\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, angles, quotes, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.6, thick]
  % a, b 取值
  \def\ax{3}
  \def\by{2}
  \pgfmathsetmacro{\Rval}{sqrt(\ax*\ax + \by*\by)}
  \pgfmathsetmacro{\phideg}{atan2(\by,\ax)}
  % 坐标轴
  \draw[->, gray] (-0.8,0) -- (4.0,0) node[right, black] {$x$};
  \draw[->, gray] (0,-0.8) -- (0,2.8) node[above, black] {$y$};
  % 向量 (a, b)
  \draw[blue, very thick, -{Stealth[length=3mm]}] (0,0) -- (\ax,\by);
  % 投影
  \draw[gray, dashed] (\ax,\by) -- (\ax,0);
  \draw[gray, dashed] (\ax,\by) -- (0,\by);
  % 长度 R 标注
  \node[above left, font=\footnotesize, blue] at ({\ax/2},{\by/2+0.05}) {$R=\sqrt{a^2+b^2}$};
  % phi 角弧
  \draw[red, thick] (0.6,0) arc (0:\phideg:0.6);
  \node[red, font=\footnotesize] at ({0.85*cos(\phideg/2)}, {0.85*sin(\phideg/2)}) {$\varphi$};
  % 标 a, b
  \node[below, font=\footnotesize] at ({\ax/2},0) {$a$};
  \node[left, font=\footnotesize] at (0,{\by/2}) {$b$};
  % 终点
  \fill[blue] (\ax,\by) circle (0.05);
  \node[above right, font=\footnotesize] at (\ax,\by) {$(a,b)$};
  % 原点
  \node[below left, font=\footnotesize] at (0,0) {$O$};
  % 直角符号
  \draw (\ax,0.18) -- ({\ax-0.18},0.18) -- ({\ax-0.18},0);
  % 公式
  \node[font=\footnotesize, align=center] at (2.0,-0.55) {$a\sin x + b\cos x = R\sin(x+\varphi)$，其中 $\tan\varphi=\tfrac{b}{a}$};
\end{tikzpicture}
\end{document}
