% gz-p8-05-1: 全概率/贝叶斯树状图——根 → 袋1/袋2 → 红/白，4 条路径
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  scale=1.0, thick,
  every node/.style={font=\small},
]
  % 根节点
  \node (root) at (0, 0) {开始};

  % 第一层：袋 1 / 袋 2
  \node (b1) at (-3.5, -2.5) {袋 1};
  \node (b2) at (3.5, -2.5) {袋 2};

  % 第二层：红 / 白
  \node (r1) at (-5.0, -5.0) {红};
  \node (w1) at (-2.0, -5.0) {白};
  \node (r2) at (2.0, -5.0) {红};
  \node (w2) at (5.0, -5.0) {白};

  % 第一层边（概率）
  \draw[-{Stealth[length=4pt]}, blue, thick] (root) -- (b1)
    node[midway, above left, font=\footnotesize, blue] {$P(B_1)=\frac{1}{2}$};
  \draw[-{Stealth[length=4pt]}, blue, thick] (root) -- (b2)
    node[midway, above right, font=\footnotesize, blue] {$P(B_2)=\frac{1}{2}$};

  % 第二层边
  \draw[-{Stealth[length=4pt]}, red, thick] (b1) -- (r1)
    node[midway, left, font=\footnotesize, red] {$\frac{3}{5}$};
  \draw[-{Stealth[length=4pt]}, red, thick] (b1) -- (w1)
    node[midway, right, font=\footnotesize, red] {$\frac{2}{5}$};
  \draw[-{Stealth[length=4pt]}, red, thick] (b2) -- (r2)
    node[midway, left, font=\footnotesize, red] {$\frac{2}{4}$};
  \draw[-{Stealth[length=4pt]}, red, thick] (b2) -- (w2)
    node[midway, right, font=\footnotesize, red] {$\frac{2}{4}$};

  % 路径概率
  \node[font=\footnotesize, below] at (r1.south) {$\frac{1}{2}\cdot\frac{3}{5}=\frac{3}{10}$};
  \node[font=\footnotesize, below] at (w1.south) {$\frac{1}{2}\cdot\frac{2}{5}=\frac{2}{10}$};
  \node[font=\footnotesize, below] at (r2.south) {$\frac{1}{2}\cdot\frac{1}{2}=\frac{1}{4}$};
  \node[font=\footnotesize, below] at (w2.south) {$\frac{1}{2}\cdot\frac{1}{2}=\frac{1}{4}$};

  % 标题/说明
  \node[font=\footnotesize, align=left] at (-6, -1) {选袋};
  \node[font=\footnotesize, align=left] at (-6, -3.7) {摸球};
\end{tikzpicture}
\end{document}
