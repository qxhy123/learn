\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=12cm, height=9cm,
    xlabel={$x$}, ylabel={$y$},
    xmin=-3.2, xmax=3.2,
    ymin=-0.6, ymax=8.4,
    axis lines=middle,
    xtick={-3,-2,-1,1,2,3},
    ytick={1,2,4,6,8},
    enlargelimits=false,
    tick label style={font=\footnotesize},
    label style={font=\small},
    legend style={font=\small, at={(0.62,0.97)}, anchor=north, draw=black!40, fill=white, fill opacity=0.85, text opacity=1},
    legend cell align=left,
    clip=false,
]
% 水平渐近线 x 轴（不入图例）
\addplot[gray, dashed, domain=-3.2:3.2, samples=2, forget plot]{0};
% y=2^x  (a>1 递增)
\addplot[red, very thick, domain=-3.2:3, samples=150]{2^x};
\addlegendentry{$y=2^x\ (a>1)$}
% y=e^x  (a>1 递增, 更快)
\addplot[blue, very thick, domain=-3.2:2.12, samples=150]{e^x};
\addlegendentry{$y=e^x\ (a=e)$}
% y=(1/2)^x  (0<a<1 递减)
\addplot[teal, very thick, domain=-3:3.2, samples=150]{(1/2)^x};
\addlegendentry{$y=(\tfrac12)^x\ (0<a<1)$}
% 公共点 (0,1)
\addplot[only marks, mark=*, mark size=2.2pt, black] coordinates {(0,1)};
\node[anchor=north west, font=\footnotesize] at (axis cs:0.08,1.0) {$(0,1)$};
% 标注：恒过 (0,1)
\node[fill=black!8, fill opacity=0.85, text opacity=1, draw=black!40, rounded corners, inner sep=3pt, font=\footnotesize, align=center]
    at (axis cs:1.15,3.0) {全部恒过 $(0,1)$};
\draw[black!45, ->, thin] (axis cs:1.15,2.55) -- (axis cs:0.12,1.05);
% 标注：恒正 + 渐近线
\node[fill=gray!12, fill opacity=0.85, text opacity=1, draw=gray!55, rounded corners, inner sep=3pt, font=\footnotesize, align=center]
    at (axis cs:-1.9,0.95) {$a^x>0$ 恒正\\ $x$ 轴为水平渐近线};
\end{axis}
\end{tikzpicture}
\end{document}
