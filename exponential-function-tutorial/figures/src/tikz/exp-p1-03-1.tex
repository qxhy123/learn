\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=12cm, height=9cm,
    xlabel={$x$}, ylabel={$y$},
    xmin=-3.4, xmax=3.4,
    ymin=-0.6, ymax=8.6,
    axis lines=middle,
    xtick={-3,-2,-1,1,2,3},
    ytick={1,2,4,6,8},
    enlargelimits=false,
    tick label style={font=\footnotesize},
    label style={font=\small},
    legend style={font=\small, at={(0.98,0.97)}, anchor=north east, draw=black!40, fill=white, fill opacity=0.85, text opacity=1},
    legend cell align=left,
    clip=false,
]
% y=2^x
\addplot[red, very thick, domain=-3.4:3, samples=150]{2^x};
\addlegendentry{$y=2^x$}
% y=2^{-x} = (1/2)^x  对称副本
\addplot[blue, very thick, dashed, domain=-3:3.4, samples=150]{2^(-x)};
\addlegendentry{$y=2^{-x}=(\tfrac12)^x$}
% y 轴对称镜像虚线（强调）
\addplot[gray!60, dotted, domain=-0.6:8.6, samples=2] coordinates {(0,-0.6) (0,8.6)};
% 对称点对：x=2 处
\addplot[only marks, mark=*, mark size=2pt, red] coordinates {(2,4)};
\addplot[only marks, mark=*, mark size=2pt, blue] coordinates {(-2,4)};
\draw[gray, <->, thin] (axis cs:-2,4) -- (axis cs:2,4);
\node[anchor=south, font=\scriptsize, red] at (axis cs:2,4) {$(2,\,2^2)$};
\node[anchor=south, font=\scriptsize, blue] at (axis cs:-2,4) {$(-2,\,2^2)$};
% 公共点
\addplot[only marks, mark=*, mark size=2pt, black] coordinates {(0,1)};
\node[anchor=north, font=\footnotesize] at (axis cs:-0.55,0.72) {$(0,1)$};
% 标注：关于 y 轴对称
\node[fill=blue!8, fill opacity=0.85, text opacity=1, draw=blue!45, rounded corners, inner sep=3pt, font=\footnotesize, align=center]
    at (axis cs:-1.55,6.6) {$2^{-x}=(\tfrac12)^x$\\ 与 $2^x$ 关于 $y$ 轴对称};
\end{axis}
\end{tikzpicture}
\end{document}
