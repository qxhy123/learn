\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
% ===== 左图：a>1 (底数2) 2^x>4 => x>2 =====
\begin{axis}[
  name=left,
  width=9cm,height=8cm,
  axis lines=middle,
  title={$a>1$：$2^{x}>4 \iff x>2$},
  xlabel={$x$},ylabel={$y$},
  xmin=-1.2,xmax=4.4,ymin=-1.0,ymax=9,
  xtick={-1,1,3,4},ytick={2,4,6,8},
  enlargelimits=false,clip=true,samples=200,domain=-1.2:3.05,
]
% y=2^x
\addplot[very thick,blue,domain=-1.2:3.05]{2^x};
\addlegendentry{$y=2^{x}$}
% y=4
\addplot[very thick,red,dashed,domain=-1.2:4.4]{4};
\addlegendentry{$y=4$}
% 满足 2^x>4 的曲线段加粗高亮 (x>2)
\addplot[ultra thick,blue!55!green,domain=2:3.05]{2^x};
% 交点
\addplot[only marks,mark=*,mark size=2.4pt,black] coordinates {(2,4)};
\draw[gray,dashed] (axis cs:2,4) -- (axis cs:2,0);
\node[below,font=\small] at (axis cs:2,-0.05) {$x=2$};
\end{axis}

% 左图数轴：x>2 阴影
\begin{scope}[shift={($(left.south)+(0,-1.7cm)$)}]
  \draw[->,thick] (-1.5,0) -- (4.0,0) node[right]{$x$};
  \foreach \x in {-1,0,1,2,3} \draw (\x,0.08)--(\x,-0.08) node[below,font=\small]{$\x$};
  % 阴影区间 x>2
  \fill[blue!55!green,opacity=0.30] (2,-0.18) rectangle (4.0,0.18);
  \draw[very thick,blue!55!green] (2,0)--(3.95,0);
  \draw[fill=white,very thick,blue!55!green] (2,0) circle (0.07);
  \node[above,font=\small] at (3.0,0.2) {解集 $x>2$};
\end{scope}

% ===== 右图：0<a<1 (底数1/2) (1/2)^x>4 => x<-2 (不等号反向) =====
\begin{axis}[
  name=right,
  at={($(left.east)+(3.2cm,0)$)},anchor=west,
  width=9cm,height=8cm,
  axis lines=middle,
  title={$0<a<1$：$\left(\tfrac12\right)^{x}>4 \iff x<-2$},
  xlabel={$x$},ylabel={$y$},
  xmin=-4.4,xmax=1.2,ymin=-1.0,ymax=9,
  xtick={-4,-3,-1,1},ytick={2,4,6,8},
  enlargelimits=false,clip=true,samples=200,domain=-3.05:1.2,
]
% y=(1/2)^x
\addplot[very thick,orange,domain=-3.05:1.2]{(1/2)^x};
\addlegendentry{$y=\left(\tfrac12\right)^{x}$}
% y=4
\addplot[very thick,red,dashed,domain=-4.4:1.2]{4};
\addlegendentry{$y=4$}
% 满足部分高亮 x<-2
\addplot[ultra thick,purple,domain=-3.05:-2]{(1/2)^x};
\addplot[only marks,mark=*,mark size=2.4pt,black] coordinates {(-2,4)};
\draw[gray,dashed] (axis cs:-2,4) -- (axis cs:-2,0);
\node[below,font=\small] at (axis cs:-2,-0.05) {$x=-2$};
\node[draw=black,fill=yellow!18,opacity=0.9,text opacity=1,rounded corners,
      align=left,font=\small] at (axis cs:-2.9,7.4)
  {底数 $<1$ 递减\\ 不等号反向};
\end{axis}

% 右图数轴：x<-2 阴影
\begin{scope}[shift={($(right.south)+(0,-1.7cm)$)}]
  \draw[->,thick] (-4.0,0) -- (1.5,0) node[right]{$x$};
  \foreach \x in {-4,-3,-2,-1,0,1} \draw (\x,0.08)--(\x,-0.08) node[below,font=\small]{$\x$};
  \fill[purple,opacity=0.28] (-4.0,-0.18) rectangle (-2,0.18);
  \draw[very thick,purple] (-3.95,0)--(-2,0);
  \draw[fill=white,very thick,purple] (-2,0) circle (0.07);
  \node[above,font=\small] at (-3.0,0.2) {解集 $x<-2$};
\end{scope}
\end{tikzpicture}
\end{document}
