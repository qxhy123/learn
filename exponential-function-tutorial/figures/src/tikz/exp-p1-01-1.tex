\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=12cm, height=9cm,
    xlabel={$x$}, ylabel={$y$},
    xmin=-0.4, xmax=6.4,
    ymin=-2, ymax=66,
    axis lines=left,
    xtick={0,1,2,3,4,5,6},
    ytick={0,16,32,48,64},
    enlargelimits=false,
    tick label style={font=\footnotesize},
    label style={font=\small},
    legend style={font=\small, at={(0.03,0.97)}, anchor=north west, draw=black!40, fill=white, fill opacity=0.85, text opacity=1},
    clip=false,
]
% 线性：重复相加 y=2x
\addplot[blue, very thick, domain=0:6, samples=2]{2*x};
\addlegendentry{$y=2x$（重复相加）}
% 指数：重复乘法 y=2^x（连续曲线）
\addplot[red, very thick, domain=0:6, samples=120]{2^x};
\addlegendentry{$y=2^x$（重复乘法）}
% 指数离散整点
\addplot[only marks, red, mark=*, mark size=2pt, domain=0:6, samples=7]{2^x};
% 关键标注：差距拉开
\node[fill=red!12, fill opacity=0.85, text opacity=1, draw=red!50, rounded corners, inner sep=3pt, font=\footnotesize, align=center]
    at (axis cs:3.7,52) {重复乘法\\远快于重复相加};
% 离散整点数值标注
\node[font=\scriptsize, red, anchor=south] at (axis cs:6,64) {$2^6=64$};
\node[font=\scriptsize, red, anchor=west] at (axis cs:5,32) {$2^5=32$};
\node[font=\scriptsize, blue, anchor=north west] at (axis cs:6,12) {$2\times6=12$};
\node[font=\scriptsize, anchor=south west] at (axis cs:0,1) {$2^0=1$};
\end{axis}
\end{tikzpicture}
\end{document}
