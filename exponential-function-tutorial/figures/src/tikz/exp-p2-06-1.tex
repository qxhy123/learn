\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=13cm, height=10cm,
  xmin=0, xmax=4.3,
  ymin=0, ymax=9,
  axis lines=left,
  xlabel={时间 $t$}, ylabel={数量 $N$},
  xlabel style={at={(ticklabel* cs:1)}, anchor=west},
  ylabel style={at={(ticklabel* cs:1)}, anchor=south},
  xtick={0,1,2,3,4},
  ytick={1,2,4,8},
  tick label style={font=\small},
  samples=200,
  clip=true,
  legend cell align=left,
  legend style={font=\small, draw=black!40, fill=white, fill opacity=0.85, text opacity=1, at={(0.97,0.55)}, anchor=east},
]

% 增长 y=2^t
\addplot[very thick, red!75!black, domain=0:3.13] {2^x};
\addlegendentry{增长 $y=2^{t}$（每过 $1$ 翻倍）}

% 衰减 y=2^{-t} = (1/2)^t
\addplot[very thick, blue!70!black, domain=0:4.3] {2^(-x)};
\addlegendentry{衰减 $y=2^{-t}=\left(\tfrac12\right)^{t}$（每过 $1$ 减半）}

% 共同起点 (0,1)
\addplot[only marks, mark=*, mark size=2.2pt, black] coordinates {(0,1)};
\node[anchor=south west, font=\small] at (axis cs:0.05,1.05) {$N_0=1$};

% --- 翻倍时间标注 doubling time: t=0->1, 值 1->2 ---
\addplot[red!75!black, only marks, mark=*, mark size=2pt] coordinates {(1,2)};
\draw[red!60!black, densely dashed] (axis cs:1,0) -- (axis cs:1,2);
\draw[red!60!black, densely dashed] (axis cs:0,2) -- (axis cs:1,2);
\node[red!70!black, font=\small, anchor=south west, fill=white, fill opacity=0.8, text opacity=1, inner sep=1.5pt]
  at (axis cs:1.05,2.35) {翻倍时间 $T_d=1$};

% --- 半衰期标注 half-life: t=0->1, 值 1->1/2 ---
\addplot[blue!70!black, only marks, mark=*, mark size=2pt] coordinates {(1,0.5)};
\draw[blue!60!black, densely dashed] (axis cs:0,0.5) -- (axis cs:1,0.5);
\node[blue!70!black, font=\small, anchor=west, fill=white, fill opacity=0.8, text opacity=1, inner sep=1.5pt]
  at (axis cs:1.55,0.62) {半衰期 $T_{1/2}=1$};

% 后续翻倍点 (2,4),(3,8) 提示快速上升
\addplot[red!75!black, only marks, mark=*, mark size=1.6pt] coordinates {(2,4) (3,8)};
\node[red!70!black, font=\small, anchor=south east] at (axis cs:2.9,8.05) {$(3,8)$};
\node[red!70!black, font=\small, anchor=west] at (axis cs:2.05,4.1) {$(2,4)$};

\end{axis}
\end{tikzpicture}
\end{document}
