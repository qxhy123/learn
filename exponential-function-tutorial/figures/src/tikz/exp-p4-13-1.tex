\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=12.5cm, height=9cm,
    xlabel={$t$}, ylabel={$y$},
    xmin=0, xmax=4, ymin=0, ymax=8,
    samples=200,
    axis lines=left,
    grid=major, grid style={gray!18},
    tick label style={font=\small},
    label style={font=\small},
    legend pos=south east, legend cell align=left,
]
% 同一 k=0.5 的解曲线族 y = C e^{0.5 t}，不同 C
\addplot[red,        very thick, domain=0:4] {0.5*exp(0.5*x)};
\addlegendentry{$C=0.5$}
\addplot[orange!90!black, very thick, domain=0:4] {1.0*exp(0.5*x)};
\addlegendentry{$C=1$}
\addplot[green!55!black,  very thick, domain=0:4] {1.5*exp(0.5*x)};
\addlegendentry{$C=1.5$}
\addplot[blue,       very thick, domain=0:4] {2.2*exp(0.5*x)};
\addlegendentry{$C=2.2$}
% 起点标记 (t=0, y=C)
\addplot[only marks, mark=*, mark size=1.6pt, black] coordinates {(0,0.5)(0,1.0)(0,1.5)(0,2.2)};
% 曲线右端标签
\node[red, anchor=west, font=\small] at (axis cs:4.02,3.7) {$y=0.5\,e^{0.5t}$};
\node[orange!90!black, anchor=west, font=\small] at (axis cs:4.02,7.4) {$y=e^{0.5t}$};
% 注释框
\node[draw=blue!55, fill=blue!7, fill opacity=0.85, text opacity=1,
      rounded corners, align=left, font=\small, inner sep=4pt, anchor=north west]
  at (axis cs:0.15,7.7)
  {方程 $y'=ky$（取 $k=0.5$）\\
   通解为 $y=Ce^{kt}$\\[2pt]
   不同 $C$ 给出一族解曲线\\
   变化率 $y'=ky$ 始终正比于自身};
\end{axis}
\end{tikzpicture}
\end{document}
