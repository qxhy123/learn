\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=11cm, height=11cm,
  xmin=-2.3, xmax=5.3,
  ymin=-2.3, ymax=5.3,
  axis equal image=true,
  axis lines=middle,
  xlabel={$x$}, ylabel={$y$},
  xlabel style={at={(ticklabel* cs:1)}, anchor=west},
  ylabel style={at={(ticklabel* cs:1)}, anchor=south},
  xtick={-2,-1,1,2,3,4,5},
  ytick={-2,-1,1,2,3,4,5},
  tick label style={font=\small},
  samples=200,
  clip=true,
  legend cell align=left,
  legend style={font=\small, draw=black!40, fill=white, fill opacity=0.85, text opacity=1, at={(0.97,0.03)}, anchor=south east},
]

% 对称轴 y=x
\addplot[gray, densely dashed, thick, domain=-2.3:5.3] {x};
\addlegendentry{$y=x$（对称轴）}

% 指数函数 y=2^x  (限制 ymax 内)
\addplot[very thick, blue!70!black, domain=-2.3:2.32] {2^x};
\addlegendentry{$y=2^{x}$（指数函数）}

% 对数函数 y=log_2 x  domain x>0
\addplot[very thick, red!75!black, domain=0.0625:5.3, samples=250] {ln(x)/ln(2)};
\addlegendentry{$y=\log_2 x$（反函数）}

% 对称点对
\addplot[only marks, mark=*, mark size=2pt, blue!70!black] coordinates {(0,1) (1,2)};
\addplot[only marks, mark=*, mark size=2pt, red!75!black] coordinates {(1,0) (2,1)};

% 连接对称点对的细虚线（垂直于 y=x）
\addplot[gray!70, densely dotted] coordinates {(0,1) (1,0)};
\addplot[gray!70, densely dotted] coordinates {(1,2) (2,1)};

% 标注
\node[anchor=south, font=\small, blue!70!black] at (axis cs:0,1.05) {$(0,1)$};
\node[anchor=west, font=\small, red!75!black] at (axis cs:1.08,-0.1) {$(1,0)$};
\node[anchor=south east, font=\small, blue!70!black] at (axis cs:1.0,2.1) {$(1,2)$};
\node[anchor=west, font=\small, red!75!black] at (axis cs:2.1,1.0) {$(2,1)$};

% 竖直渐近线 x=0 提示（沿 y 轴负方向，红色 log 曲线一侧）
\node[red!60!black, font=\small, anchor=west, fill=white, fill opacity=0.8, text opacity=1, inner sep=1.5pt] at (axis cs:0.12,-1.7) {$\log_2 x$ 渐近线 $x=0$};

\end{axis}
\end{tikzpicture}
\end{document}
