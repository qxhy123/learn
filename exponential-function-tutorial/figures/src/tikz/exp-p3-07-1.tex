\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=12cm,height=9cm,
  axis lines=middle,
  xlabel={$x$},ylabel={$y$},
  xlabel style={at={(axis description cs:1,0.08)},anchor=west},
  ylabel style={at={(axis description cs:0.5,1)},anchor=south},
  xmin=-3.2,xmax=3.4,
  ymin=-1.2,ymax=8.5,
  xtick={-3,-2,-1,1,3},
  ytick={1,2,4,6,8},
  enlargelimits=false,
  clip=true,
  samples=200,
  domain=-3.2:3.05,
]
% 指数曲线 y=2^x
\addplot[very thick,blue,domain=-3.2:3.05]{2^x};
\addlegendentry{$y=2^{x}$}
% 水平线 y=5（b=5）
\addplot[very thick,red,dashed,domain=-3.2:3.4]{5};
\addlegendentry{$y=b\ (b=5)$}
% 交点：2^x=5 => x=log2(5)≈2.3219, y=5
\addplot[only marks,mark=*,mark size=2.6pt,black] coordinates {(2.3219,5)};
% 竖直虚线投影到 x 轴
\draw[gray,dashed] (axis cs:2.3219,5) -- (axis cs:2.3219,0);
\node[below] at (axis cs:2.3219,-0.1) {$x_0=\log_2 5$};
% 唯一交点标注框（半透明，避开峰值区域）
\node[draw=black,fill=yellow!18,opacity=0.92,text opacity=1,
      rounded corners,align=left,font=\small]
  at (axis cs:-1.55,6.4)
  {单调递增 $\Rightarrow$\\ 与水平线 $y=b$\\ 恰有一个交点\\ $\Rightarrow$ 唯一解 $x_0$};
\node[above right,font=\small,black] at (axis cs:2.35,5.05) {唯一交点};
\end{axis}
\end{tikzpicture}
\end{document}
