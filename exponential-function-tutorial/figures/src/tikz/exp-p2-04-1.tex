\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=13cm, height=10cm,
  xmin=-3.2, xmax=3.2,
  ymin=-0.6, ymax=8.5,
  axis lines=middle,
  xlabel={$x$}, ylabel={$y$},
  xlabel style={at={(ticklabel* cs:1)}, anchor=west},
  ylabel style={at={(ticklabel* cs:1)}, anchor=south},
  xtick={-3,-2,-1,1,2,3},
  ytick={1,2,4,6,8},
  tick label style={font=\small},
  samples=200,
  domain=-3.2:3.2,
  clip=true,
  legend cell align=left,
  legend style={font=\small, draw=black!40, fill=white, fill opacity=0.85, text opacity=1, at={(0.03,0.97)}, anchor=north west},
]

% y=3^x 比 2^x 更陡
\addplot[very thick, red!75!black, domain=-3.2:1.95] {3^x};
\addlegendentry{$y=3^{x}$（更陡，$a$ 越大越快）}

% y=2^x 基准 增函数
\addplot[very thick, blue!70!black, domain=-3.2:3.1] {2^x};
\addlegendentry{$y=2^{x}$（$a>1$ 递增）}

% y=2^{x-1} 向右平移1
\addplot[very thick, teal, dashed, domain=-2.2:3.2] {2^(x-1)};
\addlegendentry{$y=2^{\,x-1}$（右移 $1$）}

% y=(1/2)^x 减函数
\addplot[very thick, orange!90!black, domain=-3.0:3.2] {(1/2)^x};
\addlegendentry{$y=\left(\tfrac12\right)^{x}$（$0<a<1$ 递减）}

% 公共定点 (0,1)
\addplot[only marks, mark=*, mark size=2.2pt, black] coordinates {(0,1)};
\node[anchor=north west, font=\small] at (axis cs:0.08,1.0) {$(0,1)$ 公共定点};

% 渐近线 y=0
\addplot[gray, densely dotted, thick, domain=-3.2:3.2] {0};
\node[gray, font=\small, anchor=north east, fill=white, fill opacity=0.75, text opacity=1, inner sep=1.5pt] at (axis cs:3.1,-0.05) {渐近线 $y=0$};

\end{axis}
\end{tikzpicture}
\end{document}
