\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,angles,quotes,arrows.meta,positioning,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=12cm,height=11cm,
  axis lines=middle,
  axis equal image,
  xlabel={$x_1$},ylabel={$x_2$},
  xlabel style={at={(axis description cs:1,0.5)},anchor=west},
  ylabel style={at={(axis description cs:0.5,1)},anchor=south},
  xmin=-3.2,xmax=3.2,
  ymin=-3.2,ymax=3.2,
  xtick={-3,-2,-1,1,2,3},
  ytick={-3,-2,-1,1,2,3},
  enlargelimits=false,
  clip=true,
  samples=400,
  domain=0:6.2,
  variable=t,
]
% 系统 x' = A x, A = [[-0.35, -1],[1, -0.35]]
% 特征值 -0.35 ± i  => 收敛螺旋（稳定焦点）
% 解：x(t) = e^{-0.35 t} * R(t) x0，绕原点旋转并收敛
% 三条不同初值的轨迹（参数化）
% 轨迹1：初值半径 3，相位 0
\addplot[very thick,blue,domain=0:6.2]
  ({3*exp(-0.35*t)*cos(deg(t))},{3*exp(-0.35*t)*sin(deg(t))});
% 轨迹2：初值半径 3，相位 120 度
\addplot[very thick,red!80!black,domain=0:6.2]
  ({3*exp(-0.35*t)*cos(deg(t)+120)},{3*exp(-0.35*t)*sin(deg(t)+120)});
% 轨迹3：初值半径 3，相位 240 度
\addplot[very thick,green!55!black,domain=0:6.2]
  ({3*exp(-0.35*t)*cos(deg(t)+240)},{3*exp(-0.35*t)*sin(deg(t)+240)});
% 起点标记（t=0）
\addplot[only marks,mark=*,mark size=2pt,blue] coordinates {(3,0)};
\addplot[only marks,mark=*,mark size=2pt,red!80!black]
  coordinates {({3*cos(120)},{3*sin(120)})};
\addplot[only marks,mark=*,mark size=2pt,green!55!black]
  coordinates {({3*cos(240)},{3*sin(240)})};
% 平衡点（原点，收敛目标）
\addplot[only marks,mark=*,mark size=2.6pt,black] coordinates {(0,0)};
\node[below right,font=\footnotesize] at (axis cs:0.05,-0.05) {平衡点 $\mathbf 0$};
% 方向箭头：在轨迹1中段加一个箭头示意流向
\draw[->,blue,line width=1.2pt]
  (axis cs:{3*exp(-0.35*0.78)*cos(deg(0.78))},{3*exp(-0.35*0.78)*sin(deg(0.78))})
  -- (axis cs:{3*exp(-0.35*0.95)*cos(deg(0.95))},{3*exp(-0.35*0.95)*sin(deg(0.95))});
% 起点标注
\node[right,font=\footnotesize] at (axis cs:3,0) {$\mathbf x_0$};
% 半透明说明框，置于左下空白处
\node[draw=black,fill=yellow!18,opacity=0.9,text opacity=1,
      rounded corners,align=left,font=\footnotesize]
  at (axis cs:-1.95,-2.35)
  {$\mathbf x'=A\mathbf x,\ \ \mathbf x(t)=e^{At}\mathbf x_0$\\[1pt]
   特征值 $\lambda=-0.35\pm i$\\
   实部 $<0\Rightarrow$ 螺旋收敛\\
   虚部 $\neq0\Rightarrow$ 旋转振荡};
\end{axis}
\end{tikzpicture}
\end{document}
