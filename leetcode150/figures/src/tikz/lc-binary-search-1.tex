% LC 35 Binary Search — 标准二分迭代演化
% 数组 [1,3,5,7,9,11,13]，target=9
% 3步收敛：Step1 mid<target→l右移，Step2 mid>target→r左移，Step3 mid=target命中

\begin{document}
\begin{tikzpicture}

\def\cellsize{0.75}
\newcommand{\drawcell}[4]{%
  \node[draw, minimum size=\cellsize cm, inner sep=0pt,
        fill=#4, font=\small\bfseries] at (#1,#2) {#3};
}

%% ── 标题 ──────────────────────────────────────────────────
\node[font=\large\bfseries] at (3.5, 0.7)
  {LC 35：标准二分搜索（Binary Search）};
\node[font=\small,gray] at (3.5, 0.25)
  {数组 $[1,3,5,7,9,11,13]$，target = 9};
\node[font=\scriptsize,gray] at (3.5,-0.05)
  {颜色：绿 = l（左边界），蓝 = mid，红 = r（右边界），灰 = 已排除};

%% ═══════════════════════════════════════════════════════════
%% Step 1：l=0, r=6, mid=3, nums[3]=7 < 9 → l=mid+1=4
%% ═══════════════════════════════════════════════════════════
\begin{scope}[yshift=-0.8cm]
  \node[font=\small\bfseries,gray!70,anchor=east] at (-0.2,0) {Step 1};
  \drawcell{0.00}{0}{ 1}{green!10}
  \drawcell{0.75}{0}{ 3}{white}
  \drawcell{1.50}{0}{ 5}{white}
  \drawcell{2.25}{0}{ 7}{blue!15}
  \drawcell{3.00}{0}{ 9}{white}
  \drawcell{3.75}{0}{11}{white}
  \drawcell{4.50}{0}{13}{red!10}
  \foreach \i in {0,...,6}{
    \node[font=\tiny,gray] at (\i*0.75,-0.50) {\i};
  }
  \draw[->,>=Stealth,green!60!black,thick](0.00,-0.82)--(0.00,-0.42);
  \node[font=\scriptsize,green!60!black,below] at (0.00,-0.87) {l=0};
  \draw[->,>=Stealth,blue!70!black,thick](2.25,-0.82)--(2.25,-0.42);
  \node[font=\scriptsize,blue!70!black,below] at (2.25,-0.87) {mid=3};
  \draw[->,>=Stealth,red!70!black,thick](4.50,-1.15)--(4.50,-0.42);
  \node[font=\scriptsize,red!70!black,below] at (4.50,-1.20) {r=6};
  \node[draw=gray!50,fill=white,thin,rounded corners=2pt,
        font=\scriptsize,inner sep=4pt,align=left]
    at (7.3,-0.55)
    {mid=(0+6)/2=\textbf{3}\\
     nums[3]=7 $<$ target=9\\
     $\Rightarrow$ l = mid+1 = 4};
\end{scope}

%% ═══════════════════════════════════════════════════════════
%% Step 2：l=4, r=6, mid=5, nums[5]=11 > 9 → r=mid-1=4
%% ═══════════════════════════════════════════════════════════
\begin{scope}[yshift=-3.3cm]
  \node[font=\small\bfseries,gray!70,anchor=east] at (-0.2,0) {Step 2};
  \drawcell{0.00}{0}{ 1}{gray!15}
  \drawcell{0.75}{0}{ 3}{gray!15}
  \drawcell{1.50}{0}{ 5}{gray!15}
  \drawcell{2.25}{0}{ 7}{gray!15}
  \drawcell{3.00}{0}{ 9}{green!10}
  \drawcell{3.75}{0}{11}{blue!15}
  \drawcell{4.50}{0}{13}{red!10}
  \foreach \i in {0,...,6}{
    \node[font=\tiny,gray] at (\i*0.75,-0.50) {\i};
  }
  \draw[->,>=Stealth,green!60!black,thick](3.00,-0.82)--(3.00,-0.42);
  \node[font=\scriptsize,green!60!black,below] at (3.00,-0.87) {l=4};
  \draw[->,>=Stealth,blue!70!black,thick](3.75,-0.82)--(3.75,-0.42);
  \node[font=\scriptsize,blue!70!black,below] at (3.75,-0.87) {mid=5};
  \draw[->,>=Stealth,red!70!black,thick](4.50,-1.15)--(4.50,-0.42);
  \node[font=\scriptsize,red!70!black,below] at (4.50,-1.20) {r=6};
  \node[draw=gray!50,fill=white,thin,rounded corners=2pt,
        font=\scriptsize,inner sep=4pt,align=left]
    at (7.3,-0.55)
    {mid=(4+6)/2=\textbf{5}\\
     nums[5]=11 $>$ target=9\\
     $\Rightarrow$ r = mid-1 = 4};
\end{scope}

%% ═══════════════════════════════════════════════════════════
%% Step 3：l=4, r=4, mid=4, nums[4]=9 == 9 → 命中！
%% ═══════════════════════════════════════════════════════════
\begin{scope}[yshift=-5.8cm]
  \node[font=\small\bfseries,gray!70,anchor=east] at (-0.2,0) {Step 3};
  \drawcell{0.00}{0}{ 1}{gray!15}
  \drawcell{0.75}{0}{ 3}{gray!15}
  \drawcell{1.50}{0}{ 5}{gray!15}
  \drawcell{2.25}{0}{ 7}{gray!15}
  \drawcell{3.00}{0}{ 9}{blue!15}
  \drawcell{3.75}{0}{11}{gray!15}
  \drawcell{4.50}{0}{13}{gray!15}
  \foreach \i in {0,...,6}{
    \node[font=\tiny,gray] at (\i*0.75,-0.50) {\i};
  }
  % l、mid、r 三者重合，稍微偏移显示
  \draw[->,>=Stealth,green!60!black,thick](2.85,-0.82)--(3.00,-0.42);
  \node[font=\scriptsize,green!60!black,below] at (2.70,-0.87) {l=4};
  \draw[->,>=Stealth,blue!70!black,thick](3.00,-0.82)--(3.00,-0.42);
  \node[font=\scriptsize,blue!70!black,below] at (3.00,-0.87) {mid=4};
  \draw[->,>=Stealth,red!70!black,thick](3.15,-0.82)--(3.00,-0.42);
  \node[font=\scriptsize,red!70!black,below] at (3.30,-0.87) {r=4};
  % 命中绿框
  \draw[green!60!black,very thick,rounded corners=3pt]
    (2.625,-0.42) rectangle (3.375,0.42);
  \node[font=\small\bfseries,green!60!black] at (3.00,0.65)
    {命中！返回 4};
  \node[draw=gray!50,fill=white,thin,rounded corners=2pt,
        font=\scriptsize,inner sep=4pt,align=left]
    at (7.3,-0.55)
    {mid=(4+4)/2=\textbf{4}\\
     nums[4]=9 $=$ target=9\\
     $\Rightarrow$ \textbf{返回 mid = 4}};
\end{scope}

%% ── 图例 ──────────────────────────────────────────────────
\begin{scope}[yshift=-8.1cm]
  \node[draw,minimum size=0.5cm,fill=green!10,inner sep=0pt] at (0.0,0) {};
  \node[font=\scriptsize,right] at (0.35,0) {l 左边界（绿）};
  \node[draw,minimum size=0.5cm,fill=blue!15,inner sep=0pt] at (3.2,0) {};
  \node[font=\scriptsize,right] at (3.55,0) {mid 中点（蓝）};
  \node[draw,minimum size=0.5cm,fill=red!10,inner sep=0pt]  at (6.4,0) {};
  \node[font=\scriptsize,right] at (6.75,0) {r 右边界（红）};
  \node[draw,minimum size=0.5cm,fill=gray!15,inner sep=0pt] at (9.6,0) {};
  \node[font=\scriptsize,right] at (9.95,0) {已排除范围};
\end{scope}

\end{tikzpicture}
