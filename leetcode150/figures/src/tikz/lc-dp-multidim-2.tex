%% ── Layout constants ──────────────────────────────────────
% Table: cs=1.1, rs=0.88, ox=0.5 (left margin), oy=-0.8 (top margin)

%   BUT cases env makes box ~7cm wide => right edge ~5.9
%   => no conflict with table (table ends at 4.70) -- GOOD
% The dashed separator and boxes below table are fine as long as
%   formula box center x + half-width < canvas right edge.
%   Canvas right edge determined by rightmost element = right-side labels
%   "替换（对角）" at anchor=west x=4.90 => ~6.5cm wide string => right ~6.5
%   Formula box: center 2.4 + half-width ~3.5 => right ~5.9
%   => canvas right edge ~6.5, formula box fits.

\def\cs{1.1}
\def\rs{0.88}
\def\ox{0.5}
\def\oy{-0.8}

%% ── 标题 ──────────────────────────────────────────────────
% oy=-0.8; col headers now at oy+0.65=-0.15, so title at +0.55 and +1.2
\node[font=\large\bfseries] at (2.4, 1.3)
  {编辑距离（LC 72 Edit Distance）};
\node[font=\small, gray] at (2.4, 0.7)
  {"horse" $\to$ "ros"，dp[i][j] = 最少操作数（插入/删除/替换）};

%% ── 列标签（word2: ∅ r o s）──────────────────────────────
% Character bold at oy+0.65, j-index tiny at oy+0.38
% Table top edge at oy=-0.80; j-index is 0.38cm above => clear
\foreach \j/\ch in {0/$\varnothing$, 1/r, 2/o, 3/s} {

  \node[font=\tiny, gray]     at (\ox+\j*\cs+0.45, \oy+0.38) {j=\j};
}

%% ── 行标签（word1: ∅ h o r s e）──────────────────────────
\foreach \i/\ch in {0/$\varnothing$, 1/h, 2/o, 3/r, 4/s, 5/e} {

%% ── dp 表格数值 ────────────────────────────────────────────

%% ── 转移公式框（居中于表格 x 范围，即 x=2.4）
% cases env 约 6.5cm 宽 => right edge ≈ 2.4+3.25=5.65 < 7.0 safe
\node[draw=gray!50, fill=white, rounded corners=3pt, font=\small,
      inner sep=8pt, align=left]
  at (2.4, -7.9)
  {\textbf{转移公式}\quad
   若 \texttt{word1[i]}$=$\texttt{word2[j]}：
   $\;\text{dp}[i][j]=\text{dp}[i\!-\!1][j\!-\!1]$\\[5pt]
   否则取三种操作最小值：\\[2pt]
   $\quad\text{dp}[i][j]=1+\min\!\begin{cases}
     \text{dp}[i\!-\!1][j]       & \text{（}\uparrow\text{删除）}\\
     \text{dp}[i][j\!-\!1]       & \text{（}\leftarrow\text{插入）}\\
     \text{dp}[i\!-\!1][j\!-\!1] & \text{（}\nwarrow\text{替换）}
   \end{cases}$};

%% ── 结果框
\node[draw=gray!50, fill=white, rounded corners=3pt, font=\small,
      inner sep=6pt, align=left]
  at (2.4, -11.2)
  {\textbf{完整 dp 表（"horse"$\to$"ros"）：}\\[3pt]
   \quad$j{=}0$: [0,1,2,3,4,5]\quad
   $j{=}1$: [1,1,2,2,3,4]\quad
   $j{=}2$: [2,2,1,2,3,4]\quad
   $j{=}3$: [3,3,2,2,3,3]\\[4pt]
   \textbf{最终答案：}dp[5][3] $= \mathbf{3}$
   \quad（horse $\to$ rorse $\to$ rose $\to$ ros）\\[3pt]
   \textbf{复杂度：}时间 $O(m{\times}n)$，空间 $O(m{\times}n)$ 或 $O(n)$ 滚动数组};
