\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}
\usetikzlibrary{calc,positioning,arrows.meta}

% LC 3 Longest Substring Without Repeating Characters — 可变滑窗
% 字符串 "abcabcbb"，5 个快照纵向排列

\begin{document}
\begin{tikzpicture}

\def\cellsize{0.8}  % 格子边长
\def\cs{0.85}       % 格子间距（cellstep）

%% ── 辅助：画一个字符格子 ─────────────────────────────────
% \charcell{x}{y}{char}{fillcolor}
\newcommand{\charcell}[4]{%
  \node[draw, minimum size=\cellsize cm, inner sep=0pt,
        fill=#4, font=\small\bfseries] at (#1,#2) {#3};
}

%% ── 标题 ─────────────────────────────────────────────────
\node[font=\large\bfseries] at (4.0, 0.8)
  {LC 3：可变滑窗（Longest Substring Without Repeating）};
\node[font=\small, gray] at (4.0, 0.3)
  {字符串 ``abcabcbb''，蓝色窗 [l, r]，逐步扩张/收缩};

%% 字符串数据（下标 0..7）
%% a  b  c  a  b  c  b  b
%% 0  1  2  3  4  5  6  7

%% ═════════════════════════════════════════════════════════
%% 快照 0：l=0, r=2，窗 [a,b,c]，无重复，max_len=3
%% ═════════════════════════════════════════════════════════
\begin{scope}[yshift=-0.5cm]
  \charcell{0*\cs}{0}{a}{blue!15}
  \charcell{1*\cs}{0}{b}{blue!15}
  \charcell{2*\cs}{0}{c}{blue!15}
  \charcell{3*\cs}{0}{a}{white}
  \charcell{4*\cs}{0}{b}{white}
  \charcell{5*\cs}{0}{c}{white}
  \charcell{6*\cs}{0}{b}{white}
  \charcell{7*\cs}{0}{b}{white}
  % 下标
  \foreach \i in {0,...,7}{
    \node[font=\tiny,gray] at (\i*\cs,-0.55) {\i};
  }
  % 蓝色窗框
  \draw[blue!60!black,thick,rounded corners=2pt]
    (-0.43,-0.43) rectangle (2.13,0.43);
  % l 和 r 指针
  \draw[->,>=Stealth,blue!80!black,thick] (0,-0.9) -- (0,-0.5);
  \node[font=\scriptsize,blue!80!black,below] at (0,-0.95) {l=0};
  \draw[->,>=Stealth,blue!80!black,thick] (2*\cs,-0.9) -- (2*\cs,-0.5);
  \node[font=\scriptsize,blue!80!black,below] at (2*\cs,-0.95) {r=2};
  % 注释框
  \node[draw=gray!50,fill=white,thin,rounded corners=2pt,
        font=\scriptsize,inner sep=4pt,align=left]
    at (8.5,-0.3)
    {窗 = ``abc''（无重复）\\
     max\_len = \textbf{3}};
\end{scope}

%% ═════════════════════════════════════════════════════════
%% 快照 1：r=3，遇到重复 'a'（窗内已有 a at 0）→ 收缩
%% ═════════════════════════════════════════════════════════
\begin{scope}[yshift=-2.7cm]
  \charcell{0*\cs}{0}{a}{red!15}   % 重复字符（旧）
  \charcell{1*\cs}{0}{b}{blue!15}
  \charcell{2*\cs}{0}{c}{blue!15}
  \charcell{3*\cs}{0}{a}{red!20}   % 重复字符（新，r 指向）
  \charcell{4*\cs}{0}{b}{white}
  \charcell{5*\cs}{0}{c}{white}
  \charcell{6*\cs}{0}{b}{white}
  \charcell{7*\cs}{0}{b}{white}
  \foreach \i in {0,...,7}{
    \node[font=\tiny,gray] at (\i*\cs,-0.55) {\i};
  }
  % 窗框（含冲突位置）
  \draw[red!60!black,thick,rounded corners=2pt]
    (-0.43,-0.43) rectangle (3*\cs+0.43,0.43);
  % 冲突标注
  \draw[red!70!black,thick,<->,>=Stealth]
    (0,0.52) -- (3*\cs,0.52);
  \node[font=\tiny,red!70!black] at (1.5*\cs,0.72) {重复 'a'！};
  \draw[->,>=Stealth,blue!80!black,thick] (0,-0.9) -- (0,-0.5);
  \node[font=\scriptsize,blue!80!black,below] at (0,-0.95) {l=0};
  \draw[->,>=Stealth,blue!80!black,thick] (3*\cs,-0.9) -- (3*\cs,-0.5);
  \node[font=\scriptsize,blue!80!black,below] at (3*\cs,-0.95) {r=3};
  \node[draw=gray!50,fill=white,thin,rounded corners=2pt,
        font=\scriptsize,inner sep=4pt,align=left]
    at (8.5,-0.3)
    {r=3 遇重复 'a'（窗内 index 0）\\
     $\Rightarrow$ l++ 移动到 1，跳过旧 'a'};
\end{scope}

%% ═════════════════════════════════════════════════════════
%% 快照 2：l=1, r=3，窗 [b,c,a]，无重复，max_len 仍 3
%% ═════════════════════════════════════════════════════════
\begin{scope}[yshift=-5.4cm]
  \charcell{0*\cs}{0}{a}{gray!15}  % 已排除
  \charcell{1*\cs}{0}{b}{blue!15}
  \charcell{2*\cs}{0}{c}{blue!15}
  \charcell{3*\cs}{0}{a}{blue!15}
  \charcell{4*\cs}{0}{b}{white}
  \charcell{5*\cs}{0}{c}{white}
  \charcell{6*\cs}{0}{b}{white}
  \charcell{7*\cs}{0}{b}{white}
  \foreach \i in {0,...,7}{
    \node[font=\tiny,gray] at (\i*\cs,-0.55) {\i};
  }
  \draw[blue!60!black,thick,rounded corners=2pt]
    (\cs-0.43,-0.43) rectangle (3*\cs+0.43,0.43);
  \draw[->,>=Stealth,blue!80!black,thick] (\cs,-0.9) -- (\cs,-0.5);
  \node[font=\scriptsize,blue!80!black,below] at (\cs,-0.95) {l=1};
  \draw[->,>=Stealth,blue!80!black,thick] (3*\cs,-0.9) -- (3*\cs,-0.5);
  \node[font=\scriptsize,blue!80!black,below] at (3*\cs,-0.95) {r=3};
  \node[draw=gray!50,fill=white,thin,rounded corners=2pt,
        font=\scriptsize,inner sep=4pt,align=left]
    at (8.5,-0.3)
    {收缩后窗 = ``bca''（无重复）\\
     max\_len = \textbf{3}（未更新）};
\end{scope}

%% ═════════════════════════════════════════════════════════
%% 快照 3：r=4，遇重复 'b'（窗内 index 1）→ l 移到 2
%% ═════════════════════════════════════════════════════════
\begin{scope}[yshift=-8.1cm]
  \charcell{0*\cs}{0}{a}{gray!15}
  \charcell{1*\cs}{0}{b}{red!15}   % 旧重复
  \charcell{2*\cs}{0}{c}{blue!15}
  \charcell{3*\cs}{0}{a}{blue!15}
  \charcell{4*\cs}{0}{b}{red!20}   % 新 r 位置（重复）
  \charcell{5*\cs}{0}{c}{white}
  \charcell{6*\cs}{0}{b}{white}
  \charcell{7*\cs}{0}{b}{white}
  \foreach \i in {0,...,7}{
    \node[font=\tiny,gray] at (\i*\cs,-0.55) {\i};
  }
  \draw[red!60!black,thick,rounded corners=2pt]
    (\cs-0.43,-0.43) rectangle (4*\cs+0.43,0.43);
  \draw[red!70!black,thick,<->,>=Stealth]
    (\cs,0.52) -- (4*\cs,0.52);
  \node[font=\tiny,red!70!black] at (2.5*\cs,0.72) {重复 'b'！};
  \draw[->,>=Stealth,blue!80!black,thick] (\cs,-0.9) -- (\cs,-0.5);
  \node[font=\scriptsize,blue!80!black,below] at (\cs,-0.95) {l=1};
  \draw[->,>=Stealth,blue!80!black,thick] (4*\cs,-0.9) -- (4*\cs,-0.5);
  \node[font=\scriptsize,blue!80!black,below] at (4*\cs,-0.95) {r=4};
  \node[draw=gray!50,fill=white,thin,rounded corners=2pt,
        font=\scriptsize,inner sep=4pt,align=left]
    at (8.5,-0.3)
    {r=4 遇重复 'b'（窗内 index 1）\\
     $\Rightarrow$ l++ 移动到 2};
\end{scope}

%% ═════════════════════════════════════════════════════════
%% 快照 4：l=2, r=4，窗 [c,a,b]，max_len 仍 3（最终结果）
%% ═════════════════════════════════════════════════════════
\begin{scope}[yshift=-10.8cm]
  \charcell{0*\cs}{0}{a}{gray!15}
  \charcell{1*\cs}{0}{b}{gray!15}
  \charcell{2*\cs}{0}{c}{blue!15}
  \charcell{3*\cs}{0}{a}{blue!15}
  \charcell{4*\cs}{0}{b}{blue!15}
  \charcell{5*\cs}{0}{c}{white}
  \charcell{6*\cs}{0}{b}{white}
  \charcell{7*\cs}{0}{b}{white}
  \foreach \i in {0,...,7}{
    \node[font=\tiny,gray] at (\i*\cs,-0.55) {\i};
  }
  \draw[blue!60!black,thick,rounded corners=2pt]
    (2*\cs-0.43,-0.43) rectangle (4*\cs+0.43,0.43);
  \draw[->,>=Stealth,blue!80!black,thick] (2*\cs,-0.9) -- (2*\cs,-0.5);
  \node[font=\scriptsize,blue!80!black,below] at (2*\cs,-0.95) {l=2};
  \draw[->,>=Stealth,blue!80!black,thick] (4*\cs,-0.9) -- (4*\cs,-0.5);
  \node[font=\scriptsize,blue!80!black,below] at (4*\cs,-0.95) {r=4};
  % 最终结果框
  \node[draw=gray!50,fill=white,thin,rounded corners=2pt,
        font=\scriptsize,inner sep=4pt,align=left]
    at (8.5,-0.3)
    {窗 = ``cab''（无重复）\\
     max\_len = \textbf{3}（最终答案）};
\end{scope}

%% ── 图例 ─────────────────────────────────────────────────
\begin{scope}[yshift=-13.2cm]
  \node[draw,minimum size=0.5cm,fill=blue!15,inner sep=0pt] at (0.0,0) {};
  \node[font=\scriptsize,right] at (0.35,0) {窗内字符（无重复）};
  \node[draw,minimum size=0.5cm,fill=red!15,inner sep=0pt]  at (3.5,0) {};
  \node[font=\scriptsize,right] at (3.85,0) {重复字符（触发收缩）};
  \node[draw,minimum size=0.5cm,fill=gray!15,inner sep=0pt] at (7.0,0) {};
  \node[font=\scriptsize,right] at (7.35,0) {已排除（l 左侧）};
\end{scope}

\end{tikzpicture}
\end{document}
