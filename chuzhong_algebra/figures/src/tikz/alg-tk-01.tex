% thinking-toolkit/01 何时代入（substitution）
% 示意：已知条件形如 a+b=5, a*b=6，间接求 a^2+b^2
\begin{tikzpicture}[scale=1.1, thick]
  % 左侧：条件框
  \draw[rounded corners=4pt, fill=cyan!20, draw=black]
    (-3.5, 0.5) rectangle (-0.5, -1.5);
  \node[font=\small, align=center] at (-2, -0.5) {
    已知条件\\[4pt]
    $a + b = 5$\\
    $ab = 6$
  };

  % 箭头
  \draw[->, >=Stealth, thick] (-0.4,0) -- (0.4,0) node[midway, above, font=\small] {代入};

  % 右侧：目标框
  \draw[rounded corners=4pt, fill=orange!20, draw=black]
    (0.5, 0.5) rectangle (3.5, -1.5);
  \node[font=\small, align=center] at (2, -0.5) {
    目标\\[4pt]
    $a^2 + b^2 = ?$\\
    $= (a+b)^2 - 2ab$\\
    $= 25 - 12 = 13$
  };

  % 说明
  \node[font=\footnotesize, gray] at (0, -2.1)
    {整体代入：利用已知组合式，而非逐一求$a, b$};
\end{tikzpicture}
