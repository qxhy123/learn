\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% p10-05b: 拱桥问题示意图（坐标系中的抛物线，跨度20m，最高4m）
% 以桥洞中心为原点，y=-(1/25)x^2+4
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=9cm, height=6cm,
  axis lines=center,
  xlabel={$x$/m}, ylabel={$y$/m},
  every axis x label/.style={at={(axis cs:11,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,5.0)}, anchor=south},
  xmin=-11, xmax=11,
  ymin=-0.5, ymax=5.0,
  xtick={-10,-5,5,10},
  ytick={1,2,3,4},
  tick style={color=black},
  ticklabel style={font=\small},
  clip=false,
]
  % y = -(1/25)x^2 + 4
  \addplot[blue, thick, domain=-10:10, samples=80] {-(1/25)*x^2 + 4};
  \node[right, blue] at (axis cs:8.5, 1.5) {$y=-\dfrac{1}{25}x^2+4$};

  % 端点 (-10,0), (10,0)
  \addplot[only marks, mark=*, mark size=2.5pt] coordinates {(-10,0) (10,0)};
  \node[above] at (axis cs:-10,0) {\small$A(-10,0)$};
  \node[above] at (axis cs:10,0) {\small$B(10,0)$};

  % 顶点 (0,4)
  \addplot[only marks, mark=*, mark size=3pt, red] coordinates {(0,4)};
  \node[above right, red] at (axis cs:0,4) {最高点$(0,4)$};

  % 跨度标注
  \draw[gray, <->] (axis cs:-10,-0.4) -- (axis cs:10,-0.4) node[midway, below] {\small 跨度 = 20m};

  % 高度标注
  \draw[gray, dashed] (axis cs:0,0) -- (axis cs:0,4);
  \draw[gray, <->] (axis cs:0.5,0) -- (axis cs:0.5,4) node[midway, right] {\small 高 4m};
\end{axis}
\end{tikzpicture}
\end{document}
