\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
% 不等式解集的数轴表示：四种情形
\begin{tikzpicture}[scale=1.0, thick]

  \def\axlen{3.5}
  \def\a{1.2}  % 端点 a 位置

  %--- 情形1：x > a（空心圆，向右）---
  \begin{scope}[yshift=0cm]
    \draw[->] (-\axlen,0) -- (\axlen,0) node[right] {};
    \foreach \x in {-3,-2,-1,0,1,2,3} {
      \draw (\x*\a/1.2, 2pt) -- (\x*\a/1.2, -2pt);
    }
    % 端点空心圆
    \draw ({\a}, 0) circle (2.5pt);
    % 向右加粗射线
    \draw[very thick, blue] ({\a}, 0) -- (\axlen, 0);
    \node[below] at (\a, -3pt) {\small$a$};
    \node[left] at (-\axlen, 0) {\small $x>a$:};
    \node[right, blue] at (\axlen+0.1, 0) {\small$\longrightarrow$};
  \end{scope}

  %--- 情形2：x >= a（实心圆，向右）---
  \begin{scope}[yshift=-1.2cm]
    \draw[->] (-\axlen,0) -- (\axlen,0) node[right] {};
    \foreach \x in {-3,-2,-1,0,1,2,3} {
      \draw (\x*\a/1.2, 2pt) -- (\x*\a/1.2, -2pt);
    }
    \fill ({\a}, 0) circle (2.5pt);
    \draw[very thick, blue] ({\a}, 0) -- (\axlen, 0);
    \node[below] at (\a, -3pt) {\small$a$};
    \node[left] at (-\axlen, 0) {\small $x\geq a$:};
  \end{scope}

  %--- 情形3：x < a（空心圆，向左）---
  \begin{scope}[yshift=-2.4cm]
    \draw[->] (-\axlen,0) -- (\axlen,0) node[right] {};
    \foreach \x in {-3,-2,-1,0,1,2,3} {
      \draw (\x*\a/1.2, 2pt) -- (\x*\a/1.2, -2pt);
    }
    \draw ({\a}, 0) circle (2.5pt);
    \draw[very thick, red] (-\axlen, 0) -- ({\a}, 0);
    \node[below] at (\a, -3pt) {\small$a$};
    \node[left] at (-\axlen, 0) {\small $x<a$:};
  \end{scope}

  %--- 情形4：x <= a（实心圆，向左）---
  \begin{scope}[yshift=-3.6cm]
    \draw[->] (-\axlen,0) -- (\axlen,0) node[right] {};
    \foreach \x in {-3,-2,-1,0,1,2,3} {
      \draw (\x*\a/1.2, 2pt) -- (\x*\a/1.2, -2pt);
    }
    \fill ({\a}, 0) circle (2.5pt);
    \draw[very thick, red] (-\axlen, 0) -- ({\a}, 0);
    \node[below] at (\a, -3pt) {\small$a$};
    \node[left] at (-\axlen, 0) {\small $x\leq a$:};
  \end{scope}

  % 图例说明
  \begin{scope}[yshift=-5.0cm]
    \draw (0,0) circle (2.5pt);
    \node[right] at (0.2, 0) {\small 空心圆 = 不含端点（$>$ 或 $<$）};
  \end{scope}
  \begin{scope}[yshift=-5.7cm]
    \fill (0,0) circle (2.5pt);
    \node[right] at (0.2, 0) {\small 实心圆 = 包含端点（$\geq$ 或 $\leq$）};
  \end{scope}

\end{tikzpicture}
\end{document}
