\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% p10-01a: 抛物线开口方向对比（a>0开口向上 vs a<0开口向下）
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=8cm, height=7cm,
  axis lines=center,
  xlabel={$x$}, ylabel={$y$},
  every axis x label/.style={at={(axis cs:3.5,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,5.5)}, anchor=south},
  xmin=-3.5, xmax=3.5,
  ymin=-1.5, ymax=5.5,
  xtick={-3,-2,-1,1,2,3},
  ytick={1,2,3,4,5},
  tick style={color=black},
  ticklabel style={font=\small},
  clip=true,
]
  % a>0: y = x^2（开口向上，顶点(0,0)，最低点）
  \addplot[blue, thick, domain=-2.2:2.2, samples=80] {x^2};
  \node[right, blue] at (axis cs:1.5, 2.5) {$y=x^2$};
  \node[right, blue, font=\small] at (axis cs:1.5, 1.8) {$(a>0)$};

  % a<0: y = -x^2 + 4（开口向下，顶点(0,4)，最高点）
  \addplot[red, thick, domain=-2.2:2.2, samples=80] {-x^2+4};
  \node[left, red] at (axis cs:-0.8, 4.5) {$a<0$};

  % 顶点标注
  \addplot[only marks, mark=*, mark size=2.5pt, blue] coordinates {(0,0)};
  \node[below right, blue, font=\small] at (axis cs:0,0) {最低点};
  \addplot[only marks, mark=*, mark size=2.5pt, red] coordinates {(0,4)};
  \node[above right, red, font=\small] at (axis cs:0,4) {最高点};
\end{axis}
\end{tikzpicture}
\end{document}
