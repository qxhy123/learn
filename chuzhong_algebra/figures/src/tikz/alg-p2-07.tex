% alg-p2-07: 平方差公式几何拼图（逆用）
% a^2 - b^2 = (a+b)(a-b)
% 从 a×a 正方形中切去 b×b 角，重排成 (a+b)×(a-b) 矩形
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]

  \pgfmathsetmacro{\a}{2.5}
  \pgfmathsetmacro{\b}{1.0}
  \pgfmathsetmacro{\amb}{\a-\b}   % a-b
  \pgfmathsetmacro{\apb}{\a+\b}   % a+b

  % ---- 左图：a×a 正方形，右下角切去 b×b ----
  \begin{scope}[shift={(0,0)}]
    % a^2 大正方形（浅蓝）
    \fill[blue!15] (0,0) rectangle (\a,\a);
    % 右上角 b×b（切去，灰色）
    \fill[gray!30] (\amb,\amb) rectangle (\a,\a);

    % 外框
    \draw (0,0) rectangle (\a,\a);
    % 内部分割线
    \draw (\amb,0) -- (\amb,\a);
    \draw (0,\amb) -- (\a,\amb);

    % 标注
    \node at (\amb/2, \amb/2) {$a^2-b^2$};
    \node[gray] at (\amb+\b/2, \amb+\b/2) {$b^2$};
    \node[font=\footnotesize, gray, above] at (\amb+\b/2, \a) {（切去）};

    % 边长标注
    \draw[|<->|, thin, gray] (0,-0.35) -- node[below, font=\footnotesize] {$a$} (\a,-0.35);
    \draw[|<->|, thin, gray] (-0.35,0) -- node[left, font=\footnotesize] {$a$} (-0.35,\a);
    \node[font=\footnotesize, gray] at (\a+0.2, \amb+\b/2) {$b$};
    \node[font=\footnotesize, gray] at (\amb+\b/2, \a+0.25) {$b$};

    % 左图标题
    \node[font=\footnotesize] at (\a/2, -0.75) {$a^2$ 面积（去掉角落 $b^2$）};
  \end{scope}

  % 箭头：重排
  \draw[-{Stealth[length=6pt]}, thick, red]
    (\a+0.5, \a/2) -- node[above, font=\footnotesize, red] {重排} (\a+1.5, \a/2);

  % ---- 右图：(a+b)×(a-b) 矩形 ----
  \begin{scope}[shift={(\a+2, 0)}]
    \fill[blue!20] (0,0) rectangle (\apb,\amb);
    \draw (0,0) rectangle (\apb,\amb);

    % 标注面积
    \node at (\apb/2, \amb/2) {$(a+b)(a-b)$};

    % 边长标注
    \draw[|<->|, thin, gray] (0,-0.35) -- node[below, font=\footnotesize] {$a+b$} (\apb,-0.35);
    \draw[|<->|, thin, gray] (-0.35,0) -- node[left, font=\footnotesize] {$a-b$} (-0.35,\amb);

    % 右图标题
    \node[font=\footnotesize] at (\apb/2, -0.75) {$(a+b)(a-b)$ 矩形};
  \end{scope}

  % 公式总结
  \pgfmathsetmacro{\cx}{(\a/2 + \a + 2 + \apb/2)/2}
  \node[font=\normalsize] at (\a/2, -1.4)
    {$a^2 - b^2 = (a+b)(a-b)$};

\end{tikzpicture}
\end{document}
