% thinking-toolkit/05 整体思想示意
% 把复杂表达式当作一个整体 u，避免展开
\begin{tikzpicture}[scale=1.0, thick]
  % 示例：令 u = x+1/x，则 u^2 = x^2 + 2 + 1/x^2
  \draw[rounded corners=4pt, fill=cyan!20, draw=black]
    (-4, 1) rectangle (4, -1);
  \node[font=\small, align=center] at (0, 0.3) {令 $u = x + \dfrac{1}{x}$（整体代换）};
  \node[font=\small, align=center] at (0, -0.4)
    {$u^2 = x^2 + 2 + \dfrac{1}{x^2}\ \Rightarrow\ x^2+\dfrac{1}{x^2} = u^2 - 2$};

  \draw[->,>=Stealth,thick] (0,-1.1) -- (0,-1.9);

  \draw[rounded corners=4pt, fill=orange!20, draw=black]
    (-4, -2.0) rectangle (4, -3.5);
  \node[font=\small, align=center] at (0, -2.4) {例：若 $x+\dfrac{1}{x}=3$，求 $x^2+\dfrac{1}{x^2}$};
  \node[font=\small, align=center] at (0, -3.1)
    {$x^2+\dfrac{1}{x^2} = (x+\dfrac{1}{x})^2 - 2 = 9 - 2 = \mathbf{7}$};

  \node[font=\footnotesize, gray] at (0,-4.0) {整体思想：不展开求$x$，而是把$u=x+1/x$当整体直接运算};
\end{tikzpicture}
