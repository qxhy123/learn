\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
% 判别式 Delta 三情形：抛物线与 x 轴的位置关系
\begin{tikzpicture}[scale=1.0]

  %--- 情形1：Delta > 0，两交点 ---
  \begin{scope}[xshift=0cm]
    \begin{axis}[
      width=4.5cm, height=4.2cm,
      axis lines=center,
      xlabel={\small$x$}, ylabel={\small$y$},
      every axis x label/.style={at={(axis cs:2.6,0)},anchor=west},
      every axis y label/.style={at={(axis cs:0,3.2)},anchor=south},
      xmin=-2.5, xmax=2.6, ymin=-1.5, ymax=3.2,
      xtick=\empty, ytick=\empty,
      tick style={draw=none},
    ]
      \addplot[blue, thick, domain=-2.2:2.2, samples=80] {x^2 - 1.5};
      % 零点
      \addplot[mark=*, mark options={fill=black, scale=0.8}]
        coordinates {(-1.225,0) (1.225,0)};
      \node[below, font=\small] at (axis cs:-1.225,0) {$x_1$};
      \node[below, font=\small] at (axis cs:1.225,0) {$x_2$};
      \node[above right, red, font=\small] at (axis cs:0.5,2.5) {$\Delta>0$};
      \node[font=\small] at (axis cs:0,-1.2) {两个实数根};
    \end{axis}
  \end{scope}

  %--- 情形2：Delta = 0，切点 ---
  \begin{scope}[xshift=4.8cm]
    \begin{axis}[
      width=4.5cm, height=4.2cm,
      axis lines=center,
      xlabel={\small$x$}, ylabel={\small$y$},
      every axis x label/.style={at={(axis cs:2.6,0)},anchor=west},
      every axis y label/.style={at={(axis cs:0,3.2)},anchor=south},
      xmin=-2.5, xmax=2.6, ymin=-0.5, ymax=3.5,
      xtick=\empty, ytick=\empty,
      tick style={draw=none},
    ]
      \addplot[blue, thick, domain=-2.2:2.2, samples=80] {x^2};
      \addplot[mark=*, mark options={fill=black, scale=0.8}]
        coordinates {(0,0)};
      \node[below left, font=\small] at (axis cs:0,0) {$x_1\!=\!x_2$};
      \node[above right, red, font=\small] at (axis cs:0.5,2.5) {$\Delta=0$};
      \node[font=\small] at (axis cs:0,0.8) {重根（顶点切 $x$ 轴）};
    \end{axis}
  \end{scope}

  %--- 情形3：Delta < 0，不相交 ---
  \begin{scope}[xshift=9.6cm]
    \begin{axis}[
      width=4.5cm, height=4.2cm,
      axis lines=center,
      xlabel={\small$x$}, ylabel={\small$y$},
      every axis x label/.style={at={(axis cs:2.6,0)},anchor=west},
      every axis y label/.style={at={(axis cs:0,3.5)},anchor=south},
      xmin=-2.5, xmax=2.6, ymin=-0.5, ymax=3.8,
      xtick=\empty, ytick=\empty,
      tick style={draw=none},
    ]
      \addplot[blue, thick, domain=-2.2:2.2, samples=80] {x^2 + 1.2};
      \node[above right, red, font=\small] at (axis cs:0.5,3.0) {$\Delta<0$};
      \node[font=\small] at (axis cs:0,0.4) {无实数根};
    \end{axis}
  \end{scope}

\end{tikzpicture}
\end{document}
