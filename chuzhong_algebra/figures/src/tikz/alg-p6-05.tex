\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
% 韦达定理几何意义：x1+x2 与 x1*x2 在抛物线上的体现
\begin{tikzpicture}[scale=1.0]

  \begin{axis}[
    width=8cm, height=6cm,
    axis lines=center,
    xlabel={$x$}, ylabel={$y$},
    every axis x label/.style={at={(axis cs:3.5,0)},anchor=west},
    every axis y label/.style={at={(axis cs:0,5.5)},anchor=south},
    xmin=-3.5, xmax=3.5, ymin=-2.5, ymax=5.5,
    xtick={-3,-2,-1,0,1,2,3},
    ytick={-2,-1,0,1,2,3,4,5},
    tick style={thin},
    tick label style={font=\small},
    grid=none,
  ]
    % 抛物线 y = x^2 - x - 2 = (x-2)(x+1)，根为 x1=-1, x2=2
    \addplot[blue, thick, domain=-2.5:3.2, samples=100] {x^2 - x - 2};
    \node[right, blue] at (axis cs:2.8, 4.5) {$y=x^2-x-2$};

    % 零点
    \addplot[mark=*, mark options={fill=black, scale=1.0}]
      coordinates {(-1,0) (2,0)};
    \node[below left, font=\small] at (axis cs:-1,0) {$x_1=-1$};
    \node[below right, font=\small] at (axis cs:2,0) {$x_2=2$};

    % 对称轴 x = (x1+x2)/2 = 0.5
    \addplot[gray, dashed, thick, domain=-2.5:5.5] coordinates {(0.5,-2.5)(0.5,5.5)};
    \node[above, gray, font=\small] at (axis cs:0.5,5.5) {对称轴 $x=\dfrac{1}{2}$};

    % 顶点
    \fill[red] (axis cs:0.5,-2.25) circle (2.5pt);
    \node[right, red, font=\small] at (axis cs:0.5,-2.25)
      {顶点 $\left(\dfrac{x_1+x_2}{2},\ \cdot\right)$};

    % 韦达标注：和
    \draw[<->, red, thick] (axis cs:-1,0.5) -- (axis cs:2,0.5)
      node[midway, above, red, font=\small] {$x_1+x_2 = -\dfrac{b}{a} = 1$};

    % 韦达标注：积
    \node[below, font=\small, red] at (axis cs:0.5,-0.5)
      {$x_1 x_2 = \dfrac{c}{a} = -2$};

  \end{axis}

\end{tikzpicture}
\end{document}
