  % 音乐：144°-252°
  \fill[orange!30] (0,0) -- ({1.5*cos(144)},{1.5*sin(144)}) arc(144:252:1.5) -- cycle;
  % 阅读：252°-324°
  \fill[green!30] (0,0) -- ({1.5*cos(252)},{1.5*sin(252)}) arc(252:324:1.5) -- cycle;
  % 绘画：324°-360°
  \fill[pink!30] (0,0) -- ({1.5*cos(324)},{1.5*sin(324)}) arc(324:360:1.5) -- cycle;
  % 圆边框
  \draw[thick] (0,0) circle(1.5);
  % 分割线
  \draw[thick] (0,0) -- (1.5,0);
  \draw[thick] (0,0) -- ({1.5*cos(144)},{1.5*sin(144)});
  \draw[thick] (0,0) -- ({1.5*cos(252)},{1.5*sin(252)});
  \draw[thick] (0,0) -- ({1.5*cos(324)},{1.5*sin(324)});
  % 标签
  \node[font=\small] at ({1.9*cos(72)},{1.9*sin(72)}) {体育};
  \node[font=\small] at ({1.9*cos(72)+0.2},{1.9*sin(72)-0.3}) {40\%};
  \node[font=\small] at ({1.9*cos(198)},{1.9*sin(198)}) {音乐};
  \node[font=\small] at ({1.9*cos(198)},{1.9*sin(198)-0.3}) {30\%};
  \node[font=\small] at ({1.9*cos(288)},{1.9*sin(288)}) {阅读 20\%};
  \node[font=\small] at ({2.1*cos(342)},{2.1*sin(342)}) {绘画 10\%};
  \node[font=\small, above] at (0,1.7) {兴趣爱好扇形图};
\end{tikzpicture}
