\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% p10-02a: 顶点式 y=a(x-h)^2+k，平移关系示意
% y=x^2 平移到 y=(x-2)^2+1（顶点从原点移到(2,1)）
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=8cm, height=7cm,
  axis lines=center,
  xlabel={$x$}, ylabel={$y$},
  every axis x label/.style={at={(axis cs:4.5,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,5.0)}, anchor=south},
  xmin=-2.5, xmax=4.5,
  ymin=-0.5, ymax=5.0,
  xtick={-2,-1,1,2,3,4},
  ytick={1,2,3,4},
  tick style={color=black},
  ticklabel style={font=\small},
  clip=true,
]
  % y = x^2（实线，原函数）
  \addplot[blue, thick, domain=-2.2:2.2, samples=80] {x^2};
  \node[left, blue] at (axis cs:-1.5, 2.5) {$y=x^2$};

  % y = (x-2)^2 + 1（虚线，平移后）
  \addplot[blue, dashed, thick, domain=-0.3:4.3, samples=80] {(x-2)^2+1};
  \node[right, blue] at (axis cs:3.5, 2.5) {$y=(x-2)^2+1$};

  % 顶点标注
  \addplot[only marks, mark=*, mark size=2.5pt, blue] coordinates {(0,0)};
  \node[below right, blue, font=\small] at (axis cs:0,0) {$(0,0)$};

  \addplot[only marks, mark=*, mark size=2.5pt, blue] coordinates {(2,1)};
  \node[above right, blue, font=\small] at (axis cs:2,1) {$(2,1)$};

  % 平移箭头
  \draw[->, gray, thick] (axis cs:0.2, 0.3) -- (axis cs:1.7, 1.0) node[midway, above] {\small 右移2, 上移1};
\end{axis}
\end{tikzpicture}
\end{document}
