\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% p10-06b: Delta=0（相切，一个交点）
% y = x^2 - 4x + 4 = (x-2)^2，Delta=0，切点(2,0)
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=7cm, height=6.5cm,
  axis lines=center,
  xlabel={$x$}, ylabel={$y$},
  every axis x label/.style={at={(axis cs:4.5,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,2.5)}, anchor=south},
  xmin=-0.5, xmax=4.5,
  ymin=-0.5, ymax=2.5,
  xtick={1,2,3,4},
  ytick={1,2},
  tick style={color=black},
  ticklabel style={font=\small},
  clip=false,
]
  % y = (x-2)^2
  \addplot[blue, thick, domain=-0.3:4.3, samples=100] {(x-2)^2};
  \node[right, blue] at (axis cs:3.5, 1.2) {$\Delta=0$};
  \node[right, blue, font=\small] at (axis cs:3.5, 0.5) {相切一点};

  % 切点 (2,0)
  \addplot[only marks, mark=*, mark size=3pt, red] coordinates {(2,0)};
  \node[above right] at (axis cs:2,0) {\small$(2,0)$};
\end{axis}
\end{tikzpicture}
\end{document}
