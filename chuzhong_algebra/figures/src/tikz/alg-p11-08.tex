\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc}

\begin{document}
% part11/03 加权平均数示意（茎叶图风格可视化）
% 数据：70x2, 80x3, 90x5，加权平均=83
\begin{tikzpicture}
\begin{axis}[
  ybar,
  bar width=0.8cm,
  axis lines=left,
  xlabel={成绩（分）},
  ylabel={人数（人）},
  symbolic x coords={70,80,90},
  xtick=data,
  ymin=0, ymax=7,
  ytick={0,1,2,3,4,5},
  nodes near coords,
  nodes near coords align={vertical},
  every node near coord/.style={font=\small},
  width=7cm, height=6cm,
  enlarge x limits=0.4,
  title={加权平均数示例（$\bar{x}=83$）},
  title style={font=\small},
  xlabel style={font=\small},
  ylabel style={font=\small},
  tick label style={font=\small},
]
  \addplot[fill=cyan!30, draw=black] coordinates {
    (70,2) (80,3) (90,5)
  };
  % 加权平均线
  \draw[red, dashed, thick] (axis cs:70,0) -- (axis cs:90,0);
\end{axis}
% 计算注释
\node[font=\small, align=center] at (3.5,-0.8) {
  $\bar{x} = \frac{70\times2+80\times3+90\times5}{2+3+5} = \frac{830}{10} = 83$
};
\end{tikzpicture}
\end{document}
