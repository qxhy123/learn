\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% p10-06a: Delta三种情形（Delta>0: 两交点）
% y = x^2 - 4x + 3，Delta=4>0，交点(1,0)和(3,0)
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=7cm, height=6.5cm,
  axis lines=center,
  xlabel={$x$}, ylabel={$y$},
  every axis x label/.style={at={(axis cs:4.5,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,2.5)}, anchor=south},
  xmin=-0.5, xmax=4.5,
  ymin=-1.5, ymax=2.5,
  xtick={1,2,3,4},
  ytick={-1,1,2},
  tick style={color=black},
  ticklabel style={font=\small},
  clip=false,
]
  % y = x^2 - 4x + 3
  \addplot[blue, thick, domain=-0.3:4.3, samples=100] {x^2 - 4*x + 3};
  \node[right, blue] at (axis cs:3.8, 1.0) {$\Delta>0$};
  \node[right, blue, font=\small] at (axis cs:3.8, 0.3) {两个交点};

  % 零点 (1,0) (3,0)
  \addplot[only marks, mark=*, mark size=3pt, red] coordinates {(1,0) (3,0)};
  \node[above] at (axis cs:1,0) {\small$(1,0)$};
  \node[above] at (axis cs:3,0) {\small$(3,0)$};

  % 顶点
  \addplot[only marks, mark=*, mark size=2pt, blue] coordinates {(2,-1)};
  \node[below] at (axis cs:2,-1) {\small$(2,-1)$};
\end{axis}
\end{tikzpicture}
\end{document}
