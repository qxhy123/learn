\begin{document}
% 一元二次方程应用题：矩形面积模型
\begin{tikzpicture}[scale=1.0, thick]

  % 矩形
  \def\W{4.0}
  \def\H{2.5}
  \fill[blue!10] (0,0) rectangle (\W, \H);
  \draw (0,0) rectangle (\W, \H);

  % 标注宽 x，长 x+3
  \node[below] at (\W/2, 0) {$x + 3$（米）};
  \node[left] at (0, \H/2) {$x$（米）};
  \node at (\W/2, \H/2) {面积 $= 28$ m$^2$};

  % 列方程
  \node[below, align=center, font=\small] at (\W/2, -0.7)
    {$x(x+3) = 28$};
  \node[below, align=center, font=\small] at (\W/2, -1.2)
    {$\Rightarrow x^2 + 3x - 28 = 0$};
  \node[below, align=center, font=\small] at (\W/2, -1.7)
    {$\Rightarrow (x+7)(x-4) = 0$};
  \node[below, align=center, font=\small] at (\W/2, -2.2)
    {$\Rightarrow x = 4$（舍去负值 $x=-7$）};

\end{tikzpicture}
