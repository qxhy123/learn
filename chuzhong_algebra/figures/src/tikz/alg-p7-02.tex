\begin{document}
% 解一元一次不等式步骤示意（以 -2x + 4 > 0 为例）
\begin{tikzpicture}[
  scale=1.0, thick,
  sbox/.style={rectangle, draw=black!70, rounded corners=3pt,
               minimum width=5.5cm, minimum height=0.65cm,
               align=left, font=\small, fill=blue!5,
               text width=5.3cm, inner sep=4pt},
  arr/.style={->, >=Stealth, thick},
  node distance=0.6cm,
]

  \node[sbox] (s0) {原不等式：$-2x + 4 > 0$};
  \node[sbox, below=of s0] (s1) {移项：$-2x > -4$};
  \node[sbox, below=of s1] (s2)
    {系数化1（除以 $-2$，负数 $\Rightarrow$ \textbf{变号}）：$x < 2$};
  \node[sbox, below=of s2, fill=green!10] (s3)
    {解集：$x < 2$，即 $(-\infty,\, 2)$};

  \draw[arr] (s0) -- (s1);
  \draw[arr] (s1) -- (s2);
  \draw[arr] (s2) -- (s3);

  % 数轴展示
  \begin{scope}[yshift=-4.0cm]
    \draw[->] (-2.5,0) -- (3.5,0) node[right] {$x$};
    \foreach \x in {-2,-1,0,1,2,3} {
      \draw (\x, 2pt) -- (\x, -2pt) node[below] {\small$\x$};
    }
    \draw (2, 0) circle (2.5pt);
    \draw[very thick, red] (-2.5, 0) -- (2, 0);
    \node[below left] at (0,0) {\small$O$};
    \node[below, font=\small] at (0.5, -0.55)
      {解集数轴表示（空心圆在 $2$，向左）};
  \end{scope}

  % 警示
  \node[below, font=\small, red] at (1.2, -5.4)
    {\textbf{核心提醒}：除以负数必须变号！};

\end{tikzpicture}
\end{document}
