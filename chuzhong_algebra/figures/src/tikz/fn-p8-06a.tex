\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% p8-06a: 含参一次函数——参数a影响斜率，b影响截距，图象变化示意
% y=(2a-1)x+(a+1)，不同a值下的直线族
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=8cm, height=7cm,
  axis lines=center,
  xlabel={$x$}, ylabel={$y$},
  every axis x label/.style={at={(axis cs:3.5,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,4.5)}, anchor=south},
  xmin=-2.5, xmax=3.5,
  ymin=-2.5, ymax=4.5,
  xtick={-2,-1,1,2,3},
  ytick={-2,-1,1,2,3,4},
  tick style={color=black},
  ticklabel style={font=\small},
  clip=true,
]
  % a=1: y=(2-1)x+(1+1) = x+2（k=1,b=2）
  \addplot[blue!40, thick, domain=-2.5:2.5, samples=20] {x+2};
  \node[right, blue!60, font=\small] at (axis cs:2.2, 4.0) {$a=1$};

  % a=2: y=(4-1)x+(2+1) = 3x+3（k=3,b=3）
  \addplot[blue, thick, domain=-2.0:0.5, samples=20] {3*x+3};
  \node[right, blue, font=\small] at (axis cs:0.5, 4.5) {$a=2$};

  % a=0: y=(-1)x+(0+1)=-x+1（k=-1,b=1）
  \addplot[red, thick, domain=-2.5:3.5, samples=20] {-x+1};
  \node[right, red, font=\small] at (axis cs:2.5, -1.5) {$a=0$};
\end{axis}
\end{tikzpicture}
\end{document}
