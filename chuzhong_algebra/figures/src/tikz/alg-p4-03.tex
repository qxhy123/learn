% alg-p4-03: 分式方程解题流程图（三步走 + 验根）
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{shapes.geometric, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick,
  fbox/.style={rectangle, draw, thick, rounded corners=4pt,
               minimum width=3.4cm, minimum height=0.72cm,
               font=\footnotesize, align=center},
  sbox/.style={rectangle, draw, thick, rounded corners=4pt,
               minimum width=3.4cm, minimum height=0.72cm,
               font=\footnotesize, align=center, fill=blue!12},
  dia/.style={diamond, draw, thick, aspect=3.2,
              minimum width=3.4cm, minimum height=0.9cm,
              font=\footnotesize, align=center, fill=yellow!25},
  gbox/.style={rectangle, draw, thick, rounded corners=4pt,
               minimum width=2.6cm, minimum height=0.65cm,
               font=\footnotesize, align=center, fill=green!20},
  rbox/.style={rectangle, draw, thick, rounded corners=4pt,
               minimum width=2.6cm, minimum height=0.65cm,
               font=\footnotesize, align=center, fill=red!15},
  arr/.style={-{Stealth[length=4pt]}, thick}
]

  % 所有节点使用绝对坐标
  \node[fbox, fill=gray!20] (start) at (0, 0)   {分式方程};
  \node[sbox]               (s1)    at (0,-1.2)  {第一步：去分母\\两边同乘 LCD};
  \node[sbox]               (s2)    at (0,-2.4)  {第二步：解整式方程\\（一次或二次）};
  \node[dia]                (q1)    at (0,-3.7)  {整式方程有解？};
  \node[rbox]               (nosol) at (4.0,-3.7){整式方程无解\\原方程无解};
  \node[sbox]               (s3)    at (0,-5.0)  {第三步：验根\\代回原分母检验};
  \node[dia]                (q2)    at (0,-6.3)  {代入后分母 $=0$？};
  \node[rbox]               (exrt)  at (3.8,-6.3){增根，舍去};
  \node[gbox]               (keep)  at (-3.8,-6.3){合法根，保留};
  \node[fbox, fill=green!18](done)  at (0,-7.6)  {写出最终答案};
  \node[font=\footnotesize, red, align=center]
                            (warn)  at (0,-8.3)  {验根不可省略！};

  % 箭头
  \draw[arr] (start) -- (s1);
  \draw[arr] (s1) -- (s2);
  \draw[arr] (s2) -- (q1);
  \draw[arr] (q1) -- node[left, font=\tiny] {有} (s3);
  \draw[arr] (q1) -- node[above, font=\tiny] {无} (nosol);
  \draw[arr] (s3) -- (q2);
  \draw[arr] (q2) -- node[above, font=\tiny] {是} (exrt);
  \draw[arr] (q2) -- node[above, font=\tiny] {否} (keep);
  \draw[arr] (exrt.south) |- (done.east);
  \draw[arr] (keep.south) |- (done.west);

  % 旁注：若所有根均增根
  \node[font=\tiny, gray, align=center] at (4.8,-7.2)
    {若所有根均增根\\$\to$ 原方程无解};

\end{tikzpicture}
\end{document}
