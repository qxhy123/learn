\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta}

% p8-02d: 平移规律示意图（原函数 y=2x+1 向下平移3单位得 y=2x-2）
\begin{document}
\begin{tikzpicture}[scale=0.9, thick]
  % 坐标轴
  \draw[->] (-0.5,0) -- (3.5,0) node[right] {$x$};
  \draw[->] (0,-2.8) -- (0,3.5) node[above] {$y$};
  \node[below left] at (0,0) {$O$};

  % 刻度
  \foreach \y in {-2,-1,1,2,3} {
    \draw (2pt,\y) -- (-2pt,\y) node[left] {\small$\y$};
  }
  \foreach \x in {1,2,3} {
    \draw (\x,2pt) -- (\x,-2pt) node[below] {\small$\x$};
  }

  % 原函数 y = 2x + 1（实线蓝色）
  \draw[blue, thick] (-0.2, 0.6) -- (1.2, 3.4);
  \fill[blue] (0,1) circle (2.5pt) node[right] {\small$(0,1)$};
  \node[right, blue] at (1.0, 3.2) {$y=2x+1$};

  % 平移后 y = 2x - 2（虚线蓝色）
  \draw[blue, dashed, thick] (0,-2) -- (2.5, 3.0);
  \fill[blue] (0,-2) circle (2.5pt) node[right] {\small$(0,-2)$};
  \node[right, blue] at (2.2, 2.8) {$y=2x-2$};

  % 平移箭头（向下3单位）
  \draw[->, gray, thick] (0.1,1) -- (0.1,-2) node[midway, right] {\small 下移3};
\end{tikzpicture}
\end{document}
