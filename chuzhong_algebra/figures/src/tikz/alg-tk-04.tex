% thinking-toolkit/04 设元法示意（设未知量为x，建立方程）
\begin{tikzpicture}[scale=1.0, thick]
  % 流程框宏
  \def\bw{2.5}
  \def\bh{0.65}

  % 步骤1
  \draw[rounded corners=3pt, fill=cyan!20, draw=black]
    ({-\bw/2}, {4+\bh/2}) rectangle ({\bw/2}, {4-\bh/2});
  \node[font=\small] at (0,4) {分析：题中的未知量是什么？};

  \draw[->,>=Stealth] (0,{4-\bh/2}) -- (0,{3+\bh/2});

  % 步骤2
  \draw[rounded corners=3pt, fill=cyan!20, draw=black]
    ({-\bw/2}, {3+\bh/2}) rectangle ({\bw/2}, {3-\bh/2});
  \node[font=\small] at (0,3) {设 $x$（或 $x, y$）表示未知量};

  \draw[->,>=Stealth] (0,{3-\bh/2}) -- (0,{2+\bh/2});

  % 步骤3
  \draw[rounded corners=3pt, fill=orange!20, draw=black]
    ({-\bw/2}, {2+\bh/2}) rectangle ({\bw/2}, {2-\bh/2});
  \node[font=\small] at (0,2) {找等量关系，建立方程};

  \draw[->,>=Stealth] (0,{2-\bh/2}) -- (0,{1+\bh/2});

  % 步骤4
  \draw[rounded corners=3pt, fill=orange!20, draw=black]

  \node[font=\small] at (0,1) {解方程，求出 $x$};

  \draw[->,>=Stealth] (0,{1-\bh/2}) -- (0,{0+\bh/2});

  % 步骤5
  \draw[rounded corners=3pt, fill=green!20, draw=black]
    ({-\bw/2}, {0+\bh/2}) rectangle ({\bw/2}, {0-\bh/2});
  \node[font=\small] at (0,0) {检验 + 写出答案};

  % 右侧示例
  \node[right, font=\footnotesize, align=left] at (1.4, 2.0) {

    解：$x=7$，检验 ✓
  };
