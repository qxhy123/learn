\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% p10-06d: Delta三种情形对比总图
\begin{document}
\begin{tikzpicture}

  % ====== 左：Delta>0 ======
  \begin{scope}[xshift=-5.5cm]
  \begin{axis}[
    width=4.5cm, height=4cm,
    axis lines=center,
    xlabel={$x$}, ylabel={$y$},
    every axis x label/.style={at={(axis cs:3.2,0)}, anchor=west, font=\tiny},
    every axis y label/.style={at={(axis cs:0,2.0)}, anchor=south, font=\tiny},
    xmin=-0.5, xmax=3.2,
    ymin=-1.0, ymax=2.0,
    xtick=\empty, ytick=\empty,
    clip=true,
    title={\small $\Delta>0$},
  ]
    \addplot[blue, thick, domain=-0.2:3.2, samples=60] {x^2 - 3*x + 2};
    \addplot[only marks, mark=*, mark size=2pt, red] coordinates {(1,0) (2,0)};
  \end{axis}
  \end{scope}

  % ====== 中：Delta=0 ======
  \begin{scope}[xshift=0cm]
  \begin{axis}[
    width=4.5cm, height=4cm,
    axis lines=center,
    xlabel={$x$}, ylabel={$y$},
    every axis x label/.style={at={(axis cs:3.2,0)}, anchor=west, font=\tiny},
    every axis y label/.style={at={(axis cs:0,2.0)}, anchor=south, font=\tiny},
    xmin=-0.5, xmax=3.2,
    ymin=-0.3, ymax=2.0,
    xtick=\empty, ytick=\empty,
    clip=true,
    title={\small $\Delta=0$},
  ]
    \addplot[blue, thick, domain=-0.2:3.2, samples=60] {(x-1.5)^2};
    \addplot[only marks, mark=*, mark size=2pt, red] coordinates {(1.5,0)};
  \end{axis}
  \end{scope}

  % ====== 右：Delta<0 ======
  \begin{scope}[xshift=5.5cm]
  \begin{axis}[
    width=4.5cm, height=4cm,
    axis lines=center,
    xlabel={$x$}, ylabel={$y$},
    every axis x label/.style={at={(axis cs:3.2,0)}, anchor=west, font=\tiny},
    every axis y label/.style={at={(axis cs:0,2.0)}, anchor=south, font=\tiny},
    xmin=-0.5, xmax=3.2,
    ymin=-0.3, ymax=2.0,
    xtick=\empty, ytick=\empty,
    clip=true,
    title={\small $\Delta<0$},
  ]
    \addplot[blue, thick, domain=-0.3:3.2, samples=60] {(x-1.5)^2 + 0.5};
  \end{axis}
  \end{scope}

\end{tikzpicture}
\end{document}
