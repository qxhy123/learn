\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% p10-08b: 铅垂高法简化版（底在x轴上）
% y=x^2-2x-3，A(-1,0)，B(3,0)，P为底部动点，铅垂高=|y_P|
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=8.5cm, height=7.5cm,
  axis lines=center,
  xlabel={$x$}, ylabel={$y$},
  every axis x label/.style={at={(axis cs:4.5,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,2.5)}, anchor=south},
  xmin=-2.5, xmax=4.5,
  ymin=-4.5, ymax=2.5,
  xtick={-2,-1,1,2,3,4},
  ytick={-4,-3,-2,-1,1,2},
  tick style={color=black},
  ticklabel style={font=\small},
  clip=false,
]
  % y = x^2 - 2x - 3
  \addplot[blue, thick, domain=-1.8:3.8, samples=100] {x^2 - 2*x - 3};
  \node[right, blue] at (axis cs:3.5, 1.5) {$y=x^2-2x-3$};

  % 交点 A(-1,0), B(3,0)
  \addplot[only marks, mark=*, mark size=3pt] coordinates {(-1,0) (3,0)};
  \node[above] at (axis cs:-1,0) {$A(-1,0)$};
  \node[above] at (axis cs:3,0) {$B(3,0)$};

  % 底边 AB（红色，在x轴上）
  \draw[red, thick] (axis cs:-1,0) -- (axis cs:3,0);
  \node[above, red] at (axis cs:1.0, 0) {\small 底$AB$（水平宽=4）};

  % 顶点 P(1,-4)
  \addplot[only marks, mark=*, mark size=3pt, red] coordinates {(1,-4)};
  \node[below] at (axis cs:1,-4) {$P(1,-4)$};

  % 铅垂高（P的纵坐标绝对值）
  \draw[red, thick, <->] (axis cs:1,0) -- (axis cs:1,-4) node[right, midway, red] {铅垂高$h=|y_P|=4$};

  % 辅助
  \draw[gray, dashed] (axis cs:1,0) -- (axis cs:1,-4);
  \node[below, gray] at (axis cs:1, 0) {\small$1$};
\end{axis}
\end{tikzpicture}
\end{document}
