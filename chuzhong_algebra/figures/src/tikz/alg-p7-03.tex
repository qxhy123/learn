\begin{tikzpicture}[scale=1.0, thick]

\def\axlen{4.0}
\def\a{-0.8}  % 较小的界 a
\def\b{1.2}   % 较大的界 b

%========= 情形1：同大取大（x>a 且 x>b，a<b，解集 x>b）=========
\begin{scope}[yshift=0cm]
  \node[left, font=\small\bfseries] at (-4.8, 0.3) {情形1};
  \node[left, font=\small] at (-4.8, -0.1) {同大取大};
  % 数轴1：x > a
  \begin{scope}[yshift=0.5cm]
    \draw[->] (-\axlen,0) -- (\axlen,0);
    \draw (\a, 2pt) -- (\a, -2pt) node[below] {\small$a$};
    \draw (\b, 2pt) -- (\b, -2pt) node[below] {\small$b$};
    \draw (\a, 0) circle (2.5pt);
    \draw[very thick, blue!60] (\a, 0) -- (\axlen, 0);
    \node[left, font=\small, blue!80] at (-\axlen, 0) {$x>a$:};
  \end{scope}
  % 数轴2：x > b
  \begin{scope}[yshift=-0.1cm]
    \draw[->] (-\axlen,0) -- (\axlen,0);
    \draw (\a, 2pt) -- (\a, -2pt) node[below] {\small$a$};
    \draw (\b, 2pt) -- (\b, -2pt) node[below] {\small$b$};
    \draw (\b, 0) circle (2.5pt);
    \draw[very thick, red!60] (\b, 0) -- (\axlen, 0);
    \node[left, font=\small, red!80] at (-\axlen, 0) {$x>b$:};
  \end{scope}
  % 解集
  \begin{scope}[yshift=-0.7cm]
    \draw[->] (-\axlen,0) -- (\axlen,0);
    \draw (\a, 2pt) -- (\a, -2pt) node[below] {\small$a$};
    \draw (\b, 2pt) -- (\b, -2pt) node[below] {\small$b$};
    \draw (\b, 0) circle (2.5pt);
    \draw[very thick, green!60!black] (\b, 0) -- (\axlen, 0);
    \node[left, font=\small, green!60!black] at (-\axlen, 0) {解集$x>b$:};
  \end{scope}
\end{scope}
