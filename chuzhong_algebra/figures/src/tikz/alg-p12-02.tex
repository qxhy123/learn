\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
% part12/02 树状图（3层手工布局）
% 场景：袋中红R、白W、蓝B三球，无放回取2球
\begin{tikzpicture}[scale=1.0, thick,
  arr/.style={->, >=Stealth, thin}]

  % ====== 层1：根节点 ======
  \fill (0,0) circle(2pt);
  \node[left, font=\small] at (-0.15,0) {开始};

  % ====== 层2：第一次取球（R/W/B）======
  % R分支
  \node[font=\small] (R) at (-3,-2) {红$R$};
  \draw[arr] (0,0) -- (R) node[midway, above left, font=\footnotesize] {$\frac{1}{3}$};

  % W分支
  \node[font=\small] (W) at (0,-2) {白$W$};
  \draw[arr] (0,0) -- (W) node[midway, right, font=\footnotesize] {$\frac{1}{3}$};

  % B分支
  \node[font=\small] (B) at (3,-2) {蓝$B$};
  \draw[arr] (0,0) -- (B) node[midway, above right, font=\footnotesize] {$\frac{1}{3}$};

  % ====== 层3：第二次取球（无放回，各2种）======
  % R之后取W或B
  \node[font=\small] (RW) at (-4,-4) {白$W$};
  \node[font=\small] (RB) at (-2,-4) {蓝$B$};
  \draw[arr] (R) -- (RW) node[midway, left, font=\footnotesize] {$\frac{1}{2}$};
  \draw[arr] (R) -- (RB) node[midway, right, font=\footnotesize] {$\frac{1}{2}$};

  % W之后取R或B
  \node[font=\small] (WR) at (-1,-4) {红$R$};
  \node[font=\small] (WB) at (1,-4) {蓝$B$};
  \draw[arr] (W) -- (WR) node[midway, left, font=\footnotesize] {$\frac{1}{2}$};
  \draw[arr] (W) -- (WB) node[midway, right, font=\footnotesize] {$\frac{1}{2}$};

  % B之后取R或W
  \node[font=\small] (BR) at (2,-4) {红$R$};
  \node[font=\small] (BW) at (4,-4) {白$W$};
  \draw[arr] (B) -- (BR) node[midway, left, font=\footnotesize] {$\frac{1}{2}$};
  \draw[arr] (B) -- (BW) node[midway, right, font=\footnotesize] {$\frac{1}{2}$};

  % ====== 结果标注 ======
  \node[right, font=\footnotesize, gray] at (4.6,-4) {← 共6种结果};
  \node[font=\footnotesize, gray, right] at (-4.8,-4) {$RW$};
  \node[font=\footnotesize, gray, right] at (-2.6,-4) {$RB$};
  \node[font=\footnotesize, gray, right] at (-1.6,-4) {$WR$};
  \node[font=\footnotesize, gray, right] at (0.4,-4) {$WB$};
  \node[font=\footnotesize, gray, right] at (1.4,-4) {$BR$};
  \node[font=\footnotesize, gray, right] at (3.4,-4) {$BW$};

  % 层标签
  \node[font=\footnotesize, gray] at (5.5,0) {第0层};
  \node[font=\footnotesize, gray] at (5.5,-2) {第1次};
  \node[font=\footnotesize, gray] at (5.5,-4) {第2次};

  % 图题
  \node[font=\small] at (0,-5.0) {无放回取2球的树状图（$n=6$）};
\end{tikzpicture}
\end{document}
