\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{shapes.geometric, arrows.meta, positioning}

\tikzset{
  box/.style={rectangle, draw=black, rounded corners=3pt,
              minimum width=3.5cm, minimum height=0.65cm,
              align=center, font=\small},
  diam/.style={diamond, draw=black, aspect=2.2,
               minimum width=4cm, minimum height=1cm,
               align=center, font=\small},
  arr/.style={->, >=Stealth, thick},
}

\begin{document}
\begin{tikzpicture}[node distance=1.0cm]

  \node[box, fill=blue!10] (eq)
    {含参不等式 $ax > b$（$a$ 为参数）};

  \node[box, below=of eq] (step1)
    {移项得 $ax > c$};

  \node[diam, below=1.1cm of step1, fill=yellow!20] (q1)
    {$a$ 的符号？};

  % 三个分支
  \node[box, below left=1.4cm and 2.0cm of q1, fill=green!15] (pos)
    {$a>0$：除以正数\\不等号不变\\$x > \dfrac{c}{a}$};

  \node[box, below=1.4cm of q1, fill=orange!20] (zero)
    {$a=0$：\\不等式变为 $0>c$\\判断是否恒成立};

  \node[box, below right=1.4cm and 2.0cm of q1, fill=red!15] (neg)
    {$a<0$：除以负数\\不等号\textbf{变向}\\$x < \dfrac{c}{a}$};

  % 结论框
  \node[box, below=1.2cm of zero, fill=blue!10] (ans)
    {综合三种情形，写出完整答案};

  % 箭头
  \draw[arr] (eq) -- (step1);
  \draw[arr] (step1) -- (q1);
  \draw[arr] (q1) -- node[above left, font=\small]{$a>0$} (pos);
  \draw[arr] (q1) -- node[right, font=\small]{$a=0$} (zero);
  \draw[arr] (q1) -- node[above right, font=\small]{$a<0$} (neg);
  \draw[arr] (pos) -- (ans);
  \draw[arr] (zero) -- (ans);
  \draw[arr] (neg) -- (ans);

\end{tikzpicture}
\end{document}
