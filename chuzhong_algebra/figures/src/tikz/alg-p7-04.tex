% 不等式应用题的五步解题流程
\begin{tikzpicture}[
  scale=1.0, thick,
  flowbox/.style={rectangle, draw=black, rounded corners=3pt,
                  minimum width=5.0cm, minimum height=0.65cm,
                  align=center, font=\small, fill=blue!8},
  arr/.style={->, >=Stealth, thick},
  node distance=0.65cm,
]

  \node[flowbox] (s1) {第一步：审题，找关键词（至少/至多/不超过/不少于）};
  \node[flowbox, below=of s1] (s2) {第二步：设元，设未知量 $x$，注明单位和含义};
  \node[flowbox, below=of s2] (s3) {第三步：列不等式，用 $\geq/\leq/>/< $ 建立不等式};
  \node[flowbox, below=of s3] (s4) {第四步：解不等式，负数系数变号};
  \node[flowbox, below=of s4, fill=green!10] (s5) {第五步：检验 + 写答，核实合理性};

  \draw[arr] (s1) -- (s2);
  \draw[arr] (s2) -- (s3);
  \draw[arr] (s3) -- (s4);
  \draw[arr] (s4) -- (s5);

  % 关键词对照
  \node[below=0.8cm of s5, font=\small\bfseries] (title) {关键词对照};
  \node[below=0.2cm of title, font=\small, align=left]
    {"至少/不少于" $\to \geq$\quad "至多/不超过" $\to \leq$\quad
     "超过" $\to >$\quad "不到/小于" $\to <$};

\end{tikzpicture}
