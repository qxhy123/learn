% part11/01 数据收集示意：总体/样本/个体关系图
\begin{tikzpicture}[scale=1.0, thick]
  % 大圆：总体
  \draw[thick, fill=cyan!10] (0,0) ellipse (3.8 and 2.5);
  \node[font=\small, above] at (0,2.5) {总体（所有个体）};

  % 小圆：样本
  \draw[thick, fill=orange!20] (0,-0.2) ellipse (1.8 and 1.2);
  \node[font=\small] at (0,-0.2) {样本（抽取的部分）};

  % 个体点（散点示意总体中的个体）
  \foreach \x/\y in {-3/1.5, -2.5/-0.5, -3/-1.5, 3/1.2, 2.8/-0.8, 3.2/0.2,
                     -2/2.0, 2/2.0, 0/2.3}{
    \fill[gray!60] (\x,\y) circle(2pt);
  }

  % 标注
  \node[font=\footnotesize, gray] at (-3.0, 0.5) {个体};
  \draw[->, gray, thin] (-2.8,0.6) -- (-3.0,1.5);

  % 样本容量说明
  \node[font=\footnotesize] at (0,-1.8) {样本容量 = 抽取的个体数};

  % 双箭头（推断）
  \draw[->, thick, blue] (2.0,-0.3) to[bend left=20]
    node[right, font=\small, blue] {用样本} (3.2,-1.2);
  \draw[->, thick, blue] (3.0,-1.5) to[bend left=20]
    node[right, font=\small, blue, yshift=-4pt] {估计总体} (1.8,-0.8);
\end{tikzpicture}
