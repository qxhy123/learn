\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
% thinking-toolkit/03 判别式：三种情形
% 左：Δ>0，中：Δ=0，右：Δ<0
\begin{tikzpicture}[scale=1.0, thick]
  % ===== 情形1：Δ>0，两个根 =====
  \begin{scope}[xshift=0cm]
    \draw[->] (-1.8,0) -- (2.8,0) node[right,font=\footnotesize] {$x$};
    \draw[->] (0,-0.5) -- (0,2.8) node[above,font=\footnotesize] {$y$};
    \draw[blue,thick,domain=-1.2:2.3,samples=60] plot(\x, {(\x+0.6)*(\x-1.8)});
    \fill (-0.6,0) circle(2pt);
    \fill (1.8,0) circle(2pt);
    \node[below,font=\footnotesize] at (-0.6,0) {$x_1$};
    \node[below,font=\footnotesize] at (1.8,0) {$x_2$};
    \node[below left,font=\footnotesize] at (0,0) {$O$};
    \node[font=\small,blue,above] at (0.5,2.6) {$\Delta>0$};
    \node[font=\footnotesize,align=center] at (0.5,-1.0) {两个不等实根};
  \end{scope}

  % ===== 情形2：Δ=0，重根 =====
  \begin{scope}[xshift=5.5cm]
    \draw[->] (-1.8,0) -- (2.8,0) node[right,font=\footnotesize] {$x$};
    \draw[->] (0,-0.5) -- (0,2.8) node[above,font=\footnotesize] {$y$};
    \draw[blue,thick,domain=-1.3:2.3,samples=60] plot(\x, {(\x-0.5)^2});
    \fill (0.5,0) circle(2pt);
    \node[below,font=\footnotesize] at (0.5,0) {$x_1=x_2$};
    \node[below left,font=\footnotesize] at (0,0) {$O$};
    \node[font=\small,blue,above] at (0.8,2.6) {$\Delta=0$};
    \node[font=\footnotesize,align=center] at (0.5,-1.0) {两个相等实根};
  \end{scope}

  % ===== 情形3：Δ<0，无实根 =====
  \begin{scope}[xshift=11cm]
    \draw[->] (-1.8,0) -- (2.8,0) node[right,font=\footnotesize] {$x$};
    \draw[->] (0,-0.5) -- (0,2.8) node[above,font=\footnotesize] {$y$};
    \draw[blue,thick,domain=-1.2:2.2,samples=60] plot(\x, {(\x-0.5)^2+0.6});
    \node[below left,font=\footnotesize] at (0,0) {$O$};
    \node[font=\small,blue,above] at (0.8,2.6) {$\Delta<0$};
    \node[font=\footnotesize,align=center] at (0.5,-1.0) {无实数根};
  \end{scope}
\end{tikzpicture}
\end{document}
