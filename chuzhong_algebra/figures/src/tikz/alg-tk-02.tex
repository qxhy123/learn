\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc, arrows.meta}

\begin{document}
% thinking-toolkit/02 配方法（complete the square）
% 示意：y=x^2-4x+3，配方后 y=(x-2)^2-1，顶点(2,-1)
\begin{tikzpicture}
\begin{axis}[
  axis lines=center,
  xlabel={$x$}, ylabel={$y$},
  every axis x label/.style={at={(axis cs:4.5,0)}, anchor=west},
  every axis y label/.style={at={(axis cs:0,5.5)}, anchor=south},
  xmin=-1, xmax=5, ymin=-2, ymax=6,
  xtick={-1,0,1,2,3,4},
  ytick={-2,-1,0,1,2,3,4,5},
  tick label style={font=\small},
  width=7cm, height=7cm,
  thick,
]
  \addplot[blue, thick, domain=-0.5:4.5, samples=100] {x^2-4*x+3};
  \node[blue, right, font=\small] at (axis cs:3.5,3.5) {$y=x^2-4x+3$};
  \node[blue, right, font=\small] at (axis cs:3.5,2.5) {$=(x-2)^2-1$};

  % 顶点
  \fill[red] (axis cs:2,-1) circle(2.5pt);
  \node[below right, red, font=\small] at (axis cs:2,-1) {顶点$(2,-1)$};

  % 对称轴
  \draw[gray, dashed] (axis cs:2,-1.8) -- (axis cs:2,5.5) node[above, gray, font=\small] {$x=2$};

  % 与x轴的交点
  \fill (axis cs:1,0) circle(2pt);
  \fill (axis cs:3,0) circle(2pt);
  \node[above left, font=\small] at (axis cs:1,0) {$(1,0)$};
  \node[above right, font=\small] at (axis cs:3,0) {$(3,0)$};

  \node[below left, font=\small] at (axis cs:0,0) {$O$};
\end{axis}
\end{tikzpicture}
\end{document}
