\begin{tikzpicture}[
  node distance=0.6cm and 0.8cm,
  box/.style={rectangle, rounded corners=3pt, draw, fill=white,
              text centered, font=\small, minimum width=2.8cm, minimum height=0.7cm},
  qbox/.style={diamond, draw, fill=yellow!20, text centered, font=\small,
               minimum width=2.8cm, minimum height=0.7cm, aspect=2.5},
  ans/.style={rectangle, rounded corners=3pt, draw, fill=blue!12,
              text centered, font=\small, minimum width=3.0cm, minimum height=0.8cm},
  arr/.style={-Stealth, thick},
  lab/.style={font=\footnotesize, fill=white, inner sep=1pt}
]

% Root question
\node[qbox] (q0) {方程形式？};

% Branch 1: separable
\node[ans, below left=1.0cm and 2.8cm of q0] (sep)
  {可分离\\$y'=f(x)g(y)$\\$\to$ 分离变量};

% Branch 2: homogeneous
\node[ans, below left=1.0cm and 0.3cm of q0] (hom)
  {齐次型\\$y'=f(y/x)$\\$\to$ 令 $u=y/x$};

% Branch 3: linear
\node[ans, below right=1.0cm and 0.3cm of q0] (lin)
  {一阶线性\\$y'+p(x)y=q(x)$\\$\to$ 积分因子 $\mu$};

% Branch 4: bernoulli
\node[ans, below right=1.0cm and 2.6cm of q0] (ber)
  {Bernoulli\\$y'+py=qy^n$\\$\to$ 令 $u=y^{1-n}$};

% Branch 5: exact
\node[ans, below=3.6cm of q0] (ext)
  {全微分\\$M\,dx+N\,dy=0$\\$M_y=N_x$ $\to$ 求原函数};

% Arrows from root
\draw[arr] (q0) -- node[lab, above left]{$f(x)g(y)$} (sep);
\draw[arr] (q0) -- node[lab, left]{$f(y/x)$} (hom);
\draw[arr] (q0) -- node[lab, right]{$p(x)y+q(x)$} (lin);
\draw[arr] (q0) -- node[lab, above right]{$qy^n$ 项} (ber);
\draw[arr] (q0) -- node[lab, right]{$M\,dx+N\,dy$} (ext);

% Title
\node[font=\normalsize\bfseries] at (0,1.2) {一阶 ODE 五类识别决策树};
