\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usetikzlibrary{calc,arrows.meta}
\begin{document}
\begin{tikzpicture}[
  x={(1cm,0cm)}, y={(0cm,1cm)}, z={(-0.48cm,-0.38cm)},
  line cap=round, line join=round,
  >=Stealth
]
% 坐标轴
\draw[->] (0,0,0) -- (3.8,0,0) node[right] {$x$};
\draw[->] (0,0,0) -- (0,3.8,0) node[above] {$y$};
\draw[->] (0,0,0) -- (0,0,3.5) node[above left] {$z$};

% 球坐标参数
\def\rr{2.2}   % rho
\def\ph{35}    % phi (天顶角) degrees
\def\th{50}    % theta (方位角) degrees

% 点 P = (rho*sin(phi)*cos(theta), rho*sin(phi)*sin(theta), rho*cos(phi))
\pgfmathsetmacro{\px}{\rr*sin(\ph)*cos(\th)}
\pgfmathsetmacro{\py}{\rr*sin(\ph)*sin(\th)}
\pgfmathsetmacro{\pz}{\rr*cos(\ph)}

% 球面（部分，简化为几条纬线和经线）
\foreach \phai in {20,40,60,80} {
  \draw[blue!20]
    plot[domain=0:90, samples=20, variable=\thetai]
      ({\rr*sin(\phai)*cos(\thetai)},{\rr*sin(\phai)*sin(\thetai)},{\rr*cos(\phai)});
}
\foreach \thetai in {0,30,60,90} {
  \draw[blue!20]
    plot[domain=0:90, samples=20, variable=\phai]
      ({\rr*sin(\phai)*cos(\thetai)},{\rr*sin(\phai)*sin(\thetai)},{\rr*cos(\phai)});
}

% 向量 rho
\draw[->, thick, red] (0,0,0) -- (\px,\py,\pz) node[right] {$\rho$};
\fill[red] (\px,\py,\pz) circle (1.5pt) node[above right, font=\small] {$P$};

% 天顶角 phi（从 z 轴到 rho 的角度）
\draw[->, blue!70!black] (0,0,0.7) arc[start angle=90, delta angle=-\ph, radius=0.7]
  node[midway, right, font=\small] {$\varphi$};
% 方位角 theta（xy 平面内）
\draw[->, green!60!black]
  (0.6,0,0) arc[start angle=0, delta angle=\th, radius=0.6]
  node[midway, below, font=\small] {$\theta$};

% 投影到 xy 平面
\pgfmathsetmacro{\ppx}{\rr*sin(\ph)*cos(\th)}
\pgfmathsetmacro{\ppy}{\rr*sin(\ph)*sin(\th)}
\draw[dashed, gray] (\px,\py,\pz) -- (\ppx,\ppy,0);
\draw[dashed, gray] (0,0,0) -- (\ppx,\ppy,0);

% dV 小体积元（示意标注）
\node[font=\small, align=center, draw=gray!50, fill=gray!10, rounded corners=2pt]
  at (-0.3,2.8,2.8)
  {$dV = \rho^2\sin\varphi\,d\rho\,d\varphi\,d\theta$\\
   $\rho\in[0,R]$，$\varphi\in[0,\pi]$，$\theta\in[0,2\pi]$\\
   {\color{red}$\rho^2\sin\varphi$ 不能漏！}};

% 径向弧标注
\draw[<->, gray!70] (0,0,0) -- ({\rr*0.45*sin(\ph)*cos(\th)},{\rr*0.45*sin(\ph)*sin(\th)},{\rr*0.45*cos(\ph)});
\node[font=\small, gray!80!black] at (0.5,0.3,0.9) {$d\rho$};
\end{tikzpicture}
\end{document}
