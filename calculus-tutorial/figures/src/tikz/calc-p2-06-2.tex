\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=10cm, height=6cm,
    xmin=-0.5, xmax=4.5, ymin=-2, ymax=4,
    xlabel={$x$}, ylabel={$y$},
    samples=200, grid=both, grid style={gray!20},
]
\addplot[blue, thick, domain=0:4] {x^2 - 4*x + 2};
% 端点
\fill[red] (axis cs:0, 2) circle (2pt);
\fill[red] (axis cs:4, 2) circle (2pt);
\node[anchor=south west, red] at (axis cs:0, 2) {\small $f(a)=2$};
\node[anchor=south east, red] at (axis cs:4, 2) {\small $f(b)=2$};
% 介值 c
\draw[gray, dashed] (axis cs:0, -1) -- (axis cs:4, -1);
\node[anchor=west] at (axis cs:4, -1) {\small $C$（介值）};
% 零点
\fill[green!50!black] (axis cs:0.59, 0) circle (2pt);
\fill[green!50!black] (axis cs:3.41, 0) circle (2pt);
\node[anchor=south, font=\small, green!50!black] at (axis cs:0.59, 0.1) {$\xi_1$};
\node[anchor=south, font=\small, green!50!black] at (axis cs:3.41, 0.1) {$\xi_2$};
\node[anchor=west] at (axis cs:0.3, 3.5) {\small 介值定理：$f$ 在 $[a,b]$ 连续 $\Rightarrow$ 取遍 $f(a)$ 与 $f(b)$ 之间所有值};
\end{axis}
\end{tikzpicture}
\end{document}
