\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=11cm, height=7cm,
    xmin=-2.5, xmax=2.5, ymin=-0.5, ymax=8,
    xlabel={$x$}, ylabel={$y$},
    samples=200, grid=both, grid style={gray!20},
    legend pos=north west, legend style={font=\small},
    title={$e^x$ 的幂级数部分和逼近（$x\in[-2,2]$）},
]
\addplot[black, very thick, domain=-2.5:2.5] {exp(x)};
\addlegendentry{$e^x$（精确）};
\addplot[blue, thick, dashed, domain=-2.5:2.5] {1+x};
\addlegendentry{$S_1=1+x$};
\addplot[red, thick, dashed, domain=-2.5:2.5] {1+x+x^2/2};
\addlegendentry{$S_2=1+x+x^2/2$};
\addplot[orange, thick, dashed, domain=-2.5:2.5] {1+x+x^2/2+x^3/6};
\addlegendentry{$S_3$};
\addplot[green!60!black, thick, dashed, domain=-2.5:2.5]
    {1+x+x^2/2+x^3/6+x^4/24};
\addlegendentry{$S_4$};
\end{axis}
\end{tikzpicture}
\end{document}
