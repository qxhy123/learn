\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=10cm, height=6cm,
    xmin=0, xmax=22, ymin=-0.6, ymax=0.6,
    xlabel={$n$}, ylabel={$a_n$},
    grid=both, grid style={gray!20},
]
% 极限带 [-eps, eps]
\fill[orange!20] (axis cs:0,-0.15) rectangle (axis cs:22,0.15);

  \pgfmathsetmacro{\val}{(-1)^\n / \n}
  \pgfmathsetmacro{\nn}{int(\n)}
  \edef\temp{\noexpand\fill[blue] (axis cs:\nn,\val) circle (1.5pt);}
  \temp
}
% N 位置（约 n=7 起进入 ε 带）
\draw[red, thick, dashed] (axis cs:7,-0.6) -- (axis cs:7,0.6);
\node[anchor=south, red] at (axis cs:7, 0.5) {$N=7$};
\node[anchor=west] at (axis cs:1, 0.45) {\small $a_n=\frac{(-1)^n}{n}\to 0$};
\end{axis}
