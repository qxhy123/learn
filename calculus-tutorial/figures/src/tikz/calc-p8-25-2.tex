\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}
\begin{document}
\begin{tikzpicture}
\begin{groupplot}[
  group style={
    group size=3 by 1,
    horizontal sep=1.2cm,
  },
  width=5.2cm, height=5.0cm,
  xlabel style={font=\small},
  ylabel style={font=\small},
  tick label style={font=\footnotesize},
  axis lines=center,
  xlabel={$x_1$},
  ylabel={$x_2$},
]

% Plot 1: Positive definite Hessian => elliptic contours (bowl)
\nextgroupplot[
  title={正定 Hessian\\（唯一极小值）},
  title style={font=\small, align=center},
  xmin=-2.5, xmax=2.5,
  ymin=-2.5, ymax=2.5,
]
% Elliptic contours: 2x^2 + y^2 = c
\foreach \c in {0.5, 1.5, 3.0, 5.0}{
  \addplot[blue!60, domain=-2.3:2.3, samples=200] ({sqrt(\c/2)*cos(deg(x))}, {sqrt(\c)*sin(deg(x))});
}
\addplot[mark=*, mark size=3pt, red] coordinates {(0,0)};
\node[font=\footnotesize, below right] at (axis cs:0.1,-0.2) {全局最小};
\node[font=\footnotesize, fill=white] at (axis cs:0,2.1) {$H\succ 0$};
\node[font=\footnotesize] at (axis cs:1.5,1.8) {$f=2x_1^2+x_2^2$};

% Plot 2: Indefinite Hessian => saddle
\nextgroupplot[
  title={不定 Hessian\\（鞍点）},
  title style={font=\small, align=center},
  xmin=-2.5, xmax=2.5,
  ymin=-2.5, ymax=2.5,
]
% Hyperbolic contours: x^2 - y^2 = c (positive and negative)
\foreach \c in {0.5, 1.5, 3.0}{
  \addplot[blue!60, domain=0.1:2.2, samples=100] ({sqrt(\c + x^2)}, {x});
  \addplot[blue!60, domain=0.1:2.2, samples=100] ({-sqrt(\c + x^2)}, {x});
}
\foreach \c in {0.5, 1.5, 3.0}{
  \addplot[red!60, domain=0.1:2.2, samples=100] ({x}, {sqrt(\c + x^2)});
  \addplot[red!60, domain=0.1:2.2, samples=100] ({x}, {-sqrt(\c + x^2)});
}
\addplot[mark=square*, mark size=3pt, orange] coordinates {(0,0)};
\node[font=\footnotesize, above right] at (axis cs:0.1,0.1) {鞍点};
\node[font=\footnotesize, fill=white] at (axis cs:0,2.1) {$H$ 不定};
\node[font=\footnotesize] at (axis cs:1.6,1.8) {$f=x_1^2-x_2^2$};

% Plot 3: Positive semidefinite => ridge (flat direction)
\nextgroupplot[
  title={半正定 Hessian\\（含平坦方向）},
  title style={font=\small, align=center},
  xmin=-2.5, xmax=2.5,
  ymin=-2.5, ymax=2.5,
]
% Parabolic cylinders: x^2 = c (lines in x2 direction)
\foreach \c in {0.3, 0.8, 1.6, 2.5}{
  \addplot[blue!60] coordinates {(sqrt(\c),-2.4)(sqrt(\c),2.4)};
  \addplot[blue!60] coordinates {(-sqrt(\c),-2.4)(-sqrt(\c),2.4)};
}
% flat minimum line
\addplot[red, very thick] coordinates {(0,-2.4)(0,2.4)};
\node[font=\footnotesize, rotate=90, above] at (axis cs:0.2,0) {极小线};
\node[font=\footnotesize, fill=white] at (axis cs:0,2.1) {$H\succeq 0$};
\node[font=\footnotesize] at (axis cs:1.5,1.8) {$f=x_1^2$};

\end{groupplot}

\node[font=\normalsize\bfseries] at (9.5, 5.7) {Hessian 矩阵形态与优化景观};
\end{tikzpicture}
\end{document}
