\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usetikzlibrary{calc,3d,arrows.meta}
\begin{document}
\begin{tikzpicture}[
  x={(1cm,0cm)}, y={(0cm,1cm)}, z={(-0.5cm,-0.4cm)},
  line cap=round, line join=round,
  >=Stealth
]
% 坐标轴
\draw[->] (0,0,0) -- (4.2,0,0) node[right] {$x$};
\draw[->] (0,0,0) -- (0,4.2,0) node[above] {$y$};
\draw[->] (0,0,0) -- (0,0,3.5) node[above left] {$z$};

% 曲面网格 z = 1 + 0.3*(x-2)^2 + 0.2*(y-2)^2
\foreach \xi in {0.5,1,...,3.5} {
  \draw[blue!30, thin]
    plot[domain=0.5:3.5, samples=7, variable=\yi]
      ({\xi},{\yi},{1.2+0.3*(\xi-2)*(\xi-2)+0.2*(\yi-2)*(\yi-2)});
}
\foreach \yi in {0.5,1,...,3.5} {
  \draw[blue!30, thin]
    plot[domain=0.5:3.5, samples=7, variable=\xi]
      ({\xi},{\yi},{1.2+0.3*(\xi-2)*(\xi-2)+0.2*(\yi-2)*(\yi-2)});
}

% 点 P0 = (2, 2, z0)
\def\xz{2} \def\yz{2} \def\zz{1.2}
\fill[red] (\xz,\yz,\zz) circle (1.5pt);
\node[red, above right] at (\xz,\yz,\zz) {$P_0$};

% 截面 y = y0 (平行 xOz)
\fill[orange!20, opacity=0.6]
  (0.5,\yz,0) -- (3.5,\yz,0) -- (3.5,\yz,2.6) -- (0.5,\yz,2.6) -- cycle;
\draw[orange!70] (0.5,\yz,0) -- (3.5,\yz,0);
\draw[orange, thick]
  plot[domain=0.5:3.5, samples=20, variable=\xi]
    ({\xi},{\yz},{1.2+0.3*(\xi-2)*(\xi-2)});
\node[orange!80!black, right] at (3.5,\yz,1.9) {$z=f(x,y_0)$};

% 切线（偏导 f_x 方向）
\draw[red, very thick, ->]
  (\xz,\yz,\zz) -- ++(1,0,{0.3*2*1}) node[right] {切线斜率$=f_x$};

% 截面 x = x0 (平行 yOz)
\fill[green!20, opacity=0.5]
  (\xz,0.5,0) -- (\xz,3.5,0) -- (\xz,3.5,2.4) -- (\xz,0.5,2.4) -- cycle;
\draw[green!70] (\xz,0.5,0) -- (\xz,3.5,0);
\draw[green!60!black, thick]
  plot[domain=0.5:3.5, samples=20, variable=\yi]
    ({\xz},{\yi},{1.2+0.2*(\yi-2)*(\yi-2)});

% 切线（偏导 f_y 方向）
\draw[green!50!black, very thick, ->]
  (\xz,\yz,\zz) -- ++(0,1,{0.2*2*1}) node[above] {切线斜率$=f_y$};

\node[font=\small, align=center] at (1,0.2,3.3)
  {$f_x$：平面 $y=y_0$ 截曲面\\所得曲线的切线斜率};
\end{tikzpicture}
\end{document}
