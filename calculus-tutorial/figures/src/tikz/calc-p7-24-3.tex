\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=12cm, height=7.5cm,
  xmin=0, xmax=20,
  ymin=-12, ymax=12,
  xlabel={$t$}, ylabel={$y$},
  title={强迫振动与共振：$y''+y=\sin t$（共振）vs $y''+y=\sin 0.8t$（非共振）},
  title style={font=\small},
  xlabel style={font=\small},
  ylabel style={font=\small},
  tick label style={font=\footnotesize},
  grid=both, grid style={gray!20},
  axis lines=left,
  samples=400,
  legend pos=north west,
  legend style={font=\footnotesize},
]

% Resonance: y'' + y = sin t, y(0)=0, y'(0)=0
% y = 1/2*sin t - t/2*cos t
\addplot[red, very thick, domain=0:20]
  {0.5*sin(deg(x)) - 0.5*x*cos(deg(x))};
\addlegendentry{共振 $\omega_0=1$：$y=\frac{1}{2}\sin t-\frac{t}{2}\cos t$（振幅 $\sim t/2$）}

% Envelope for resonance
\addplot[red, dashed, domain=0:20] {0.5*x};
\addplot[red, dashed, domain=0:20] {-0.5*x};
\addlegendentry{共振包络 $\pm t/2$}

% Non-resonance: y'' + y = sin(0.8t)
% yp = sin(0.8t)/(1-0.64) = sin(0.8t)/0.36
\addplot[blue, thick, domain=0:20]
  {(1/0.36)*sin(deg(0.8*x)) - (0.8/0.36)*sin(deg(x))};
\addlegendentry{非共振 $\omega_0=0.8$：有界拍频振荡}

% Annotation
\node[font=\footnotesize, fill=yellow!20, draw, rounded corners=2pt, inner sep=3pt,
      text width=4.5cm, align=center]
  at (axis cs:12, -8) {共振：特解须额外乘 $t$\\（与齐次解重合时）\\$y^*=-\frac{t}{2}\cos t$};

\node[font=\footnotesize, fill=white, inner sep=2pt]
  at (axis cs:14, 10) {振幅线性增长 $\sim t/2$};

\end{axis}
\end{tikzpicture}
\end{document}
