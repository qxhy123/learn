\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usepackage{esint}
\usetikzlibrary{calc,arrows.meta}
\begin{document}
\begin{tikzpicture}[
  x={(1cm,0cm)}, y={(0cm,1cm)}, z={(-0.45cm,-0.38cm)},
  line cap=round, line join=round,
  >=Stealth,
  font=\small
]
% 坐标轴（向右下让出顶部空间给标题）
\draw[->] (-3,0,0) -- (3.3,0,0) node[right] {$x$};
\draw[->] (0,-3,0) -- (0,3,0) node[above] {$y$};
\draw[->] (0,0,-3) -- (0,0,3.3) node[above left] {$z$};

% 闭曲面 Σ = ∂Ω（球形）
\foreach \ph in {25,55,90,125,155} {
  \draw[blue!30, very thin]
    plot[domain=0:360, samples=40, variable=\th]
      ({2*sin(\ph)*cos(\th)},{2*sin(\ph)*sin(\th)},{2*cos(\ph)});
}
\foreach \th in {0,45,90,135,180,225,270,315} {
  \draw[blue!30, very thin]
    plot[domain=0:180, samples=20, variable=\ph]
      ({2*sin(\ph)*cos(\th)},{2*sin(\ph)*sin(\th)},{2*cos(\ph)});
}
\node[blue!60!black, anchor=west] at (2.3, 1.5, 1.8) {$\Sigma=\partial\Omega$};
\node[blue!50!black, font=\footnotesize] at (0.8, 0.3, -1.0) {$\Omega$};

% 内部"源"散度向量（红色）
\foreach \dx/\dy/\dz in {0.6/0/0, -0.6/0/0, 0/0.6/0, 0/-0.6/0, 0/0/0.6, 0/0/-0.6} {
  \draw[->, red!70, thick] (0,0,0) -- ({0.85*\dx},{0.85*\dy},{0.85*\dz});
}
\fill[red] (0,0,0) circle (1.5pt);
\node[red!80!black, font=\footnotesize, anchor=east] at (-2.5, 0, -1.5)
  {$\nabla\!\cdot\!\mathbf{F}>0$（源）};

% 外法向通量（远离 z 轴上方与 y 轴）
\foreach \ph/\th in {40/40, 70/130, 110/220, 130/320} {
  \pgfmathsetmacro{\nx}{sin(\ph)*cos(\th)}
  \pgfmathsetmacro{\ny}{sin(\ph)*sin(\th)}
  \pgfmathsetmacro{\nz}{cos(\ph)}
  \draw[->, orange, very thick]
    ({2*\nx},{2*\ny},{2*\nz}) -- ++({0.85*\nx},{0.85*\ny},{0.85*\nz});
  \fill[orange] ({2*\nx},{2*\ny},{2*\nz}) circle (0.8pt);
}
\node[orange!80!black, font=\footnotesize, anchor=west] at (2.5, -2.2, 0)
  {$\mathbf{F}\cdot\mathbf{n}\,dS$（外向通量）};

% 公式标注（独立标题区在左下，半透明，远离峰值）
\node[font=\small, draw=gray!50, fill=gray!10, fill opacity=0.85,
      text opacity=1, rounded corners=2pt, align=left]
  at (-4.5, 3.8, 0)
  {\textbf{Gauss 散度定理}\\[2pt]
   $\displaystyle\iiint_\Omega\nabla\!\cdot\!\mathbf{F}\,dV=\!\!\!\oiint_{\partial\Omega}\!\!\!\mathbf{F}\!\cdot\!d\mathbf{S}$\\[2pt]
   \footnotesize 内部源积分 = 边界通量};
\end{tikzpicture}
\end{document}
