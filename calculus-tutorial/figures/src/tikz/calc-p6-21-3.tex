\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usetikzlibrary{calc,arrows.meta}
\begin{document}
\begin{tikzpicture}[
  x={(1cm,0cm)}, y={(0cm,1cm)}, z={(-0.45cm,-0.38cm)},
  line cap=round, line join=round,
  >=Stealth,
  font=\small
]
% ===== 左：非封闭曲面 z = z(x,y) 的上侧 / 下侧 =====
\begin{scope}[xshift=0cm]
\draw[->] (0,0,0) -- (3.5,0,0) node[right] {$x$};
\draw[->] (0,0,0) -- (0,3,0) node[above] {$y$};
\draw[->] (0,0,0) -- (0,0,3) node[above left] {$z$};

% 曲面
\foreach \ui in {0.2,0.7,...,2.7} {
  \draw[blue!30, thin]
    plot[domain=0.2:2.5, samples=8, variable=\vi]
      ({\ui},{\vi},{0.5+0.12*\ui*\ui*0.4+0.10*\vi*\vi*0.4});
}
\foreach \vi in {0.2,0.7,...,2.5} {
  \draw[blue!30, thin]
    plot[domain=0.2:2.7, samples=8, variable=\ui]
      ({\ui},{\vi},{0.5+0.12*\ui*\ui*0.4+0.10*\vi*\vi*0.4});
}
\node[blue!50!black, font=\small] at (2.5, 0.4, 1.6) {$S: z=z(x,y)$};

% 上侧法向（红色，向上）
\def\u{1.5} \def\v{1.5}
\pgfmathsetmacro{\pz}{0.5+0.12*\u*\u*0.4+0.10*\v*\v*0.4}
\draw[->, very thick, red] (\u,\v,\pz) -- ++(0,0,1.2);
\node[red, font=\small, anchor=south west] at (\u+0.15, \v, \pz+1.2) {$\mathbf{n}^{+}$ 上侧};

% 下侧法向（绿色，向下）
\draw[->, very thick, green!50!black] (\u,\v,\pz) -- ++(0,0,-1.0);
\node[green!40!black, font=\small, anchor=north west] at (\u+0.15, \v, \pz-1.0) {$\mathbf{n}^{-}$ 下侧};

\node[font=\small] at (1.5, 0.0, -0.8) {(a) 非封闭曲面：上/下侧};
\end{scope}

% ===== 右：封闭曲面（球面）外侧 / 内侧 =====
\begin{scope}[xshift=9cm]
\draw[->] (-2.2,0,0) -- (2.7,0,0) node[right] {$x$};
\draw[->] (0,-2.2,0) -- (0,2.7,0) node[above] {$y$};
\draw[->] (0,0,-2.2) -- (0,0,2.5) node[above left] {$z$};

% 球面
\foreach \ph in {25,55,90,125,155} {
  \draw[blue!30]
    plot[domain=0:360, samples=40, variable=\th]
      ({1.7*sin(\ph)*cos(\th)},{1.7*sin(\ph)*sin(\th)},{1.7*cos(\ph)});
}
\foreach \th in {0,45,90,135,180,225,270,315} {
  \draw[blue!30]
    plot[domain=0:180, samples=20, variable=\ph]
      ({1.7*sin(\ph)*cos(\th)},{1.7*sin(\ph)*sin(\th)},{1.7*cos(\ph)});
}

% 外法向（橙色，三个示例点，避开标签区域）
\foreach \ph/\th in {40/30, 70/100, 130/200} {
  \pgfmathsetmacro{\nx}{sin(\ph)*cos(\th)}
  \pgfmathsetmacro{\ny}{sin(\ph)*sin(\th)}
  \pgfmathsetmacro{\nz}{cos(\ph)}
  \fill[orange] ({1.7*\nx},{1.7*\ny},{1.7*\nz}) circle (1pt);
  \draw[->, orange, very thick]
    ({1.7*\nx},{1.7*\ny},{1.7*\nz}) -- ++({0.8*\nx},{0.8*\ny},{0.8*\nz});
}
\node[orange!80!black, font=\small, anchor=west] at (3.0, 2.5, 0)
  {$\mathbf{n}$ 外侧};

% 内法向（红色虚线，远离外法向标签区）
\pgfmathsetmacro{\nx}{sin(60)*cos(310)}
\pgfmathsetmacro{\ny}{sin(60)*sin(310)}
\pgfmathsetmacro{\nz}{cos(60)}
\fill[red] ({1.7*\nx},{1.7*\ny},{1.7*\nz}) circle (1pt);
\draw[->, red, very thick, dashed]
  ({1.7*\nx},{1.7*\ny},{1.7*\nz}) -- ++({-0.8*\nx},{-0.8*\ny},{-0.8*\nz});
\node[red, font=\small, anchor=west] at (3.0, -2.0, 0) {内法向};

\node[font=\small] at (0, 0, -3) {(b) 封闭曲面：默认外侧};
\end{scope}

% ===== 顶部独立说明区（远离曲面峰值）=====
\node[font=\small, draw=gray!50, fill=gray!10, fill opacity=0.85,
      text opacity=1, rounded corners=2pt, align=left]
  at (4.5, 4.5, 0)
  {\textbf{曲面的定向}：反向 $\Rightarrow$ 第二类曲面积分变号：$\iint_{\Sigma^-}\mathbf{F}\cdot d\mathbf{S} = -\iint_{\Sigma}\mathbf{F}\cdot d\mathbf{S}$};
\end{tikzpicture}
\end{document}
