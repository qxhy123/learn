% 标注
\draw[<->] (2.0,1.65) -- (2.7,1.65) node[midway, below, font=\small] {$dx$};
\draw[<->] (2.85,1.8) -- (2.85,2.4) node[midway, right, font=\small] {$dy$};
\node[blue!80!black, font=\small] at (2.35,2.1) {$dA$};
\node[font=\small] at (2,-0.5) {$dA = dx\,dy$};

% ——— 右图：极坐标 dA ———
\begin{scope}[xshift=5.5cm]

\node[font=\small, above right] at (0.2,3.2) {极坐标};

% 弧线（极坐标网格）

% dA 扇环填充（用折线近似弧）

  ({1.8*cos(25)},{1.8*sin(25)}) --
