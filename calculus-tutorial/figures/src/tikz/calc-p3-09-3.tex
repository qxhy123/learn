\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}

\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=10cm, height=6cm,
    xmin=-1, xmax=2, ymin=-0.5, ymax=2,
    xlabel={$x$}, ylabel={$y$},
    samples=200, grid=both, grid style={gray!20},
    legend pos=north west,
]
\addplot[blue, thick, domain=-1:2] {sin(deg(x))};
\addlegendentry{$f(x)=\sin x$}
\addplot[red, thick, domain=-1:2] {x};
\addlegendentry{$g(x)=x$}
\node[anchor=west] at (axis cs:0.1, 1.7) {\small $\lim_{x\to 0}\frac{\sin x}{x}$};
\node[anchor=west] at (axis cs:0.1, 1.4) {\small $= \lim\frac{(\sin x)'}{x'} = \lim\frac{\cos x}{1} = 1$};
\fill[black] (axis cs:0, 0) circle (2pt);
\node[anchor=north, font=\tiny] at (axis cs:0, -0.05) {两曲线在 $0$ 处切线相同};
