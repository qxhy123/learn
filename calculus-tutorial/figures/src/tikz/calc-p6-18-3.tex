\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=9cm, height=9cm,
  xmin=-0.5, xmax=3, ymin=-0.5, ymax=3,
  xlabel={$x$}, ylabel={$y$},
  axis lines=center,
  title={方向导数与最速下降方向：$f(x,y)=x^2+\frac{1}{2}y^2$},
  title style={font=\small},
  xlabel style={right},
  ylabel style={above},
  tick label style={font=\small},
  clip=false,
]
% 等高线（椭圆）
\foreach \c in {0.5,1.5,3,5} {
  \addplot[blue!50, domain=0:360, samples=80]
    ({sqrt(\c)*cos(x)}, {sqrt(2*\c)*sin(x)});
}

% 基点 P = (1.5, 2)
\def\px{1.5} \def\py{2}
\def\gx{3.0} \def\gy{2.0}  % 梯度 = (2x, y) = (3, 2)
\def\gn{3.6056}  % sqrt(3^2+2^2)

% 梯度方向（最速上升）
\draw[->, very thick, red]
  (axis cs:\px,\py) --
  (axis cs:{1.5+0.8*3/3.6056},{2+0.8*2/3.6056})
  node[right, font=\small] {$\nabla f$（最速上升）};

% 负梯度方向（最速下降）
\draw[->, very thick, green!60!black]
  (axis cs:\px,\py) --
  (axis cs:{1.5-0.8*3/3.6056},{2-0.8*2/3.6056})
  node[left, font=\small] {$-\nabla f$（最速下降）};

% 任意方向 l（非梯度方向）
\draw[->, thick, orange]
  (axis cs:\px,\py) --
  (axis cs:{1.5+0.7},{2+0.2})
  node[right, font=\small] {$\mathbf{l}$（任意方向）};

% 投影标注
\draw[dashed, orange!80]
  (axis cs:{1.5+0.7},{2+0.2}) --
  (axis cs:{1.5+0.7*3/3.6056*3/3.6056+0.2*2/3.6056*3/3.6056},
            {2+0.7*3/3.6056*2/3.6056+0.2*2/3.6056*2/3.6056});

\fill[black] (axis cs:\px,\py) circle (2pt) node[above left, font=\small] {$P_0$};

% 公式注解
\node[font=\small, align=left, draw=gray, fill=gray!10, rounded corners=2pt]
  at (axis cs:2.2,0.5)
  {$\frac{\partial f}{\partial\mathbf{l}}=\nabla f\cdot\mathbf{l}=|\nabla f|\cos\theta$\\最大值在 $\mathbf{l}\!\parallel\!\nabla f$ 时取得};
\end{axis}
\end{tikzpicture}
\end{document}
