\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}
\begin{document}
\begin{tikzpicture}
\begin{groupplot}[
  group style={
    group size=2 by 1,
    horizontal sep=1.8cm,
  },
  width=6.5cm, height=5.5cm,
  xlabel={$x$}, ylabel={$f(x)$},
  xlabel style={font=\small},
  ylabel style={font=\small},
  tick label style={font=\footnotesize},
  grid=both, grid style={gray!20},
  axis lines=left,
  samples=150,
  legend style={font=\footnotesize, at={(0.98,0.98)}, anchor=north east},
]

% Plot 1: Convex (quadratic) gradient descent trajectory
\nextgroupplot[
  title={凸函数（$f=x^2$）：稳定收敛},
  title style={font=\small},
  xmin=-0.5, xmax=4.5,
  ymin=-0.5, ymax=17,
]
\addplot[blue, thick, domain=-0.2:4.3] {x^2};
\addlegendentry{$f(x)=x^2$}
% GD steps from x0=4, lr=0.3: x_{n+1} = x_n - 0.3*2*x_n = 0.4*x_n
% 4, 1.6, 0.64, 0.256, 0.1024
\def\xa{4.0}\def\xb{1.6}\def\xc{0.64}\def\xd{0.256}\def\xe{0.102}
\addplot[red, thick, mark=*, mark size=2pt] coordinates {
  (\xa, \xa*\xa)
  (\xb, \xb*\xb)
  (\xc, \xc*\xc)
  (\xd, \xd*\xd)
  (\xe, \xe*\xe)
};
\addlegendentry{GD 轨迹}
% vertical drops
\foreach \xi/\xj in {\xa/\xb, \xb/\xc, \xc/\xd, \xd/\xe}{
  \addplot[gray!60, dashed] coordinates {(\xi, \xi*\xi)(\xj, \xi*\xi)};
  \addplot[gray!60, dashed] coordinates {(\xj, \xi*\xi)(\xj, \xj*\xj)};
}
\node[font=\footnotesize, red] at (axis cs:2.5, 5) {单调下降至全局最优};
\addplot[mark=star, mark size=4pt, green!60!black] coordinates {(0,0)};
\node[font=\footnotesize, below right, green!60!black] at (axis cs:0.1, 0.3) {全局最优};

% Plot 2: Nonconvex x^4 - 3x^2 gradient descent
\nextgroupplot[
  title={非凸函数（$f=x^4-3x^2$）：陷局部极值},
  title style={font=\small},
  xmin=-2.5, xmax=2.5,
  ymin=-3, ymax=4,
]
\addplot[blue, thick, domain=-2.2:2.2] {x^4 - 3*x^2};
\addlegendentry{$f(x)=x^4-3x^2$}
% GD from x0=0.4, lr=0.05: grad=4x^3-6x
% x0=0.4: g=4*0.064-2.4=-2.144, x1=0.4+0.107=0.507
% approximate trajectory converging to local at x=sqrt(1.5)~1.22
\addplot[red, thick, mark=*, mark size=2pt] coordinates {
  (0.4, -0.144)
  (0.51, -0.509)
  (0.7,  -1.029)
  (0.95, -1.571)
  (1.1,  -1.849)
  (1.2,  -2.004)
  (1.22, -2.25)
};
\addlegendentry{GD 轨迹（初值偏右）}
\addplot[orange, thick, mark=square*, mark size=2pt] coordinates {
  (-0.4, -0.144)
  (-0.51, -0.509)
  (-0.7,  -1.029)
  (-0.95, -1.571)
  (-1.1,  -1.849)
  (-1.22, -2.25)
};
\addlegendentry{GD 轨迹（初值偏左）}
% local minima
\addplot[mark=star, mark size=4pt, green!60!black] coordinates {(1.2247,-2.25)(-1.2247,-2.25)};
\node[font=\footnotesize, below] at (axis cs:1.2247, -2.25) {局部极小};
% saddle at 0
\addplot[mark=diamond*, mark size=4pt, purple] coordinates {(0,0)};
\node[font=\footnotesize, above] at (axis cs:0, 0.1) {鞍点};
\node[font=\footnotesize, fill=white] at (axis cs:-1.8, 3.5) {不同初值→不同极值};

\end{groupplot}

\node[font=\normalsize\bfseries] at (8.2, 6.3) {梯度下降轨迹：凸 vs 非凸};
\end{tikzpicture}
