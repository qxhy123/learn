\begin{tikzpicture}

    axis lines=middle, width=10cm, height=6cm,
    xmin=0, xmax=20, ymin=0, ymax=3,
    xlabel={$n$}, ylabel={$a_n$},
    grid=both, grid style={gray!20},

  \pgfmathsetmacro{\nn}{int(\n)}
  \edef\temp{\noexpand\fill[blue] (axis cs:\nn,\val) circle (1.5pt);}
  \temp
}
\node[anchor=west] at (axis cs:2, 1.5) {\small $a_n=(1+\frac{1}{n})^n$};
\node[anchor=west] at (axis cs:2, 1.0) {\small 单调递增 + 有上界 $\Rightarrow$ 收敛};
