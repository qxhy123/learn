\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\usepgfplotslibrary{groupplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{groupplot}[
    group style={group size=1 by 2, vertical sep=1.2cm},
    width=10cm, height=4.5cm, axis lines=middle, samples=200,
    grid=both, grid style={gray!20},
    xmin=-2.5, xmax=4,
]
\nextgroupplot[ymin=-3, ymax=6, ylabel={$y=f(x)$}, title={函数与单调性}]
\addplot[blue, thick, domain=-2:3.5] {x^3/3 - x^2 - 3*x + 1};
\fill[red] (axis cs:-1, 8/3) circle (2pt);
\fill[red] (axis cs:3, -8) circle (2pt);
\node[anchor=south, red] at (axis cs:-1, 8/3) {\small 极大};
\node[anchor=north, red, font=\tiny] at (axis cs:3, -8) {极小};

\nextgroupplot[ymin=-4, ymax=4, ylabel={$y=f'(x)$}, title={导数符号决定单调性}, xlabel={$x$}]
\addplot[red, thick, domain=-2.5:4] {x^2 - 2*x - 3};
\draw[gray, dashed] (axis cs:-2.5, 0) -- (axis cs:4, 0);
\fill[blue] (axis cs:-1, 0) circle (2pt);
\fill[blue] (axis cs:3, 0) circle (2pt);
\node[anchor=south] at (axis cs:0, -2.5) {\small $f'<0$（减）};
\node[anchor=south] at (axis cs:-2, 1) {\small $f'>0$（增）};
\node[anchor=south] at (axis cs:3.5, 1) {\small $f'>0$（增）};
\end{groupplot}
\end{tikzpicture}
