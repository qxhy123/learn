\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usetikzlibrary{calc,arrows.meta}
\begin{document}
\begin{tikzpicture}[
  x={(1cm,0cm)}, y={(0cm,1cm)}, z={(-0.45cm,-0.38cm)},
  line cap=round, line join=round,
  >=Stealth
]
% 坐标轴
\draw[->] (-0.5,0,0) -- (4,0,0) node[right] {$x$};
\draw[->] (0,-0.5,0) -- (0,3.8,0) node[above] {$y$};
\draw[->] (0,0,-0.5) -- (0,0,3.5) node[above left] {$z$};

% 曲面（半球面上部，简化为若干曲线）
\foreach \ph in {20,40,60,80} {
  \draw[blue!30]
    plot[domain=0:90, samples=15, variable=\th]
      ({2.5*sin(\ph)*cos(\th)},{2.5*sin(\ph)*sin(\th)},{2.5*cos(\ph)});
}
\foreach \th in {0,30,60,90} {
  \draw[blue!30]
    plot[domain=0:90, samples=12, variable=\ph]
      ({2.5*sin(\ph)*cos(\th)},{2.5*sin(\ph)*sin(\th)},{2.5*cos(\ph)});
}
\node[blue!60!black, font=\small] at (2.0,2.0,1.5) {$S$};

% 场向量（均匀竖直向上，过曲面各点）
\foreach \xf/\yf in {0.5/0.5, 1.5/0.5, 0.5/1.5, 1.5/1.5, 1.0/1.0} {
  \pgfmathsetmacro{\zbase}{0.3}
  \draw[->, red!60, thick]
    (\xf,\yf,\zbase) -- (\xf,\yf,{\zbase+0.9});
}
\node[red!80!black, right, font=\small] at (2.5,0.5,1.2) {$\mathbf{F}$（向量场）};

% 法向量（单位外法向）
\def\ph{45} \def\th{45}
\pgfmathsetmacro{\nx}{sin(\ph)*cos(\th)}
\pgfmathsetmacro{\ny}{sin(\ph)*sin(\th)}
\pgfmathsetmacro{\nz}{cos(\ph)}
\def\rr{2.5}
\fill[orange] ({\rr*\nx},{\rr*\ny},{\rr*\nz}) circle (1.5pt);
\draw[->, orange, very thick]
  ({\rr*\nx},{\rr*\ny},{\rr*\nz}) --
  ++({0.7*\nx},{0.7*\ny},{0.7*\nz})
  node[right, font=\small] {$\mathbf{n}$（外法向）};

% 底面圆（封口）
\draw[blue!50, dashed]
  plot[domain=0:360, samples=60]
    ({2.5*cos(\x)},{2.5*sin(\x)},0);

% 通量公式
\node[font=\small, draw=gray!50, fill=gray!10, rounded corners=2pt, align=left]
  at (-0.5,3,3.0)
  {$\Phi = \iint_S \mathbf{F}\cdot d\mathbf{S} = \iint_S \mathbf{F}\cdot\mathbf{n}\,dS$\\
   法向反向 $\Rightarrow$ 通量变号};
\end{tikzpicture}
\end{document}
