\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usetikzlibrary{calc,arrows.meta}
\begin{document}
\begin{tikzpicture}[
  x={(1cm,0cm)}, y={(0cm,1cm)}, z={(-0.45cm,-0.38cm)},
  line cap=round, line join=round,
  >=Stealth
]
% 坐标轴
\draw[->] (0,0,0) -- (4.5,0,0) node[right] {$x$};
\draw[->] (0,0,0) -- (0,4,0) node[above] {$y$};
\draw[->] (0,0,0) -- (0,0,3.5) node[above left] {$z$};

% 曲面（马鞍面近似） z = 2 - 0.15*x^2 - 0.1*y^2
\foreach \ui in {0,0.5,...,3} {
  \draw[blue!30, thin]
    plot[domain=0:3, samples=7, variable=\vi]
      ({\ui},{\vi},{2.2-0.12*\ui*\ui-0.1*\vi*\vi});
}
\foreach \vi in {0,0.5,...,3} {
  \draw[blue!30, thin]
    plot[domain=0:3, samples=7, variable=\ui]
      ({\ui},{\vi},{2.2-0.12*\ui*\ui-0.1*\vi*\vi});
}

% 参数点 P = (u0, v0)
\def\u{1.5} \def\v{1.5}
\pgfmathsetmacro{\pz}{2.2-0.12*\u*\u-0.1*\v*\v}
\fill[red] (\u,\v,\pz) circle (1.5pt);
\node[red, right, font=\small] at (\u,\v,\pz) {$P(\mathbf{r}(u_0,v_0))$};

% r_u 方向（沿 u 方向切向量）
\draw[->, thick, red]
  (\u,\v,\pz) -- ++(1,0,{-0.12*2*\u*1})
  node[right, font=\small] {$\mathbf{r}_u$};

% r_v 方向（沿 v 方向切向量）
\draw[->, thick, green!60!black]
  (\u,\v,\pz) -- ++(0,1,{-0.1*2*\v*1})
  node[above, font=\small] {$\mathbf{r}_v$};

% 法向量 n = r_u × r_v
\pgfmathsetmacro{\nx}{0.12*2*\u}
\pgfmathsetmacro{\ny}{0.1*2*\v}
\pgfmathsetmacro{\nz}{1}
\pgfmathsetmacro{\nn}{sqrt(\nx*\nx+\ny*\ny+\nz*\nz)}
\draw[->, ultra thick, orange]
  (\u,\v,\pz) --
  ++({0.8*\nx/\nn},{0.8*\ny/\nn},{0.8*\nz/\nn})
  node[above right, font=\small] {$\mathbf{n}=\mathbf{r}_u\times\mathbf{r}_v$};

% dS 面积元标注
\fill[blue!20, opacity=0.5]
  (\u,\v,\pz) -- ++(0.5,0,{-0.12*2*\u*0.5}) -- ++(0,0.5,{-0.1*2*\v*0.5}) --
  ++(-0.5,0,{0.12*2*\u*0.5}) -- cycle;

\node[font=\small, draw=gray!50, fill=gray!10, rounded corners=2pt, align=left]
  at (-0.5,3.5,3.2)
  {$dS = |\mathbf{r}_u\times\mathbf{r}_v|\,du\,dv$\\
   曲面 $z=z(x,y)$：$dS=\sqrt{1+z_x^2+z_y^2}\,dx\,dy$};
\end{tikzpicture}
\end{document}
