\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}
\begin{document}
\begin{tikzpicture}
\begin{groupplot}[
  group style={
    group size=3 by 1,
    horizontal sep=1.4cm,
  },
  width=5.5cm, height=5cm,
  xmin=0, xmax=4,
  ymin=-2.5, ymax=3,
  xlabel={$x$}, ylabel={$y$},
  xlabel style={font=\small},
  ylabel style={font=\small},
  tick label style={font=\footnotesize},
  grid=both, grid style={gray!20},
  axis lines=left,
  samples=120,
  legend style={font=\footnotesize},
]

% Plot 1: two distinct real roots, e.g. r1=1, r2=-2 => y = C1 e^x + C2 e^{-2x}
\nextgroupplot[title={实异根 $r_1\neq r_2$\\$y''-y'-2y=0$},
               title style={font=\footnotesize, align=center}]
\addplot[red, thick, domain=0:3.5] {exp(x) - exp(-2*x)};
\addlegendentry{$C_1=1,C_2=-1$}
\addplot[blue, thick, domain=0:3.5] {0.5*exp(x) + 0.5*exp(-2*x)};
\addlegendentry{$C_1=C_2=0.5$}
\addplot[gray, dashed, domain=0:3.5] {exp(-2*x)};
\addlegendentry{$C_1=0$（衰减）}
\node[font=\footnotesize, fill=white] at (axis cs:3,2.3) {$r_1=2,r_2=-1$};

% Plot 2: repeated root, e.g. y'' - 2y' + y = 0, r=1 => y=(C1+C2 x)e^x
\nextgroupplot[title={重根 $r_1=r_2=r$\\$y''-2y'+y=0$},
               title style={font=\footnotesize, align=center}]
\addplot[red, thick, domain=0:2.8] {(1 + x)*exp(x)};
\addlegendentry{$C_1=C_2=1$}
\addplot[blue, thick, domain=0:2.8] {exp(x)};
\addlegendentry{$C_1=1,C_2=0$}
\addplot[green!70!black, thick, domain=0:2.8] {x*exp(x)};
\addlegendentry{$C_1=0,C_2=1$}
\node[font=\footnotesize, fill=white] at (axis cs:2.5,2.3) {$r=1$，临界};

% Plot 3: complex roots alpha=-1, beta=2 => e^{-x}(C1 cos2x + C2 sin2x)
\nextgroupplot[title={复根 $\alpha\pm\beta i$\\$y''+2y'+5y=0$},
               title style={font=\footnotesize, align=center},
               ymin=-2.5, ymax=2.5]
\addplot[red, thick, domain=0:4] {exp(-x)*cos(deg(2*x))};
\addlegendentry{$C_1=1,C_2=0$}
\addplot[blue, thick, domain=0:4] {exp(-x)*sin(deg(2*x))};
\addlegendentry{$C_1=0,C_2=1$}
\addplot[orange, thick, domain=0:4] {exp(-x)*(cos(deg(2*x))+sin(deg(2*x)))};
\addlegendentry{$C_1=C_2=1$}
% envelope
\addplot[gray, dashed, domain=0:4] {exp(-x)};
\addplot[gray, dashed, domain=0:4] {-exp(-x)};
\node[font=\footnotesize, fill=white] at (axis cs:3.2,2.2) {衰减振荡};

\end{groupplot}

\node[font=\normalsize\bfseries] at (9.5, 5.4) {二阶常系数齐次 ODE：三类特征根对应解形态};
\end{tikzpicture}
\end{document}
