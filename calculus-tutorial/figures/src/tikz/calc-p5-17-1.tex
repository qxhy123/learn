\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=12cm, height=7cm,
    xmin=-3.5, xmax=3.5, ymin=-1.4, ymax=1.4,
    xlabel={$x$}, ylabel={$y$},
    samples=400, grid=both, grid style={gray!20},
    legend pos=south east, legend style={font=\small},
    title={方波 Fourier 级数部分和（Gibbs 现象）},
]
% Square wave (target)
\addplot[black, very thick, domain=-3.14159:-0.001] {-1};
\addplot[black, very thick, domain=0.001:3.14159] {1};
\addlegendentry{方波};

% N=1: (4/pi)*sin(x)
\addplot[blue, thick, domain=-3.5:3.5]
    {4/3.14159*(sin(deg(x)))};
\addlegendentry{$N=1$};

% N=3: + sin(3x)/3
\addplot[red, thick, domain=-3.5:3.5]
    {4/3.14159*(sin(deg(x)) + sin(deg(3*x))/3)};
\addlegendentry{$N=3$};

% N=9: higher approx
\addplot[green!60!black, thick, domain=-3.5:3.5]
    {4/3.14159*(sin(deg(x)) + sin(deg(3*x))/3 + sin(deg(5*x))/5
    + sin(deg(7*x))/7 + sin(deg(9*x))/9)};
\addlegendentry{$N=9$};

% Annotation for Gibbs
\node[font=\small, red!70, anchor=west] at (axis cs:0.05, 1.25)
    {约$9\%$过冲（Gibbs）};
\draw[->, red!70] (axis cs:0.4,1.18) -- (axis cs:0.18,1.09);
\end{axis}
\end{tikzpicture}
\end{document}
