\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\usepgfplotslibrary{groupplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{groupplot}[
    group style={group size=1 by 2, vertical sep=1.4cm},
    width=11cm, height=4.5cm,
    axis lines=middle, samples=300,
    xmin=-3.5, xmax=3.5, ymin=-1.3, ymax=1.3,
    xlabel={$x$}, grid=both, grid style={gray!20},
    xtick={-3.14159,-1.5708,0,1.5708,3.14159},
    xticklabels={$-\pi$,$-\frac{\pi}{2}$,$0$,$\frac{\pi}{2}$,$\pi$},
]
\nextgroupplot[
    title={正弦函数正交性：$\int_{-\pi}^{\pi}\sin(mx)\sin(nx)\,dx$},
    ylabel={$y$},
]
\addplot[blue, thick, domain=-3.5:3.5] {sin(deg(x))};
\addlegendentry{$\sin x$};
\addplot[red, thick, dashed, domain=-3.5:3.5] {sin(deg(2*x))};
\addlegendentry{$\sin 2x$};
\addplot[green!60!black, thick, dotted, domain=-3.5:3.5]
    {sin(deg(x))*sin(deg(2*x))};
\addlegendentry{$\sin x\cdot\sin 2x$（乘积，积分$=0$）};

\nextgroupplot[
    title={余弦函数正交性：$\int_{-\pi}^{\pi}\cos(mx)\cos(nx)\,dx$},
    ylabel={$y$},
]
\addplot[blue, thick, domain=-3.5:3.5] {cos(deg(x))};
\addlegendentry{$\cos x$};
\addplot[red, thick, dashed, domain=-3.5:3.5] {cos(deg(2*x))};
\addlegendentry{$\cos 2x$};
\addplot[green!60!black, thick, dotted, domain=-3.5:3.5]
    {cos(deg(x))*cos(deg(2*x))};
\addlegendentry{$\cos x\cdot\cos 2x$（乘积，积分$=0$）};
\end{groupplot}
\end{tikzpicture}
\end{document}
