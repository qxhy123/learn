\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=10cm, height=7cm,
    xmin=-0.5, xmax=3, ymin=-1, ymax=5,
    xlabel={$x$}, ylabel={$y$},
    samples=100, grid=both, grid style={gray!20},
]
\addplot[blue, thick, domain=-0.5:3] {x^2-2};
% 切线 at x_0=1: y = 2(x-1) + (-1) = 2x - 3
\addplot[red, thick, dashed, domain=0.3:2.5] {2*x - 3};
% 割线 at x_0=1, x_1=2: slope = (2-(-1))/(2-1) = 3, y = 3(x-1)-1
\addplot[green!50!black, thick, dashed, domain=0.3:2.5] {3*x - 4};
\fill[red] (axis cs:1,-1) circle (2pt);
\fill[green!50!black] (axis cs:2,2) circle (2pt);
\node[anchor=south west, red] at (axis cs:1,-1) {\small $x_0$};
\node[anchor=south west, green!50!black] at (axis cs:2,2) {\small $x_0+\Delta x$};
\node[anchor=west, red] at (axis cs:1.6,0.5) {\small 切线 $k=f'(x_0)=2$};
\node[anchor=west, green!50!black] at (axis cs:1.6,1.5) {\small 割线 $k=3$};
\node[anchor=west] at (axis cs:0,4.5) {\small $f(x)=x^2-2$};
\node[anchor=west] at (axis cs:0,4) {\small $\Delta x\to 0 \Rightarrow$ 割线 $\to$ 切线};
\end{axis}
\end{tikzpicture}
\end{document}
