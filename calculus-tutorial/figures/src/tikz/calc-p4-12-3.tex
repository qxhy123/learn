\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\usepgfplotslibrary{fillbetween}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=10cm, height=6cm,
    xmin=-3, xmax=3, ymin=-2.5, ymax=2.5,
    xlabel={$x$}, ylabel={$y$},
    samples=200, grid=both, grid style={gray!20},
]
\addplot[name path=A, blue, thick, domain=-2.5:2.5] {sin(deg(x))};
\addplot[name path=Z, draw=none, domain=-2.5:0] {0};
\addplot[red!30] fill between[of=A and Z, soft clip={domain=-2.5:0}];
\addplot[name path=Z2, draw=none, domain=0:2.5] {0};
\addplot[green!30] fill between[of=A and Z2, soft clip={domain=0:2.5}];
\node[anchor=west, red] at (axis cs:-2.4, -1.2) {\small 负面积};
\node[anchor=west, green!50!black] at (axis cs:0.5, 1.2) {\small 正面积};
\node[anchor=west] at (axis cs:-2.9, 2.2) {\small **定积分含正负**：$\int_{-a}^a \sin x\,dx = 0$（奇函数对称）};
\end{axis}
\end{tikzpicture}
\end{document}
