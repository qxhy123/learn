\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=10cm, height=6cm,
    xmin=-1.2, xmax=2, ymin=-3, ymax=2,
    xlabel={$x$}, ylabel={$y$},
    samples=300, grid=both, grid style={gray!20},
    legend pos=south east,
]
\addplot[blue, thick, domain=-0.99:2] {ln(1+x)};
\addlegendentry{$\ln(1+x)$}
\addplot[red, thick, dashed, domain=-0.99:2] {x};
\addlegendentry{$P_1=x$}
\addplot[green!50!black, thick, dashed, domain=-0.99:2] {x - x^2/2};
\addlegendentry{$P_2=x-\frac{x^2}{2}$}
\addplot[purple, thick, dashed, domain=-0.99:2] {x - x^2/2 + x^3/3};
\addlegendentry{$P_3=P_2+\frac{x^3}{3}$}
\draw[red, very thick, dashed] (axis cs:-1, -3) -- (axis cs:-1, 2);
\draw[red, very thick, dashed] (axis cs:1, -3) -- (axis cs:1, 2);
\node[anchor=west, red] at (axis cs:-0.9, -1.5) {\small 收敛域 $-1<x\le 1$};
\end{axis}
\end{tikzpicture}
\end{document}
