\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usetikzlibrary{arrows.meta, positioning, fit, backgrounds, shapes.geometric}
\begin{document}
\begin{tikzpicture}[
  node distance=0.4cm and 0.9cm,
  neuron/.style={circle, draw=blue!70, fill=blue!12,
                 font=\footnotesize, minimum size=0.75cm, text centered},
  layer/.style={rectangle, draw=gray!40, fill=gray!6, rounded corners,
                minimum width=1.0cm, inner sep=0.3cm},
  op/.style={rectangle, rounded corners=2pt, draw=orange!70, fill=orange!10,
             font=\footnotesize, minimum width=2.8cm, minimum height=0.6cm, text centered},
  farr/.style={{Stealth[scale=0.9]}-{Stealth[scale=0.9]}, blue!60, thick},
  barr/.style={-{Stealth[scale=0.9]}, red!70, thick, dashed},
  garr/.style={-{Stealth[scale=0.9]}, green!60!black, thick},
]

% ---- Network layers ----
% Input layer
\node[neuron] (x1) {$x_1$};
\node[neuron, below=0.4cm of x1] (x2) {$x_2$};
\node[neuron, below=0.4cm of x2] (x3) {$x_3$};
\node[font=\footnotesize, below=0.15cm of x3] {输入层};

% Hidden layer
\node[neuron, right=1.6cm of x1, yshift=-0.4cm] (h1) {$h_1$};
\node[neuron, below=0.4cm of h1] (h2) {$h_2$};
\node[neuron, below=0.4cm of h2] (h3) {$h_3$};
\node[font=\footnotesize, below=0.15cm of h3] {隐层 ($W_1$)};

% Output layer
\node[neuron, right=1.6cm of h1, yshift=0.4cm] (o1) {$o_1$};
\node[neuron, below=0.4cm of o1] (o2) {$o_2$};
\node[font=\footnotesize, below=0.15cm of o2] {输出层 ($W_2$)};

% Loss
\node[op, right=1.2cm of o1, yshift=-0.2cm] (loss) {损失 $L$};

% Forward arrows (blue)
\foreach \src in {x1,x2,x3}
  \foreach \dst in {h1,h2,h3}
    \draw[blue!30, thin] (\src) -- (\dst);
\foreach \src in {h1,h2,h3}
  \foreach \dst in {o1,o2}
    \draw[blue!30, thin] (\src) -- (\dst);
\draw[blue!60, thick] (o1) -- (loss);
\draw[blue!60, thick] (o2) -- (loss);

% Forward label
\node[font=\footnotesize, blue!70] at (1.5, 0.6) {前向传播};

% Backward arrows (red dashed)
\draw[barr] (loss.north) -- ++(0,0.5) -- ++(-1.6,0)
  node[above, font=\footnotesize, red!70]{$\frac{\partial L}{\partial o}$} -- ++(-0.6,0);
\draw[barr] ([yshift=-4pt]h1.east) -- ++(0.3,0) node[above right, font=\footnotesize, red!70]{$\frac{\partial L}{\partial W_2}\!=\!\frac{\partial L}{\partial o}h^\top$};
\draw[barr] ([yshift=-4pt]x1.east) -- ++(0.3,0) node[above right, font=\footnotesize, red!70]{$\frac{\partial L}{\partial W_1}\!=\!\frac{\partial L}{\partial h}x^\top$};

% Backward label
\node[font=\footnotesize, red!70] at (3.5, -3.5) {反向传播（链式法则）};

% Chain rule box
\node[op, draw=green!60!black, fill=green!8, align=left, minimum width=4.5cm, minimum height=1.6cm]
  at (8.0, -1.2)
  {\textbf{链式法则核心公式}\\
   $\dfrac{\partial L}{\partial W_1}=\dfrac{\partial L}{\partial h}\cdot\dfrac{\partial h}{\partial W_1}$\\[4pt]
   $\dfrac{\partial L}{\partial x}=W_1^\top\dfrac{\partial L}{\partial h}$};

\node[font=\normalsize\bfseries] at (4.5, 1.2) {反向传播：神经网络中的矩阵链式法则};

\end{tikzpicture}
