\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usepackage{esint}
\usetikzlibrary{arrows.meta,calc}
\begin{document}
\begin{tikzpicture}[>=Stealth, line join=round]

% ——— 左图：散度（发散源）———
\begin{scope}[xshift=0cm]
\node[font=\small\bfseries, align=center] at (2,4.2)
  {散度 $\nabla\cdot\mathbf{F}$\\（源/汇）};

% 外向辐射场（正散度 = 源）
\foreach \ang in {0,30,60,90,120,150,180,210,240,270,300,330} {
  \pgfmathsetmacro{\cx}{2}
  \pgfmathsetmacro{\cy}{2}
  \pgfmathsetmacro{\fx}{cos(\ang)*0.9}
  \pgfmathsetmacro{\fy}{sin(\ang)*0.9}
  \draw[->, blue!70, thick]
    (\cx,\cy) -- ++(\fx,\fy);
}
\fill[red!80] (2,2) circle (4pt);
\node[red!80!black, right, font=\small] at (2.1,2) {源点};
\node[blue!70!black, font=\small] at (3.4,3.4) {$\mathbf{F}$向外};

% 方框
\draw[gray!50, dashed] (0.5,0.5) rectangle (3.5,3.8);
\node[gray!60, font=\small] at (2,0.2)
  {$\nabla\cdot\mathbf{F}>0$：流出大于流入};

% 闭合面积分示意
\draw[orange, thick]
  plot[domain=0:360, samples=40]
    ({2+1.5*cos(\x)},{2+1.2*sin(\x)});
\node[orange!80!black, font=\tiny] at (3.9,2) {$\oiint_S$};
\end{scope}

% ——— 右图：旋度（涡旋）———
\begin{scope}[xshift=5.5cm]
\node[font=\small\bfseries, align=center] at (2,4.2)
  {旋度 $\nabla\times\mathbf{F}$\\（涡旋轴）};

% 旋转场（正旋度 = 逆时针涡旋）
\foreach \ang in {0,30,60,90,120,150,180,210,240,270,300,330} {
  \pgfmathsetmacro{\rx}{2+1.2*cos(\ang)}
  \pgfmathsetmacro{\ry}{2+1.2*sin(\ang)}
  \pgfmathsetmacro{\fx}{-sin(\ang)*0.5}
  \pgfmathsetmacro{\fy}{cos(\ang)*0.5}
  \draw[->, green!60!black, thick]
    (\rx,\ry) -- ++(\fx,\fy);
}
\fill[purple!80] (2,2) circle (4pt);
% 旋度轴（指向屏幕外，即 z 方向）
\node[purple, font=\small, above] at (2,2.15) {$\nabla\times\mathbf{F}\,\uparrow$};
\draw[purple!80, ->, very thick] (2,2) -- ++(0,0.8);
\node[green!60!black, font=\small] at (3.5,3.5) {涡旋};

\draw[gray!50, dashed] (0.5,0.5) rectangle (3.5,3.8);
\node[gray!60, font=\small] at (2,0.2)
  {$\nabla\times\mathbf{F}\neq\mathbf{0}$：有旋场};
\end{scope}

% 关系说明
\node[font=\small, draw=gray!40, fill=gray!10, rounded corners=2pt,
      align=center, minimum width=10cm]
  at (5.5,-0.5)
  {梯度场无旋：$\nabla\times(\nabla f)=\mathbf{0}$\quad
   旋度场无散：$\nabla\cdot(\nabla\times\mathbf{F})=0$};
\end{tikzpicture}
\end{document}
