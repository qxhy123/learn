\usepackage{ctex}

\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=10cm, height=6cm,
    xmin=-0.5, xmax=4, ymin=-0.5, ymax=4,
    xlabel={$x$}, ylabel={$y$},
    samples=200, grid=both, grid style={gray!20},
]
\addplot[name path=A, blue, thick, domain=0.5:3] {0.3*(x-1)^2 + 0.5};
\addplot[name path=B, draw=none, domain=0.5:3] {0};
\addplot[orange!30] fill between[of=A and B];
\draw[blue, thick] (axis cs:0.5,0) -- (axis cs:0.5,0.575);
\draw[blue, thick] (axis cs:3,0) -- (axis cs:3,1.7);
\node[anchor=north] at (axis cs:0.5,0) {$a$};
\node[anchor=north] at (axis cs:3,0) {$b$};
\node[anchor=west] at (axis cs:0,3.5) {\small **牛顿-莱布尼茨公式**};
\node[anchor=west] at (axis cs:0,3) {\small $\int_a^b f(x)\,dx = F(b)-F(a)$};
\node[anchor=west] at (axis cs:0,2.5) {\small（$F$ 是 $f$ 的任意原函数）};
\end{axis}
