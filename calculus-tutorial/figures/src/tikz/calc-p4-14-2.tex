\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\usepgfplotslibrary{fillbetween}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=10cm, height=6cm,
    xmin=-0.3, xmax=2.5, ymin=-0.5, ymax=5,
    xlabel={$x$}, ylabel={$y$},
    samples=200, grid=both, grid style={gray!20},
]
\addplot[name path=A, blue, thick, domain=0.05:2, restrict y to domain=0:5] {1/sqrt(x)};
\addplot[name path=Z, draw=none, domain=0:2] {0};
\addplot[orange!30] fill between[of=A and Z, soft clip={domain=0.05:2}];
\draw[red, dashed] (axis cs:0,0) -- (axis cs:0,5);
\node[anchor=south, red] at (axis cs:0.05, 4.5) {瑕点 $x=0$};
\node[anchor=north] at (axis cs:2,0) {$1$};
\node[anchor=west] at (axis cs:0.8,4) {\small **瑕积分**};
\node[anchor=west] at (axis cs:0.8,3.4) {\small $\int_0^1\frac{1}{\sqrt{x}}\,dx = \lim_{a\to 0^+}\int_a^1 = 2$};
\node[anchor=west] at (axis cs:0.8,2.7) {\small 收敛（$p=1/2<1$）};
\end{axis}
\end{tikzpicture}
\end{document}
