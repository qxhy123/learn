\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=11cm, height=6cm,
    xmin=-7, xmax=7, ymin=-2.5, ymax=2.5,
    xlabel={$x$}, ylabel={$y$},
    samples=300, grid=both, grid style={gray!20},
    legend pos=south east,
]
\addplot[blue, thick, domain=-7:7] {sin(deg(x))};
\addlegendentry{$\sin x$}
\addplot[red, thick, dashed, domain=-2.5:2.5] {x};
\addlegendentry{$P_1=x$}
\addplot[green!50!black, thick, dashed, domain=-3.5:3.5] {x - x^3/6};
\addlegendentry{$P_3=x-\frac{x^3}{6}$}
\addplot[purple, thick, dashed, domain=-5:5] {x - x^3/6 + x^5/120};
\addlegendentry{$P_5=x-\frac{x^3}{6}+\frac{x^5}{120}$}
\node[anchor=west] at (axis cs:-6.5, 2) {\small $\sin x$ 的 Taylor 截断};
\node[anchor=west] at (axis cs:-6.5, 1.5) {\small 阶数越高，逼近范围越大};
\end{axis}
\end{tikzpicture}
\end{document}
