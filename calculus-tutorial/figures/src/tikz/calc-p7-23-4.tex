\documentclass[tikz,border=6pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usetikzlibrary{arrows.meta, positioning, shapes.geometric}
\begin{document}
\begin{tikzpicture}[
  node distance=0.8cm,
  eqbox/.style={rectangle, draw, rounded corners=3pt, fill=orange!10,
                text centered, font=\small, minimum width=5.8cm, minimum height=1.0cm,
                inner sep=4pt},
  subbox/.style={rectangle, draw, rounded corners=3pt, fill=blue!10,
                 text centered, font=\small, minimum width=5.8cm, minimum height=1.0cm,
                 inner sep=4pt},
  arrowstyle/.style={-Stealth, thick, blue!60!black},
  titlenode/.style={font=\normalsize\bfseries}
]

% Left column: original Bernoulli
\node[titlenode] (ltitle) at (-3.5, 5.5) {原 Bernoulli 方程};
\node[eqbox] (beq) at (-3.5,4.5)
  {$y' + P(x)\,y = Q(x)\,y^n$\\[2pt]（$n\neq0,1$，非线性）};
\node[eqbox] (bstep1) at (-3.5,3.0)
  {两边除以 $y^n$：\\[2pt]$y^{-n}y' + P(x)\,y^{1-n} = Q(x)$};
\node[eqbox, fill=green!10] (bsub) at (-3.5,1.5)
  {换元 $z = y^{1-n}$\\[2pt]$z' = (1-n)\,y^{-n}\,y'$};

% Right column: linear equation
\node[titlenode] (rtitle) at (3.5, 5.5) {换元后线性方程};
\node[subbox] (leq) at (3.5,4.5)
  {$z' + (1-n)P(x)\,z = (1-n)Q(x)$\\[2pt]（关于 $z$ 的一阶线性，可套公式）};
\node[subbox] (lsol) at (3.5,3.0)
  {积分因子 $\mu = e^{\int (1-n)P\,dx}$\\[2pt]$(\mu z)' = (1-n)Q\mu$};
\node[subbox, fill=red!10] (lres) at (3.5,1.5)
  {解出 $z$，再回代 $y = z^{1/(1-n)}$};

% Arrows left column
\draw[arrowstyle] (beq) -- (bstep1);
\draw[arrowstyle] (bstep1) -- (bsub);

% Arrows right column
\draw[arrowstyle] (leq) -- (lsol);
\draw[arrowstyle] (lsol) -- (lres);

% Central arrow (conversion)
\draw[arrowstyle, very thick, red!70!black]
  (bsub) -- node[above, font=\small, fill=white, inner sep=2pt]
  {代入 $z$} (leq);

% Bottom example
\node[rectangle, draw, fill=gray!10, rounded corners=3pt, font=\small,
      text width=11cm, align=center, inner sep=5pt] (ex) at (0,-0.2)
  {例：$y'+\frac{1}{x}y=x^2y^2$ （$n=2$）$\xrightarrow{z=y^{-1}}$ $z'-\frac{1}{x}z=-x^2$ $\xrightarrow{\text{解}}$ $z=Cx-\frac{x^3}{2}$};

\draw[arrowstyle] (lres) -- (ex);
\draw[arrowstyle] (bsub) -- (ex);

% Overall title
\node[font=\large\bfseries] at (0, 6.6) {Bernoulli 方程换元前后对比};

\end{tikzpicture}
\end{document}
