\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{calc,arrows.meta,decorations.markings}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=12cm, height=8cm,
  xmin=-0.5, xmax=4.2,
  ymin=-0.5, ymax=3.5,
  xlabel={$x$}, ylabel={$y$},
  axis lines=center,
  xlabel style={right},
  ylabel style={above},
  title={\textbf{保守场：势函数 $u=xy$ 的等高线 + 路径无关}},
  title style={font=\small},
  view={0}{90},
  clip=true,
  restrict y to domain=-0.5:3.5,
]
% 势函数 u = xy 的等高线（手动绘制几条 u=c 双曲线，限制 y 范围避免拖尾）
\foreach \c in {0.5, 1, 1.5, 2, 2.5, 3, 4, 5, 6, 7} {
  \pgfmathsetmacro{\xstart}{\c/3.4}
  \addplot[gray!40, samples=60, domain=\xstart:4]
    {\c/x};
}
% 等高线 u = 3 高亮
\addplot[blue!30, very thick, samples=80, domain=1:4]
  {3/x};
\node[blue!50!black, font=\footnotesize] at (axis cs: 3.6, 1.05) {$u=3$};

% 起点 A、终点 B
\node[red] at (axis cs: 1, 1) {$\bullet$};
\node[red, anchor=north east, font=\small] at (axis cs: 1, 1) {$A(1,1)$};
\node[red] at (axis cs: 3, 2) {$\bullet$};
\node[red, anchor=south west, font=\small] at (axis cs: 3, 2) {$B(3,2)$};

% 路径 1：折线（先 x 后 y）
\draw[->, very thick, orange!80!black,
  decoration={markings, mark=at position 0.5 with {\arrow{>}}},
  postaction={decorate}]
  (axis cs: 1, 1) -- (axis cs: 3, 1);
\draw[->, very thick, orange!80!black,
  decoration={markings, mark=at position 0.5 with {\arrow{>}}},
  postaction={decorate}]
  (axis cs: 3, 1) -- (axis cs: 3, 2);
\node[orange!80!black, font=\small] at (axis cs: 2.0, 0.7) {路径 $L_1$ (折线)};

% 路径 2：直线 A→B
\draw[->, very thick, purple,
  decoration={markings, mark=at position 0.5 with {\arrow{>}}},
  postaction={decorate}]
  (axis cs: 1, 1) -- (axis cs: 3, 2);
\node[purple, font=\small] at (axis cs: 1.5, 2.1) {路径 $L_2$ (直线)};

% 路径 3：抛物线 A→B
\addplot[
  ->, very thick, teal!70!black,
  domain=1:3, samples=30,
  decoration={markings, mark=at position 0.5 with {\arrow{>}}},
  postaction={decorate}
] {1 + 0.25*(x-1)*(x-1)};
\node[teal!70!black, font=\small] at (axis cs: 3.3, 2.6) {路径 $L_3$ (曲线)};

% 结论框
\node[draw=gray!60, fill=gray!10, fill opacity=0.92,
      text opacity=1, rounded corners=2pt, font=\footnotesize, align=left]
  at (axis cs: 1.0, 3.1)
  {$\mathbf{F}=(y,x)$，势函数 $u=xy$\\[2pt]
   $\displaystyle\int_L y\,dx+x\,dy=u(B)-u(A)=6-1=5$\\[2pt]
   \textbf{与路径无关}：$L_1,L_2,L_3$ 同值};
\end{axis}
\end{tikzpicture}
\end{document}
