\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=10cm, height=7cm,
    xmin=0.5, xmax=5, ymin=-0.1, ymax=2,
    xlabel={$n$}, ylabel={$1/n^p$},
    samples=200, grid=both, grid style={gray!20},
    xtick={1,2,3,4,5}, ytick={0,0.5,1,1.5,2},
    legend pos=north east, legend style={font=\small},
    title={p-级数 $\sum 1/n^p$ 的通项衰减速度},
]
\addplot[blue, thick, domain=0.6:5] {x^(-2)};
\addlegendentry{$p=2>1$（收敛）};
\addplot[red, thick, domain=0.6:5] {x^(-1)};
\addlegendentry{$p=1$（调和，发散）};
\addplot[orange, thick, domain=0.6:5] {x^(-0.5)};
\addlegendentry{$p=0.5<1$（发散）};
\draw[dashed, gray] (axis cs:0.5,0) -- (axis cs:5,0);
\node[anchor=north west, font=\small, blue] at (axis cs:1.5,0.25) {衰减快$\Rightarrow$收};
\node[anchor=south west, font=\small, orange] at (axis cs:2,0.72) {衰减慢$\Rightarrow$散};
\end{axis}
\end{tikzpicture}
\end{document}
