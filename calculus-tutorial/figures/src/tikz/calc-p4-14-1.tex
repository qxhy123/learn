\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\usepgfplotslibrary{fillbetween}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=10cm, height=6cm,
    xmin=0, xmax=8, ymin=-0.5, ymax=2,
    xlabel={$x$}, ylabel={$y$},
    samples=200, grid=both, grid style={gray!20},
]
\addplot[name path=A, blue, thick, domain=0.5:8] {1/x^2};
\addplot[name path=Z, draw=none, domain=1:8] {0};
\addplot[orange!30] fill between[of=A and Z, soft clip={domain=1:8}];
\draw[red, dashed] (axis cs:8,0) -- (axis cs:8,1);
\node[anchor=north, red] at (axis cs:8,0) {$+\infty$};
\node[anchor=north] at (axis cs:1,0) {$1$};
\node[anchor=west] at (axis cs:3,1.5) {\small **无穷积分**};
\node[anchor=west] at (axis cs:3,1.2) {\small $\int_1^{+\infty}\frac{1}{x^2}\,dx = \lim_{t\to+\infty}\int_1^t = 1$};
\node[anchor=west] at (axis cs:3,0.85) {\small 收敛！（$p=2>1$）};
\end{axis}
\end{tikzpicture}
\end{document}
