\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=10cm, height=7cm,
  xmin=-0.2, xmax=3.5,
  ymin=-1.5, ymax=1.8,
  xlabel={$x$}, ylabel={$y$},
  axis lines=center,
  title={曲线参数化与弧长元 $ds$：$\mathbf{r}(t)=(t,\sin t)$},
  title style={font=\small},
  xlabel style={right},
  ylabel style={above},
  clip=false,
]
% 参数曲线: x=t, y=sin(t), t in [0, pi]
\addplot[blue, very thick, domain=0:3.14, samples=60]
  ({x},{sin(deg(x))});
% 箭头（方向标记）
\addplot[->, blue, very thick, domain=1.55:1.57, samples=3]
  ({x},{sin(deg(x))});

\node[blue, font=\small] at (axis cs:1.57,1.5) {$\mathbf{r}(t)=(t,\sin t)$};
\node[blue, font=\small] at (axis cs:2.8,0.5) {$t$ 增大方向 $\rightarrow$};

% 点 P = t=1.2
\addplot[mark=*, mark size=2.5pt, red] coordinates {(1.2, 0.9320)};
\node[red, above left, font=\small] at (axis cs:1.2,0.932) {$P(t_0)$};

% 切向量（手动计算）
\addplot[->, red, very thick] coordinates {(1.2,0.932) (1.7,1.222)};
\node[red, above right, font=\small] at (axis cs:1.7,1.22) {$\mathbf{r}'=(1,\cos t)$};

% ds 弧段
\addplot[orange, very thick, domain=1.2:1.7, samples=15]
  ({x},{sin(deg(x))});
\node[orange!80!black, below, font=\small] at (axis cs:1.45,0.85) {$\Delta s$};

% 公式框
\node[font=\small, fill=gray!10, draw=gray!50, rounded corners=2pt, align=left]
  at (axis cs:2.8,-0.9)
  {$ds = \sqrt{x'^2+y'^2}\,dt$\\
   $\int_L f\,ds = \int_a^b f\sqrt{x'^2+y'^2}\,dt$};
\end{axis}
\end{tikzpicture}
\end{document}
