\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines=middle, width=10cm, height=6cm,
    xmin=-3, xmax=3, ymin=-3, ymax=5,
    xlabel={$x$}, ylabel={$y$},
    samples=200, grid=both, grid style={gray!20},
]
\addplot[blue, thick, domain=-2.5:2.5] {x^3 - 3*x};
% 极大 (-1, 2), 极小 (1, -2)
\fill[red] (axis cs:-1, 2) circle (2pt);
\fill[red] (axis cs:1, -2) circle (2pt);
\node[anchor=south west, red] at (axis cs:-1, 2) {\small 极大值 $f''<0$ 凹};
\node[anchor=north west, red] at (axis cs:1, -2) {\small 极小值 $f''>0$ 凸};
% 拐点 (0, 0)
\fill[green!50!black] (axis cs:0, 0) circle (2pt);
\node[anchor=west, green!50!black] at (axis cs:0.2, -0.5) {\small 拐点 $f''=0$};
\node[anchor=west] at (axis cs:-2.8, 4.5) {\small $f(x)=x^3-3x$};
\node[anchor=west] at (axis cs:-2.8, 4) {\small 极值（$f'=0$ + 变号）+ 拐点（$f''$ 变号）};
\end{axis}
\end{tikzpicture}
\end{document}
