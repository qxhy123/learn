\begin{tikzpicture}
\begin{axis}[
  width=11cm, height=7cm,
  xmin=0, xmax=7,
  ymin=-1.2, ymax=1.3,
  xlabel={$t$}, ylabel={$y$},
  title={阻尼振动：$y''+2y'+5y=0$，初值 $y(0)=1$，$y'(0)=0$},
  title style={font=\small},
  xlabel style={font=\small},
  ylabel style={font=\small},
  tick label style={font=\footnotesize},
  grid=both, grid style={gray!20},
  axis lines=left,
  samples=300,
  legend pos=north east,
  legend style={font=\footnotesize},
]

% Solution: alpha=-1, beta=2
% y(0)=1: C1=1; y'(0)=0 => -C1 + 2*C2 = 0 => C2=1/2
\addplot[blue, very thick, domain=0:7]
  {exp(-x)*(cos(deg(2*x)) + 0.5*sin(deg(2*x)))};
\addlegendentry{$y=e^{-t}(\cos 2t + \frac{1}{2}\sin 2t)$}

% Envelope curves
\addplot[red, dashed, thick, domain=0:7] {exp(-x)};
\addlegendentry{包络 $\pm e^{-t}$}
\addplot[red, dashed, thick, domain=0:7] {-exp(-x)};

% Zero line
\addplot[gray, thin, domain=0:7] {0};

% Annotations
\node[font=\footnotesize, fill=yellow!20, draw, rounded corners=2pt, inner sep=3pt]
  at (axis cs:5, 0.9)
  {衰减因子 $e^{-t}$：振荡趋零};

\node[font=\footnotesize, fill=white, inner sep=2pt]
  at (axis cs:3, -0.9)
  {特征根 $-1\pm2i$，振动周期 $T=\pi$};

\end{axis}
