\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath,amssymb}
\usetikzlibrary{arrows.meta,calc}
\begin{document}
\begin{tikzpicture}[>=Stealth, line join=round]

% ——— 左图：y 优先（先对 y 积分）———
\begin{scope}[xshift=0cm]
\draw[->] (-0.3,0) -- (3.8,0) node[right] {$x$};
\draw[->] (0,-0.3) -- (0,3.5) node[above] {$y$};
\node[font=\small, above, align=center] at (1.7,3.3) {$y$ 优先（先积 $y$）};

% 区域 D: 0<=x<=3, 0<=y<=x (三角形)
\fill[blue!15] (0,0) -- (3,0) -- (3,3) -- cycle;
\draw[blue!70, thick] (0,0) -- (3,0) -- (3,3) -- cycle;
\node[blue!60!black, font=\small] at (2.8,2.0) {$y=x$};
\draw[blue!60, dashed] (0,0) -- (3,3);

% 垂直切片（固定 x，y 从 0 到 x）
\fill[red!25] (1.8-0.08,0) rectangle (1.8+0.08,1.8);
\draw[red!70, very thick] (1.8,0) -- (1.8,1.8);
\draw[<->] (2.1,0) -- (2.1,1.8) node[midway, right, font=\small] {$y:0\to x$};
\node[red, font=\small, below] at (1.8,-0.15) {$x_0$};

% 积分式
\node[font=\small, align=center] at (0.9,0.6)
  {$\int_0^3\!\int_0^x\! f\,dy\,dx$};
\end{scope}

% ——— 右图：x 优先（先对 x 积分）———
\begin{scope}[xshift=5cm]
\draw[->] (-0.3,0) -- (3.8,0) node[right] {$x$};
\draw[->] (0,-0.3) -- (0,3.5) node[above] {$y$};
\node[font=\small, above, align=center] at (1.7,3.3) {$x$ 优先（先积 $x$）};

% 同一区域
\fill[green!15] (0,0) -- (3,0) -- (3,3) -- cycle;
\draw[green!60!black, thick] (0,0) -- (3,0) -- (3,3) -- cycle;
\node[green!50!black, font=\small] at (0.6,1.5) {$x=y$};
\draw[green!60, dashed] (0,0) -- (3,3);

% 水平切片（固定 y，x 从 y 到 3）
\fill[orange!30] (1.5,1.5-0.08) rectangle (3,1.5+0.08);
\draw[orange, very thick] (1.5,1.5) -- (3,1.5);
\draw[<->] (1.5,1.8) -- (3,1.8) node[midway, above, font=\small] {$x:y\to 3$};
\node[orange!80!black, font=\small, left] at (-0.05,1.5) {$y_0$};

% 积分式
\node[font=\small, align=center] at (2.4,2.6)
  {$\int_0^3\!\int_y^3\! f\,dx\,dy$};
\end{scope}

% 等号
\node[font=\LARGE] at (4.55,1.5) {$=$};

% 底部说明
\node[font=\small, align=center] at (5.0,-0.55)
  {换序前须画出积分区域，重新确定上下限};
\end{tikzpicture}
