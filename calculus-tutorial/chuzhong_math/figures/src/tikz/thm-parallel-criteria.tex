% 平行判定：三种判定的子图
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta, decorations.markings}
\begin{document}
\begin{tikzpicture}[scale=0.85, thick,
  arrowone/.style={postaction={decorate, decoration={markings,
    mark=at position 0.5 with {\arrow{Stealth}}}}}]
  % ---- 判定 1：同位角相等 ----
  \begin{scope}[shift={(0,0)}]
    \draw[arrowone] (-1.5,1.8) -- (2.5,1.8);
    \draw[arrowone] (-1.5,0) -- (2.5,0);
    \draw (-0.5,2.6) -- (1.1,-0.8);
    % 同位角（同侧、同方向）：左下侧
    \coordinate (P) at (0,1.8);
    \coordinate (Q) at (0.6,0);
    \draw[red] ($(P)+(0.4,0)$) arc (0:-65:0.4);
    \node[red, font=\footnotesize] at ($(P)+(0.55,-0.45)$) {};
    \draw[red] ($(Q)+(0.4,0)$) arc (0:-65:0.4);
    \node[below, font=\small] at (0.5,-1.2) {同位角相等 $\Rightarrow a\parallel b$};
  \end{scope}
  % ---- 判定 2：内错角相等 ----
  \begin{scope}[shift={(5.5,0)}]
    \draw[arrowone] (-1.5,1.8) -- (2.5,1.8);
    \draw[arrowone] (-1.5,0) -- (2.5,0);
    \draw (-0.5,2.6) -- (1.1,-0.8);
    \coordinate (P) at (0,1.8);
    \coordinate (Q) at (0.6,0);
    % 内错角：P 右下侧，Q 左上侧
    \draw[red] ($(P)+(0.4,0)$) arc (0:-65:0.4);
    \draw[red] ($(Q)+(-0.4,0)$) arc (180:115:0.4);
    \node[below, font=\small] at (0.5,-1.2) {内错角相等 $\Rightarrow a\parallel b$};
  \end{scope}
  % ---- 判定 3：同旁内角互补 ----
  \begin{scope}[shift={(11,0)}]
    \draw[arrowone] (-1.5,1.8) -- (2.5,1.8);
    \draw[arrowone] (-1.5,0) -- (2.5,0);
    \draw (-0.5,2.6) -- (1.1,-0.8);
    \coordinate (P) at (0,1.8);
    \coordinate (Q) at (0.6,0);
    % 同旁内角：P 右下、Q 右上
    \draw[red] ($(P)+(0.4,0)$) arc (0:-65:0.4);
    \draw[red] ($(Q)+(0.4,0)$) arc (0:115:0.4);
    \node[below, font=\small] at (0.5,-1.2) {同旁内角互补 $\Rightarrow a\parallel b$};
  \end{scope}
\end{tikzpicture}
\end{document}
