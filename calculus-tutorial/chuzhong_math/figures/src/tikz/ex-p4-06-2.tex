% 例 2：梯形 AD=3, BC=9, 延长 BA, CD 交于 P，PA=2，求 PB
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[scale=0.7, thick]
  % BC = 9 on y=0; AD = 3, P 上方；选 P 在 (4.5, 6)，则 △PBC 中 BC=9
  % 由相似 PA/PB=1/3, A 在 PB 上 1/3 处；类似 D 在 PC 上 1/3 处
  \coordinate (B) at (0,0);
  \coordinate (C) at (9,0);
  \coordinate (P) at (4.5,6);
  \coordinate (A) at ($(P)!2/3!(B)$);
  \coordinate (D) at ($(P)!2/3!(C)$);

  \draw (B)--(C);
  \draw (A)--(D);
  \draw[red, dashed] (P)--(A);
  \draw[red, dashed] (P)--(D);
  \draw (A)--(B);
  \draw (D)--(C);

  \node[above]       at (P) {$P$};
  \node[left]        at (A) {$A$};
  \node[right]       at (D) {$D$};
  \node[below left]  at (B) {$B$};
  \node[below right] at (C) {$C$};
  \node[above] at ($(A)!0.5!(D)$) {\footnotesize $3$};
  \node[below] at (4.5,0) {\footnotesize $9$};
  \node[left, red]  at ($(P)!0.5!(A)$) {\footnotesize $2$};
\end{tikzpicture}
\end{document}
