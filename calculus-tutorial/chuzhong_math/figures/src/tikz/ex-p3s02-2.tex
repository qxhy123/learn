% X字 例2：两直线交于 O，截两平行线，OA=4,OC=6,OB=5 求 OD
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[scale=0.7, thick]
  \coordinate (O) at (0,0);
  % l1 在上方 y=2, l2 在下方 y=-3
  % A 在 l1 上 距 O 为 4: 方向 atan2(2, ?)... 简单：A = O + 4*(unit upward-left)
  \coordinate (A) at ({4*cos(135)}, {4*sin(135)});  % 上左 (-2.83, 2.83)
  \coordinate (B) at ({5*cos(45)},{5*sin(45)});      % 上右
  % 对顶: C 在 A 反方向距 6
  \coordinate (C) at ({-6*cos(135)},{-6*sin(135)});  % 下右
  \coordinate (D) at ({-7.5*cos(45)},{-7.5*sin(45)}); % 下左 OD=15/2
  % 画两平行线 l1（过 A,B), l2（过 C,D）
  \draw[gray] ($(A)!-0.3!(B)$) -- ($(B)!-0.3!(A)$);
  \draw[gray] ($(C)!-0.3!(D)$) -- ($(D)!-0.3!(C)$);
  \draw (A)--(C);
  \draw (B)--(D);
  \draw[red, dashed] (A)--(B);
  \draw[red, dashed] (C)--(D);
  \filldraw (O) circle (1.5pt);
  \node[above right] at (O) {$O$};
  \node[above left] at (A) {$A$};
  \node[above right] at (B) {$B$};
  \node[below right] at (C) {$C$};
  \node[below left] at (D) {$D$};
  \node at ($(O)!0.5!(A) + (-0.3,0.1)$) {\footnotesize $4$};
  \node at ($(O)!0.5!(B) + (0.3,0)$) {\footnotesize $5$};
  \node at ($(O)!0.5!(C) + (0.3,0)$) {\footnotesize $6$};
\end{tikzpicture}
\end{document}
