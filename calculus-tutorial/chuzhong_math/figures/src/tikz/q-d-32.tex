% D.32 矩形 ABCD（AB=6, BC=8），△BCD 沿 BD 折叠，C → C'
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[scale=0.5, thick]
  \coordinate (A) at (0, 6);
  \coordinate (B) at (0, 0);
  \coordinate (C) at (8, 0);
  \coordinate (D) at (8, 6);
  % C' 是 C 关于直线 BD 的对称点
  % BD: from (0,0) to (8,6), 方向(8,6)/10
  % 投影 C 到 BD：t = (8*8+0*6)/100 = 0.64
  % 投影点 = (5.12, 3.84)
  % C' = 2*(5.12, 3.84) - (8, 0) = (2.24, 7.68)
  \coordinate (Cp) at (2.24, 7.68);
  \coordinate (E) at (intersection of A--Cp and B--D);

  \draw (A) -- (B) -- (C) -- (D) -- cycle;
  \draw (B) -- (D);
  \draw[blue, dashed] (B) -- (Cp) -- (D);
  \draw[red] (A) -- (Cp);

  \fill (E) circle (1.5pt);
  \node[above left]  at (A)  {$A$};
  \node[below left]  at (B)  {$B$};
  \node[below right] at (C)  {$C$};
  \node[above right] at (D)  {$D$};
  \node[above]       at (Cp) {$C'$};
  \node[right]       at (E)  {$E$};
\end{tikzpicture}
\end{document}
