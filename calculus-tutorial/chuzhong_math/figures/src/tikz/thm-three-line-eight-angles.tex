% 三线八角标准图：两直线 a, b 被截线 l 所截，8 个角编号 1..8
% 编号约定（每个交点）：1=左上, 2=右上, 3=右下, 4=左下
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.1, thick]
  % a 上方水平，b 下方水平，l 斜穿
  \draw[-{Stealth[length=3pt]}] (-2.5, 2.2) -- (5, 2.2) node[right] {$a$};
  \draw[{Stealth[length=3pt]}-] (-2.5,2.2) -- (-2.6,2.2);
  \draw[-{Stealth[length=3pt]}] (-2.5,-0.5) -- (5,-0.5) node[right] {$b$};
  \draw[{Stealth[length=3pt]}-] (-2.5,-0.5) -- (-2.6,-0.5);
  % 截线 l：斜率，过 P=(0,2.2) 与 Q=(1.8,-0.5)
  \coordinate (P) at (0,2.2);
  \coordinate (Q) at (1.8,-0.5);
  % 延伸至外侧
  \draw[-{Stealth[length=3pt]}] (Q) -- ($(P)!-0.6!(Q)$) node[above] {$l$};
  \draw (P) -- (Q);
  % 角度计算：l 方向角
  \pgfmathsetmacro{\ang}{atan2(-0.5-2.2, 1.8-0)} % from P 指向 Q
  % 在 P 处 8 个区：用 P 处与水平线（a）的夹角
  % a 方向：右 = 0°，左 = 180°
  % l 方向：朝 Q 角度 \ang（负，约 -56°）；反向 \ang+180
  % 8 个角 = 4 each
  % 角弧半径
  \def\r{0.55}
  % --- P 处 ---
  % ∠1 左上：在 a 左方向 (180°) 与 l 反方向 (\ang+180) 之间 [扫 \ang+180 → 180]
  \draw[red] ($(P)+({\r*cos(\ang+180)},{\r*sin(\ang+180)})$) arc (\ang+180:180:\r);
  \node[red, font=\footnotesize] at ($(P)+({0.7*\r*cos((\ang+180+180)/2)},{0.7*\r*sin((\ang+180+180)/2)})$) {$1$};
  % ∠2 右上：在 0° 与 \ang+180 之间 (即 0 → \ang+180)
  \draw[red] ($(P)+({\r},0)$) arc (0:\ang+180:\r);
  \node[red, font=\footnotesize] at ($(P)+({0.7*\r*cos((0+\ang+180)/2)},{0.7*\r*sin((0+\ang+180)/2)})$) {$2$};
  % ∠3 右下：在 \ang 与 0 之间
  \draw[red] ($(P)+({\r*cos(\ang)},{\r*sin(\ang)})$) arc (\ang:0:\r);
  \node[red, font=\footnotesize] at ($(P)+({0.75*\r*cos(\ang/2)},{0.75*\r*sin(\ang/2)})$) {$3$};
  % ∠4 左下：在 180° 与 \ang+360 之间 (即 180 → \ang+360)
  \draw[red] ($(P)+({\r*cos(180)},0)$) arc (180:\ang+360:\r);
  \node[red, font=\footnotesize] at ($(P)+({0.7*\r*cos((180+\ang+360)/2)},{0.7*\r*sin((180+\ang+360)/2)})$) {$4$};
  % --- Q 处（同样规则）---
  \draw[blue] ($(Q)+({\r*cos(\ang+180)},{\r*sin(\ang+180)})$) arc (\ang+180:180:\r);
  \node[blue, font=\footnotesize] at ($(Q)+({0.7*\r*cos((\ang+180+180)/2)},{0.7*\r*sin((\ang+180+180)/2)})$) {$5$};
  \draw[blue] ($(Q)+({\r},0)$) arc (0:\ang+180:\r);
  \node[blue, font=\footnotesize] at ($(Q)+({0.7*\r*cos((0+\ang+180)/2)},{0.7*\r*sin((0+\ang+180)/2)})$) {$6$};
  \draw[blue] ($(Q)+({\r*cos(\ang)},{\r*sin(\ang)})$) arc (\ang:0:\r);
  \node[blue, font=\footnotesize] at ($(Q)+({0.75*\r*cos(\ang/2)},{0.75*\r*sin(\ang/2)})$) {$7$};
  \draw[blue] ($(Q)+({\r*cos(180)},0)$) arc (180:\ang+360:\r);
  \node[blue, font=\footnotesize] at ($(Q)+({0.7*\r*cos((180+\ang+360)/2)},{0.7*\r*sin((180+\ang+360)/2)})$) {$8$};

  \filldraw (P) circle (1.2pt) node[above right, font=\footnotesize] {$P$};
  \filldraw (Q) circle (1.2pt) node[below right, font=\footnotesize] {$Q$};

  \node[below, font=\footnotesize, align=center] at (1.2,-1.0)
    {同位角：$\angle 1\text{-}\angle 5,\ \angle 2\text{-}\angle 6,\ \angle 3\text{-}\angle 7,\ \angle 4\text{-}\angle 8$\\
     内错角：$\angle 3\text{-}\angle 5,\ \angle 4\text{-}\angle 6$\quad 同旁内角：$\angle 3\text{-}\angle 6,\ \angle 4\text{-}\angle 5$};
\end{tikzpicture}
\end{document}
