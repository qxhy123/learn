% 折叠模型：矩形 ABCD 沿对角线 BD 折，△ABD 折到 △A'BD
% A=(0,0)、B=(8,0)、C=(8,6)、D=(0,6)；A 关于 BD 的对称点 A'
\documentclass[tikz,border=4pt]{standalone}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=0.55, thick]
  \coordinate (A) at (0, 0);
  \coordinate (B) at (8, 0);
  \coordinate (C) at (8, 6);
  \coordinate (D) at (0, 6);

  % A 关于直线 BD 的对称点
  % BD 方向：D - B = (-8, 6)，单位 u = (-0.8, 0.6)；BD 法向 n = (0.6, 0.8)
  % A - B = (-8, 0)；沿 n 的分量 = (-8)*0.6 + 0*0.8 = -4.8
  % A' = A - 2*(-4.8)*n = A + 9.6*(0.6, 0.8) = (5.76, 7.68)
  \coordinate (Ap) at (5.76, 7.68);

  % 浅灰填充原 △ABD
  \fill[black!10] (A) -- (B) -- (D) -- cycle;
  % 浅红填充折后 △A'BD
  \fill[red!15] (Ap) -- (B) -- (D) -- cycle;

  % 矩形原始边
  \draw (A) -- (B) -- (C) -- (D) -- cycle;
  % 折痕 BD（粗）
  \draw[very thick] (B) -- (D);
  % 折后三角形边
  \draw[red] (B) -- (Ap) -- (D);

  % A → A' 箭头（虚线）
  \draw[->, dashed, gray] (A) to[bend left=30] (Ap);
  \node[gray] at (2.0, 4.0) {折痕};

  \node[below left]  at (A)  {$A$};
  \node[below right] at (B)  {$B$};
  \node[above right] at (C)  {$C$};
  \node[above left]  at (D)  {$D$};
  \node[above]       at (Ap) {$A'$};
\end{tikzpicture}
\end{document}
