% 例 3（将军饮马 + 坐标）：A(1,2), B(5,4), A'(1,-2), P=(7/3,0)
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[thick, scale=0.7]
  \draw[->, gray] (-0.5,0) -- (6.5,0) node[right, font=\footnotesize] {$x$};
  \draw[->, gray] (0,-3) -- (0,5) node[above, font=\footnotesize] {$y$};
  \node[font=\footnotesize, below left, gray] at (0,0) {$O$};
  \coordinate (A) at (1,2);
  \coordinate (B) at (5,4);
  \coordinate (Ap) at (1,-2);
  \coordinate (P) at (2.333,0);
  \filldraw (A) circle (1.8pt) node[above left, font=\footnotesize] {$A(1,2)$};
  \filldraw (B) circle (1.8pt) node[above right, font=\footnotesize] {$B(5,4)$};
  \filldraw[red] (Ap) circle (1.8pt) node[below left, red, font=\footnotesize] {$A'(1,-2)$};
  \filldraw (P) circle (1.8pt) node[above right, font=\footnotesize] {$P$};
  \draw[red, thick] (Ap) -- (B);
  \draw[thick] (A) -- (P) -- (B);
  \draw[gray, dashed] (A) -- (Ap);
  % 对称箭头
  \node[font=\tiny, gray] at (1.3,0) {关于 $x$ 轴};
\end{tikzpicture}
\end{document}
