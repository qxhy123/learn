% 直线/射线/线段三者对比
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  % 直线 AB（两端箭头）
  \draw[{Stealth[length=4pt]}-{Stealth[length=4pt]}] (-0.6,3) -- (3.6,3);
  \filldraw (0.4,3) circle (1.5pt) node[above] {$A$};
  \filldraw (2.6,3) circle (1.5pt) node[above] {$B$};
  \node[left] at (-0.6,3) {直线 $AB$};

  % 射线 OA（端点 + 一端箭头）
  \draw[-{Stealth[length=4pt]}] (0.4,1.5) -- (3.6,1.5);
  \filldraw (0.4,1.5) circle (1.5pt) node[above] {$O$};
  \filldraw (2.6,1.5) circle (1.5pt) node[above] {$A$};
  \node[left] at (-0.6,1.5) {射线 $OA$};

  % 线段 AB（两端点，无延伸）
  \draw (0.4,0) -- (2.6,0);
  \filldraw (0.4,0) circle (1.5pt) node[above] {$A$};
  \filldraw (2.6,0) circle (1.5pt) node[above] {$B$};
  \node[left] at (-0.6,0) {线段 $AB$};
\end{tikzpicture}
\end{document}
