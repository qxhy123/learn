% 例 2：塔高问题（仰角 30°，水平距离 50 米）
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \coordinate (O) at (0,0);
  \coordinate (F) at (4.5,0);
  \coordinate (T) at (4.5,2.6);
  \draw (O) -- (F) -- (T) -- cycle;
  \draw (4.25,0) -- (4.25,0.25) -- (4.5,0.25);
  \draw (0.9,0) arc (0:30:0.9);
  \node at (1.1, 0.25) {\footnotesize $30^\circ$};
  \fill (O) circle (1.2pt); \fill (F) circle (1.2pt); \fill (T) circle (1.2pt);
  \node[below left] at (O) {\footnotesize 人};
  \node[above] at (T) {\footnotesize 塔顶};
  \node[below] at (2.25,0) {\footnotesize $50$ 米};
  \node[right] at (4.5,1.3) {\footnotesize $h$};
\end{tikzpicture}
\end{document}
