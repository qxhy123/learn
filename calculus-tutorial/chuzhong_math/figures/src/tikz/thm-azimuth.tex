% 方位角示意：北偏东 30°
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \coordinate (O) at (0,0);
  \draw[->] (O) -- (0,2.4) node[above]{\footnotesize 北};
  \draw[->] (O) -- (0,-2.4) node[below]{\footnotesize 南};
  \draw[->] (O) -- (2.4,0) node[right]{\footnotesize 东};
  \draw[->] (O) -- (-2.4,0) node[left]{\footnotesize 西};
  % 北偏东 30° 方向（与 y 轴正向逆时针 -30°，即向东转 30°）
  \coordinate (P) at ({2.4*sin(30)}, {2.4*cos(30)});
  \draw[red, thick, -{Stealth[length=2.5mm]}] (O) -- (P);
  % 角弧
  \draw (0, 1.1) arc (90:60:1.1);
  \node at (0.32, 1.4) {\footnotesize $30^\circ$};
  \fill (O) circle (1.3pt);
  \node[below left] at (O) {$O$};
  \node[above right] at (P) {\footnotesize 目标};
  \node[below] at (0,-2.8) {\footnotesize 北偏东 $30^\circ$};
\end{tikzpicture}
\end{document}
