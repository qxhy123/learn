% 半角 例1：正方形边1，∠EAF=45°，求△CEF 周长 (=2)
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[scale=3.0, thick]
  \coordinate (A) at (0,0);
  \coordinate (B) at (1,0);
  \coordinate (C) at (1,1);
  \coordinate (D) at (0,1);
  \coordinate (E) at (1,0.4);
  \coordinate (F) at (0.3,1);
  \draw (A)--(B)--(C)--(D)--cycle;
  \draw (A)--(E)--(F)--cycle;
  \draw[red, very thick] (E)--(F);
  % 阴影 △CEF
  \fill[red, opacity=0.1] (C)--(E)--(F)--cycle;
  \draw (0.2,0) arc[start angle=0, end angle=21.8, radius=0.2];
  \node at (0.16,0.05) {\tiny $45^\circ$};
  \node[below left] at (A) {$A$};
  \node[below right] at (B) {$B$};
  \node[above right] at (C) {$C$};
  \node[above left] at (D) {$D$};
  \node[right] at (E) {$E$};
  \node[above] at (F) {$F$};
  \node[below=8pt] at (0.5,-0.05) {\footnotesize 边长 $=1$};
\end{tikzpicture}
\end{document}
