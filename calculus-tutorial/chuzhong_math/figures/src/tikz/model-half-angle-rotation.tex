% 半角模型的旋转过程图：单独突出 △ADF → △ABF' 的旋转
% 仅保留正方形作为底图，强调旋转关系
\documentclass[tikz,border=4pt]{standalone}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \coordinate (A) at (0, 0);
  \coordinate (B) at (4, 0);
  \coordinate (C) at (4, 4);
  \coordinate (D) at (0, 4);
  \coordinate (F)  at (1, 4);     % F 在 CD 上，DF=1
  \coordinate (Fp) at (-1, 0);    % F' 在 BA 延长线上，BF'=DF=1

  \draw (A) -- (B) -- (C) -- (D) -- cycle;

  % △ADF 红色实线（待旋转）
  \fill[red!10] (A) -- (D) -- (F) -- cycle;
  \draw[red, thick] (A) -- (D) -- (F) -- cycle;

  % △ABF' 蓝虚线（旋转后）
  \fill[blue!10] (A) -- (B) -- (Fp) -- cycle;
  \draw[blue, dashed, thick] (A) -- (B) -- (Fp) -- cycle;

  % 旋转箭头（从 D 弧形指向 B），中心在 A，半径 1.0
  \draw[->, gray, thick] ({1.0*cos(95)}, {1.0*sin(95)})
       arc[start angle=95, end angle=-5, radius=1.0];
  \node[gray] at (1.1, 1.4) {\footnotesize $90^\circ$ 旋转};

  \node[below left]  at (A)  {$A$};
  \node[below right] at (B)  {$B$};
  \node[above right] at (C)  {$C$};
  \node[above left]  at (D)  {$D$};
  \node[above]       at (F)  {$F$};
  \node[below]       at (Fp) {$F'$};
\end{tikzpicture}
\end{document}
