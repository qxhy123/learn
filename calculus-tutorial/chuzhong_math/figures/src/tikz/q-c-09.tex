% C.9 △ABC, ∠A=50°, ∠B=60°, 延长 BC 至 D，外角 ∠ACD
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \coordinate (B) at (0,0);
  \coordinate (C) at (4,0);
  \coordinate (D) at (5.5,0);
  % A located by ∠B=60°, ∠C=70° (since 180-50-60=70 inside)
  \pgfmathsetmacro{\hh}{4*tan(60)*tan(70)/(tan(60)+tan(70))}
  \pgfmathsetmacro{\xa}{\hh/tan(60)}
  \coordinate (A) at (\xa, \hh);
  \draw (B) -- (C) -- (A) -- cycle;
  \draw[dashed] (C) -- (D);
  \node[below left] at (B) {$B$};
  \node[below] at (C) {$C$};
  \node[right] at (D) {$D$};
  \node[above] at (A) {$A$};
  % angles
  \draw (0.5,0) arc (0:60:0.5);
  \node[font=\footnotesize] at (0.5,0.25) {$60°$};
  \node[font=\footnotesize, below right] at (A) {$50°$};
  % external angle at C
  \draw ({4+0.5*cos(180-70)}, {0.5*sin(180-70)}) arc (110:0:0.5);
  \node[font=\footnotesize] at (4.55, 0.32) {?};
\end{tikzpicture}
\end{document}
