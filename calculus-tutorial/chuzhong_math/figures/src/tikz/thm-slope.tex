% 坡度示意：坡角 α、铅直高度 h、水平距离 l
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \coordinate (A) at (0,0);
  \coordinate (B) at (4,0);
  \coordinate (C) at (4,1.6);
  \draw (A) -- (B) -- (C) -- cycle;
  \draw (3.75,0) -- (3.75,0.25) -- (4,0.25);
  % 坡角 α
  \draw (0.7,0) arc (0:22:0.7);
  \node at (0.92, 0.16) {\footnotesize $\alpha$};
  \fill (A) circle (1.2pt); \fill (B) circle (1.2pt); \fill (C) circle (1.2pt);
  \node[above] at (C) {\footnotesize 坡顶};
  \node[below] at (2,0) {\footnotesize $l$（水平距离）};
  \node[right] at (4,0.8) {\footnotesize $h$（铅直高度）};
  \node[above left] at (2,0.8) {\footnotesize 坡面};
  \node[below] at (2,-0.6) {\footnotesize $\tan\alpha=\dfrac{h}{l}=i$};
\end{tikzpicture}
\end{document}
