% p3b-01 例3：等腰三角形，腰 x, 底 20-2x
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, decorations.markings}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick,
  tick/.style={postaction={decorate, decoration={markings,
    mark=at position 0.5 with {\draw (-1.5pt,-3pt) -- (1.5pt,3pt);}}}}]
  \coordinate (A) at (0, 2.8);
  \coordinate (B) at (-2, 0);
  \coordinate (C) at (2, 0);
  \draw[tick] (A) -- (B);
  \draw[tick] (A) -- (C);
  \draw (B) -- (C);
  \node[above] at (A) {$A$};
  \node[below left] at (B) {$B$};
  \node[below right] at (C) {$C$};
  \node[left] at ($(A)!0.5!(B)$) {\footnotesize 腰 $x$};
  \node[right] at ($(A)!0.5!(C)$) {\footnotesize 腰 $x$};
  \node[below] at ($(B)!0.5!(C)$) {\footnotesize 底 $20-2x$};
\end{tikzpicture}
\end{document}
