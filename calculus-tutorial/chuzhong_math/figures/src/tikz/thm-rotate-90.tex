% 绕原点逆时针旋转 90°：A(1,2)→A'(-2,1), B(4,1)→B'(-1,4), C(3,4)→C'(-4,3)
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[thick, scale=0.55]
  \draw[->, gray] (-5.5,0) -- (5.5,0) node[right, font=\footnotesize] {$x$};
  \draw[->, gray] (0,-1.5) -- (0,5.5) node[above, font=\footnotesize] {$y$};
  \foreach \i in {-5,...,5} \draw[gray, very thin] (\i,-1.5) -- (\i,5.5);
  \foreach \j in {-1,...,5} \draw[gray, very thin] (-5.5,\j) -- (5.5,\j);
  \node[font=\footnotesize, below left, gray] at (0,0) {$O$};
  % 原三角形
  \draw[blue, thick] (1,2) -- (4,1) -- (3,4) -- cycle;
  \node[blue, font=\footnotesize, below] at (1,2) {$A$};
  \node[blue, font=\footnotesize, below right] at (4,1) {$B$};
  \node[blue, font=\footnotesize, above] at (3,4) {$C$};
  % 旋转后
  \draw[red, thick, dashed] (-2,1) -- (-1,4) -- (-4,3) -- cycle;
  \node[red, font=\footnotesize, below left] at (-2,1) {$A'$};
  \node[red, font=\footnotesize, right] at (-1,4) {$B'$};
  \node[red, font=\footnotesize, above left] at (-4,3) {$C'$};
  % 旋转箭头
  \draw[->, orange, thick] (2.5,0.3) arc (-20:120:1.6);
  \node[font=\footnotesize, orange] at (1.5,-0.8) {绕 $O$ 逆时针 $90°$};
\end{tikzpicture}
\end{document}
