% 中心投影：路灯下两人/物体影子从光源发散
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[thick, scale=1.0]
  % 地面
  \draw[gray] (-0.3,0) -- (9.5,0);
  % 灯柱 (高 4) 在 x=0
  \coordinate (B) at (0,0);
  \coordinate (A) at (0,4);
  \draw[very thick] (B) -- (A);
  \filldraw[orange] (A) circle (3pt);
  \node[left, font=\footnotesize] at (A) {$A$（光源）};
  \node[below left, font=\footnotesize] at (B) {$B$};

  % 物体 1：高 1.5，位于 x=2
  \coordinate (D1) at (2,0);
  \coordinate (C1) at (2,1.5);
  \draw[very thick] (D1) -- (C1);
  % 由 A 经过 C1 延长到地面：参数化
  % 直线 A->C1 方向 (2-0, 1.5-4) = (2, -2.5)。到 y=0 时 t = 4/2.5 = 1.6, x = 2*1.6 = 3.2
  \coordinate (E1) at (3.2, 0);
  \draw[red, thick] (D1) -- (E1);
  \draw[dashed, orange] (A) -- (E1);
  \node[below, font=\footnotesize, red] at ($(D1)!0.5!(E1)$) {影};

  % 物体 2：高 1.5，位于 x=5
  \coordinate (D2) at (5,0);
  \coordinate (C2) at (5,1.5);
  \draw[very thick] (D2) -- (C2);
  % A=(0,4), C2=(5,1.5)，方向 (5,-2.5)，到 y=0 时 t=1.6, x = 5*1.6 = 8
  \coordinate (E2) at (8, 0);
  \draw[red, thick] (D2) -- (E2);
  \draw[dashed, orange] (A) -- (E2);
  \node[below, font=\footnotesize, red] at ($(D2)!0.5!(E2)$) {影};

  \node[below, font=\footnotesize, gray] at (4.5, -0.5) {点光源 $\to$ 光线从 $A$ 发散；离灯越远影越长};
\end{tikzpicture}
\end{document}
