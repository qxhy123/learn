% 倍长中线模型：△ABC，D 为 BC 中点，延长 AD 到 E 使 DE=AD，连 CE
% 结果 △ABD ≅ △ECD（SAS），AB ∥ CE 且 AB = CE
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, decorations.markings}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick,
  tickI/.style={postaction={decorate, decoration={markings,
    mark=at position 0.5 with {\draw (-1.5pt,-3pt) -- (1.5pt,3pt);}}}},
  tickII/.style={postaction={decorate, decoration={markings,
    mark=at position 0.5 with {\draw (-2.5pt,-3pt) -- (-0.5pt,3pt);
                               \draw ( 0.5pt,-3pt) -- ( 2.5pt,3pt);}}}}]

  \coordinate (A) at (0.5, 3);
  \coordinate (B) at (-2, 0);
  \coordinate (C) at (3, 0);
  \coordinate (D) at ($(B)!0.5!(C)$);   % BC 中点
  \coordinate (E) at ($2*(D) - (A)$);   % AD 延长到 E，DE=AD

  \draw (A) -- (B);
  \draw (B) -- (C);
  \draw (A) -- (C);

  % BD、DC 标单短线（中点）
  \draw[tickI] (B) -- (D);
  \draw[tickI] (D) -- (C);

  % AD、DE 标双短线（构造等长）
  \draw[tickII] (A) -- (D);
  \draw[tickII, dashed] (D) -- (E);

  % CE 红色实线
  \draw[red] (C) -- (E);

  \node[above]       at (A) {$A$};
  \node[below left]  at (B) {$B$};
  \node[below right] at (C) {$C$};
  \node[above right] at (D) {$D$};
  \node[below right] at (E) {$E$};
\end{tikzpicture}
\end{document}
