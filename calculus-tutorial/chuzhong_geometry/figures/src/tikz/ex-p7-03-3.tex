% 例 3：方位角 + 勾股 — 甲船北偏东30° 4海里(A)，乙船北偏西60° 3海里(B)
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{arrows.meta, calc}
\begin{document}
\begin{tikzpicture}[scale=0.7, thick]
  \coordinate (O) at (0,0);
  \draw[->] (O) -- (0,4.5) node[above]{\footnotesize 北};
  \draw[->] (O) -- (0,-2) node[below]{\footnotesize 南};
  \draw[->] (O) -- (4.5,0) node[right]{\footnotesize 东};
  \draw[->] (O) -- (-4.5,0) node[left]{\footnotesize 西};
  % A: 北偏东 30°，距离 4
  \coordinate (A) at ({4*sin(30)}, {4*cos(30)}); % (2, 3.464)
  % B: 北偏西 60°，距离 3
  \coordinate (B) at ({-3*sin(60)}, {3*cos(60)}); % (-2.598, 1.5)
  \draw[red, thick] (O) -- (A);
  \draw[red, thick] (O) -- (B);
  \draw[blue, dashed, thick] (A) -- (B);
  % 角弧
  \draw (0, 1.0) arc (90:60:1.0);
  \node at (0.32, 1.27) {\footnotesize $30^\circ$};
  \draw (0, 1.3) arc (90:150:1.3);
  \node at (-0.5, 1.55) {\footnotesize $60^\circ$};

  \fill (O) circle (1.5pt);
  \fill (A) circle (1.5pt);
  \fill (B) circle (1.5pt);
  \node[below right] at (O) {$O$};
  \node[right] at (A) {\footnotesize $A$（甲）};
  \node[left] at (B) {\footnotesize $B$（乙）};
  \node[above] at ($(O)!0.5!(A)$) {\footnotesize $4$};
  \node[above left] at ($(O)!0.5!(B)$) {\footnotesize $3$};
\end{tikzpicture}
\end{document}
