% 常见中心对称图形：平行四边形 + 圆
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  % 平行四边形
  \coordinate (A) at (0,0);
  \coordinate (B) at (2.8,0);
  \coordinate (C) at (3.6,1.6);
  \coordinate (D) at (0.8,1.6);
  \coordinate (O1) at (1.8,0.8);
  \draw (A) -- (B) -- (C) -- (D) -- cycle;
  \draw[gray, dashed] (A) -- (C);
  \draw[gray, dashed] (B) -- (D);
  \fill (O1) circle (1.3pt);
  \node[below right] at (O1) {\footnotesize $O$};
  \node[below left] at (A) {$A$};
  \node[below right] at (B) {$B$};
  \node[above right] at (C) {$C$};
  \node[above left] at (D) {$D$};
  \node[below] at (1.4,-0.4) {\footnotesize 平行四边形};

  % 圆
  \coordinate (O2) at (6.5, 0.8);
  \draw (O2) circle (1.2);
  \fill (O2) circle (1.3pt);
  \node[above right] at (O2) {\footnotesize $O$};
  \node[below] at (6.5,-0.6) {\footnotesize 圆};
\end{tikzpicture}
\end{document}
