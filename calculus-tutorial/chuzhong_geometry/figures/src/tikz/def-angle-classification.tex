% 角的分类：锐角/直角/钝角/平角/周角
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=0.85, thick]
  \def\R{1.1}
  % 锐角 50
  \begin{scope}[shift={(0,0)}]
    \draw[-{Stealth[length=3pt]}] (0,0) -- (1.8,0);
    \draw[-{Stealth[length=3pt]}] (0,0) -- ({1.8*cos(50)}, {1.8*sin(50)});
    \draw (\R,0) arc (0:50:\R);
    \node[below, font=\small] at (0.9,-0.5) {锐角};
    \node[font=\footnotesize] at ({0.7*cos(25)}, {0.7*sin(25)}) {$<90°$};
    \filldraw (0,0) circle (1pt);
  \end{scope}
  % 直角
  \begin{scope}[shift={(4,0)}]
    \draw[-{Stealth[length=3pt]}] (0,0) -- (1.8,0);
    \draw[-{Stealth[length=3pt]}] (0,0) -- (0,1.8);
    \draw (0.3,0) -- (0.3,0.3) -- (0,0.3);
    \node[below, font=\small] at (0.9,-0.5) {直角};
    \node[font=\footnotesize] at (0.6,0.6) {$=90°$};
    \filldraw (0,0) circle (1pt);
  \end{scope}
  % 钝角 130
  \begin{scope}[shift={(8,0)}]
    \draw[-{Stealth[length=3pt]}] (0,0) -- (1.8,0);
    \draw[-{Stealth[length=3pt]}] (0,0) -- ({1.8*cos(130)}, {1.8*sin(130)});
    \draw (\R,0) arc (0:130:\R);
    \node[below, font=\small] at (0.5,-0.5) {钝角};
    \node[font=\footnotesize] at (0.1,1.0) {$<180°$};
    \filldraw (0,0) circle (1pt);
  \end{scope}
  % 平角
  \begin{scope}[shift={(0,-2.8)}]
    \draw[-{Stealth[length=3pt]}] (0,0) -- (1.8,0);
    \draw[-{Stealth[length=3pt]}] (0,0) -- (-1.8,0);
    \draw (\R,0) arc (0:180:\R);
    \node[below, font=\small] at (0,-0.5) {平角 $=180°$};
    \filldraw (0,0) circle (1pt);
  \end{scope}
  % 周角
  \begin{scope}[shift={(5,-2.8)}]
    \draw[-{Stealth[length=3pt]}] (0,0) -- (1.8,0);
    \draw[->, thick] (\R,0) arc (0:355:\R);
    \node[below, font=\small] at (0,-1.6) {周角 $=360°$};
    \filldraw (0,0) circle (1pt);
  \end{scope}
\end{tikzpicture}
\end{document}
