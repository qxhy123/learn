% F / Z / U 形位置识别
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=0.9, thick]
  % ---------- F 形（同位角）----------
  \begin{scope}[shift={(0,0)}]
    \draw[-{Stealth[length=3pt]}] (-1.5,1.8) -- (2.5,1.8);
    \draw[-{Stealth[length=3pt]}] (-1.5,0) -- (2.5,0);
    \draw (0,2.6) -- (0,-0.8);
    % 高亮 F 形（红粗）：截线上一小段 + 两条横线各一截
    \draw[red, very thick] (0,1.8) -- (0,0);
    \draw[red, very thick] (0,1.8) -- (1.5,1.8);
    \draw[red, very thick] (0,0) -- (1.5,0);
    % 两个同位角的角弧
    \draw[red] (0.5,1.8) arc (0:-65:0.5);
    \draw[red] (0.5,0) arc (0:-65:0.5);
    \node[below, font=\small] at (0.5,-1.0) {同位角 \textbf{F} 形};
  \end{scope}
  % ---------- Z 形（内错角）----------
  \begin{scope}[shift={(5,0)}]
    \draw[-{Stealth[length=3pt]}] (-1.5,1.8) -- (2.5,1.8);
    \draw[-{Stealth[length=3pt]}] (-1.5,0) -- (2.5,0);
    \draw (-0.7,2.6) -- (1.3,-0.8);
    % Z：上方 a 上的一段（左 → 截点）+ 截线段 + 下方 b 上一段（截点 → 右）
    \coordinate (Pt) at (0,1.8);
    \coordinate (Qt) at (0.6,0);
    \draw[red, very thick] (-1.2,1.8) -- (Pt);
    \draw[red, very thick] (Pt) -- (Qt);
    \draw[red, very thick] (Qt) -- (1.8,0);
    \node[below, font=\small] at (0.5,-1.0) {内错角 \textbf{Z} 形};
  \end{scope}
  % ---------- U 形（同旁内角）----------
  \begin{scope}[shift={(10,0)}]
    \draw[-{Stealth[length=3pt]}] (-1.5,1.8) -- (2.5,1.8);
    \draw[-{Stealth[length=3pt]}] (-1.5,0) -- (2.5,0);
    \draw (0,2.6) -- (0,-0.8);
    % U：a 上一段（截点 → 右）+ 截线段 + b 上一段（截点 → 右）
    \draw[red, very thick] (0,1.8) -- (1.6,1.8);
    \draw[red, very thick] (0,1.8) -- (0,0);
    \draw[red, very thick] (0,0) -- (1.6,0);
    \node[below, font=\small] at (0.5,-1.0) {同旁内角 \textbf{U} 形};
  \end{scope}
\end{tikzpicture}
\end{document}
