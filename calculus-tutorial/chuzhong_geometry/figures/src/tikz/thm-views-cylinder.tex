% 圆柱三视图
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[thick, scale=1.0]
  \pgfmathsetmacro{\r}{0.9}
  \pgfmathsetmacro{\h}{2.4}
  \pgfmathsetmacro{\yr}{0.3}  % 椭圆短半径
  % --- 左：圆柱立体 ---
  \draw (-\r,0) -- (-\r,\h);
  \draw (\r,0) -- (\r,\h);
  \draw (0,\h) ellipse ({\r} and {\yr});
  \draw (-\r,0) arc (180:360:{\r} and {\yr});
  \draw[dashed, gray] (-\r,0) arc (180:0:{\r} and {\yr});
  \node[below, font=\footnotesize] at (0,-0.8) {（a）圆柱};

  % --- 主视 ---
  \begin{scope}[xshift=3.5cm]
    \draw (0,0) rectangle (2*\r, \h);
    \node[font=\footnotesize, below] at (\r, 0) {主视};
    \node[font=\footnotesize, left] at (0, \h/2) {$h$};
    \node[font=\footnotesize, below] at (\r, \h+0.0) {};
    \draw[<->, gray] (0,-0.25) -- (2*\r,-0.25);
    \node[font=\footnotesize, below, gray] at (\r,-0.25) {$2r$};
  \end{scope}

  % --- 左视 ---
  \begin{scope}[xshift=6.5cm]
    \draw (0,0) rectangle (2*\r, \h);
    \node[font=\footnotesize, below] at (\r, 0) {左视};
  \end{scope}

  % --- 俯视：圆 + 圆心 ---
  \begin{scope}[xshift=9.5cm, yshift=0.6cm]
    \draw (0,\h/2) circle (\r);
    \filldraw (0,\h/2) circle (1pt);
    \node[font=\footnotesize, below] at (0, \h/2 - \r - 0.2) {俯视};
  \end{scope}
\end{tikzpicture}
\end{document}
