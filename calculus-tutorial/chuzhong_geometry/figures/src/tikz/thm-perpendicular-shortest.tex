% 垂线段最短：点 P 到直线 l 各点的距离，PA（垂直）最短
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \draw[-{Stealth[length=3pt]}] (-0.5,0) -- (6,0) node[right] {$l$};
  \coordinate (P) at (3,2.5);
  \coordinate (A) at (3,0);
  \coordinate (B1) at (1,0);
  \coordinate (B2) at (4.5,0);
  \coordinate (B3) at (5.5,0);
  \draw[red, thick] (P) -- (A);
  \draw[gray] (P) -- (B1);
  \draw[gray] (P) -- (B2);
  \draw[gray] (P) -- (B3);
  \draw (A) ++(-0.2,0) -- ++(0,0.2) -- ++(0.2,0);
  \filldraw (P) circle (1.5pt) node[above] {$P$};
  \foreach \pt/\lab/\pos in {A/A/below, B1/B_1/below, B2/B_2/below, B3/B_3/below}
    \filldraw (\pt) circle (1.2pt) node[\pos, font=\footnotesize] {$\lab$};
  \node[red, right, font=\small] at (3,1.2) {$PA$（最短）};
\end{tikzpicture}
\end{document}
