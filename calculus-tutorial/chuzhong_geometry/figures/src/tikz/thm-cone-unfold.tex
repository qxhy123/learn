% 圆锥展开图：扇形（半径 l, 弧长 2πr）+ 底圆
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[thick, scale=0.9]
  \pgfmathsetmacro{\r}{0.7}
  \pgfmathsetmacro{\h}{2.0}
  \pgfmathsetmacro{\yr}{0.22}
  % 圆锥
  \draw (-\r,0) -- (0,\h) -- (\r,0);
  \draw (-\r,0) arc (180:360:{\r} and {\yr});
  \draw[dashed,gray] (-\r,0) arc (180:0:{\r} and {\yr});
  \node[below, font=\footnotesize] at (0,-0.6) {圆锥};
  % 母线长 l 标注
  \node[font=\footnotesize, left] at ($(-\r,0)!0.5!(0,\h)$) {$l$};
  \node[font=\footnotesize, below] at (0,-\yr-0.05) {$r$};

  \draw[->, gray, thick] (\r+0.3, \h/2) -- (\r+1.3, \h/2) node[midway, above, font=\footnotesize] {展开};

  % 扇形：半径 R = 2 单位（代表 l），张角 α 取 120° 演示
  \begin{scope}[xshift=4cm, yshift=0cm]
    \pgfmathsetmacro{\Rs}{2.2}
    \pgfmathsetmacro{\angA}{60}  % 从 90-α/2 到 90+α/2，α=120
    \pgfmathsetmacro{\angB}{120}
    \coordinate (V) at (0,0);
    \draw (V) -- ({\Rs*cos(\angA)},{\Rs*sin(\angA)}) arc (\angA:\angB:\Rs) -- cycle;
    \node[font=\footnotesize, right] at ($(V)!0.5!({\Rs*cos(\angA)},{\Rs*sin(\angA)})$) {$l$};
    % 弧长标注
    \node[font=\footnotesize, above] at (0, \Rs+0.05) {弧长 $=2\pi r$};
    % 圆心角
    \node[font=\footnotesize] at (0.45,0.5) {$\alpha$};

    % 底圆
    \draw (\Rs+0.5+\r, \r+0.2) circle (\r);
    \node[font=\footnotesize] at (\Rs+0.5+\r, \r+0.2) {$r$};
  \end{scope}
\end{tikzpicture}
\end{document}
