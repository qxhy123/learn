% 阿氏圆模型：⊙O；B 在圆内、A 在圆外（OA 较大）；P 在圆上；构造 B' 使 △OPB ~ △OB'P，相似比 k=1/2
% r=2, OB=1, OA=5；B'=OP^2/OB=4 → 在 OA 方向取 B' 使 OB'=4
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \coordinate (O) at (0, 0);
  \coordinate (A) at (5, 0);
  \coordinate (B) at (1, 0);
  \coordinate (Bp) at (4, 0);
  % P 在圆上：取 P 使其在最优位置（P, B' , A 共线时取最小）；这里 A, B' 都在 x 轴上，最优 P 在线段 AB' 上的圆点 → P=(2,0)，但那是直径端点
  % 选示意位置 P=(2cos60°, 2sin60°)=(1, 1.732)
  \coordinate (P) at (1, 1.732);

  \draw[thick] (O) circle (2);
  \filldraw (O) circle (1.5pt) node[below left] {$O$};
  \filldraw (A) circle (2pt) node[below right] {$A$};
  \filldraw (B) circle (2pt) node[below] {$B$};
  \filldraw (Bp) circle (2pt) node[below right] {$B'$};
  \filldraw (P) circle (2pt) node[above] {$P$};

  \draw[gray, dashed] (O) -- (A);
  \draw[red, thick] (P) -- (A);
  \draw[red, thick] (P) -- (B);
  \draw[blue, thick] (P) -- (Bp);

  \node[red, font=\footnotesize] at (3.0, 1.2) {$PA$};
  \node[red, font=\footnotesize] at (0.8, 0.95) {$PB$};
  \node[blue, font=\footnotesize] at (2.5, 0.85) {$PB'=k\,PB$};

  \node[font=\footnotesize, gray] at (2.5, -2.4) {$\triangle OPB \sim \triangle OB'P$，相似比 $k$};
\end{tikzpicture}
\end{document}
