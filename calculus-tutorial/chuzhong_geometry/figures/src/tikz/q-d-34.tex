% D.34 方位角：A→B 北偏东 30° 行 20；B→C 北偏西 60° 行 30
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=0.3, thick]
  \coordinate (A) at (0, 0);
  % 北偏东 30° = 方位角 60°（从 x 轴起）：
  \coordinate (B) at ({20*cos(60)}, {20*sin(60)});
  % B 北偏西 60° = 从 B 沿 (cos(150°), sin(150°)) 方向，距离 30
  \coordinate (C) at ($(B) + ({30*cos(150)}, {30*sin(150)})$);

  \draw[->] (A) -- (0, 14) node[above] {北};
  \draw[->] (B) -- ($(B) + (0, 14)$) node[above] {北};
  \draw (A) -- (B) -- (C);
  \draw[gray, dashed] (A) -- (C);

  \draw (0, 4) arc[start angle=90, end angle=60, radius=4];
  \node at (1.3, 5) {\footnotesize $30^\circ$};
  \draw ($(B) + (0, 4)$) arc[start angle=90, end angle=150, radius=4];
  \node at ($(B) + (-1.8, 4.5)$) {\footnotesize $60^\circ$};

  \node[below left]  at (A) {$A$};
  \node[right]       at (B) {$B$};
  \node[above left]  at (C) {$C$};
\end{tikzpicture}
\end{document}
