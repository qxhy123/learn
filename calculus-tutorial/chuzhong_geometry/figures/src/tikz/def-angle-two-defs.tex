% 角的两种定义：静态（两射线） vs 动态（旋转）
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta, decorations.markings}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  % 左：静态定义
  \begin{scope}[shift={(0,0)}]
    \coordinate (O) at (0,0);
    \draw[-{Stealth[length=3pt]}] (O) -- (3,0) node[right] {};
    \draw[-{Stealth[length=3pt]}] (O) -- ({3*cos(55)}, {3*sin(55)});
    \draw (1.2,0) arc (0:55:1.2);
    \node at ({0.85*cos(27)}, {0.85*sin(27)}) {$\alpha$};
    \filldraw (O) circle (1.5pt);
    \node[below left] at (O) {$O$};
    \node[right] at (3,0) {$A$};
    \node[above] at ({3*cos(55)}, {3*sin(55)}) {$B$};
    \node[below=0.8cm] at (1.5,0) {\small 静态：两射线};
  \end{scope}
  % 右：动态定义
  \begin{scope}[shift={(6,0)}]
    \coordinate (O) at (0,0);
    \draw[-{Stealth[length=3pt]}, gray] (O) -- (3,0);
    \node[gray, right] at (3,0) {\small 始边};
    \draw[-{Stealth[length=3pt]}] (O) -- ({3*cos(55)}, {3*sin(55)});
    \node[right] at ({3*cos(55)}, {3*sin(55)}) {\small 终边};
    \draw[->, red, thick] (1.4,0) arc (0:55:1.4);
    \node[red] at ({1.05*cos(27)}, {1.05*sin(27)}) {\small 旋转};
    \filldraw (O) circle (1.5pt);
    \node[below left] at (O) {$O$};
    \node[below=0.8cm] at (1.5,0) {\small 动态：旋转};
  \end{scope}
\end{tikzpicture}
\end{document}
