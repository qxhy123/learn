% 圆柱展开图：矩形（长 2πr，宽 h）+ 两圆
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[thick, scale=0.9]
  \pgfmathsetmacro{\r}{0.7}
  \pgfmathsetmacro{\h}{2}
  \pgfmathsetmacro{\yr}{0.25}
  % 圆柱
  \draw (-\r,0) -- (-\r,\h);
  \draw (\r,0) -- (\r,\h);
  \draw (0,\h) ellipse ({\r} and {\yr});
  \draw (-\r,0) arc (180:360:{\r} and {\yr});
  \draw[dashed,gray] (-\r,0) arc (180:0:{\r} and {\yr});
  \node[below, font=\footnotesize] at (0,-0.6) {圆柱};
  \node[font=\footnotesize, left] at (-\r, \h/2) {$h$};
  \node[font=\footnotesize, below] at (0, -\yr-0.05) {$r$};

  % 箭头
  \draw[->, gray, thick] (\r+0.3, \h/2) -- (\r+1.3, \h/2) node[midway, above, font=\footnotesize] {展开};

  % 矩形：2πr × h；两端两圆
  \begin{scope}[xshift=3.2cm]
    \pgfmathsetmacro{\L}{2*3.1416*\r}  % 实际算出长度
    \draw (0,0) rectangle (\L, \h);
    \node[font=\footnotesize] at (\L/2, \h/2) {侧面};
    \draw[<->, gray] (0,-0.25) -- (\L,-0.25);
    \node[font=\footnotesize, below, gray] at (\L/2,-0.25) {$2\pi r$};
    \draw[<->, gray] (-0.25,0) -- (-0.25,\h);
    \node[font=\footnotesize, left, gray] at (-0.25,\h/2) {$h$};
    % 两底圆
    \draw (\L+0.5+\r, \h-\r) circle (\r);
    \node[font=\footnotesize] at (\L+0.5+\r, \h-\r) {$r$};
    \draw (\L+0.5+\r, \r) circle (\r);
    \node[font=\footnotesize] at (\L+0.5+\r, \r) {$r$};
  \end{scope}
\end{tikzpicture}
\end{document}
