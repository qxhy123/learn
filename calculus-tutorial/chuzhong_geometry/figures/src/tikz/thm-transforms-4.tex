% 4 种坐标变换：关于 x 轴、y 轴、原点对称、平移；用一个三角形演示
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[thick, scale=0.55]
  % 共享：原三角形 A(1,1), B(3,1), C(2,3)
  \newcommand{\axes}{
    \draw[->, gray] (-3.5,0) -- (3.5,0) node[right, font=\tiny] {$x$};
    \draw[->, gray] (0,-3.5) -- (0,3.5) node[above, font=\tiny] {$y$};
    \node[font=\tiny, below left, gray] at (0,0) {$O$};
  }

  % (a) 关于 x 轴对称
  \begin{scope}[xshift=0cm, yshift=0cm]
    \axes
    \draw[blue, thick] (1,1) -- (3,1) -- (2,3) -- cycle;
    \draw[red, thick, dashed] (1,-1) -- (3,-1) -- (2,-3) -- cycle;
    \node[font=\footnotesize, below] at (0,-4) {（a）关于 $x$ 轴};
  \end{scope}

  % (b) 关于 y 轴对称
  \begin{scope}[xshift=9cm, yshift=0cm]
    \axes
    \draw[blue, thick] (1,1) -- (3,1) -- (2,3) -- cycle;
    \draw[red, thick, dashed] (-1,1) -- (-3,1) -- (-2,3) -- cycle;
    \node[font=\footnotesize, below] at (0,-4) {（b）关于 $y$ 轴};
  \end{scope}

  % (c) 关于原点对称
  \begin{scope}[xshift=18cm, yshift=0cm]
    \axes
    \draw[blue, thick] (1,1) -- (3,1) -- (2,3) -- cycle;
    \draw[red, thick, dashed] (-1,-1) -- (-3,-1) -- (-2,-3) -- cycle;
    \node[font=\footnotesize, below] at (0,-4) {（c）关于原点};
  \end{scope}

  % (d) 平移 (右 2, 下 2)：从 (1,1) 到 (3,-1)
  \begin{scope}[xshift=27cm, yshift=0cm]
    \axes
    \draw[blue, thick] (1,1) -- (3,1) -- (2,3) -- cycle;
    \draw[red, thick, dashed] (3,-1) -- (5,-1) -- (4,1) -- cycle;
    \draw[->, gray, very thin] (2,2) -- (4,0);
    \node[font=\footnotesize, below] at (0,-4) {（d）平移 $(2,-2)$};
  \end{scope}
\end{tikzpicture}
\end{document}
