% 例 1：内角和 1440° → 十边形示意
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[scale=1.0, thick]
  \def\R{2.0}
  \foreach \i in {0,...,9} {
    \coordinate (P\i) at ({90 + \i*36}:\R);
  }
  \draw (P0)
    \foreach \i in {1,...,9} { -- (P\i) } -- cycle;
  \foreach \i in {0,...,9} \filldraw (P\i) circle (1.2pt);
  \node[font=\footnotesize] at (0,0) {内角和 $=1440°$};
  \node[font=\footnotesize, above] at (90:\R+0.15) {$n=10$};
\end{tikzpicture}
\end{document}
