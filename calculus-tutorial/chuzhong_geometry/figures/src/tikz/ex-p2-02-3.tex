% 将军饮马：A, B 同侧，作 A 关于 l 的对称点 A'，连 A'B 与 l 交点 P
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc, arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=0.9, thick]
  \draw[-{Stealth[length=3pt]}] (-0.5,0) -- (7.5,0) node[right] {$l$};
  \coordinate (A) at (1.5,2);
  \coordinate (B) at (5,3);
  \coordinate (Ap) at (1.5,-2);
  % 计算 P = A'B 与 x 轴交点
  % 参数：A'=(1.5,-2)，B=(5,3)，t 使 y=0：t = 2/5
  % P = A' + t*(B-A') = (1.5 + 0.4*3.5, 0) = (1.5+1.4, 0) = (2.9, 0)
  \coordinate (P) at (2.9,0);
  \draw[gray, dashed] (A) -- (Ap);
  \draw (A) -- (P) -- (B);
  \draw[red, dashed] (Ap) -- (B);
  \filldraw (A) circle (1.5pt) node[above] {$A$};
  \filldraw (B) circle (1.5pt) node[above] {$B$};
  \filldraw (Ap) circle (1.5pt) node[below] {$A'$};
  \filldraw (P) circle (1.5pt) node[above right, font=\footnotesize] {$P$};
  \node[font=\footnotesize, below right] at (3.5,1.8) {$PA+PB=A'B$（最小）};
\end{tikzpicture}
\end{document}
