% D.33 仰角：A 测塔顶 C 仰角 30°；前进 20m 至 B 测仰角 45°
\documentclass[tikz,border=4pt]{standalone}
\usepackage{ctex}
\usepackage{amsmath}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[scale=0.3, thick]
  \coordinate (A) at (0, 0);
  \coordinate (B) at (20, 0);
  % 塔基 T，B 至 T = h，A 至 T = h + ?  AB=20，仰角 A=30°，B=45°
  % h/tan30 - h/tan45 = 20 → h(√3 - 1) = 20 → h = 10(√3+1) ≈ 27.32
  \pgfmathsetmacro{\h}{10*(sqrt(3)+1)}
  \coordinate (T) at ({20 + \h}, 0);
  \coordinate (C) at ({20 + \h}, \h);

  \draw[gray] (A) -- (T) -- (C) -- cycle;
  \draw (T) -- (C);
  \draw (A) -- (C);
  \draw (B) -- (C);
  \draw (0,0) -- ({20 + \h + 5}, 0);

  \draw (3, 0) arc[start angle=0, end angle=30, radius=3];
  \node at (4.5, 0.8) {\footnotesize $30^\circ$};
  \draw (23, 0) arc[start angle=0, end angle=45, radius=3];
  \node at (24, 1.2) {\footnotesize $45^\circ$};

  \node[below] at (A) {$A$};
  \node[below] at (B) {$B$};
  \node[below right] at (T) {};
  \node[above right] at (C) {$C$};
  \node[below] at ($(A)!0.5!(B)$) {\footnotesize $20$};
\end{tikzpicture}
\end{document}
