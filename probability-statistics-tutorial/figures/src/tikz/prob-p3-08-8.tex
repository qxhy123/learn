\begin{document}
\begin{tikzpicture}[
  node distance=1.1cm,
  mainbox/.style={draw, rounded corners=6pt, fill=#1, minimum width=3.4cm,
                  minimum height=0.9cm, font=\small, align=center, text width=3.2cm},
  arrow/.style={-Latex, thick, blue!70},
  label/.style={font=\scriptsize, fill=white, inner sep=2pt},
]

% Main distributions
\node[mainbox=blue!25] (normal) {$N(\mu,\sigma^2)$\\正态分布\\中心极限定理极限};
\node[mainbox=green!25, below left=1.6cm and 0.8cm of normal] (chi2)
  {$\chi^2_n$\\卡方分布\\$=\text{Gamma}(n/2,1/2)$};
\node[mainbox=red!20, below right=1.6cm and 0.8cm of normal] (exp)
  {$\text{Exp}(\lambda)$\\指数分布\\$=\text{Gamma}(1,\lambda)$};
\node[mainbox=orange!25, below=1.5cm of chi2] (t)
  {$t_n$\\$t$ 分布\\小样本均值检验};
\node[mainbox=purple!20, below=1.5cm of exp] (gamma)
  {$\text{Gamma}(\alpha,\beta)$\\Gamma 分布\\等待$\alpha$个泊松事件};
\node[mainbox=teal!20, below right=1.5cm and 1.4cm of chi2] (F)
  {$F_{m,n}$\\$F$ 分布\\方差比检验};

% Arrows
\draw[arrow] (normal) -- node[label, left]{平方和\\$\sum Z_i^2$} (chi2);
\draw[arrow] (normal) -- node[label, right]{无记忆\\$\alpha=1$} (exp);
\draw[arrow] (chi2) -- node[label, left]{$Z/\sqrt{\chi^2_n/n}$} (t);
\draw[arrow] (chi2) -- node[label, above, font=\tiny]{$\chi^2_m/m \div \chi^2_n/n$} (F);
\draw[arrow] (exp) -- node[label, right]{$\alpha$ 个独立\\指数之和} (gamma);
\draw[arrow] (gamma) -- node[label, above]{$\alpha=n/2$\\$\beta=1/2$} (chi2);

% Reference box
\node[draw, rounded corners=4pt, fill=yellow!15, font=\scriptsize, align=left,
      below=0.6cm of F, text width=5.8cm] (ref)
  {\textbf{三大抽样分布}（统计推断基石）：\\
   $\chi^2_n=\sum_{i=1}^n Z_i^2$（方差检验）\\
   $t_n=Z/\sqrt{\chi^2_n/n}$（均值检验，小样本）\\
   $F_{m,n}=(\chi^2_m/m)/(\chi^2_n/n)$（方差比/ANOVA）};

\end{tikzpicture}
