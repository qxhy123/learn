\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}
\usetikzlibrary{calc, positioning}

\begin{document}
\begin{tikzpicture}

% 左子图：两正态分布对比（μ1=0 vs μ2=3，σ=1）
\begin{axis}[
  name=ax1,
  width=7cm, height=6.5cm,
  xmin=-4.0, xmax=7.0,
  ymin=-0.03, ymax=0.52,
  xlabel={$x$},
  ylabel={$f(x)$},
  axis lines=left,
  xtick={-3,-2,-1,0,1,2,3,4,5,6},
  xticklabels={$-3$,$-2$,$-1$,$\mu_1$,$1$,$2$,$\mu_2$,$4$,$5$,$6$},
  xticklabel style={font=\scriptsize},
  ytick={0,0.1,0.2,0.3,0.4},
  yticklabel style={font=\scriptsize},
  grid=major, grid style={gray!30},
  clip=false
]
  % 重叠阴影区域
  \addplot[fill=gray!25, opacity=0.65, draw=none, domain=1.2:1.8, samples=40]
    {exp(-x^2/2)/sqrt(2*3.14159)} \closedcycle;

  % 样本1: N(0,1)，蓝色实线
  \addplot[blue!70!black, ultra thick, domain=-4.0:4.5, samples=150]
    {exp(-x^2/2)/sqrt(2*3.14159)};

  % 样本2: N(3,1)，红色虚线
  \addplot[red!70!black, ultra thick, dashed, domain=-0.5:7.0, samples=150]
    {exp(-(x-3)^2/2)/sqrt(2*3.14159)};

  % μ1 垂线
  \addplot[blue!50, dashed, thin] coordinates {(0,0) (0,0.399)};
  \node[font=\scriptsize, blue!70!black, above] at (axis cs:0, 0.40) {$\mu_1{=}0$};

  % μ2 垂线
  \addplot[red!50, dashed, thin] coordinates {(3,0) (3,0.399)};
  \node[font=\scriptsize, red!70!black, above] at (axis cs:3, 0.40) {$\mu_2{=}3$};

  % 效应量标注
  \draw[<->, thick, gray!60!black] (axis cs:0, 0.16) -- (axis cs:3, 0.16);
  \node[font=\scriptsize, fill=white, draw=gray!50, thin, rounded corners=2pt, inner sep=2pt]
    at (axis cs:1.5, 0.22) {$d=\dfrac{\mu_1-\mu_2}{\sigma}=-3$};

  % 图例（内置在左子图内部）
  \node[font=\scriptsize, blue!70!black] at (axis cs:-2, 0.35) {$-\!-$ $N(0,1)$};
  \node[font=\scriptsize, red!70!black] at (axis cs:-2, 0.29) {- - $N(3,1)$};

\end{axis}

% 右子图：F 分布方差比检验
\begin{axis}[
  name=ax2,
  at={(ax1.east)}, anchor=west, xshift=0.6cm,
  width=7cm, height=6.5cm,
  xmin=0, xmax=5.5,
  ymin=-0.03, ymax=0.85,
  xlabel={$F$},
  ylabel={$f(F)$},
  axis lines=left,
  xtick={0,1,2,3,4,5},
  ytick={0,0.2,0.4,0.6,0.8},
  yticklabel style={font=\scriptsize},
  xticklabel style={font=\scriptsize},
  grid=major, grid style={gray!30},
  clip=false,
  legend style={at={(0.98,0.98)}, anchor=north east, font=\scriptsize,
                fill=white, draw=gray!50}
]
  % F(3,10): f ∝ x^{0.5}*(1+0.3x)^{-6.5}，归一化常数≈0.695
  \addplot[blue!70!black, ultra thick, domain=0.02:5.5, samples=150]
    {0.695*(x^0.5)*(1+0.3*x)^(-6.5)};
  \addlegendentry{$F(3,10)$}

  % F(5,20): f ∝ x^{1.5}*(1+0.25x)^{-12.5}，归一化常数≈0.95
  \addplot[red!70!black, ultra thick, dashed, domain=0.02:5.5, samples=150]
    {0.95*(x^1.5)*(1+0.25*x)^(-12.5)};
  \addlegendentry{$F(5,20)$}

  % 临界值线（F_{0.05}(3,10) ≈ 3.71）
  \addplot[dashed, gray!70, thick] coordinates {(3.71,0) (3.71,0.55)};
  \node[font=\scriptsize, gray!70!black, above] at (axis cs:3.71, 0.55) {$F_{0.05}$};

  % 拒绝域阴影
  \addplot[fill=orange!30, opacity=0.7, draw=none, domain=3.71:5.5, samples=40]
    {0.695*(x^0.5)*(1+0.3*x)^(-6.5)} \closedcycle;

  % 拒绝域标注
  \node[font=\scriptsize, orange!80!black] at (axis cs:4.5, 0.04) {拒绝域};

\end{axis}

% 子图标题（外置 node）
\node[font=\bfseries\small] at ($(ax1.north)+(0,0.4cm)$) {两总体均值比较};
\node[font=\bfseries\small] at ($(ax2.north)+(0,0.4cm)$) {$F$ 检验方差比};

% 整体标题
\node[font=\large\bfseries] at ($(ax1.north)!0.5!(ax2.north)+(0,1.3cm)$)
  {两样本检验：均值差 \& $F$ 分布};

\end{tikzpicture}
