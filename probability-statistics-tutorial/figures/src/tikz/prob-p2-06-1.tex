\begin{tikzpicture}
% Three panels: rho = 0, 0.5, 0.9 contour plots of bivariate normal
% f(x,y) = 1/(2*pi*sqrt(1-rho^2)) * exp(-1/(2*(1-rho^2))*(x^2-2*rho*x*y+y^2))
\begin{groupplot}[
  group style={
    group size=3 by 1,
    horizontal sep=0.8cm
  },
  width=4.8cm, height=4.8cm,
  xmin=-3, xmax=3,
  ymin=-3, ymax=3,
  xlabel={$x$},
  ylabel={$y$},
  axis lines=box,
  xtick={-2,0,2},
  ytick={-2,0,2},
  xticklabel style={font=\scriptsize},
  yticklabel style={font=\scriptsize},
  colormap/cool,
  view={0}{90},
  clip=false
]

% rho = 0
\nextgroupplot[title={$\rho=0$（独立）}]
\addplot3[contour filled={number=8}, domain=-3:3, samples=40]
  {exp(-(x^2+y^2)/2)/(2*pi)};

% rho = 0.5
\nextgroupplot[title={$\rho=0.5$（正相关）}]
\addplot3[contour filled={number=8}, domain=-3:3, samples=40]
  {exp(-(x^2-x*y+y^2)/1.5)/(2*pi*0.866)};

% rho = 0.9
\nextgroupplot[title={$\rho=0.9$（强相关）}]
\addplot3[contour filled={number=8}, domain=-3:3, samples=40]
  {exp(-(x^2-1.8*x*y+y^2)/0.38)/(2*pi*0.4359)};

\end{groupplot}

\node[font=\small\bfseries, align=center] at (7.2,5.5)
  {二元正态联合 PDF 等高线：不同相关系数 $\rho$};
\node[font=\scriptsize, align=center, fill=yellow!15, draw, rounded corners=2pt]
  at (7.2,-2.0) {圆形 $\to$ 椭圆（倾斜方向随 $|\rho|$ 增大而收窄）};

\end{tikzpicture}
