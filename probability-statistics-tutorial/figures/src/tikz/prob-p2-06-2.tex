\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{ctex}
\begin{document}
\begin{tikzpicture}
% Bivariate normal contour ellipses drawn analytically
% Contour ellipse for bivariate standard normal with correlation rho:
% x^2 - 2*rho*x*y + y^2 = c^2*(1-rho^2)
% For 1-sigma contour: c=1

\def\R{2.5}  % radius scale

% Helper: draw ellipse for given rho
% Parametric: rotate the circle by angle theta=atan(rho)
% Actually draw the 68.3% contour ellipse

% rho=0: circle
\draw[blue!70, ultra thick] (0,0) circle (\R);
\node[blue!80, font=\small] at (0,\R+0.3) {$\rho=0$};

% rho=0.5: ellipse tilted 45 degrees
% Semi-axes: sqrt(1+rho) and sqrt(1-rho) in rotated frame
% a = sqrt(1+0.5)=1.225, b=sqrt(0.5)=0.707, angle=45 deg
\draw[red!70, ultra thick, rotate=45]
  (0,0) ellipse ({1.225*\R/1.5} and {0.707*\R/1.5});
\node[red!80, font=\small] at (\R+0.5, 0.5) {$\rho=0.5$};

% rho=0.9: highly elongated ellipse at 45 degrees
% a=sqrt(1.9)=1.378, b=sqrt(0.1)=0.316, angle=45 deg
\draw[green!60!black, ultra thick, rotate=45]
  (0,0) ellipse ({1.378*\R/1.5} and {0.316*\R/1.5});
\node[green!50!black, font=\small] at (-\R-0.9, -0.5) {$\rho=0.9$};

% rho=-0.7: ellipse tilted -45 degrees
\draw[orange!80, thick, rotate=-45]
  (0,0) ellipse ({1.304*\R/1.5} and {0.548*\R/1.5});
\node[orange!80!black, font=\small] at (\R+0.5,-0.8) {$\rho=-0.7$};

% Axes
\draw[->] (-\R-0.8,0) -- (\R+0.8,0) node[right,font=\small] {$x$};
\draw[->] (0,-\R-0.8) -- (0,\R+0.8) node[above,font=\small] {$y$};

% Diagonal guide lines (dashed)
\draw[gray!50, dashed] (-\R-0.5,-\R-0.5) -- (\R+0.5,\R+0.5)
  node[font=\tiny,right,gray] {正相关轴};
\draw[gray!50, dashed] (-\R-0.5,\R+0.5) -- (\R+0.5,-\R-0.5)
  node[font=\tiny,right,gray] {负相关轴};

% Title
\node[font=\small\bfseries, align=center] at (0,\R+1.3)
  {二元正态 $68.3\%$ 置信椭圆（不同 $\rho$）};

\end{tikzpicture}
\end{document}
