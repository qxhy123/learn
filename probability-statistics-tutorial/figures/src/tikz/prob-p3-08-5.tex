\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=10cm, height=7cm,
  xmin=0, xmax=1,
  ymin=0, ymax=3.8,
  xlabel={$x$},
  ylabel={$f(x)$},
  title={Beta$(\alpha,\beta)$ 分布 PDF：多种形状},
  title style={font=\small\bfseries},
  xlabel style={font=\small},
  ylabel style={font=\small},
  xticklabel style={font=\scriptsize},
  yticklabel style={font=\scriptsize},
  axis lines=left,
  samples=120,
  legend style={font=\scriptsize, at={(1.02,1.0)}, anchor=north west},
  clip=false,
]

% Beta(1,1) = Uniform
\addplot[gray!70, very thick, domain=0:1] {1};
\addlegendentry{$\alpha=\beta=1$: uniform}

% Beta(2,2): symmetric bell
\addplot[blue, very thick, domain=0.01:0.99] {6*x*(1-x)};
\addlegendentry{$\alpha=\beta=2$: sym.\ bell}

% Beta(5,5): narrow symmetric
\addplot[green!60!black, very thick, domain=0.01:0.99]
  {252*x^4*(1-x)^4};
\addlegendentry{$\alpha=\beta=5$: narrow bell}

% Beta(0.5,0.5): U-shape
\addplot[orange!90, very thick, domain=0.01:0.99]
  {1/(3.14159*sqrt(x*(1-x)))};
\addlegendentry{$\alpha=\beta=0.5$: U-shape}

% Beta(3,1): right-skewed (peak near 1)
\addplot[red!80, very thick, domain=0.01:0.99] {3*x^2};
\addlegendentry{$\alpha=3,\beta=1$: right-skewed}

% Beta(1,4): left-skewed (peak near 0)
\addplot[purple!70, very thick, domain=0.01:0.99] {4*(1-x)^3};
\addlegendentry{$\alpha=1,\beta=4$: left-skewed}

\node[font=\scriptsize, fill=yellow!15, draw, rounded corners=2pt, text width=4.5cm]
  at (0.35, -0.7)
  {支撑$(0,1)$：概率的先验分布\\$E=\alpha/(\alpha+\beta)$};

\end{axis}
\end{tikzpicture}
\end{document}
