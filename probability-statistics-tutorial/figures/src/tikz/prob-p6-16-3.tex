\begin{axis}[

  xmin=0, xmax=110,
  ymin=0, ymax=0.28,
  xlabel={样本量 $n$},
  ylabel={$\text{Var}(\hat\theta)$},
  title={\textbf{Cramér-Rao 下界与各估计量方差}},
  axis lines=left,
  xtick={10,20,30,50,100},
  xticklabel style={font=\scriptsize},
  yticklabel style={font=\scriptsize},
  ytick={0,0.05,0.10,0.15,0.20,0.25},
  grid=major, grid style={gray!20},
  clip=false,
  legend pos=north east,
  legend style={font=\scriptsize}
]

  % CR 下界曲线: 1/(n * I(theta)), 设 I(theta)=1
  \addplot[gray!70!black, very thick, domain=5:105, samples=80]
    {1/(x*1.0)};
  \addlegendentry{CR 下界 $\frac{1}{n I(\theta)}$}

  % UMVUE 估计量：贴近下界
  \addplot[blue!70!black, thick, domain=5:105, samples=80]
    {1/(x*1.0) + 0.002};
  \addlegendentry{$\hat\theta_1$（UMVUE，贴近下界）}

  % 渐近有效估计量
  \addplot[green!60!black, thick, domain=5:105, samples=80]
    {1/(x*0.85)};
  \addlegendentry{$\hat\theta_2$（渐近有效）}

  % 低效估计量：明显高于下界
  \addplot[orange!80!black, thick, domain=5:105, samples=80]
    {1/(x*0.5)};
  \addlegendentry{$\hat\theta_3$（低效估计量）}

  % 标记若干数据点

  \addplot[only marks, mark=square*, mark size=2.5pt, orange!80!black] coordinates {
    (10,0.202)(20,0.102)(30,0.068)(50,0.042)(100,0.021)
  };

  % 标注 Fisher 信息量
  \node[draw, fill=yellow!15, rounded corners=3pt, font=\small, align=center]
    at (axis cs:75, 0.22) {
      Fisher 信息量\\
      $I(\theta)=-E\!\left[\frac{\partial^2 \ln f}{\partial\theta^2}\right]$
    };

  % 标注下界区域
  \node[font=\scriptsize, gray!70!black] at (axis cs:55, 0.032) {CR 下界};
