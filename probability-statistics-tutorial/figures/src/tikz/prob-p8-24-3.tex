\begin{tikzpicture}[

  hidden/.style={circle, draw=blue!70!black, fill=blue!8,
                 minimum size=14mm, thick, dashed, font=\normalsize\bfseries},

% ============================================================
% HMM 时序图：4 个时间步
% X1, X2, X3, X4：隐状态（圆形虚线）
% O1, O2, O3, O4：观测变量（实心方形）
% ============================================================

% 节点间距
\def\hstep{3.5}
\def\vgap{2.8}

% 隐状态节点
\node[hidden] (X1) at (0*\hstep, \vgap)  {$X_1$};
\node[hidden] (X2) at (1*\hstep, \vgap)  {$X_2$};
\node[hidden] (X3) at (2*\hstep, \vgap)  {$X_3$};
\node[hidden] (X4) at (3*\hstep, \vgap)  {$X_4$};

% 观测节点
\node[observe] (O1) at (0*\hstep, 0)  {$O_1$};
\node[observe] (O2) at (1*\hstep, 0)  {$O_2$};
\node[observe] (O3) at (2*\hstep, 0)  {$O_3$};
\node[observe] (O4) at (3*\hstep, 0)  {$O_4$};

% 横向转移箭头 X_t -> X_{t+1}
\draw[->, thick, blue!70!black] (X1) -- node[midway, above, font=\scriptsize] {$A$} (X2);
\draw[->, thick, blue!70!black] (X2) -- node[midway, above, font=\scriptsize] {$A$} (X3);
\draw[->, thick, blue!70!black] (X3) -- node[midway, above, font=\scriptsize] {$A$} (X4);

% 纵向发射箭头 X_t -> O_t
\draw[->, thick, red!60!black] (X1) -- node[midway, right, font=\scriptsize] {$B$} (O1);
\draw[->, thick, red!60!black] (X2) -- node[midway, right, font=\scriptsize] {$B$} (O2);
\draw[->, thick, red!60!black] (X3) -- node[midway, right, font=\scriptsize] {$B$} (O3);
\draw[->, thick, red!60!black] (X4) -- node[midway, right, font=\scriptsize] {$B$} (O4);

% 时间轴标注
\foreach \t/\x in {1/0, 2/1, 3/2, 4/3}{
  \node[font=\scriptsize, gray!70!black] at (\x*\hstep, -1.1) {$t=\t$};
}
\draw[gray!50, thin, ->] (-1.0, -1.1) -- (3*\hstep+1.0, -1.1)
  node[right, font=\small] {时间};

% 初始状态分布
\node[font=\scriptsize, blue!60!black] at (-1.6, \vgap) {$\pi$};
\draw[->, thin, blue!60!black] (-1.0, \vgap) -- (X1.west);

% 整体标题
\node[font=\large\bfseries] at (1.5*\hstep, \vgap+1.6) {隐马尔可夫模型（HMM）时序图};

% 图例
\node[hidden, minimum size=9mm, font=\small] (leg_h) at (-0.5, -2.2) {};
\node[font=\small, right] at (0.1, -2.2) {隐状态 $X_t$（圆形虚线）};

\node[observe, minimum size=9mm, font=\small] (leg_o) at (-0.5, -3.2) {};
\node[font=\small, right] at (0.1, -3.2) {观测变量 $O_t$（实心方形）};

\draw[->, blue!70!black, thick] (4.0, -2.2) -- (5.2, -2.2);

% 算法说明框

      fill=white, draw=gray!50, thin, rounded corners=2pt, inner sep=6pt]
  at (1.5*\hstep, -4.5) {

    后向变量 $\beta_t(i)=P(O_{t+1}\cdots O_T\mid X_t=i,\lambda)$
  };
