\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  width=10cm, height=6.5cm,
  xmin=-5.5, xmax=7.5,
  ymin=0, ymax=0.85,
  xlabel={$x$},
  ylabel={$f(x)$},
  title={正态分布 $N(\mu,\sigma^2)$：不同参数对比},
  title style={font=\small\bfseries},
  xlabel style={font=\small},
  ylabel style={font=\small},
  xticklabel style={font=\scriptsize},
  yticklabel style={font=\scriptsize},
  axis lines=center,
  samples=120,
  legend pos=north west,
  legend style={font=\scriptsize},
  clip=false,
]

% N(0,1): standard normal
\addplot[blue, very thick, domain=-5.5:5.5]
  {exp(-x^2/2)/sqrt(2*pi)};
\addlegendentry{$N(0,1)$：$\mu=0,\sigma=1$}

% N(0, 0.5^2): narrow
\addplot[red!80, very thick, domain=-3:3]
  {exp(-x^2/(2*0.25))/(0.5*sqrt(2*pi))};
\addlegendentry{$N(0,0.25)$：窄峰，$\sigma=0.5$}

% N(2, 1): shifted
\addplot[green!60!black, very thick, domain=-3:7]
  {exp(-(x-2)^2/2)/sqrt(2*pi)};
\addlegendentry{$N(2,1)$：右移，$\mu=2$}

% N(0, 2^2): wide
\addplot[orange!90, very thick, domain=-5:5]
  {exp(-x^2/8)/(2*sqrt(2*pi))};
\addlegendentry{$N(0,4)$：宽峰，$\sigma=2$}

\node[font=\scriptsize, fill=yellow!15, draw, rounded corners=2pt, text width=5.5cm]
  at (4.5, 0.30) {标准化：$Z=\frac{X-\mu}{\sigma}\sim N(0,1)$\\均值控制位置，方差控制宽窄};

\end{axis}
\end{tikzpicture}
