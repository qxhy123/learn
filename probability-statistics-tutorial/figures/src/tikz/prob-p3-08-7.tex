\begin{tikzpicture}
\begin{axis}[
  width=10cm, height=7cm,
  xmin=-5, xmax=5,
  ymin=0, ymax=0.42,
  xlabel={$x$},
  ylabel={$f(x)$},
  title={$t_n$ 分布 vs 标准正态：自由度影响尾部厚度},
  title style={font=\small\bfseries},
  xlabel style={font=\small},
  ylabel style={font=\small},
  xticklabel style={font=\scriptsize},

% Standard Normal N(0,1)
\addplot[black, very thick, domain=-5:5]
  {exp(-x^2/2)/2.5066};
\addlegendentry{$N(0,1)$}

% t_30: nearly normal
\addplot[blue, thick, domain=-5:5]
  {(1+(x^2/30))^(-15.5)/5.9270};
\addlegendentry{$t_{30}$}

% t_5
\addplot[red!80, thick, domain=-5:5]
  {(1+x^2/5)^(-3)*0.3796};
\addlegendentry{$t_5$}

% t_1 = Cauchy(0,1)
\addplot[orange!90, thick, domain=-5:5]
  {1/(3.14159*(1+x^2))};
\addlegendentry{$t_1$ (Cauchy)}

\node[draw, fill=yellow!15, rounded corners=2pt, font=\scriptsize, align=left,
      text width=5cm]
  at (axis cs:-2.8, 0.12)
  {$t_n = Z/\sqrt{\chi^2_n/n}$\\
   自由度增大趋近 $N(0,1)$\\
   $n < 30$：尾部比正态厚};
