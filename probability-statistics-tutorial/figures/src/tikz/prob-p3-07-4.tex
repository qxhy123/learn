\node[box=blue!20] (bern) {伯努利分布\\$\text{Bernoulli}(p)$\\$E=p,\ \text{Var}=p(1-p)$};

\node[box=orange!20, below=1.6cm of binom] (poisson) {泊松分布\\$\text{Poisson}(\lambda)$\\$E=\text{Var}=\lambda$};

\node[box=purple!15, below=1.6cm of geom] (nb) {负二项分布\\$\text{NB}(r,p)$\\$E=r/p$（第$r$次成功）};
\node[box=teal!15, right=2.8cm of binom] (hyp) {超几何分布\\$\text{Hyp}(N,K,n)$\\有限总体不放回};

% Arrows
\draw[arrow] (bern) -- node[label, right]{$n$ 次独立叠加} (binom);

   \textbf{首次成功} $\to$ 几何};
