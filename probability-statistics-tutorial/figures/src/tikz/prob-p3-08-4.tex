\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{positioning}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}
\begin{document}
\begin{tikzpicture}
\begin{axis}[
  name=myaxis,
  width=10cm, height=7cm,
  xmin=0, xmax=8,
  ymin=0, ymax=0.55,
  xlabel={$x$},
  ylabel={$f(x)$},
  title={\textbf{Gamma}$(\alpha,\beta)$ \textbf{分布 PDF}：$f(x)=\dfrac{\beta^\alpha}{\Gamma(\alpha)}x^{\alpha-1}e^{-\beta x}$},
  title style={font=\small, yshift=4pt},
  xlabel style={font=\small},
  ylabel style={font=\small},
  xticklabel style={font=\scriptsize},
  yticklabel style={font=\scriptsize},
  axis lines=left,
  samples=120,
  legend pos=north east,
  legend style={font=\scriptsize},
  clip=true,
  restrict y to domain=0:0.56,
]

% Gamma(1,1) = Exp(1): monotone decreasing
\addplot[blue, very thick, domain=0.05:8] {exp(-x)};
\addlegendentry{$\alpha=1,\beta=1$ (Exp)}

% Gamma(2,1)
\addplot[red!80, very thick, domain=0.01:8] {x*exp(-x)};
\addlegendentry{$\alpha=2,\beta=1$}

% Gamma(3,1)
\addplot[green!60!black, very thick, domain=0.01:8] {x^2*exp(-x)/2};
\addlegendentry{$\alpha=3,\beta=1$}

% Gamma(5,1)
\addplot[orange!90, very thick, domain=0.01:8] {x^4*exp(-x)/24};
\addlegendentry{$\alpha=5,\beta=1$}

% Gamma(2,2): beta changes scale — restrict to ymax to avoid overflow
\addplot[purple!70, very thick, dashed, domain=0.01:8] {4*x*exp(-2*x)};
\addlegendentry{$\alpha=2,\beta=2$ (mean$=1$)}

\end{axis}

% Annotation box placed in tikz coordinates below the axis
\node[font=\scriptsize, fill=yellow!15, draw, rounded corners=2pt, text width=9cm, align=left,
      below=0.5cm of myaxis.south]
  {$E[X]=\alpha/\beta$，$\text{Var}(X)=\alpha/\beta^2$；$\alpha$ 控制形状（偏度），$\beta$ 控制尺度；$\alpha=n/2,\beta=1/2\Rightarrow\chi^2_n$};

\end{tikzpicture}
\end{document}
