\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{calc,arrows.meta,matrix,positioning}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}

\begin{document}
\begin{tikzpicture}

\begin{axis}[
  name=mainax,
  width=10cm, height=6.5cm,
  xmin=0, xmax=8,
  ymin=-0.04, ymax=0.38,
  xlabel={$\theta$},
  ylabel={后验密度 $\pi(\theta\mid x)$},
  title={\textbf{损失函数与贝叶斯最优估计}},
  axis lines=left,
  xtick={0,1,2,3,4,5,6,7,8},
  xticklabels={$0$,,,,,,,,$8$},
  xticklabel style={font=\scriptsize},
  yticklabel style={font=\scriptsize},
  ytick={0,0.1,0.2,0.3},
  grid=major, grid style={gray!20},
  clip=false
]

  % 后验分布：Gamma(3, 0.6) 形状，不对称，右偏
  % f(x) ≈ x^2 * exp(-x/0.6) / (2 * 0.6^3)，峰值在 x = 2*0.6 = 1.2
  % 用 pgfplots 表达式 (alpha=3, beta=0.6): f = x^2 * exp(-x/0.6) / (2 * 0.216)
  \addplot[blue!70!black, ultra thick, domain=0.01:7.5, samples=120]
    {(x^2)*exp(-x/0.6)/(2*0.216)};
  \addplot[fill=blue!15, opacity=0.5, draw=none, domain=0.01:7.5, samples=120]
    {(x^2)*exp(-x/0.6)/(2*0.216)} \closedcycle;

  % 众数 MAP：峰值在 x = (alpha-1)*beta = 2*0.6 = 1.2
  \addplot[red!80!black, very thick] coordinates {(1.2, 0) (1.2, 0.368)};
  \addplot[only marks, mark=*, mark size=3.5pt, red!80!black] coordinates {(1.2, 0.368)};

  % 均值：alpha * beta = 3 * 0.6 = 1.8
  \addplot[green!60!black, very thick] coordinates {(1.8, 0) (1.8, 0.330)};
  \addplot[only marks, mark=diamond*, mark size=3.5pt, green!60!black] coordinates {(1.8, 0.330)};

  % 中位数：近似 (alpha - 1/3) * beta ≈ 2.67 * 0.6 = 1.60
  \addplot[orange!80!black, very thick] coordinates {(1.6, 0) (1.6, 0.348)};
  \addplot[only marks, mark=square*, mark size=3.5pt, orange!80!black] coordinates {(1.6, 0.348)};

  % 标注标签（纵向错开避免重叠）
  \node[font=\small, red!80!black, right, fill=white, fill opacity=0.85, text opacity=1,
        rounded corners=2pt] at (axis cs:2.0, 0.38)
    {\textbf{众数/MAP} $\hat\theta_{MAP}=1.2$};
  \node[font=\small, orange!80!black, right, fill=white, fill opacity=0.85, text opacity=1,
        rounded corners=2pt] at (axis cs:2.0, 0.31)
    {\textbf{中位数} $\approx 1.6$};
  \node[font=\small, green!60!black, right, fill=white, fill opacity=0.85, text opacity=1,
        rounded corners=2pt] at (axis cs:2.0, 0.24)
    {\textbf{均值} $E[\theta\mid x]=1.8$};

  % 分布标注（移至右侧较高处，避免与三线标注冲突）
  \node[font=\small, blue!60!black] at (axis cs:5.5, 0.34)
    {后验：Gamma$(3, 0.6)$};
  \node[font=\scriptsize, blue!50!black] at (axis cs:5.5, 0.30)
    {（右偏，不对称）};

\end{axis}

% 表格（位于图下方，用 tikz node 绘制）
\node[draw, rounded corners=4pt, fill=gray!8, font=\small, below=0.5cm of mainax.south, anchor=north]
  (tbl) {%
  \begin{tabular}{l|c|c}
    \textbf{损失函数} & \textbf{贝叶斯估计} & \textbf{位置}\\
    \hline
    平方损失 $(\hat\theta-\theta)^2$ & 后验均值 $E[\theta\mid x]$ & \textcolor{green!60!black}{均值 $=1.8$}\\
    绝对损失 $|\hat\theta-\theta|$   & 后验中位数                 & \textcolor{orange!80!black}{中位数 $\approx 1.6$}\\
    0-1 损失                          & 后验众数（MAP）            & \textcolor{red!80!black}{众数 $=1.2$}\\
  \end{tabular}%
};

\end{tikzpicture}
