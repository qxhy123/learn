\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}
\usetikzlibrary{calc, positioning, arrows.meta}

\begin{document}
\begin{tikzpicture}[
  box/.style={draw=gray!70, fill=white, rounded corners=4pt,
              minimum width=4.5cm, minimum height=1.8cm, align=center, font=\small},
  sbox/.style={draw=blue!40, fill=blue!8, rounded corners=3pt,
               minimum width=4.0cm, minimum height=1.2cm, align=center, font=\scriptsize},
  arrow/.style={-{Latex[length=3mm]}, thick, blue!60!black},
  title/.style={font=\bfseries\small}
]

% ─── 符号检验（左） ───
\node[box] (sign) at (0,0) {%
  \textbf{符号检验}\\\small Sign Test
};

\node[sbox, below=0.5cm of sign] (sign1) {%
  原始数据：$x_1,\dots,x_n$\\
  中位数假设 $M_0$
};

\node[sbox, below=0.5cm of sign1] (sign2) {%
  差值符号：$d_i = x_i - M_0$\\
  统计正号数 $S^+$
};

\node[sbox, below=0.5cm of sign2] (sign3) {%
  $S^+\sim B(n,0.5)$\\（$H_0$ 下）
};

\draw[arrow] (sign1) -- (sign2);
\draw[arrow] (sign2) -- (sign3);

% ─── Wilcoxon 秩和（中） ───
\node[box] (wil) at (5.8,0) {%
  \textbf{Wilcoxon 秩和检验}\\\small Rank-Sum Test
};

\node[sbox, below=0.5cm of wil] (wil1) {%
  两组数据合并排秩\\
  $R_1\leq R_2\leq\cdots\leq R_{m+n}$
};

\node[sbox, below=0.5cm of wil1] (wil2) {%
  计算第一组秩和\\
  $W = \sum_{i\in\text{组1}} R_i$
};

\node[sbox, below=0.5cm of wil2] (wil3) {%
  $W$ 查 Wilcoxon 表\\
  或正态近似
};

\draw[arrow] (wil1) -- (wil2);
\draw[arrow] (wil2) -- (wil3);

% ─── Mann-Whitney U（右） ───
\node[box] (mw) at (11.6,0) {%
  \textbf{Mann-Whitney $U$}\\\small Two-Sample Test
};

\node[sbox, below=0.5cm of mw] (mw1) {%
  两组成对比较\\
  $U=\#\{(i,j):x_i>y_j\}$
};

\node[sbox, below=0.5cm of mw1] (mw2) {%
  $U_1+U_2=mn$\\
  取 $U=\min(U_1,U_2)$
};

\node[sbox, below=0.5cm of mw2] (mw3) {%
  $U$ 查表或\\
  $z=\frac{U-mn/2}{\sqrt{mn(m{+}n{+}1)/12}}$
};

\draw[arrow] (mw1) -- (mw2);
\draw[arrow] (mw2) -- (mw3);

% 组间横向箭头
\draw[arrow, gray!50] (sign.east) -- (wil.west)
  node[midway, above, font=\scriptsize, gray!70!black] {推广};
\draw[arrow, gray!50] (wil.east) -- (mw.west)
  node[midway, above, font=\scriptsize, gray!70!black] {等价};

% 各组底部等价说明
\node[font=\scriptsize, gray!70!black, align=center]
  at ($(sign3.south)+(0,-0.4cm)$) {适用：单样本中位数};
\node[font=\scriptsize, gray!70!black, align=center]
  at ($(wil3.south)+(0,-0.4cm)$) {适用：两独立样本};
\node[font=\scriptsize, gray!70!black, align=center]
  at ($(mw3.south)+(0,-0.4cm)$) {与秩和检验等价};

% 整体标题
\node[font=\Large\bfseries] at ($(sign.north)!0.5!(mw.north)+(0,1.1cm)$)
  {三种秩检验方法对比};

\end{tikzpicture}
\end{document}
