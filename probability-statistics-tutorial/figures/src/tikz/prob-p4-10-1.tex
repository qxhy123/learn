\begin{tikzpicture}
\begin{axis}[
  width=11cm, height=6.5cm,
  xmin=-4, xmax=4,
  ymin=-0.08, ymax=0.46,
  xlabel={$x$},
  ylabel={$f(x)$},
  title={\textbf{切比雪夫不等式几何示意}},
  axis lines=left,
  xtick={-3,-2,-1,0,1,2,3},
  xticklabels={$\mu{-}3\sigma$,$\mu{-}2\sigma$,$\mu{-}\varepsilon$,$\mu$,$\mu{+}\varepsilon$,$\mu{+}2\sigma$,$\mu{+}3\sigma$},
  xticklabel style={font=\scriptsize},
  ytick={0,0.1,0.2,0.3,0.4},
  yticklabel style={font=\scriptsize},
  grid=major, grid style={gray!30},
  clip=false
]
  % Left tail (red) - beyond -epsilon
  \addplot[fill=red!40, opacity=0.8, draw=none, domain=-4:-1, samples=80]
    {exp(-x^2/2)/sqrt(2*3.14159)} \closedcycle;

  % Right tail (red) - beyond +epsilon
  \addplot[fill=red!40, opacity=0.8, draw=none, domain=1:4, samples=80]
    {exp(-x^2/2)/sqrt(2*3.14159)} \closedcycle;

  % Central region (blue) - within epsilon of mu
  \addplot[fill=blue!25, opacity=0.7, draw=none, domain=-1:1, samples=80]
    {exp(-x^2/2)/sqrt(2*3.14159)} \closedcycle;

  % Main normal curve
  \addplot[blue!70!black, ultra thick, domain=-4:4, samples=150]
    {exp(-x^2/2)/sqrt(2*3.14159)};

  % Vertical dashed lines at -epsilon and +epsilon
  \addplot[dashed, gray!70, thick] coordinates {(-1,0) (-1,{exp(-0.5)/sqrt(2*3.14159)})};
  \addplot[dashed, gray!70, thick] coordinates {(1,0) (1,{exp(-0.5)/sqrt(2*3.14159)})};

  % Label for central (blue) region
  \node[font=\small, blue!70!black] at (axis cs:0,0.17)
    {$P(|X-\mu|<\varepsilon)\geq 1-\frac{\sigma^2}{\varepsilon^2}$};

  % Label for tail (red) regions
  \node[font=\scriptsize, red!70!black] at (axis cs:-2.5,0.06) {"尾部"};
  \node[font=\scriptsize, red!70!black] at (axis cs:2.5,0.06) {"尾部"};

  % Braces / arrows indicating tail area
  \draw[->, thick, red!70!black] (axis cs:-1.5,0.10) -- (axis cs:-2.0,0.04);
  \draw[->, thick, red!70!black] (axis cs:1.5,0.10) -- (axis cs:2.0,0.04);

\end{axis}
% Chebyshev inequality annotation placed below the plot area
\node[font=\small, align=center, fill=white, draw=gray!50, thin, rounded corners=2pt]
  at (5.5,-1.1)
  {$P(|X-\mu|\geq\varepsilon)\leq\dfrac{\sigma^2}{\varepsilon^2}$};
