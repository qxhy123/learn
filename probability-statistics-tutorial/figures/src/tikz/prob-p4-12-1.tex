\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}
\usetikzlibrary{shapes.geometric, arrows.meta, positioning, decorations.pathreplacing, calc}
\begin{document}
\begin{tikzpicture}[
  node distance=3.5cm,
  circnode/.style={circle, draw, thick, minimum size=2.8cm, align=center, font=\normalsize},
  arrow/.style={-Stealth, thick, shorten >=3pt, shorten <=3pt},
  dasharrow/.style={-Stealth, thick, dashed, shorten >=3pt, shorten <=3pt, gray!70},
]

  % Nodes
  \node[circnode, fill=blue!15, draw=blue!70!black]
    (as) at (0,3) {几乎必然\\收敛\\$X_n\xrightarrow{a.s.}X$};

  \node[circnode, fill=orange!15, draw=orange!70!black]
    (lr) at (6,3) {$L^r$ 收敛\\（矩收敛）\\$X_n\xrightarrow{L^r}X$};

  \node[circnode, fill=green!15, draw=green!70!black]
    (pr) at (3,0) {依概率\\收敛\\$X_n\xrightarrow{P}X$};

  \node[circnode, fill=purple!15, draw=purple!70!black]
    (di) at (3,-3.2) {依分布\\收敛\\$X_n\xrightarrow{d}X$};

  % Arrows: solid = implication holds
  \draw[arrow, blue!70!black] (as) -- (pr)
    node[midway, left, font=\scriptsize, blue!70!black] {蕴含};
  \draw[arrow, orange!70!black] (lr) -- (pr)
    node[midway, right, font=\scriptsize, orange!70!black] {蕴含};
  \draw[arrow, green!70!black] (pr) -- (di)
    node[midway, right, font=\scriptsize, green!70!black] {蕴含};

  % Dashed arrows: implication does NOT hold in general (counterexample exists)
  \draw[dasharrow] (pr) to[bend right=25] (as);
  \draw[dasharrow] (di) to[bend right=25] (pr);
  \draw[dasharrow] (pr) to[bend left=25] (lr);

  % Title
  \node[font=\large\bfseries] at (3, 5.2)
    {四种收敛关系图};

  % Legend
  \node[font=\small, align=left] at (8.5, 1) {
    \textbf{图例}\\
    $\to$ 箭头：蕴含\\
    虚线箭头：不蕴含\\
    \hspace{1em}（存在反例，\\
    \hspace{1em}一般不成立）
  };

  % Special note: a.s. and L^r no direct implication
  \node[font=\scriptsize, gray!60] at (3, 3.5)
    {(a.s. 与 $L^r$ 之间无直接蕴含)};

\end{tikzpicture}
\end{document}
