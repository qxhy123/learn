\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{ctex}
\usetikzlibrary{patterns,decorations.pathreplacing}
\begin{document}
\begin{tikzpicture}[>=latex]

% ---- LEFT: Joint distribution as a "hill", with slices ----
\begin{scope}[shift={(-4,0)}]
  % Draw a schematic joint PDF surface
  % Represent as nested ellipses (contours)
  \draw[fill=blue!10, draw=blue!40] (0,0) ellipse (2.5cm and 1.8cm);
  \draw[fill=blue!20, draw=blue!50] (0,0) ellipse (1.8cm and 1.2cm);
  \draw[fill=blue!35, draw=blue!60] (0,0) ellipse (1.0cm and 0.7cm);
  \fill[blue!60] (0,0) circle (0.3cm);

  % Axes
  \draw[->] (-3,0) -- (3.2,0) node[right,font=\small] {$x$};
  \draw[->] (0,-2.3) -- (0,2.5) node[above,font=\small] {$y$};

  % Marginal: integrate over y (project to x-axis)
  \draw[red!80, thick, <->] (-2.5,-3.2) -- (2.5,-3.2)
    node[midway,below,font=\scriptsize,red] {$f_X(x)=\int f(x,y)\,dy$};
  % Marginal curve on x-axis
  \draw[red!70, thick, domain=-2.5:2.5, samples=40]
    plot (\x, {-2.7 - 0.4*exp(-\x*\x/2)});

  % Conditional: vertical slice at x=x0
  \def\xzero{0.8}
  \fill[orange!40, opacity=0.5] (\xzero,-1.8) rectangle (\xzero+0.3,1.8);
  \draw[orange!80, ultra thick] (\xzero,-2.0) -- (\xzero,2.2);
  \node[font=\scriptsize, orange!80!black, right] at (\xzero+0.05,2.0)
    {$x=x_0$ 切片};
  \node[font=\scriptsize, orange!80!black, right] at (\xzero+0.05,1.5)
    {$\to f(y|x_0)$};

  \node[font=\small\bfseries] at (0,3.1) {联合 PDF $f(x,y)$};
\end{scope}

% ---- RIGHT: Marginal + Conditional illustration ----
\begin{scope}[shift={(4,0)}]

  % Marginal f_Y(y) curve
  \draw[->] (0,-2.5) -- (0,2.5) node[above,font=\small] {$y$};
  \draw[->] (-0.3,0) -- (3.5,0) node[right,font=\small] {density};

  % f_Y(y): marginal
  \draw[red!80, ultra thick, domain=-2.2:2.2, samples=60]
    plot ({1.2*exp(-\x*\x/2)}, \x);
  \node[red!80, font=\scriptsize, right] at (1.3,2.0) {$f_Y(y)$（边缘）};

  % f(y|x=1): conditional (narrower)
  \draw[orange!80, ultra thick, domain=-1.5:1.5, samples=50]
    plot ({0.8*exp(-(\x-0.5)*(\x-0.5)/0.6)+1.8}, \x);
  \node[orange!80!black, font=\scriptsize, right] at (2.7,1.5)
    {$f(y|x_0)$（条件）};

  % Arrow showing narrower = less uncertainty
  \draw[<->, green!60!black, thick] (0.05,0.6) -- (1.1,0.6);
  \node[font=\scriptsize, green!50!black] at (0.55,0.85) {$\sigma_Y$};
  \draw[<->, green!60!black, thick] (1.85,0.6) -- (2.5,0.6);
  \node[font=\scriptsize, green!50!black] at (2.18,0.85) {$\sigma_{Y|x}$};

  \node[font=\scriptsize, fill=yellow!15, draw, rounded corners=2pt, align=center]
    at (1.8,-1.8) {条件比边缘\\不确定性更小};

  \node[font=\small\bfseries] at (1.6,3.0) {边缘 vs 条件分布};

\end{scope}

% Center: formula box
\node[font=\small, fill=yellow!10, draw, rounded corners=3pt, align=center]
  at (0,-3.5) {
    $f_X(x)=\int_{-\infty}^{+\infty}f(x,y)\,dy$ \quad（积分掉 $y$）\\[4pt]
    $f(y|x)=\dfrac{f(x,y)}{f_X(x)}$ \quad（切片归一化）
  };

\end{tikzpicture}
\end{document}
