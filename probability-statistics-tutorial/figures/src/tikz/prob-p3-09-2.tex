\begin{tikzpicture}
\begin{groupplot}[
  group style={group size=3 by 1, horizontal sep=0.8cm},
  width=4.8cm, height=4.8cm,
  xmin=-3, xmax=3,
  ymin=-3, ymax=3,
  xlabel={$x_1$},
  ylabel={$x_2$},
  axis lines=box,
  xtick={-2,0,2},
  ytick={-2,0,2},
  xticklabel style={font=\tiny},
  yticklabel style={font=\tiny},
  colormap/cool,
  view={0}{90},
  samples=40,
  clip=false,
]

% rho = 0: f = exp(-(x^2+y^2)/2)/(2*pi)
\nextgroupplot[title={$\rho=0$（独立）}, title style={font=\small\bfseries}]
\addplot3[contour filled={number=7}, domain=-3:3]
  {exp(-(x^2+y^2)/2)/(2*pi)};

% rho = 0.7: Sigma = [[1, 0.7],[0.7,1]], det=0.51, 1-rho^2=0.51
% f = 1/(2*pi*sqrt(0.51)) * exp(-1/(2*0.51)*(x^2-1.4*x*y+y^2))
\nextgroupplot[title={$\rho=0.7$（正相关）}, title style={font=\small\bfseries}]
\addplot3[contour filled={number=7}, domain=-3:3]
  {exp(-(x^2 - 1.4*x*y + y^2)/(2*0.51))/(2*pi*0.7141)};

% rho = -0.8: 1-rho^2=0.36
\nextgroupplot[title={$\rho=-0.8$（负相关）}, title style={font=\small\bfseries}]
\addplot3[contour filled={number=7}, domain=-3:3]
  {exp(-(x^2 + 1.6*x*y + y^2)/(2*0.36))/(2*pi*0.6)};

\end{groupplot}
\node[font=\small\bfseries] at (7.2,5.5) {二元正态等高线：协方差结构};
\node[font=\scriptsize, fill=yellow!15, draw, rounded corners=2pt]
  at (7.2,-1.2) {圆$\to$倾斜椭圆：$|\rho|$ 越大，椭圆越细长};
