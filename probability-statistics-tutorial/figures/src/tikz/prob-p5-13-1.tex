\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{calc,arrows.meta,positioning}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}

\begin{document}
\begin{tikzpicture}

%% ── 第一列：总体分布 ──────────────────────────────────────────────
\begin{axis}[
  name=popax,
  width=5.2cm, height=4.2cm,
  xmin=-3.5, xmax=3.5,
  ymin=0, ymax=0.48,
  xlabel={$x$}, ylabel={$f(x)$},
  title={\textbf{总体分布}},
  axis lines=left,
  xtick=\empty,
  ytick=\empty,
  clip=false,
  title style={font=\small}
]
  \addplot[blue!70!black, ultra thick, domain=-3.5:3.5, samples=120]
    {exp(-x^2/2)/sqrt(2*3.14159)};
  % μ 和 σ 标注
  \addplot[dashed, gray] coordinates {(0,0) (0,0.399)};
  \node[font=\scriptsize, below] at (axis cs:0,0) {$\mu$};
  \draw[<->, gray, thick] (axis cs:-1,0.10) -- (axis cs:1,0.10)
    node[midway, above, font=\scriptsize]{$\sigma$};
  \node[font=\scriptsize, fill=yellow!15, draw, rounded corners=2pt, align=center]
    at (axis cs:2.2,0.38) {$N(\mu,\sigma^2)$};
\end{axis}

%% ── 箭头 1：总体 → 抽样 ────────────────────────────────────────
\node[right=0.15cm of popax.east, font=\large] (arr1) {};
\draw[-{Stealth[length=8pt]}, very thick, orange!80!black]
  ($(popax.east)+(0.05,0)$) -- ++(1.0,0)
  node[midway, above, font=\scriptsize, align=center]{抽样\\$n$ 个};

%% ── 第二列：抽样过程（多组样本） ────────────────────────────────
\node[right=1.4cm of popax.east, anchor=west,
      draw, rounded corners=4pt, fill=green!6,
      minimum width=3.8cm, minimum height=4.0cm,
      font=\small, align=center] (sampbox) {};

% 画三行样本点
\foreach \row/\yoff in {1/1.2, 2/0.0, 3/-1.2}{
  \node[font=\scriptsize, left] at ($(sampbox.west)+(0.05,\yoff cm)$) {第$\row$组};
  \foreach \xi in {0.55, 0.95, 1.35, 1.75, 2.15}{
    \fill[blue!60] ($(sampbox.west)+(\xi cm, \yoff cm)+(0,{0.18*rand})$) circle (1.8pt);
  }
}
\node[font=\scriptsize, below] at (sampbox.south) {每组 $n$ 个观测};
\node[font=\small\bfseries, above] at (sampbox.north) {抽样过程};
% 省略号
\node[font=\normalsize] at ($(sampbox.center)+(0,-0.7)$) {$\vdots$};

%% ── 箭头 2：抽样 → 统计量 ───────────────────────────────────────
\draw[-{Stealth[length=8pt]}, very thick, orange!80!black]
  ($(sampbox.east)+(0.05,0)$) -- ++(1.05,0)
  node[midway, above, font=\scriptsize, align=center]{计算\\$\bar{X}$};

%% ── 第三列：$\bar X$ 抽样分布 ──────────────────────────────────
\begin{axis}[
  name=sampax,
  at={($(sampbox.east)+(1.35cm,0)$)}, anchor=west,
  width=5.2cm, height=4.2cm,
  xmin=-3.5, xmax=3.5,
  ymin=0, ymax=0.78,
  xlabel={$\bar{x}$}, ylabel={频率},
  title={\textbf{统计量分布}},
  axis lines=left,
  xtick=\empty, ytick=\empty,
  clip=false,
  title style={font=\small}
]
  % 抽样分布直方图（写死数值）
  \addplot+[ybar, bar width=0.48, fill=teal!30, draw=teal!60!black] coordinates {
    (-2.8,0.04) (-2.3,0.08) (-1.8,0.14) (-1.3,0.21) (-0.8,0.30)
    (-0.3,0.40) (0.2,0.50) (0.7,0.40) (1.2,0.30) (1.7,0.21)
    (2.2,0.14) (2.7,0.08)
  };
  % 正态拟合曲线（方差缩小：σ^2/n）
  \addplot[orange!80!black, ultra thick, domain=-3.5:3.5, samples=120]
    {exp(-x^2/(2*0.5))/sqrt(2*3.14159*0.5)};
  \node[font=\scriptsize, fill=yellow!15, draw, rounded corners=2pt, align=center]
    at (axis cs:2.0,0.68) {$N\!\left(\mu,\frac{\sigma^2}{n}\right)$};
\end{axis}

%% ── 整体标题 ────────────────────────────────────────────────────
\node[font=\large\bfseries]
  at ($(popax.north)!0.5!(sampax.north)+(0,0.55cm)$)
  {抽样分布构造示意};

\end{tikzpicture}
\end{document}
