\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}
\usetikzlibrary{calc, positioning}

\begin{document}
\begin{tikzpicture}

\begin{axis}[
  name=ax,
  width=12cm, height=7cm,
  xmin=-3.5, xmax=5.5,
  ymin=-0.04, ymax=0.48,
  xlabel={$x$},
  ylabel={$f(x)$},
  axis lines=left,
  xtick={-3,-2,-1,0,1,1.645,2,3,4,5},
  xticklabels={$-3$,$-2$,$-1$,$0$,$1$,$c$,$2$,$3$,$4$,$5$},
  ytick={0,0.1,0.2,0.3,0.4},
  yticklabel style={font=\scriptsize},
  xticklabel style={font=\scriptsize},
  grid=major, grid style={gray!30},
  clip=false,
  legend style={at={(1.02,1.0)}, anchor=north west, font=\small,
                fill=white, draw=gray!50}
]

  % α 阴影：H0 右尾（超过临界值 c=1.645）
  \addplot[fill=blue!25, opacity=0.75, draw=none, domain=1.645:4.5, samples=60]
    {exp(-x^2/2)/sqrt(2*3.14159)} \closedcycle;

  % β 阴影：H1(μ=2) 左尾（未超过临界值 c=1.645）
  \addplot[fill=red!25, opacity=0.75, draw=none, domain=-1.5:1.645, samples=60]
    {exp(-(x-2)^2/2)/sqrt(2*3.14159)} \closedcycle;

  % H0 曲线：N(0,1) 蓝色实线
  \addplot[blue!70!black, ultra thick, domain=-3.5:4.5, samples=150]
    {exp(-x^2/2)/sqrt(2*3.14159)};
  \addlegendentry{$H_0: N(0,1)$}

  % H1 曲线：N(2,1) 红色实线
  \addplot[red!70!black, ultra thick, domain=-1.5:5.5, samples=150]
    {exp(-(x-2)^2/2)/sqrt(2*3.14159)};
  \addlegendentry{$H_1: N(2,1)$}

  % 临界值垂线
  \addplot[dashed, gray!60, thick] coordinates {(1.645,-0.01) (1.645,0.40)};

  % α 标注
  \node[font=\small, blue!70!black] at (axis cs:2.4, 0.06) {$\alpha$};

  % β 标注
  \node[font=\small, red!70!black] at (axis cs:0.5, 0.06) {$\beta$};

  % power 标注
  \node[font=\small, red!50!black] at (axis cs:3.2, 0.22) {power $= 1-\beta$};

  % 临界值标注
  \node[font=\scriptsize, gray!60!black, above] at (axis cs:1.645, 0.40) {临界值 $c$};

\end{axis}

% 公式框（axis 外）
\node[font=\small, align=center, fill=white, draw=gray!50, thin,
      rounded corners=2pt, inner sep=5pt]
  at ($(ax.south west)+(3.5cm,-1.2cm)$)
  {第一类错误（$\alpha$）：$H_0$ 真但被拒绝 \quad 第二类错误（$\beta$）：$H_1$ 真但未拒绝};

% 整体标题
\node[font=\large\bfseries] at ($(ax.north)+(0,0.6cm)$)
  {第一类 / 第二类错误与检验功效};

\end{tikzpicture}
\end{document}
