\begin{axis}[
  width=10cm, height=6cm,
  xmin=-3.5, xmax=3.5,
  ymin=-0.04, ymax=0.48,
  xlabel={$x$},
  ylabel={$f(x)$},
  title={\textbf{标准正态 PDF 与区间概率（面积）}},
  axis lines=left,
  xtick={-3,-2,-1,0,1,2,3},
  ytick={0,0.1,0.2,0.3,0.4},

  % Shade area between -1 and 1
  \addplot[fill=blue!25, opacity=0.7, draw=none, domain=-1:1, samples=80]

  % Shade area between 1 and 2
  \addplot[fill=orange!30, opacity=0.7, draw=none, domain=1:2, samples=40]
    {exp(-x^2/2)/sqrt(2*pi)} \closedcycle;

  \addplot[blue!70!black, ultra thick, domain=-3.5:3.5, samples=120]
    {exp(-x^2/2)/sqrt(2*pi)};

  % Annotation for shaded areas
  \node[font=\scriptsize, blue!70!black] at (axis cs:0,0.18) {$\approx 68.3\%$};

  \node[font=\scriptsize, below] at (axis cs:1,-0.02) {$+\sigma$};
  \node[font=\scriptsize, below] at (axis cs:2,-0.02) {$+2\sigma$};

  % Formula
  \node[font=\small, align=center, fill=yellow!15, draw, rounded corners=2pt]
    at (axis cs:2.5,0.38) {$f(x)=\frac{1}{\sqrt{2\pi}}e^{-x^2/2}$};
