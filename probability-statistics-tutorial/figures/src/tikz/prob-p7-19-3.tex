\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{ctex}
\usetikzlibrary{calc, positioning}

\begin{document}
\begin{tikzpicture}

% 左子图：单侧 p 值
\begin{axis}[
  name=ax1,
  width=6.5cm, height=5.5cm,
  xmin=-3.5, xmax=3.8,
  ymin=-0.03, ymax=0.46,
  xlabel={$z$},
  ylabel={$f(z)$},
  axis lines=left,
  xtick={-3,-2,-1,0,1,1.8,2,3},
  xticklabels={$$,$-2$,$-1$,$0$,$1$,$z_0$,$2$,$3$},
  ytick={0,0.1,0.2,0.3,0.4},
  yticklabel style={font=\scriptsize},
  xticklabel style={font=\scriptsize},
  grid=major, grid style={gray!30},
  clip=false
]
  % 单侧 p 值阴影（观测值 z0=1.8 右侧）
  \addplot[fill=blue!30, opacity=0.8, draw=none, domain=1.8:3.8, samples=60]
    {exp(-x^2/2)/sqrt(2*3.14159)} \closedcycle;

  % 主曲线
  \addplot[blue!70!black, ultra thick, domain=-3.5:3.8, samples=150]
    {exp(-x^2/2)/sqrt(2*3.14159)};

  % 观测值垂线
  \addplot[dashed, red!70!black, thick] coordinates {(1.8,0) (1.8,{exp(-1.8^2/2)/sqrt(2*3.14159)})};

  % p 值标注
  \node[font=\small, blue!70!black] at (axis cs:2.5, 0.05) {$p$ 值};

  % z0 箭头
  \node[font=\scriptsize, red!70!black, above] at (axis cs:1.8, {exp(-1.8^2/2)/sqrt(2*3.14159)+0.04}) {$z_0=1.8$};

  % p 值公式
  \node[font=\small, fill=white, draw=gray!50, thin, rounded corners=2pt, inner sep=3pt]
    at (axis cs:-1.2, 0.37) {$p=P(Z\geq z_0)$};

\end{axis}

% 右子图：双侧 p 值
\begin{axis}[
  name=ax2,
  at={(ax1.east)}, anchor=west, xshift=0.8cm,
  width=6.5cm, height=5.5cm,
  xmin=-3.5, xmax=3.8,
  ymin=-0.03, ymax=0.46,
  xlabel={$z$},
  ylabel={},
  axis lines=left,
  xtick={-3,-1.8,-1,0,1,1.8,3},
  xticklabels={$$,$-z_0$,$-1$,$0$,$1$,$z_0$,$$},
  ytick={0,0.1,0.2,0.3,0.4},
  yticklabel style={font=\scriptsize},
  xticklabel style={font=\scriptsize},
  grid=major, grid style={gray!30},
  clip=false
]
  % 双侧 p 值阴影：右尾
  \addplot[fill=blue!30, opacity=0.8, draw=none, domain=1.8:3.8, samples=60]
    {exp(-x^2/2)/sqrt(2*3.14159)} \closedcycle;

  % 双侧 p 值阴影：左尾
  \addplot[fill=blue!30, opacity=0.8, draw=none, domain=-3.8:-1.8, samples=60]
    {exp(-x^2/2)/sqrt(2*3.14159)} \closedcycle;

  % 主曲线
  \addplot[blue!70!black, ultra thick, domain=-3.5:3.8, samples=150]
    {exp(-x^2/2)/sqrt(2*3.14159)};

  % 观测值垂线（两侧）
  \addplot[dashed, red!70!black, thick] coordinates {(1.8,0) (1.8,{exp(-1.8^2/2)/sqrt(2*3.14159)})};
  \addplot[dashed, red!70!black, thick] coordinates {(-1.8,0) (-1.8,{exp(-1.8^2/2)/sqrt(2*3.14159)})};

  % 双侧 p 标注
  \node[font=\small, blue!70!black] at (axis cs:2.5, 0.05) {$p/2$};
  \node[font=\small, blue!70!black] at (axis cs:-2.5, 0.05) {$p/2$};

  % 公式
  \node[font=\small, fill=white, draw=gray!50, thin, rounded corners=2pt, inner sep=3pt]
    at (axis cs:0, 0.37) {$p=2P(Z\geq|z_0|)$};

\end{axis}

% 子图标题（外置 node）
\node[font=\bfseries\small] at ($(ax1.north)+(0,0.35cm)$) {单侧 $p$ 值};
\node[font=\bfseries\small] at ($(ax2.north)+(0,0.35cm)$) {双侧 $p$ 值};

% 整体标题
\node[font=\large\bfseries] at ($(ax1.north)!0.5!(ax2.north)+(0,1.2cm)$)
  {$p$ 值的几何示意};

\end{tikzpicture}
\end{document}
