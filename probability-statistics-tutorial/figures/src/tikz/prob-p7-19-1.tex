\begin{document}
\begin{tikzpicture}

\begin{axis}[
  name=ax,
  width=11cm, height=6.5cm,
  xmin=-3.8, xmax=4.5,
  ymin=-0.05, ymax=0.50,
  xlabel={$z$},
  ylabel={$f(z)$},
  axis lines=left,
  xtick={-3,-2,-1,0,1,1.645,2,3},
  xticklabels={$-3$,$-2$,$-1$,$0$,$1$,$z_\alpha$,$2$,$3$},
  ytick={0,0.1,0.2,0.3,0.4},
  yticklabel style={font=\scriptsize},
  xticklabel style={font=\scriptsize},
  grid=major, grid style={gray!30},
  clip=false
]
  % 右尾拒绝域蓝色阴影
  \addplot[fill=blue!30, opacity=0.8, draw=none, domain=1.645:3.8, samples=80]
    {exp(-x^2/2)/sqrt(2*3.14159)} \closedcycle;

  % 主曲线 N(0,1)
  \addplot[blue!70!black, ultra thick, domain=-3.8:3.8, samples=150]
    {exp(-x^2/2)/sqrt(2*3.14159)};

  % 临界值垂线
  \addplot[dashed, red!70!black, thick] coordinates {(1.645,0) (1.645,{exp(-1.645^2/2)/sqrt(2*3.14159)})};

  % H0 标注
  \node[font=\small, blue!70!black] at (axis cs:-1.2, 0.22) {$H_0$ 成立区域};

  % H1 标注
  \node[font=\small, blue!50!white!70!black] at (axis cs:2.6, 0.12) {拒绝$H_0$};

  % z_alpha 箭头
  \draw[->, thick, red!70!black] (axis cs:1.645, 0.28) -- (axis cs:1.645, 0.08);
  \node[font=\small, red!70!black, above] at (axis cs:1.645, 0.28) {$z_\alpha=1.645$};

  % 检验统计量 Z 示例
  \addplot[mark=*, mark size=3pt, only marks, orange!80!black]
    coordinates {(2.1, 0)};
  \node[font=\scriptsize, orange!80!black, above right] at (axis cs:2.1, 0.02) {观测 $Z$};

  % alpha=0.05 标注
  \node[font=\scriptsize, blue!60!black] at (axis cs:2.4, 0.05) {$\alpha=0.05$};

\end{axis}

% 公式框 + 说明（放在 axis 外）
\node[font=\small, align=center, fill=white, draw=gray!50, thin,
      rounded corners=2pt, inner sep=5pt]
  at ($(ax.south east)+(-1.5cm,-1.2cm)$)
  {若 $Z > z_\alpha$，则 $Z$ 落入阴影区 $\Rightarrow$ 拒绝 $H_0$};

% 整体标题
\node[font=\large\bfseries] at ($(ax.north)+(0,0.6cm)$)
  {假设检验框架：右侧检验（$\alpha=0.05$）};
