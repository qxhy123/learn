% Rational approximation for Phi(x) = standard normal CDF
% Using Abramowitz & Stegun 26.2.17, accurate to ~1e-4
\pgfmathdeclarefunction{normcdf}{1}{%
  \pgfmathparse{%
    (1/(1+0.47047*(abs(#1))))^3%
  }%
  \let\t=\pgfmathresult
  \pgfmathparse{%
    (#1>0) ?
      1 - (0.3480242*\t - 0.0958798*\t*\t + 0.7478556*\t*\t*\t)*exp(-#1*#1/2)/sqrt(2*pi)
    :
      (0.3480242*\t - 0.0958798*\t*\t + 0.7478556*\t*\t*\t)*exp(-#1*#1/2)/sqrt(2*pi)
  }%
}
